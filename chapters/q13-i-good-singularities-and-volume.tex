\chapter{I-good singularities and volume}
\label{chap:q13}

Let $X$ be a connected compact K\"ahler manifold of dimension $n\geq1$. Multiplier ideals can identify potentials whose non-pluripolar masses differ, as the polar example in \cref{ex:multiplier-and-model-types-differ} shows. The multiplier-envelope volume measures the mass of the largest potential with the prescribed ideal type. Equality with the original mass characterizes those positive-mass types that are limits of analytic singularities in $d_S$.

\section{Analytic closure and the mass gap}
\label{sec:i-good-analytic-closure}
Fix a big background $\theta$. Write $M_\theta(u)=\int_X\langle\theta_u^n\rangle$. A quasi-equisingular approximation keeps the multiplier ideals in the limit. Its analytic model projections therefore approach the multiplier-ideal envelope, which may have larger mass than the potential being approximated.

\begin{proposition}\label{prop:analytic-regularizations-limit-at-multiplier-envelope}
Let $\theta_u$ be a K\"ahler current, and let $u_j\in\PSH(X,\theta)$ be a decreasing quasi-equisingular approximation of $u$. Then
\[
 P_\theta[u_j]\downarrow P_\theta[u]_{\mathcal I},\qquad
 d_{S,\theta}(u_j,P_\theta[u]_{\mathcal I})\longrightarrow0.
\]
Such approximations can be chosen with every $\theta_{u_j}$ a K\"ahler current.
\end{proposition}
\begin{proof}
Normalize $u\leq0$. Put $p_j=P_\theta[u_j]=P_\theta[u_j]_{\mathcal I}$, using analytic envelope equality. The $p_j$ decrease, have supremum zero and lie above $u$. Their limit $p$ is model, has mass at least $M_\theta(u)>0$, and satisfies $p=P_\theta[p]_{\mathcal I}$ by decreasing multiplier-envelope continuity.

The inequality $u\leq p$ gives $\mathcal I(\lambda u)\subseteq\mathcal I(\lambda p)$. For each $\lambda>0$, strong openness and compactness give one $\lambda'>\lambda$ with $\mathcal I(\lambda p)=\mathcal I(\lambda'p)$ on $X$. Indeed use finite generators on relatively compact coordinate neighborhoods and take the minimum of the finitely many openness margins. Since $p\leq p_j$, quasi-equisingularity gives, for large $j$,
\[
 \mathcal I(\lambda p)=\mathcal I(\lambda'p)
     \subseteq\mathcal I(\lambda'p_j)
     =\mathcal I(\lambda'u_j)\subseteq\mathcal I(\lambda u).
\]
Thus $p\sim_{\mathcal I}u$ and $p=P_\theta[u]_{\mathcal I}$. Decreasing positive-mass model continuity gives $p_j\to p$ in $d_S$, and $u_j\sim_Pp_j$ gives the asserted limit.

If $\theta_u\geq\delta\omega$, apply gentle approximation with background $\theta-\delta\omega$. Its curvature loss tends to zero. After discarding finitely many terms, the resulting currents in background $\theta$ are at least $\delta\omega/2$. This proves the last assertion.
\end{proof}

\begin{theorem}\label{thm:i-good-analytic-closure-characterization}
For $u\in\PSH(X,\theta)$ with $M_\theta(u)>0$, the following conditions are equivalent:
\begin{enumerate}
 \item $u$ is a $d_{S,\theta}$ limit of potentials with rational analytic singularities in $\PSH(X,\theta)$;
 \item $M_\theta(u)=\operatorname{vol}_{\mathcal I}(\theta_u)$;
 \item $P_\theta[u]=P_\theta[u]_{\mathcal I}$.
\end{enumerate}
In the first condition the approximating currents may all be K\"ahler currents. If $\theta_u$ is a K\"ahler current and the equivalent conditions hold, every decreasing quasi-equisingular approximation lying in $\PSH(X,\theta)$ converges to $u$ in $d_S$.
\end{theorem}
\begin{proof}
For analytic potentials the mass equals the multiplier-envelope volume. If they converge to $u$, positive-limit continuity of both quantities gives the second condition. This argument applies to sequences or nets.

Put $a=P_\theta[u]$ and $b=P_\theta[u]_{\mathcal I}$. Both are model, $a\leq b$, and $M_\theta(a)=M_\theta(u)$. If the mass equals the multiplier volume, then $M_\theta(a)=M_\theta(b)>0$. The positive-mass maximum characterization forces $b\leq a$. Conversely envelope equality and mass preservation give the second condition.

Assume envelope equality. Choose the strict minorant $v\leq u$ of \cref{lem:positive-mass-strict-minorant} and set $z_k=(1-k^{-1})u+k^{-1}v$ for $k\geq2$. These increase to $u$, have strict currents and have positive mass. Increasing multiplier-envelope continuity gives
\[
 P_\theta[z_k]_{\mathcal I}\uparrow P_\theta[u]_{\mathcal I}=P_\theta[u]
\]
canonically, and increasing distance continuity gives convergence in $d_S$. The preceding proposition makes every $P_\theta[z_k]_{\mathcal I}$ a distance limit of analytic potentials with strict currents in this same class. Choose one such potential $a_k$ at distance below $1/k$ from that envelope. The triangle inequality and $u\sim_PP_\theta[u]$ give $a_k\to u$. The final assertion is the preceding proposition and envelope equality.
\end{proof}

\begin{definition}\label{def:i-good-singularity}
A quasi-psh potential $u$ is \emph{$\mathcal I$-good} if in some big background $\theta$ it has positive mass and satisfies $P_\theta[u]=P_\theta[u]_{\mathcal I}$. Its \emph{mass gap} in any admitting background is
\[
 G_\theta(u)=\operatorname{vol}_{\mathcal I}(\theta_u)-M_\theta(u)\geq0.
\]
\end{definition}
\index{singularity type!I-good@$\mathcal I$-good}\index{Monge--Ampere mass!gap from multiplier volume}

\begin{proposition}\label{prop:i-good-background-independence}
If $u$ is $\mathcal I$-good, then $G_\alpha(u)=0$ for every smooth closed background $\alpha$ admitting $u$. Whenever $\{\alpha\}$ is big and $M_\alpha(u)>0$, one also has
\[
 P_\alpha[u]=P_\alpha[u]_{\mathcal I}.
\]
Goodness depends only on the global $P$-singularity type.
\end{proposition}
\begin{proof}
First enlarge a big positive-mass background $\theta$ by a K\"ahler form. The analytic approximating sequence from the characterization still converges in the larger distance, by background independence. It remains analytic and admissible, and its limit has positive mass. The characterization therefore proves goodness in every sufficiently large K\"ahler background.

For an arbitrary admitting $\alpha$, choose such a background $\kappa$ with $\Omega=\kappa-\alpha$ K\"ahler, and put $b=P_\alpha[u]_{\mathcal I}$. It is ideal-equivalent to $u$, so
$M_\kappa(b)\leq\operatorname{vol}_{\mathcal I}(\kappa_u)=M_\kappa(u)$.
On the other hand $u\preceq b$. Mixed-mass monotonicity and the positive binomial expansion give
\[
 M_\kappa(b)-M_\kappa(u)
   =\sum_{j=0}^n\binom nj\int_X
       \bigl(\langle\alpha_b^j\wedge\Omega^{n-j}\rangle
                -\langle\alpha_u^j\wedge\Omega^{n-j}\rangle\bigr)\geq0.
\]
Every summand is nonnegative. Hence each vanishes; the term $j=n$ is $G_\alpha(u)$. Positive mass in a big $\alpha$ then gives envelope equality by the characterization.

If $u\sim_Pv$, take one large K\"ahler background giving both positive mass and in which $u$ is good. Their model envelopes agree. Their multiplier ideals and therefore multiplier envelopes agree as well. Thus $v$ has equal envelopes there and is good.
\end{proof}

\begin{corollary}\label{cor:i-good-singularity-closedness}
The $\mathcal I$-good potentials form a closed set for $d_S$ convergence in every fixed big class, including at zero-mass limits.
\end{corollary}
\begin{proof}
Enlarge the background by a K\"ahler form. All masses are then uniformly positive, convergence is preserved, and every member is good in that background. The mass gap is zero for the members. Positive-limit mass and multiplier-volume continuity make the limit gap zero. The characterization proves that the limit is good.
\end{proof}

\section{Operations and examples}
\label{sec:i-good-operations}
Analytic singularities survive sums and maxima. Distance continuity transfers these operations to their closure; rational approximation of coefficients then supplies arbitrary positive real weights.

\begin{proposition}\label{prop:i-good-positive-algebra-and-suprema}
If $u,v$ are $\mathcal I$-good and $a,b>0$, then $au+bv$ and $\max\{u,v\}$ are $\mathcal I$-good. Positive scaling and adding a smooth function preserve goodness. If $(u_i)$ is a nonempty locally uniformly upper bounded family of good potentials in a common background, its canonical upper envelope $(\sup_i u_i)^*$ is good.
\end{proposition}
\begin{proof}
Choose a large K\"ahler background giving $u,v$ positive mass, and approximate them in its distance by analytic potentials. Sum and maximum continuity give analytic approximations of $u+v$ and $\max\{u,v\}$, by \cref{prop:rational-analytic-operations}. Increase the fixed background by a positive form if needed to give these limits positive mass. The characterization proves both assertions. For a positive rational coefficient, scaling the analytic approximation and using distance scaling proves goodness of that multiple.

For arbitrary $a>0$, shift $u\leq0$, take rationals $a_j\downarrow a$, and choose one large K\"ahler background admitting all $a_ju$ with a positive margin. They increase to $au$, hence converge in its distance. Closedness proves goodness of $au$. Apply the sum result to $au,bv$. A smooth addition preserves ordinary singularity type, so it preserves $P$-type and goodness.

Finite maxima are now good. The finite subsets of the index set, ordered by inclusion, give an increasing net of finite maxima with the stated canonical envelope. Increasing distance continuity in an enlarged big background gives convergence of this net. Closedness proves goodness of the envelope.
\end{proof}

\begin{example}\label{ex:i-good-basic-types-and-mass-gap}
Every potential with rational analytic singularities is good: take a large positive background and use analytic envelope equality. Every positive-mass $\mathcal I$-model potential in a big class is good, because it is model and is its own multiplier envelope. Every full-mass potential about $V_\theta$ is good, since both its envelopes equal $V_\theta$. Thus the unbounded log-log potential in \cref{ex:loglog-full-mass-envelope} is good.

For the polar-Cantor potential $q$ of \cref{ex:multiplier-and-model-types-differ}, all multiplier ideals equal those of zero. In background $2\omega$, with $\int_{\mathbb P^1}\omega=1$, one has
\[
 M_{2\omega}(q)=1,\qquad
 P_{2\omega}[q]_{\mathcal I}=0,\qquad G_{2\omega}(q)=1.
\]
Indeed the ordinary current is $\omega+\mu$, and its polar-supported part $\mu$ is removed in the non-pluripolar restriction. Hence $q$ is not good, although it has positive mass in this background and is ideal-equivalent to the good potential zero.
\end{example}

\begin{definition}\label{def:elementary-psh-metric}
A psh metric $h=h_0\mathrm e^{-u}$ on a pseudo-effective line bundle $L$ is \emph{elementary} if $u$ has the same global $P$-singularity type as the weight of a generalized Fubini--Study metric on $L$. Write $\operatorname{Ele}(L)$ for their set.
\end{definition}
\index{Hermitian metric!elementary}

\begin{corollary}\label{cor:elementary-metrics-i-good}
Elementary metrics have $\mathcal I$-good weights. The definition and this property are independent of the choice of the smooth metric $h_0$.
\end{corollary}
\begin{proof}
The generalized Fubini--Study weights have rational analytic singularities by \cref{prop:fubini-study-finite-operations}, so are good. Goodness depends only on $P$-type. Changing $h_0$ subtracts the same smooth function from both compared weights and preserves their types.
\end{proof}

\section{Quasi-equisingularity and distance}
\label{sec:quasi-equisingular-distance}
The analytic-closure theorem distinguishes convergence to a potential from convergence to its multiplier envelope. For a decreasing analytic approximation, the two-weight ideal condition identifies the latter limit even when its curvature has a vanishing loss.

\begin{proposition}\label{prop:quasi-equisingular-distance-characterization}
Let $u\in\PSH(X,\theta)$ have positive mass. Suppose $u_j\downarrow u$ have rational analytic singularities and lie in $\PSH(X,\theta+\varepsilon_j\omega)$, where $\varepsilon_j\downarrow0$. Then the sequence is quasi-equisingular if and only if
\[
 u_j\longrightarrow P_\theta[u]_{\mathcal I}\quad\text{in }d_{S,\kappa}
\]
for a fixed sufficiently large K\"ahler background $\kappa$. The same convergence then holds in every fixed big background admitting the sequence and this limit.
\end{proposition}
\begin{proof}
Choose $\kappa$ so that $\kappa-\theta-(\varepsilon_1+1)\omega>0$. The original potential is strict in this background, and all sequence members are admissible. If the sequence is quasi-equisingular, the first proposition gives
$u_j\to P_\kappa[u]_{\mathcal I}$ in its distance. Put $p=P_\theta[u]_{\mathcal I}$. It has positive mass, is good and is ideal-equivalent to $u$. Background independence of goodness and multiplier-envelope equivalence give
\[
 P_\kappa[p]=P_\kappa[p]_{\mathcal I}=P_\kappa[u]_{\mathcal I}.
\]
Thus $p$ is at zero distance from that limit, proving the forward implication.

Conversely the two-weight multiplier stability theorem gives, for each $\lambda'>\lambda>0$, eventual inclusion
$\mathcal I(\lambda'u_j)\subseteq\mathcal I(\lambda p)=\mathcal I(\lambda u)$.
This is the remaining quasi-equisingular condition, since analyticity, decreasing convergence and the vanishing curvature losses are already assumed. The last assertion is distance background independence.
\end{proof}

\section{Mixed multiplier volumes}
\label{sec:mixed-multiplier-volumes}
The multiplier envelope removes the mass lost within one ideal type. For several currents, the mixed product of their envelopes has the same role. Its additivity follows from goodness of the envelopes, rather than linearity of the envelope operator. A K\"ahler perturbation then extends the construction to currents with zero volume.

\Needspace{22\baselineskip}
\begin{lemma}\label{lem:positive-multiplier-volume-polarization}
Write $T_i=\theta_i+\ddc u_i\geq0$, with $\{\theta_i\}$ big and $\operatorname{vol}_{\mathcal I}(T_i)>0$, and put $p_i=P_{\theta_i}[u_i]_{\mathcal I}$. The quantity
\[
 F(T_1,\ldots,T_n)=\int_X\left\langle\bigwedge_{i=1}^n(\theta_i+\ddc p_i)\right\rangle
\]
is independent of the smooth decompositions and symmetric. It is additive and positively homogeneous in each argument whenever the currents under consideration satisfy these hypotheses. In fixed classes it increases with multiplier-ideal order. Moreover
\[
 F(T_1+\varepsilon\omega,\ldots,T_n+\varepsilon\omega)
 =\int_X\left\langle\bigwedge_{i=1}^n(\theta_i+\varepsilon\omega+\ddc p_i)\right\rangle
\]
for every $\varepsilon>0$ and K\"ahler form $\omega$.
\end{lemma}
\begin{proof}
Background-change bounds from \cref{prop:envelopes-change-background-modification} change the envelope potentials by bounded terms after the corresponding smooth shifts. Mixed bounded-type invariance gives independence; symmetry is that of the non-pluripolar product. Multiplier order orders the normalized envelopes, so mixed-mass monotonicity gives the asserted order.

Every $p_i$ is good, by its positive mass. For an additional current $T_1'=\theta_1'+\ddc u_1'$ of the stated kind, put $p_1'=P_{\theta_1'}[u_1']_{\mathcal I}$. The sum $p_1+p_1'$ is good and ideal-equivalent to $u_1+u_1'$ by \cref{prop:preorders-algebra-and-envelopes}. Its mass in background $\theta_1+\theta_1'$ is positive: its expanded top product contains the positive term $M_{\theta_1}(p_1)$. That class is big, since the sum of two strict reference currents is strict. Goodness in this background gives
\[
 p_1+p_1'\sim_P P_{\theta_1+\theta_1'}[u_1+u_1']_{\mathcal I}.
\]
Replacing this argument preserves every mixed mass by \cref{thm:p-order-mixed-mass-monotonicity}. Positive multilinearity proves additivity of $F$. For $a>0$, the same reasoning uses goodness and ideal-equivalence of $ap_1$ to $au_1$ in background $a\theta_1$; it proves homogeneity for all real $a>0$.

Finally $p_i$ remains good with positive mass in $\theta_i+\varepsilon\omega$. It is ideal-equivalent to $u_i$, so it is $P$-equivalent there to $P_{\theta_i+\varepsilon\omega}[u_i]_{\mathcal I}$. Mixed $P$-invariance proves the last formula. Its right side is a polynomial with nonnegative coefficients in $\varepsilon$, whose constant term is $F(T_1,\ldots,T_n)$.
\end{proof}

\begin{definition}\label{def:mixed-multiplier-volume}
For arbitrary closed positive $(1,1)$-currents $T_1,\ldots,T_n$, their \emph{mixed multiplier volume} is
\[
 \operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)
  =\lim_{\varepsilon\downarrow0}F(T_1+\varepsilon\omega,\ldots,T_n+\varepsilon\omega),
\]
where $\omega$ is a K\"ahler form.
\end{definition}
\index{volume!mixed multiplier volume}\index{mixed volume!multiplier envelopes}
Each perturbed current is strict, so its class is big and its multiplier volume is at least $\varepsilon^n\int_X\omega^n$. Thus $F$ is defined at every positive parameter.

\begin{proposition}\label{prop:mixed-multiplier-volume-calculus}
The limit is finite, independent of $\omega$, and extends $F$. Mixed multiplier volume is symmetric, additive and nonnegatively homogeneous in every argument. It increases with multiplier-ideal order in fixed classes and satisfies
\[
 \int_X\langle T_1\wedge\cdots\wedge T_n\rangle
                 \leq\operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n).
\]
\end{proposition}
\begin{proof}
The preceding lemma implies monotonicity when a K\"ahler form is added to an argument: use its last formula with just that argument perturbed. Hence the defining numbers decrease as $\varepsilon\downarrow0$. They are nonnegative. Choose smooth representatives $\theta_i$ and $L>0$ with $\beta_i=\theta_i+L\omega$ K\"ahler. For $0<\varepsilon\leq1$, adding $L\omega$ and comparing the resulting normalized envelopes with zero bounds the defining numbers by
$\int_X\bigwedge_i(\beta_i+\omega)$.
This proves finiteness. Two K\"ahler forms satisfy $c^{-1}\omega\leq\omega'\leq c\omega$ for some $c\geq1$. For a semipositive increment $\Omega$, a positive-mass envelope $p_i$ remains good and has positive mass in the big background $\theta_i+\Omega$. Its multiplier envelope there is $P$-equivalent to $p_i$. The resulting mixed formula expands against $\Omega$ into nonnegative terms, giving monotonicity for such increments. Sandwiching proves independence. The polynomial formula proves agreement with $F$ on its original domain.

The same sandwich shows that different fixed positive multiples of $\varepsilon\omega$ in the arguments have the same limiting value. Apply positive-domain additivity to $T_1+\varepsilon\omega$ and $T_1'+\varepsilon\omega$; the sum contains $2\varepsilon\omega$. Taking limits gives additivity. Scaling a perturbed argument and taking limits gives homogeneity for $a>0$. An argument $\varepsilon\omega$ bounds the mixed product by $\varepsilon$ times a fixed smooth mixed bound, so an argument zero gives value zero. This proves homogeneity at $a=0$. Symmetry and ideal-order monotonicity pass to the limit.

The original potentials lie below their multiplier envelopes up to constants. Mixed-mass monotonicity therefore bounds the mixed product of the perturbed original currents by $F$. Its limit is the original mixed mass by positive multilinearity and global finiteness. This proves the lower bound.
\end{proof}

\begin{lemma}\label{lem:mixed-multiplier-volume-perturbation-bound}
For fixed classes $\{T_1\},\ldots,\{T_n\}$ and a K\"ahler form $\omega$, there is $C$ depending only on these data such that
\[
 0\leq\operatorname{vol}_{\mathcal I}(T_1+\varepsilon\omega,\ldots,T_n+\varepsilon\omega)
       -\operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)\leq C\varepsilon
\]
for $0\leq\varepsilon\leq1$. The same $C$ works for all currents in these classes.
\end{lemma}
\begin{proof}
Choose the harmonic representatives $\theta_i$ of the classes and one $L>0$ with $\beta_i=\theta_i+L\omega$ K\"ahler. For every nonempty subset $J\subseteq\{1,\ldots,n\}$, monotonicity under addition and the smooth upper bound give
\[
 \operatorname{vol}_{\mathcal I}\bigl((\omega)_{i\in J},(T_i)_{i\notin J}\bigr)
               \leq\int_X\omega^{|J|}\wedge\bigwedge_{i\notin J}\beta_i.
\]
Here the notation denotes an $n$-tuple in the indicated positions. To see the smooth bound, first perturb every argument, add $L\omega$ to the currents $T_i$, and compare their normalized envelopes in the K\"ahler backgrounds with zero; then let the perturbation vanish. Expand by additivity and homogeneity. The difference in the statement is the sum of these coefficients times $\varepsilon^{|J|}$. Since $\varepsilon^{|J|}\leq\varepsilon$, one may take
\[
 C=\sum_{\varnothing\ne J\subseteq\{1,\ldots,n\}}
          \int_X\omega^{|J|}\wedge\bigwedge_{i\notin J}\beta_i.
\]
The harmonic representative construction and $L$ depend only on the fixed classes and geometry.
\end{proof}

\begin{proposition}\label{prop:multiplier-envelope-boundary-background-limit}
For every admitting background $\theta$ and $u\in\PSH(X,\theta)$,
\[
 P_{\theta+\varepsilon\omega}[u]_{\mathcal I}\downarrow P_\theta[u]_{\mathcal I}
                         \quad(\varepsilon\downarrow0).
\]
Consequently the mixed construction extends the pure volume from \cref{def:multiplier-envelope-volume}:
\[
 \operatorname{vol}_{\mathcal I}(T,\ldots,T)=\operatorname{vol}_{\mathcal I}(T).
\]
\end{proposition}
\begin{proof}
Put $p=P_\theta[u]_{\mathcal I}$ and $p_\varepsilon=P_{\theta+\varepsilon\omega}[u]_{\mathcal I}$. The latter decrease, are normalized and dominate $p$. Their limit $q$ is therefore a nonpositive $\theta$-psh potential. For every divisor $E$ over $X$, they all have Lelong number $\nu(u,E)$. Decreasing-limit upper semicontinuity gives $\nu(q,E)\geq\nu(u,E)$. The order $q\geq p$ gives the reverse inequality. Thus $q\sim_{\mathcal I}u$ and, being a normalized competitor, $q\leq p$. This proves the limit.

Each $p_\varepsilon$ has positive mass and is a fixed point of its ordinary envelope. \Cref{prop:perturbed-model-limit-mass} applies even when the limiting class is not big or the mass is zero. It gives
\[
 F(T+\varepsilon\omega,\ldots,T+\varepsilon\omega)
       \longrightarrow\int_X\langle(\theta+\ddc p)^n\rangle,
\]
which proves the pure-volume identity.
\end{proof}

\begin{theorem}\label{thm:mixed-multiplier-volume-distance-continuity}
Let $\theta_1,\ldots,\theta_n$ represent big classes. If $u_i^j\to u_i$ in $d_{S,\theta_i}$, for nets with one directed index set, then
\[
 \operatorname{vol}_{\mathcal I}(\theta_1+\ddc u_1^j,\ldots,\theta_n+\ddc u_n^j)
       \longrightarrow\operatorname{vol}_{\mathcal I}(\theta_1+\ddc u_1,\ldots,\theta_n+\ddc u_n).
\]
In particular pure multiplier volume is continuous at every mass, including zero.
\end{theorem}
\begin{proof}
Fix $\varepsilon>0$. Background independence preserves each distance convergence in $\theta_i+\varepsilon\omega$, where all original masses are uniformly positive. Multiplier-projection distance continuity and mixed-mass continuity give convergence of the mixed volumes at this parameter. The perturbation bound supplies one $C$ independent of $j$, because all classes are fixed. The difference from the desired values is at most $C\varepsilon$ for both the net and its limit. Given a tolerance, first choose $\varepsilon$ with $2C\varepsilon$ below half that tolerance and then use the perturbed convergence. The pure case follows by repeating one current in every argument.
\end{proof}
\index{volume!continuity at zero mass}

\begin{corollary}\label{cor:i-good-mixed-volume-equality}
If the potentials of $T_1,\ldots,T_n$ are good, then
\[
 \operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)
                          =\int_X\langle T_1\wedge\cdots\wedge T_n\rangle.
\]
\end{corollary}
\begin{proof}
After a positive K\"ahler perturbation every background is big and every original mass is positive. Intrinsic goodness identifies the multiplier envelope with the model envelope in each background. Replacing the arguments preserves the mixed mass. Letting the parameter decrease to zero proves the equality by positive multilinearity, including when original masses vanish.
\end{proof}

\begin{proposition}\label{prop:mixed-multiplier-volume-divisorial-removal}
A positive divisorial current contributes zero to any argument of mixed multiplier volume. If $R_i$ is the non-divisorial part of $T_i$ in Siu's decomposition, then
\[
 \operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)
                              =\operatorname{vol}_{\mathcal I}(R_1,\ldots,R_n).
\]
\end{proposition}
\begin{proof}
For a prime divisor $D$, its logarithmic potential is analytic and hence good. The preceding corollary, applied after perturbing the other arguments and replacing them by their good multiplier envelopes, gives
\[
 F([D]+\varepsilon\omega,T_2+\varepsilon\omega,\ldots,T_n+\varepsilon\omega)
     =\varepsilon\int_X\omega\wedge
         \left\langle\bigwedge_{i=2}^n(\theta_i+\varepsilon\omega+\ddc p_{i,\varepsilon})\right\rangle.
\]
The divisor itself disappears in the non-pluripolar restriction. A fixed smooth mixed bound makes the right side tend to zero. Additivity and real positive homogeneity handle finite divisorial sums.

Now write $S=\sum_{k\geq1}c_k[D_k]\geq0$. Its tails $S-S_N$ tend weakly to zero by Siu's decomposition. Their classes therefore tend to zero. For fixed $\varepsilon>0$, the harmonic representatives $\eta_N$ of these tail classes tend smoothly to zero; for large $N$, $\omega_N=\varepsilon\omega+\eta_N$ is K\"ahler. The currents $S+\varepsilon\omega$ and $S_N+\omega_N$ have the same class. Their potentials differ by the global potential of the tail relative to $\eta_N$, which is bounded above. Thus the first is ordinarily, hence ideally, more singular. Monotonicity and removal of the finite sum give
\[
 \operatorname{vol}_{\mathcal I}(S,T_2,\ldots,T_n)
    \leq\operatorname{vol}_{\mathcal I}(\omega_N,T_2,\ldots,T_n)
    \leq\int_X\omega_N\wedge\bigwedge_{i=2}^n\beta_i,
\]
where fixed K\"ahler $\beta_i$ dominate smooth representatives of the other classes, as in the perturbation bound. Let $N\to\infty$ and then $\varepsilon\downarrow0$ to obtain zero. Additivity now removes each divisorial part of the original tuple.
\end{proof}
\index{volume!removal of divisorial currents}

\Needspace{12\baselineskip}
\begin{theorem}\label{thm:mixed-multiplier-volume-divisorial-upper-limit}
For nets $u_i^j,u_i\in\PSH(X,\theta_i)$, suppose $\nu(u_i^j,E)\to\nu(u_i,E)$ for every divisor $E$ over $X$ and every $1\leq i\leq n$.
Then
\[
 \limsup_j\operatorname{vol}_{\mathcal I}(\theta_1+\ddc u_1^j,\ldots,\theta_n+\ddc u_n^j)
      \leq\operatorname{vol}_{\mathcal I}(\theta_1+\ddc u_1,\ldots,\theta_n+\ddc u_n).
\]
\end{theorem}
\begin{proof}
First perturb by $\varepsilon\omega>0$ and write $\alpha_i=\theta_i+\varepsilon\omega$. Put $q_i^j=P_{\alpha_i}[u_i^j]_{\mathcal I}$ and $q_i=P_{\alpha_i}[u_i]_{\mathcal I}$. These are normalized good model potentials of mass at least $\varepsilon^n\int_X\omega^n$. For each tail define
\[
 s_i^j=(\sup_{k\geq j}q_i^k)^*,\qquad p_i^j=P_{\alpha_i}[s_i^j].
\]
Upper-envelope stability makes $s_i^j$ good. Its positive mass therefore gives $p_i^j=P_{\alpha_i}[s_i^j]_{\mathcal I}$. For every fixed divisor, envelope closure gives
$\nu(s_i^j,E)=\inf_{k\geq j}\nu(u_i^k,E)$.
These numbers increase to $\nu(u_i,E)$ and are at most that number. Hence $q_i\leq p_i^j$ by ideal order. The $p_i^j$ decrease to a positive-mass model limit $p_i$. Decreasing multiplier-envelope continuity makes it $\mathcal I$-model. Decreasing positive-mass divisorial continuity identifies its numbers with those of $q_i$, so $p_i=q_i$.

The decreasing models converge in $d_S$. For $k\geq j$ one has $q_i^k\leq s_i^j\leq p_i^j$. Mixed-mass monotonicity bounds every volume on that tail by the mixed mass of the $p_i^j$, which converges to that of the $q_i$. This proves the desired upper inequality for the perturbed currents. Finally the uniform perturbation bound changes the right side by at most $C\varepsilon$, while perturbing can only increase the left side. Let $\varepsilon\downarrow0$. The argument also covers backgrounds which are only pseudo-effective, since their positive perturbations are big.
\end{proof}
\index{volume!divisorial upper semicontinuity}

\begin{corollary}\label{cor:mixed-multiplier-volume-modification-invariance}
For a proper bimeromorphic morphism $\pi:Y\to X$ between connected compact K\"ahler manifolds,
\[
 \operatorname{vol}_{\mathcal I}(\pi^*T_1,\ldots,\pi^*T_n)
                        =\operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n).
\]
\end{corollary}
\begin{proof}
Symmetry and additivity give the polarization identity
\[
 n!\operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)
   =\sum_{J\subseteq\{1,\ldots,n\}}(-1)^{n-|J|}
                \operatorname{vol}_{\mathcal I}\left(\sum_{i\in J}T_i\right).
\]
The empty sum is zero. Expansion cancels every monomial omitting an index; the monomials using every index occur $n!$ times. Apply pure-volume modification invariance from \cref{cor:multiplier-envelope-volume-bounds} to each finite sum, and use linearity of pullback in the same formula on $Y$.
\end{proof}

\section{Analytic approximation on modifications}
\label{sec:analytic-volume-approximation}
Gentle approximation has a numerical interpretation on a resolution. Its logarithmic divisor accounts for the singularities, while the remaining current is smooth and semipositive. A small exceptional correction makes this residual class K\"ahler. The resulting intersection numbers approximate the multiplier volume; for minimal singularities they approximate the volume of the class.

\begin{lemma}\label{lem:exceptional-kahler-correction}
Let $\pi:Y\to X$ be a finite composition of blowups with smooth centers. For a K\"ahler form $\omega$ on $X$, there is an effective rational exceptional divisor $F$ such that $\pi^*\{\omega\}-\{F\}$ has a K\"ahler representative $\gamma$. If $S$ is smooth and $S\geq\delta\pi^*\omega$ for $\delta>0$, then
\[
 S_t=S+t(\gamma-\pi^*\omega)
\]
is K\"ahler for $0<t\leq\delta$, and $\{S_t\}=\{S\}-t\{F\}$.
\end{lemma}
\begin{proof}
For one blowup, \cref{prop:kahler-smooth-center-blowup} constructs a K\"ahler form in $\pi^*\{\omega\}-q\{E\}$ for all sufficiently small $q>0$. Choose $q$ rational. Inductively pull back the effective rational divisor already chosen and add a small positive rational multiple of the new exceptional divisor. This gives an effective rational exceptional divisor at every stage. A codimension-one blowup is an isomorphism and adds nothing. In the identity case take $F=0$ and $\gamma=\omega$.

The displayed form is at least $(\delta-t)\pi^*\omega+t\gamma$, which is positive for $0<t\leq\delta$. It is closed, and its class has the stated value. For any finite family $S_i\geq\delta_i\pi^*\omega$, the same correction works simultaneously when $t\leq\min_i\delta_i$. Their smooth mixed integrals tend to the original integrals as $t\downarrow0$, by expansion of the finite polynomial in $t$.
\end{proof}

\begin{theorem}\label{thm:analytic-fujita-multiplier-volume}
Let $T=\theta+\ddc u\geq0$ in a big class, with $\operatorname{vol}_{\mathcal I}(T)>0$. There are gentle analytic K\"ahler currents $T_k=\theta+\ddc a_k$, finite smooth-center modifications $\pi_k:Y_k\to X$, smooth K\"ahler forms $\kappa_k$ and effective rational divisors $D_k$ on $Y_k$ such that
\[
 a_k\longrightarrow P_\theta[u]_{\mathcal I}\quad\text{in }d_{S,\theta},
 \qquad \pi_k^*\{T\}=\{\kappa_k\}+\{D_k\},
 \qquad \int_{Y_k}\kappa_k^n\longrightarrow\operatorname{vol}_{\mathcal I}(T).
\]
For every $\lambda'>\lambda>0$, their ideals eventually satisfy
\[
 \mathcal I(\lambda u)\subseteq\mathcal I(\lambda a_k),\qquad
 \mathcal I(\lambda'a_k)\subseteq\mathcal I(\lambda u).
\]
\end{theorem}
\begin{proof}
Put $p=P_\theta[u]_{\mathcal I}$. It is a good model potential of positive mass. In the analytic-closure proof, choose a strict minorant $v\leq p$ and the increasing strict potentials $z_k=(1-k^{-1})p+k^{-1}v$. Their multiplier envelopes increase canonically to $p$ and converge in distance. Apply the gentle approximation theorem to each $z_k$ with a background decreased by its strict margin. The analytic regularization limit proposition makes these gentle strict currents approach $P_\theta[z_k]_{\mathcal I}$. A diagonal choice gives $a_k\to p$ in the stated distance. Mass continuity gives $M_\theta(a_k)\to M_\theta(p)=\operatorname{vol}_{\mathcal I}(T)$.

Resolve the specified gentle ideal type of $a_k$ by a finite smooth-center modification. Upstairs the current splits as
\[
 \pi_k^*T_k=S_k+[B_k],
\]
where $B_k$ is effective rational and $S_k$ is smooth. On the complement of the divisor, strict positivity gives $S_k\geq\delta_k\pi_k^*\omega$ for some $\delta_k>0$. Smoothness extends this inequality to all of $Y_k$. In particular $\{S_k\}$ is nef, and locality and modification invariance give
\[
 M_\theta(a_k)=\int_{Y_k}S_k^n.
\]
Indeed the divisor is removed in the non-pluripolar restriction, while the smooth top product charges none of its components.

Use the exceptional correction lemma, choosing a positive rational $t_k\leq\delta_k$ so small that
$|\int_{Y_k}S_{k,t_k}^n-\int_{Y_k}S_k^n|<1/k$.
Set $\kappa_k=S_{k,t_k}$ and $D_k=B_k+t_kF_k$. Their classes add to $\pi_k^*\{T\}$, and the desired intersection limit follows. Finally paired multiplier stability applies to $a_k\to p$, and $p\sim_{\mathcal I}u$, proving both inclusions. No constant or coefficient in the resolution is asserted uniform in $k$.
\end{proof}
\index{Fujita approximation!multiplier volume}\index{analytic singularities!smooth residual on a resolution}

\begin{corollary}\label{cor:transcendental-fujita-volume}
For every big class $\alpha$,
\[
 \operatorname{vol}(\alpha)
 =\sup\left\{\int_Y\kappa^n:
       \pi^*\alpha=\{\kappa\}+\{D\},\quad
       \kappa\text{ K\"ahler},\quad D\text{ effective rational}\right\},
\]
where $\pi:Y\to X$ ranges over proper bimeromorphic morphisms from connected compact K\"ahler manifolds. Finite smooth-center modifications suffice for the supremum.
\end{corollary}
\begin{proof}
Apply the theorem to the minimal current $\theta+\ddc V_\theta$, which is good and has multiplier volume $\operatorname{vol}(\alpha)$. This gives the lower bound for the supremum using finite smooth-center modifications.

For any displayed decomposition, the positive current $\pi_*(\kappa+[D])$ lies in $\alpha$. Pushforward commutes with $\ddc$, and $\pi_*\pi^*\theta=\theta$ for a smooth representative: test on the complement of the exceptional image, where $\pi$ is an isomorphism, and use that smooth top forms give analytic subsets zero mass. This identifies its class. On that same complement its non-pluripolar top product is the transform of $\kappa^n$, after removing the divisor. The exceptional image and the divisors are pluripolar downstairs and upstairs, while the smooth top product gives them zero mass. Locality therefore identifies the total mass with $\int_Y\kappa^n$. The class mass bound makes this at most $\operatorname{vol}(\alpha)$, proving the upper bound.
\end{proof}
\index{volume!Fujita approximation of a big class}

\Needspace{13\baselineskip}
\begin{theorem}\label{thm:mixed-fujita-boundary-approximation}
For arbitrary closed positive currents $T_i\in\alpha_i$, $1\leq i\leq n$, fix a K\"ahler form $\omega$ and positive rational $\varepsilon_k\downarrow0$. There are finite smooth-center modifications $\pi_k:Y_k\to X$, smooth K\"ahler forms $\kappa_{i,k}$ and effective rational divisors $D_{i,k}$ with
\[
 \pi_k^*(\alpha_i+\varepsilon_k\{\omega\})
               =\{\kappa_{i,k}\}+\{D_{i,k}\}
\]
and, for every $(j_1,\ldots,j_n)\in\{1,\ldots,n\}^n$,
\[
 \int_{Y_k}\kappa_{j_1,k}\wedge\cdots\wedge\kappa_{j_n,k}
                     \longrightarrow\operatorname{vol}_{\mathcal I}(T_{j_1},\ldots,T_{j_n}).
\]
\end{theorem}
\begin{proof}
For fixed $k$, all currents $T_i+\varepsilon_k\omega$ are strict in big classes and have positive multiplier volume. Apply the gentle analytic diagonal construction of the preceding theorem to their multiplier envelopes, in these fixed backgrounds. Mixed-mass distance continuity applies to every ordered $n$-tuple of indices, including repetitions. There are finitely many such tuples. Intersecting their product neighborhoods lets us choose one analytic current for each $i$ so that every indicated mixed mass differs from its perturbed mixed multiplier volume by less than $1/k$.

Principalize the finitely many ideal types successively, retaining each existing boundary divisor. This gives a common finite smooth-center resolution on which all currents split as $S_{i,k}+[B_{i,k}]$. Smooth pullback preserves gentleness at later steps. Every $S_{i,k}$ is at least $\delta_{i,k}\pi_k^*\omega$ for some $\delta_{i,k}>0$. Locality and modification invariance identify each smooth mixed integral, with repetitions, with its chosen downstairs mixed mass.

Apply one exceptional correction on this common resolution. Each of the finitely many mixed integrals is a polynomial in its correction parameter. Choose one positive rational $t_k\leq\min_i\delta_{i,k}$ so small that every corrected integral differs by less than $1/k$. Put $\kappa_{i,k}=S_{i,k}+t_k(\gamma_k-\pi_k^*\omega)$ and $D_{i,k}=B_{i,k}+t_kF_k$. These are the required forms and divisors. Every discrepancy from its perturbed mixed volume is below $2/k$. The maximum of the finitely many class-uniform perturbation constants bounds all remaining discrepancies by $C\varepsilon_k$. This proves all limits at every original mass.
\end{proof}
\index{Fujita approximation!mixed boundary volumes}

The same modifications now carry both the fully mixed intersections and every pure top intersection. The smooth mixed inequality therefore passes to multiplier volumes.

\begin{corollary}\label{cor:mixed-multiplier-volume-log-concavity}
For arbitrary closed positive $(1,1)$-currents $T_1,\ldots,T_n$,
\[
 \operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n)
             \geq\prod_{i=1}^n\operatorname{vol}_{\mathcal I}(T_i)^{1/n}.
\]
\end{corollary}
\begin{proof}
Apply \cref{cor:nef-mixed-intersection-geometric-mean} on $Y_k$ to the forms $\kappa_{i,k}$. The simultaneous approximation theorem gives convergence of their mixed and all pure top intersections to the corresponding multiplier volumes. The nonnegative $n$th root is continuous, including at zero, so the inequality passes to the limit. For $n=1$ it is an identity.
\end{proof}
\index{multiplier volume!log-concavity}


\section{Pushforward and quasi-equisingular pullback}
\label{sec:volume-goodness-modifications}
Divisorial removal and the analytic support theorem identify the volume of a pushed current. A K\"ahler perturbation then transfers intrinsic goodness in both directions, including currents whose original non-pluripolar mass vanishes.

\begin{proposition}\label{prop:pushforward-mixed-volume-and-mass}
For closed positive currents $T_1,\ldots,T_n$ on $Y$ and a proper bimeromorphic morphism $\pi:Y\to X$ of connected compact K\"ahler manifolds,
\[
 \operatorname{vol}_{\mathcal I}(\pi_*T_1,\ldots,\pi_*T_n)
               =\operatorname{vol}_{\mathcal I}(T_1,\ldots,T_n).
\]
Their mixed non-pluripolar top masses also agree.
\end{proposition}
\begin{proof}
Write $T_i=R_i+S_i$ in Siu's decomposition, with $R_i$ non-divisorial and $S_i$ a positive divisorial current. \Cref{lem:nondivisorial-pull-push-exceptional} and continuity of proper pushforward make $\pi_*S_i$ positive divisorial, possibly zero. Divisorial removal reduces both multiplier volumes to the tuples $R_i$ and $\pi_*R_i$.

By \cref{lem:nondivisorial-pull-push-exceptional}, every $\pi^*\pi_*R_i$ differs from $R_i$ by an effective exceptional divisor current. Divisorial removal and mixed modification invariance give
\[
 \begin{aligned}
 \operatorname{vol}_{\mathcal I}(R_1,\ldots,R_n)
  &=\operatorname{vol}_{\mathcal I}(\pi^*\pi_*R_1,\ldots,\pi^*\pi_*R_n)\\
  &=\operatorname{vol}_{\mathcal I}(\pi_*R_1,\ldots,\pi_*R_n).
 \end{aligned}
\]
For the ordinary mixed non-pluripolar masses, use the isomorphism off the exceptional image. Locality identifies the measures there, and the exceptional locus and its image are pluripolar, so both top products give them zero mass. Integration proves equality for the original tuple.
\end{proof}
\index{volume!bimeromorphic pushforward}

\begin{corollary}\label{cor:i-good-under-modifications}
Pushforward and pullback under such a morphism preserve I-good currents.
\end{corollary}
\begin{proof}
First suppose $T$ on $Y$ is good. For a K\"ahler form $\omega_X$ and $A>0$, the current $S=T+A\pi^*\omega_X$ has the same potential in its changed background. Intrinsic background independence gives
$\operatorname{vol}_{\mathcal I}(S)=M(S)$.
The proposition gives the same identity for $\pi_*S=\pi_*T+A\omega_X$. This current is strict, so its class is big and its mass positive. The positive-mass characterization makes it good. Its potential is also that of $\pi_*T$ in the original background, proving pushforward goodness.

Now let $T$ on $X$ be good and put $R=\pi^*T$. The smooth current $\omega_Y$ is good; the part just proved makes $\pi_*\omega_Y$ good. Positive algebra makes
$\pi_*(R+\omega_Y)=T+\pi_*\omega_Y$ good. Its mass gap vanishes in its admitting background. The proposition transfers this identity to $R+\omega_Y$, which is strict with positive mass in a big class. It is good by the characterization, and hence so is $R$, with its unchanged potential in the original background.
\end{proof}
\index{I-good singularity!bimeromorphic invariance}

\begin{corollary}\label{cor:quasi-equisingular-pullback}
For a proper bimeromorphic morphism $\pi:Y\to X$ of connected compact K\"ahler manifolds, a quasi-equisingular approximation $u_j$ of $u\in\PSH(X,\theta)$ pulls back to a quasi-equisingular approximation $u_j\circ\pi$ of $u\circ\pi$.
\end{corollary}
\begin{proof}
The pulled functions decrease to $u\circ\pi$ and have analytic singularities. If $u_j$ has background $\theta+\varepsilon_j\omega_X$, compactness gives the pulled curvature bound in
$\pi^*\theta+C\varepsilon_j\omega_Y$, with $C$ fixed and $\varepsilon_j\downarrow0$.

Choose $A>0$, put $\theta_A=\theta+A\omega_X$, and $p=P_{\theta_A}[u]_{\mathcal I}$. The current of $u$ in this background is strict. \Cref{prop:quasi-equisingular-distance-characterization} gives $u_j\to p$ in distance in one fixed sufficiently large background. Pullback distance invariance and \cref{prop:big-classes-under-modifications} give convergence of the pulled functions to $p\circ\pi$ in the pulled big background. The multiplier-envelope modification identity identifies
$p\circ\pi=P_{\pi^*\theta_A}[u\circ\pi]_{\mathcal I}$.
It has positive mass by modification invariance, and $\pi^*\{\theta_A\}$ is big by the same proposition. Move the convergence to a fixed sufficiently large K\"ahler background on $Y$, using background independence. The reverse implication in the quasi-equisingular distance characterization now gives all paired multiplier inclusions upstairs. Together with the first paragraph this proves the assertion in the original background $\pi^*\theta$.
\end{proof}
\index{quasi-equisingular approximation!bimeromorphic pullback}

\section{Sections of strictly positive singular metrics}
\label{sec:strict-singular-section-asymptotics}
\begin{definition}\label{def:hermitian-pseudo-effective-line-bundle}
A \emph{Hermitian pseudo-effective line bundle} on a complex manifold is a holomorphic line bundle $L$ together with a quadratic psh metric $h$.
\end{definition}
On $X$, in a smooth reference metric write $h=h_0\mathrm e^{-u}$ and $\theta=c_1(L,h_0)$, so its curvature is $\theta_u$. We write $\mathcal I(h^k)=\mathcal I(ku)$; a change of smooth reference adds a smooth function and leaves this ideal unchanged. The metric's multiplier volume is $\operatorname{vol}_{\mathcal I}(\theta_u)$.
\index{line bundle!Hermitian pseudo-effective}

\begin{proposition}\label{prop:strict-analytic-type-section-asymptotic}
Let $L,T$ be holomorphic line bundles, let $h=h_0\mathrm e^{-u}$, and suppose $u$ has a specified rational analytic type $(c,\mathfrak a)$ with $\theta_u\geq\delta\omega$ for $\delta>0$. Then
\begin{equation}\label{eq:strict-analytic-section-asymptotic}
 \lim_{k\to\infty}\frac{n!}{k^n}
       h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       =M_\theta(u)=\operatorname{vol}_{\mathcal I}(\theta_u).
\end{equation}
\end{proposition}
\begin{proof}
Resolve the specified coherent ideal by a finite smooth-center modification $\pi:Y\to X$. Its resolved logarithmic divisor is an effective rational simple-normal-crossing divisor $D$. Thus
\[
 \pi^*\theta_u=[D]+R,
 \qquad \alpha=\{R\}=c_1(\pi^*L)-\{D\},
\]
where $R$ has bounded local potentials. Apply the divisorial decomposition to the positive current $\pi^*(\theta_u-\delta\omega)$. Smooth subtraction leaves the logarithmic coefficients unchanged. Its residual is positive, so $R\geq\delta\pi^*\omega$. Bounded-potential regularization makes $\alpha$ nef.

Choose the rational exceptional correction $F$ and K\"ahler form $\gamma\in\pi^*\{\omega\}-\{F\}$ from \cref{lem:exceptional-kahler-correction}. For rational $0<t\leq\delta$,
\[
 R+t(\gamma-\pi^*\omega)\geq t\gamma
\]
has bounded local potentials, so \cref{prop:bounded-potential-nef} makes $\alpha-t\{F\}$ K\"ahler. Choose $b$ clearing the coefficients of $D+tF$. The class $b(\alpha-t\{F\})$ is the first Chern class of
$(\pi^*L)^b\otimes\mathcal O_Y(-bD-btF)$.
The global $\ddc$ lemma gives it a positive metric, establishing the nef section theorem's hypothesis on $Y$.

The birational multiplier transformation \cref{prop:birational-weighted-change-variables}, in local frames of $T\otimes L^k$, gives an identification of section spaces
\[
 \begin{aligned}
 &H^0(X,T\otimes L^k\otimes\mathcal I(ku))\\
 &\quad\simeq H^0\bigl(Y,\mathcal O_Y(K_{Y/X})\otimes\pi^*T
            \otimes(\pi^*L)^k\otimes\mathcal O_Y(-\lfloor kD\rfloor)\bigr).
 \end{aligned}
\]
The bounded remainder changes no integrability, so normal-crossing polar integration gives $\mathcal I(ku\circ\pi)=\mathcal O_Y(-\lfloor kD\rfloor)$. The cited transformation includes holomorphic descent across the exceptional image.

Choose $q\geq1$ with $qD$ integral, and write $k=qm+s$, $0\leq s<q$. Set
\[
 B=(\pi^*L)^q\otimes\mathcal O_Y(-qD),\qquad
 T_s=\mathcal O_Y(K_{Y/X})\otimes\pi^*T
         \otimes(\pi^*L)^s\otimes\mathcal O_Y(-\lfloor sD\rfloor).
\]
The upstairs bundle is $B^m\otimes T_s$, and $c_1(B)=q\alpha$ is nef. \Cref{thm:projective-nef-section-asymptotic} gives the limit $\int_Y\alpha^n$ on every residue class. Bounded global Stokes identifies this intersection with $\int_YR^n$. Divisorial removal and modification invariance for non-pluripolar mass identify the latter with $M_\theta(u)$. Finally analytic singularities are I-good, so this mass equals the multiplier volume.
\end{proof}
\index{sections!strict analytic metric asymptotics}

\begin{theorem}\label{thm:strict-psh-metric-section-asymptotic}
Let $(L,h_0\mathrm e^{-u})$ be a Hermitian pseudo-effective line bundle with $\theta_u\geq\delta_0\omega$ for $\delta_0>0$, and let $T$ be any fixed holomorphic line bundle. Then
\begin{equation}\label{eq:strict-psh-section-asymptotic}
 \lim_{k\to\infty}\frac{n!}{k^n}
       h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       =\operatorname{vol}_{\mathcal I}(\theta_u).
\end{equation}
For analytic $u$, the right side is also $M_\theta(u)$.
\end{theorem}
\begin{proof}
Apply gentle quasi-equisingular approximation with background $\theta-\delta_0\omega$. After discarding finitely many terms, it gives $u_j\downarrow u$ with $\theta_{u_j}\geq\delta\omega$ for one $\delta>0$. A common constant makes $u_j,u\leq0$. Put $p=P_\theta[u]_{\mathcal I}$. The regularization-limit proposition gives
\[
 d_{S,\theta}(u_j,p)\to0,\qquad
 M_\theta(u_j)\to M_\theta(p)>0,
 \qquad p\leq P_\theta[u_j].
\]
The ideals of $p$ and $u$ agree for every positive coefficient. Since $u\leq u_j$, the preceding proposition gives
\[
 \limsup_k\frac{n!}{k^n}
      h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       \leq M_\theta(u_j).
\]
Let $j\to\infty$ to obtain the desired upper bound.

For the lower bound enlarge the auxiliary K\"ahler form so that $-\omega\leq\theta\leq\omega$, retaining a positive common strict margin $\delta$. Fix an integer $N\geq2$ with $N\delta>1$. Choose a strict minorant $w\leq p$ by \cref{lem:positive-mass-strict-minorant}; its class is big because $u$ is strict. For large $j$, \cref{thm:model-order-rooftop-extrapolation} with coefficient $\varepsilon=1/(2N)$ supplies $\eta_j\in\PSH(X,\theta)$ such that
\[
 (1-\varepsilon)u_j+\varepsilon\eta_j\leq p.
\]
Its mass-gap condition holds because $M_\theta(u_j)\to M_\theta(p)>0$. Put $\beta=1/N$ and $a=(\beta-\varepsilon)/(1-\varepsilon)\in(0,1)$. Combining this inequality with $w\leq p$ gives
\[
 (1-\beta)u_j+\beta\psi_j\leq p,\qquad
 \psi_j=\frac{(1-a)\varepsilon\eta_j+a w}{\beta}.
\]
These coefficients sum to one and the coefficient of $w$ is positive, so $\psi_j$ is strict. Fix such a $j$. \Cref{thm:compact-point-extension} gives a nonzero section
$t_m\in H^0(X,L^m\otimes\mathcal I(m\psi_j))$ for every sufficiently large $m$, by prescribing nonzero data at a finite point of $\psi_j$ once its curvature exceeds the extension threshold.

Write $k=Nm-r$, $0\leq r<N$, so $m=\lceil k/N\rceil$. The analytic integrability-to-growth estimate \cref{lem:analytic-weighted-integrability-growth}, applied to $u_j$ with coefficient $Nm$ and gap $1/N$, shows that every local member $f$ of $\mathcal I(Nm u_j)$ satisfies
\[
 |f|^2\leq C_f\mathrm e^{m(N-1)u_j}
\]
for all sufficiently large $m$. Therefore multiplication by $t_m$ is injective from
\[
 H^0(X,T\otimes L^{-r}\otimes L^{m(N-1)}
                   \otimes\mathcal I(Nm u_j))
\]
into the space with bundle $T\otimes L^k$ and weight
$m((N-1)u_j+\psi_j)$.
The product is integrable because the displayed bound leaves precisely the integrable factor $|t_m|^2\mathrm e^{-m\psi_j}$ in local frames. The convex inequality makes this ideal a subsheaf of $\mathcal I(Nm p)=\mathcal I(Nm u)$. Since $u\leq0$ and $Nm\geq k$, that ideal is contained in $\mathcal I(ku)$. The threshold depends on the fixed $N,j$; we take $k\to\infty$ first.

For $L^{N-1}$ with analytic weight $Nu_j$, the curvature is
\[
 (N-1)\theta+N\ddc u_j=N\theta_{u_j}-\theta
       \geq(N\delta-1)\omega>0.
\]
The preceding proposition with each fixed twist $T\otimes L^{-r}$ therefore gives
\[
 \liminf_k\frac{n!}{k^n}h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       \geq\int_X\left\langle(\theta_{u_j}-N^{-1}\theta)^n\right\rangle
       \geq M_\theta(u_j)-C/N.
\]
Here $C$ is independent of $j,N$: expand the difference in powers of $N^{-1}$, writing $\theta=(\omega+\theta)-\omega$. The semipositive smooth factors and \cref{thm:compact-nonpluripolar-finiteness} bound all coefficients uniformly in $j$. Bounded-product multilinearity gives this polynomial on the common finite locus, and extension by zero gives it globally.

Let $j\to\infty$ for each fixed $N$, then $N\to\infty$. The lower bound tends to $M_\theta(p)=\operatorname{vol}_{\mathcal I}(\theta_u)$, proving \eqref{eq:strict-psh-section-asymptotic}. Analytic goodness supplies the final mass equality.
\end{proof}
\index{sections!strict psh metric asymptotics}

\section{Nonstrict metrics and line-bundle volume}
\label{sec:nonstrict-section-volume}
Keep $\theta=c_1(L,h_0)$, with $L,T$ holomorphic line bundles and $h_0$ a smooth metric. The local Morse upper bound applies to the smooth residual metric on an analytic resolution. A curvature allowance makes the residual semipositive, and the multiplier-model limit controls the allowance as it vanishes. This gives an upper bound even when the original volume is zero. In a big class a strict minorant supplies the reverse inequality.

\begin{proposition}\label{prop:section-multiplier-volume-upper-bound}
Let $L,T$ be holomorphic line bundles, let $\theta=c_1(L,h_0)$, and let $a\geq0$. For $u\in\PSH(X,\theta+a\omega)$,
\begin{equation}\label{eq:section-multiplier-volume-upper-bound}
 \limsup_{k\to\infty}\frac{n!}{k^n}
 h^0(X,T\otimes L^k\otimes\mathcal I(ku))
 \leq\operatorname{vol}_{\mathcal I}(\theta+a\omega+\ddc u).
\end{equation}
For $a=0$ and zero multiplier volume, the section limit exists and equals zero.
\end{proposition}
\begin{proof}
Put $\rho=\theta+a\omega$. Choose a decreasing gentle quasi-equisingular approximation $u_j$ with specified type and $\rho+\ddc u_j\geq-\varepsilon_j\omega/2$, where $\varepsilon_j\downarrow0$. Subtract a common constant so that $u_j,u\leq0$. Put $p_j=P_{\rho+\varepsilon_j\omega}[u_j]_{\mathcal I}$. These normalized analytic model envelopes decrease: both their backgrounds and their ideal competitors decrease. They dominate $u$, so their limit $p$ is a noncollapsed $\rho$-psh potential.

For each $\lambda>0$, global strong openness gives $\lambda'>\lambda$ with $\mathcal I(\lambda p)=\mathcal I(\lambda'p)$. Compactness and finite generators give one margin on all of $X$, as in the regularization-limit proof. For large $j$, quasi-equisingularity and $p\leq p_j$ give
\[
 \mathcal I(\lambda p)=\mathcal I(\lambda'p)
   \subseteq\mathcal I(\lambda'p_j)
   =\mathcal I(\lambda'u_j)\subseteq\mathcal I(\lambda u).
\]
The order $p\geq u$ gives the reverse ideal order, while $p$ is a normalized competitor. Hence $p=P_\rho[u]_{\mathcal I}$. Analytic envelope equality and \cref{prop:perturbed-model-limit-mass} yield
\[
 M_{\rho+\varepsilon_j\omega}(u_j)
   =M_{\rho+\varepsilon_j\omega}(p_j)
   \longrightarrow M_\rho(p)
   =\operatorname{vol}_{\mathcal I}(\rho+\ddc u).
\]

Fix $j$ and resolve its specified gentle ideal type. Upstairs write
$\pi^*(\theta+\ddc u_j)=[D_j]+R_j$, with $D_j$ rational normal crossing and $R_j$ smooth. Smoothness extends the inequality
\[
 R_j+(a+\varepsilon_j)\pi^*\omega\geq0
\]
from the divisor complement to all of $Y$. Choose $q$ clearing $D_j$. Birational multiplier transformation and $k=qm+s$, $0\leq s<q$, identify the section space with that of the $m$th power of
$B=(\pi^*L)^q\otimes\mathcal O_Y(-qD_j)$ and a fixed twist for each residue. Its class is $q\{R_j\}$; the global $\ddc$ lemma supplies a smooth metric with curvature $qR_j$. The local Morse upper bound then gives
\[
 \limsup_k\frac{n!}{k^n}
 h^0(X,T\otimes L^k\otimes\mathcal I(ku_j))
 \leq\int_{\{R_j>0\}}R_j^n
 \leq\int_Y(R_j+(a+\varepsilon_j)\pi^*\omega)^n.
\]
The second inequality follows pointwise from determinant monotonicity on positive Hermitian matrices; its right-hand integrand is nonnegative everywhere. Divisorial removal and modification invariance identify this last integral with $M_{\rho+\varepsilon_j\omega}(u_j)$. Since $u\leq u_j$, its ideals contain those of $u$. Take $j\to\infty$ to prove \eqref{eq:section-multiplier-volume-upper-bound}. Nonnegative dimensions give the final zero case.
\end{proof}

\begin{theorem}\label{thm:big-class-singular-section-asymptotic}
Suppose $c_1(L)$ is big and $u\in\PSH(X,\theta)$. For every fixed holomorphic line bundle $T$,
\[
 \lim_{k\to\infty}\frac{n!}{k^n}
 h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       =\operatorname{vol}_{\mathcal I}(\theta_u).
\]
\end{theorem}
\begin{proof}
The upper bound and zero case follow from the preceding proposition. If the volume is positive, put $p=P_\theta[u]_{\mathcal I}$. It has that positive mass and the same ideals as $u$. Choose a strict minorant $w\leq p$ by \cref{lem:positive-mass-strict-minorant}. For $0<t\leq1$, the potential $p_t=(1-t)p+t w$ is strict and lies below $p$. The strict section formula gives
\[
 \liminf_k\frac{n!}{k^n}h^0(X,T\otimes L^k\otimes\mathcal I(ku))
    \geq\operatorname{vol}_{\mathcal I}(\theta+\ddc p_t).
\]
The convex convergence in \cref{cor:singularity-convergence-criteria} makes $p_t\to p$ in $d_{S,\theta}$ as $t\downarrow0$. \Cref{thm:mixed-multiplier-volume-distance-continuity} therefore sends this lower bound to $\operatorname{vol}_{\mathcal I}(\theta_u)$, proving equality.
\end{proof}

\begin{proposition}\label{prop:projective-positive-multiplier-volume-big}
Suppose $X$ admits a positive line bundle. If $u\in\PSH(X,\theta)$ for $\theta=c_1(L,h_0)$ and $\operatorname{vol}_{\mathcal I}(\theta_u)>0$, then $c_1(L)$ is big. The same conclusion holds if $M_\theta(u)>0$.
\end{proposition}
\begin{proof}
Choose positive very ample $A$ and a fixed smooth $D\in|A|$. Let $\kappa=c_1(A,h_A)>0$ and write $v=\operatorname{vol}_{\mathcal I}(\theta_u)>0$. For every integer $q\geq1$, the metric with weight $qu$ on $L^q\otimes A$ has strict curvature $q\theta_u+\kappa$. Thus
\[
 \lim_{\ell\to\infty}\frac{n!}{(q\ell)^n}
 h^0(X,L^{q\ell}\otimes A^\ell\otimes\mathcal I(q\ell u))
   =\operatorname{vol}_{\mathcal I}(\theta_u+q^{-1}\kappa)\geq v.
\]
Apply \cref{lem:fixed-divisor-section-loss} with twist $A^{-1}$ and exponent $\ell+1$. Since the weighted space is contained in the unweighted one, it gives, for $k=q\ell$,
\[
 h^0(X,L^k\otimes A^{-1})
  \geq h^0(X,L^k\otimes A^\ell\otimes\mathcal I(ku))
       -C(\ell+1)(k+\ell+2)^{n-1}.
\]
After multiplying by $n!/k^n$, the error has upper limit at most $C'/q$, with $C'$ independent of $q$. The divisor and metrics stayed fixed. Choose $q$ with $C'/q<v/2$. For large $\ell$ there is a nonzero section $s$ of $L^k\otimes A^{-1}$. Its zero divisor $E$ gives the psh metric on $L$ with weight $k^{-1}\log|s|^2_{h_0^k\otimes h_A^{-1}}$. Poincar\'e--Lelong computes its curvature as
\[
 k^{-1}(\kappa+[E])\geq k^{-1}\kappa.
\]
It is a K\"ahler current, proving bigness. Finally \cref{prop:mixed-multiplier-volume-calculus} bounds multiplier volume below by the original mass, proving the last assertion.
\end{proof}

\begin{corollary}\label{cor:projective-psh-metric-section-asymptotic}
If $X$ admits a positive line bundle, every Hermitian pseudo-effective $(L,h_0\mathrm e^{-u})$ and every fixed holomorphic line bundle $T$ satisfy
\[
 \lim_{k\to\infty}\frac{n!}{k^n}
 h^0(X,T\otimes L^k\otimes\mathcal I(ku))
       =\operatorname{vol}_{\mathcal I}(\theta_u).
\]
\end{corollary}
\begin{proof}
At zero volume use the upper-bound proposition. At positive volume the preceding proposition establishes the big-class hypothesis of \cref{thm:big-class-singular-section-asymptotic}.
\end{proof}

\begin{lemma}\label{lem:minimal-boundary-volume-attainment}
For every pseudo-effective $\alpha=\{\theta\}$,
\[
 \operatorname{vol}(\alpha)=M_\theta(V_\theta)
   =\operatorname{vol}_{\mathcal I}(\theta+\ddc V_\theta).
\]
\end{lemma}
\begin{proof}
The normalized minimal potentials $V_{\theta+\varepsilon\omega}$ decrease and dominate $V_\theta$. Their limit is $\theta$-psh, by distributional passage and the canonical representative, and remains normalized by compactness. It is therefore at most $V_\theta$, proving equality. Each minimal potential is model: it is the largest normalized potential in its admitting background. Apply \cref{prop:perturbed-model-limit-mass}. This identifies the boundary definition of class volume with $M_\theta(V_\theta)$ and agrees with the original big-class definition. Also $P_\theta[V_\theta]_{\mathcal I}=V_\theta$, since $V_\theta$ is a competitor and dominates every normalized competitor. The pure multiplier-volume identity gives the last equality.
\end{proof}

\begin{corollary}\label{cor:big-or-zero-line-bundle-volume}
Let $L$ be pseudo-effective and put $\alpha=c_1(L)$. If $\alpha$ is big or $\operatorname{vol}(\alpha)=0$, then every fixed holomorphic line bundle $T$ satisfies
\[
 \lim_{k\to\infty}\frac{n!}{k^n}h^0(X,L^k\otimes T)
       =\operatorname{vol}(\alpha).
\]
For every pseudo-effective $L$ and every $k\geq1$,
\[
 H^0(X,L^k)=H^0(X,L^k\otimes\mathcal I(kV_\theta)).
\]
\end{corollary}
\begin{proof}
For a nonzero $s\in H^0(X,L^k)$, the section weight $k^{-1}\log|s|_{h_0^k}^2$ is $\theta$-psh by Poincar\'e--Lelong. Normalize its supremum to zero; minimality bounds it by $V_\theta$. Hence $|s|^2\mathrm e^{-kV_\theta}$ is bounded almost everywhere, proving the section-space identity, including the zero section.

For the upper bound with $T$, fix $\varepsilon>0$ and a smooth metric $h_T$ on $T$. For sufficiently large $k$, $\varepsilon\omega-k^{-1}c_1(T,h_T)$ is semipositive. The same normalized section-weight argument in background $\theta+\varepsilon\omega$ makes every section of $L^k\otimes T$ belong to $\mathcal I(kV_{\theta+\varepsilon\omega})$. Apply \cref{prop:section-multiplier-volume-upper-bound} with allowance $a=\varepsilon$. The preceding lemma gives
\[
 \limsup_k\frac{n!}{k^n}h^0(X,L^k\otimes T)
   \leq\operatorname{vol}(\alpha+\varepsilon\{\omega\}).
\]
Let $\varepsilon\downarrow0$ to obtain the upper bound. It proves the zero-volume case. In a big class, \cref{thm:big-class-singular-section-asymptotic} applied to $V_\theta$ gives the lower bound using the weighted subspace; its multiplier volume is $\operatorname{vol}(\alpha)$ by the lemma.
\end{proof}
