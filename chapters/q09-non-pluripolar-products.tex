\chapter{Non-pluripolar products}
\label{chap:q09}

Let $X$ be a complex manifold of dimension $n\geq1$. The bounded Bedford--Taylor product can concentrate near the poles of an unbounded potential. Truncating the potentials and retaining only the common finite levels removes that concentration. The resulting measures increase; a local mass bound makes their limit a current. Closedness requires a separate estimate for the derivatives of the level cutoffs. On a compact K\"ahler manifold, bounded global potentials supply the mass bound through Stokes' formula. Sources and precise comparisons are in \cref{sec:history-nonpluripolar}.

\section{Truncation and plurifine locality}
\label{sec:nonpluripolar-construction}
On a coordinate domain $U\subset X$, let $u_1,\ldots,u_p$ be plurisubharmonic, $1\leq p\leq n$. Put
\[
 \begin{gathered}
 u_{j,k}=\max\{u_j,-k\},\qquad
 O_k=\bigcap_{j=1}^p\{u_j>-k\},\\
 S_k=\mathbf1_{O_k}\ddc u_{1,k}\wedge\cdots\wedge\ddc u_{p,k}.
 \end{gathered}
\]
The bounded products are those of \cref{def:bedford-taylor-product}, and the indicators restrict their coefficient measures. The sets $O_k$ are Borel and plurifine open. They need not be open in the ordinary topology.

\Needspace{14\baselineskip}
\begin{lemma}\label{lem:nonpluripolar-truncation-limit}
The positive measure-coefficient currents $S_k$ increase. Their limit is locally finite precisely when
\begin{equation}\label{eq:nonpluripolar-local-mass-condition}
 \sup_k\int_K S_k\wedge\beta^{n-p}<\infty
 \quad\text{for every }K\Subset U,
\end{equation}
where $\beta$ is the Euclidean K\"ahler form. In that case $S$ annihilates pluripolar sets and $S_k\to S$ in coefficient variation on compact sets. For Borel $E_k\subset O_k$ with $\mathbf1_{E_k}\to1$ almost everywhere for the trace of $S$, one has $\mathbf1_{E_k}S_k\to S$ against bounded Borel tests with compact support.
\end{lemma}
\begin{proof}
For $l\geq k$, all corresponding bounded potentials agree on $O_k$. \Cref{thm:bt-plurifine-locality} gives $\mathbf1_{O_k}S_l=S_k$. On the complement of $O_k$, $S_k$ is zero and $S_l$ is positive. Thus $S_l-S_k\geq0$.

Their trace measures increase, and \eqref{eq:nonpluripolar-local-mass-condition} gives a locally finite measure limit. The positive coefficient bounds of \cref{thm:positive-current-order-zero} bound the variation of each coefficient of $S_l-S_k$ by a fixed multiple of its trace. Hence every coefficient is Cauchy in variation on compact sets and defines the positive limit $S$. Conversely a locally finite positive limit bounds the traces of all $S_k$. Every bounded product here annihilates pluripolar sets by \cref{prop:bt-poles-zero-capacity,prop:bt-capacity-basic}, so its increasing limit does too. Its restriction to $O_k$ is exactly $S_k$; it is zero on $U\setminus\bigcup_kO_k$, a pluripolar set. Finally $\mathbf1_{E_k}S_k=\mathbf1_{E_k}S$. Dominated convergence against the locally finite coefficient variations proves the last assertion. The trace determines their null sets. In particular an increasing exhaustion whose complement is pluripolar satisfies the indicator condition.
\end{proof}

\begin{definition}\label{def:nonpluripolar-product}
When \eqref{eq:nonpluripolar-local-mass-condition} holds, the limit is the \emph{non-pluripolar product}
\[
 \langle\ddc u_1\wedge\cdots\wedge\ddc u_p\rangle=S.
\]
The angle brackets distinguish this construction from a product that retains polar mass. We omit them for bounded potentials, for which it is the Bedford--Taylor product. The empty product is $1$.
\end{definition}
\index{non-pluripolar product!truncation construction}

\begin{proposition}\label{prop:nonpluripolar-strong-positivity}
Every locally finite non-pluripolar product is strongly positive: it is nonnegative on compactly supported smooth weakly positive test forms. This is the current version of the dual cones in \cref{prop:positivity-cones}.
\end{proposition}
\begin{proof}
Products of smooth decreasing regularizations of the bounded potentials are exterior products of semipositive $(1,1)$-forms, hence strongly positive. Decreasing Bedford--Taylor continuity in \cref{thm:bt-decreasing-continuity} preserves their nonnegative pairing with each weakly positive test form. This property is also preserved by restriction of coefficient measures. Indeed, relative to the trace, test the coefficient density against a countable dense family of constant weakly positive forms and nonnegative scalar tests. Outside one null set every such pairing is nonnegative; density and finite-dimensional cone duality give the strong cone condition pointwise. Restriction and an increasing locally finite limit preserve it. Apply this to the truncated products defining the non-pluripolar product.
\end{proof}
\index{positive current!strong positivity}

\begin{proposition}\label{prop:nonpluripolar-local-calculus}
Local finiteness and the product depend only on the currents $\ddc u_j$. The product is symmetric, local in the plurifine topology and multilinear in its positive current factors. For addition in one factor, the product of the sum is locally finite if and only if both summand products are locally finite. These assertions allow the local products of arbitrary closed positive $(1,1)$-currents to glue whenever their mass condition holds.
\end{proposition}
\begin{proof}
Symmetry holds at every bounded truncation. For locality, two tuples agreeing on a plurifine open $O$ have equal bounded products on
$O\cap\bigcap_j\{u_j>-k\}\cap\bigcap_j\{v_j>-k\}$.
These sets exhaust $O$ except for their pluripolar pole sets. Both limits give those sets zero mass, so the restrictions agree. The quasi-Lindel\"of argument in \cref{thm:bt-plurifine-locality} makes these restrictions measurable for the completed coefficient measures.

If $\ddc u_j=\ddc v_j$, their difference $w_j$ is smooth pluriharmonic: the trace equation and Weyl's lemma first make the distribution smooth, and its Hessian is zero. On a neighborhood $V$ of a compact test set choose $C$ with $|w_j|\leq C$. On $O_k(v)\cap V$, the bounded psh functions $\max\{v_j,-k\}$ and $\max\{u_j,-k-C\}+w_j$ coincide. Plurifine locality therefore identifies the restricted product with the corresponding restriction of the $u$-product at level $k+C$. Its trace is bounded by the mass of that product on a slightly larger compact neighborhood. Thus finiteness for $u$ implies finiteness for $v$; the reverse implication uses $-w_j$. Exhausting their common finite locus proves equality. Constants and real truncation levels cause no change, since a real level is bounded between adjacent integer levels.

\Needspace{6\baselineskip}
For additivity compare the positive bounded products on
\[
 E_k=\{u_1>-k\}\cap\{v_1>-k\}
                    \cap\bigcap_{j=2}^p\{u_j>-k\}.
\]
On this set $u_{1,k}+v_{1,k}=u_1+v_1$, whose truncation at $-2k$ has the same Hessian. Bounded multilinearity gives the sum of the two summand products there. For the other factors use level $2k$ on the sum side; they agree with their level-$k$ truncations on $E_k$. The sets $E_k$ increase and miss only the union of the pole sets. By \cref{prop:finite-locus-truncation}, each tuple has an increasing trace-measure limit even before local finiteness is known; this extended measure need not define a current. It annihilates every pluripolar set. Restricting each extended trace to $E_k$ gives its corresponding bounded-level trace, so the bounded identity just proved holds there. Monotone convergence as $k\to\infty$ gives equality of the extended traces on every Borel set. Positivity now shows that the uniform local mass condition for the sum is equivalent to the two separate conditions. Passing to the limits then gives equality of currents. Positive scaling follows by rescaling the levels, proving multilinearity. Ordinary open locality and potential independence give the gluing assertion. Local positive-current potentials exist by \cref{thm:local-positive-current-potential}.
\end{proof}

\begin{example}\label{ex:nonpluripolar-infinite-local-mass}
On a small bidisc write $L=-\log|y|$ and
\[
 u(x,y)=(|x|^2-1)\sqrt L\quad(y\ne0),\qquad u(x,0)=-\infty.
\]
Take $|x|^2<1/2$ and $|y|<\mathrm e^{-1}$. Off $y=0$ its complex Hessian has
\[
 u_{x\bar x}=\sqrt L,\qquad
 u_{y\bar y}=\frac{1-|x|^2}{16|y|^2L^{3/2}},\qquad
 u_{x\bar y}=-\frac{\bar x}{4\bar y\sqrt L},\qquad
 \det(u_{j\bar k})=\frac{1-2|x|^2}{16|y|^2L}.
\]
It is positive definite there. Bounded-above analytic removal in \cref{thm:psh-bounded-zero-extension} gives exactly the displayed value on the axis, so $u$ is psh. The smooth measure $(\ddc u)^2$ has a positive fixed multiple of this determinant as density. Its integral near the axis diverges, since normal radial integration contains $\int_0^r\mathrm dt/(t(-\log t))$. Truncation agrees with that smooth product off the axis, which the finite levels exhaust. Thus the local mass condition fails. Local non-pluripolar products of arbitrary potentials need not be finite.
\end{example}

\section{Closedness and cutoff gradients}
\label{sec:nonpluripolar-closedness}
The level indicators are not differentiable. We replace them by bounded psh cutoffs whose gradient energies are controlled by the restricted masses. Throughout this section $\mathrm d^{\mathrm c}=(\partial-\bar\partial)/(4\pi\mathrm i)$, as in \cref{sec:bt-monotone}; hence $\mathrm d\mathrm d^{\mathrm c}=\ddc$.

\Needspace{23\baselineskip}
\begin{lemma}\label{lem:bounded-psh-current-chain-rule}
Let $\Theta$ be a closed positive current of bidimension $(1,1)$ on a coordinate domain, and let $v$ be locally bounded psh. The current $\mathrm d(v\Theta)$ has measure coefficients, denoted $\mathrm dv\wedge\Theta$. There is a positive gradient measure
\begin{equation}\label{eq:bounded-current-gradient-measure}
 E_\Theta(v)=\mathrm dv\wedge\mathrm d^{\mathrm c}v\wedge\Theta
       =\tfrac12\ddc(v^2\Theta)-v\,\ddc v\wedge\Theta.
\end{equation}
Smooth decreasing approximations have gradients converging strongly in the coefficient-matrix energy seminorm on compact subsets. For every smooth real $f$, the bounded products satisfy
\begin{equation}\label{eq:bounded-current-chain-rule}
 \mathrm d(f(v)\Theta)=f'(v)\,\mathrm dv\wedge\Theta.
\end{equation}
For a compactly supported smooth real one-form $\eta$, a cutoff $\tau\geq0$ equal to one near its support, and a bounded Borel function $a$, there is a constant $C_\eta$ such that
\begin{equation}\label{eq:current-gradient-cauchy-schwarz}
 \left|\int a\eta\wedge\mathrm dv\wedge\Theta\right|^2
 \leq C_\eta\left(\int\tau E_\Theta(v)\right)
                 \left(\int_{\operatorname{supp}\eta}|a|^2\,
                                         \Theta\wedge\beta\right).
\end{equation}
\end{lemma}
\begin{proof}
All assertions are local near a compact test support. Add a constant to make $v$ and its smooth radial approximations $v_l\downarrow v$ nonnegative and uniformly bounded there. For a smooth real function $w$, the differential product rule gives
\[
 \mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta
       =\tfrac12\ddc(w^2\Theta)-w\ddc w\wedge\Theta.
\]
The measure on the left is positive. Applied to $w=v_l-v_m$ and tested with $\tau$, this identity gives energy tending to zero as $l,m\to\infty$. Indeed \cref{cor:bt-polynomial-weights} makes the squared-weight term tend to zero after integration by parts against $\ddc\tau$, and each of the four expanded Hessian terms tends to the same integral with weight $v$ and Hessian $\ddc v$. Their signs cancel. For two independent indices the convergence is uniform on tails: otherwise select violating pairs with both indices strictly increasing and apply that same decreasing-continuity theorem.

Here is the meaning of the energy seminorm, including when $\Theta$ is singular. Its positive coefficient matrix has bounded Radon--Nikodym entries relative to $\mu=\Theta\wedge\beta$, by \cref{thm:positive-current-order-zero}. Write that measurable Hermitian matrix as $H$. For this bidimension, its quadratic form is the density of $\mathrm i\xi\wedge\bar\xi\wedge\Theta$ for a constant $(1,0)$-form $\xi$. Positivity for a countable dense family of $\xi$ makes $H$ positive semidefinite almost everywhere. In coordinates the gradient energy is the squared $L^2(\mu)$ norm of $H^{1/2}\partial w$, with the fixed normalization absorbed into $H$. The preceding estimate makes these vectors for $v_l$ Cauchy on compact subsets. Their strong limit defines the gradient modulo the zero seminorm. Multiplication by a bounded measurable scalar respects that quotient. Pointwise matrix Cauchy--Schwarz bounds the derivative paired with $a\eta$ by the two norms in \eqref{eq:current-gradient-cauchy-schwarz}. It also bounds every derivative coefficient in variation. Distributional convergence of $v_l\Theta$ to $v\Theta$, by dominated convergence, identifies that measure derivative with $\mathrm d(v\Theta)$.

The squared norms converge in $L^1(\mu)$, so their limit is a positive measure. Polynomial weighted continuity identifies its distribution with the right side of \eqref{eq:bounded-current-gradient-measure}. This also proves independence of the smooth approximations. Finally $f(v_l)\Theta\to f(v)\Theta$ by dominated convergence. The products $f'(v_l)$ times the gradients converge strongly: subtract the limiting gradient first, then use dominated convergence on its squared energy with $f'(v_l)\to f'(v)$. Passing through the smooth chain rule proves \eqref{eq:bounded-current-chain-rule}. A constant shift only translates the argument of $f$ and does not change the gradients, completing the proof.
\end{proof}

\begin{theorem}\label{thm:nonpluripolar-product-closed}
Every locally finite non-pluripolar product is closed.
\end{theorem}
\begin{proof}
Closedness is local. Work on a smaller coordinate neighborhood and subtract constants so that all local potentials $u_j\leq0$. If $p=n$, closedness holds by degree. Otherwise wedge the truncated products with a constant strongly positive $(n-p-1,n-p-1)$-form $\gamma$ and write
\[
 \Theta_k=\gamma\wedge\bigwedge_j\ddc u_{j,k},\qquad
 S=\gamma\wedge\langle\bigwedge_j\ddc u_j\rangle.
\]
These $\Theta_k$ are closed positive currents of bidimension $(1,1)$, and $\mathbf1_{O_k}\Theta_k\uparrow S$ with uniformly bounded restricted mass on compact sets.

Set $u=\sum_j u_j$. Choose a smooth convex increasing $\chi$ with $\chi=0$ on $(-\infty,1/2]$ and $\chi(1)=1$, and put $w_k=\chi(\mathrm e^{u/k})$. It is bounded psh, $0\leq w_k\leq1$, increases to one at every finite point of $u$, and is zero outside $O_k$: there $u\leq-k$ and $\mathrm e^{-1}<1/2$. At the poles define its value to be zero. Composition followed by the canonical representative gives precisely these pointwise values. Choose smooth $0\leq\zeta\leq1$, zero on $(-\infty,1/2]$ and equal to one near $1$. Both $\zeta(w_k)$ and $\zeta'(w_k)$ vanish outside $O_k$. Dominated convergence against $S$ shows
\[
 \zeta(w_k)\Theta_k\longrightarrow S,
 \qquad \zeta'(w_k)\longrightarrow0
                         \quad S\text{-almost everywhere}.
\]
For the first assertion use $\mathbf1_{O_k}\Theta_k=S$ on $O_k$ and that $S$ gives the poles zero mass.

For a compactly supported one-form $\eta$ and a cutoff $\tau\geq0$ equal to one near its support, the gradient identity and positivity give
\[
 \int\tau E_{\Theta_k}(w_k)
 \leq\tfrac12\int w_k^2\ddc\tau\wedge\Theta_k
 \leq C\int_{\operatorname{supp}\tau}
                           \mathbf1_{O_k}\Theta_k\wedge\beta\leq C'.
\]
The first inequality drops $w_k\ddc w_k\wedge\Theta_k\geq0$. The next uses $w_k=0$ off $O_k$, its bound by one, and the coefficient bound for the fixed smooth $\ddc\tau$. The mass hypothesis makes $C'$ independent of $k$. Apply \eqref{eq:current-gradient-cauchy-schwarz} with $a=\zeta'(w_k)$. Its second factor tends to zero, since
\[
 |\zeta'(w_k)|^2\Theta_k\wedge\beta
       =|\zeta'(w_k)|^2\mathbf1_{O_k}\Theta_k\wedge\beta
       \leq |\zeta'(w_k)|^2 S\wedge\beta,
\]
and the latter tends to zero by dominated convergence. The chain rule now gives $\mathrm d(\zeta(w_k)\Theta_k)\to0$, hence $\mathrm dS=0$. Constant strongly positive $\gamma$ span the real forms of this bidegree; wedging them with coordinate one-forms spans the test forms for the two types of $\mathrm d\langle\bigwedge_j\ddc u_j\rangle$. Thus the original product is closed.
\end{proof}
\index{non-pluripolar product!closedness}

\section{Bounded factors and the first-degree product}
\label{sec:nonpluripolar-bounded-factors}
A bounded factor can be inserted by the recursive Bedford--Taylor construction. To identify that construction with truncation, its resulting current must still avoid the other factors' poles. A relative weighted estimate provides this control.

\begin{lemma}\label{lem:relative-weighted-cln}
Let $T$ be closed positive of bidimension $(r,r)$, $r\geq1$, on a coordinate domain $U$. For $K\Subset L^\circ\subset L\Subset U$, a nonpositive psh function $V$ integrable on $L$ for the trace of $T$, and a locally bounded psh function $v$, one has
\[
 \int_K(-V)\,\ddc v\wedge T\wedge\beta^{r-1}
       \leq C_{K,L}\|v\|_{L^\infty(L)}
                             \int_L(-V)T\wedge\beta^r.
\]
The constant depends on the neighborhoods and dimension, and is independent of $T,V,v$.
\end{lemma}
\begin{proof}
First suppose $r\geq2$ and $V$ smooth. Cover $K$ by finitely many smaller balls inside balls whose closures lie in $L^\circ$. Normalize a nonzero bounded norm of $v$ and subtract a constant so that $-2\leq v\leq-1$ on each larger ball. If its norm is zero, its local Hessian is zero. On a larger ball $B=B(a,R)$ put $\psi=|z-a|^2-R^2$, $b=\ddc\psi$, and replace $v$ by $\widetilde v=\max\{v,A\psi\}$. Choose $A$ from the fixed radii so that it equals $v$ near the smaller ball and equals $A\psi$ on a fixed boundary collar. Its bounds depend only on these radii. Set $S=\ddc\widetilde v\wedge T$, and choose a smooth nonnegative cutoff $\rho$ supported in $B$ and equal to $-\psi$ on a neighborhood of the inner ball $B_\delta=\{\psi<-\delta\}$ containing all nonsmooth changes. On its complement $S=Ab\wedge T$. Put $Q=\ddc V\wedge T$. Consequently
\[
 \begin{aligned}
 \int_{B_\delta}(-V)S\wedge b^{r-1}
 &=\int_B VS\wedge b^{r-2}\wedge\ddc\rho
       -A\int_{B\setminus B_\delta}
                          VT\wedge b^{r-1}\wedge\ddc\rho\\
 &=\int_B\rho\,\ddc\widetilde v\wedge Q\wedge b^{r-2}
       -A\int_{B\setminus B_\delta}
                          VT\wedge b^{r-1}\wedge\ddc\rho.
 \end{aligned}
\]
The integration by parts has a smooth compactly supported test and the smooth multiplier $V$; symmetry with the smooth factor gives the second line. The positive closed $Q$ has bidimension $(r-1,r-1)$. The bounded CLN estimate of \cref{thm:cln-bounded} bounds the first term by its trace mass on an intermediate compact neighborhood. Testing $Q$ with a nonnegative cutoff equal to one on that neighborhood bounds this mass by $C\int_L(-V)T\wedge\beta^r$, since the derivative of that cutoff has fixed bounded coefficients. The collar term has the same bound. Finite covering and undoing the normalization prove the displayed estimate.

For nonsmooth $V\leq0$, use smooth radial approximations $V_l\downarrow V$ on slightly smaller neighborhoods. They remain nonpositive, and $0\leq-V_l\leq-V$. Their right-hand integrals are bounded by the specified finite integral against $T$. Monotone convergence against the fixed positive product $\ddc v\wedge T$ proves the assertion. For $r=1$, pull $T,V,v$ back to the product with a plane disc using \cref{prop:submersion-current-pullback}. The pulled-back current has bidimension $(2,2)$, so the established estimate applies. Integration in a fixed smaller disc cancels its positive area factor from the two trace integrals. A finite product-neighborhood cover supplies the same kind of constant and proves the remaining case.
\end{proof}

\index{Chern--Levine--Nirenberg inequality!relative weighted}

\begin{lemma}\label{lem:bounded-product-preserves-nonpolar-current}
If the trace of a closed positive current $T$ gives every pluripolar set zero mass, then the same holds for $\ddc v\wedge T$ for every locally bounded psh $v$ of admissible degree, and for iterated bounded-potential products with base $T$.
\end{lemma}
\begin{proof}
Work near a compact set on which a given pluripolar set is contained in $\{V=-\infty\}$ for a nonpositive psh $V$. Write $\mu=T\wedge\beta^r$ on a compact larger neighborhood $L$. The measure is finite, and $t=-V$ is finite $\mu$-almost everywhere. Choose $R_j\geq2^j$ increasing so fast that
\[
 \int_L\min\{t/R_j,1\}\,\mathrm d\mu\leq2^{-j}.
\]
Dominated convergence as $R_j\to\infty$ permits these choices. The function
$g(t)=\sum_{j\geq1}\min\{t/R_j,1\}$ is finite, continuous, increasing and concave on $[0,\infty)$, and tends to infinity with $t$. Uniform convergence on bounded intervals proves continuity and concavity. Thus $\widetilde V=-g(-V)$ is nonpositive psh by convex increasing composition, remains $-\infty$ on the given pole set, and satisfies $\int_L(-\widetilde V)\,\mathrm d\mu\leq1$. \Cref{lem:relative-weighted-cln} makes it integrable for the trace of $\ddc v\wedge T$ on the smaller compact set. That trace must give its poles zero mass. A countable local cover proves the assertion for every pluripolar set. Repeating the argument proves the iterated assertion.
\end{proof}

\Needspace{12\baselineskip}
\begin{proposition}\label{prop:nonpluripolar-bounded-factor-compatibility}
Suppose $u_1,\ldots,u_p$ are local psh functions, $1\leq p\leq n$, and $u_1,\ldots,u_q$ are locally bounded, $0\leq q\leq p$. If $S=\langle\ddc u_{q+1}\wedge\cdots\wedge\ddc u_p\rangle$ is locally finite, then the full non-pluripolar product is locally finite and
\[
 \langle\ddc u_1\wedge\cdots\wedge\ddc u_p\rangle
              =\ddc u_1\wedge\cdots\wedge\ddc u_q\wedge S,
\]
where the right side is the bounded recursive product with base $S$.
\end{proposition}
\begin{proof}
For $q=0$ the assertion is the definition. Otherwise it suffices to insert one bounded factor $v$ and then iterate. Let $S$ be the remaining finite product and $R=\ddc v\wedge S$. It is closed and positive, and has no polar mass by the preceding lemma. If no other factor remains, bounded Bedford--Taylor theory already gives the assertion. Otherwise normalize the other local potentials $u_j\leq0$, put
\[
 O_k=\bigcap_j\{u_j>-k\},\qquad
 \Theta_k=\bigwedge_j\ddc\max\{u_j,-k\},\qquad
 w_k=\chi\bigl(\mathrm e^{\sum_j u_j/k}\bigr),
\]
with the same smooth convex increasing $\chi$ as in the closedness proof. These $w_k$ are bounded psh, vanish off $O_k$, and their positive loci increase to the common finite locus. Take bounded smooth decreasing regularizations $v_l\downarrow v$ near the test support. Since $S=\Theta_k$ on $O_k$ and $\ddc v_l$ is smooth,
\[
 w_k\ddc v_l\wedge S=w_k\ddc v_l\wedge\Theta_k.
\]
Apply weighted decreasing continuity from \cref{thm:bt-decreasing-continuity} with the fixed base $S$ on the left and fixed base $\Theta_k$ on the right. It gives $w_kR=w_k\ddc v\wedge\Theta_k$. Dividing on $\{w_k>0\}$ identifies the coefficient measures there. For fixed $j$ and $k\geq j$, bounded plurifine locality gives
$\ddc v\wedge\Theta_k=\ddc v\wedge\Theta_j$ on $O_j$.
The sets $\{w_k>0\}$, $k\geq j$, exhaust $O_j$, so $R=\ddc v\wedge\Theta_j$ on that entire set. For large $j$, the bounded $v$ needs no truncation on the chosen compact neighborhood. Its full finite-level product is therefore $\mathbf1_{O_j}R$. These restrictions have uniformly bounded local mass and increase to $R$, because $R$ gives the remaining pole sets zero mass. This proves the one-factor assertion and its iteration.
\end{proof}
\index{non-pluripolar product!bounded-factor compatibility}

The first-degree identification will use planar measures on almost every fibre. We first establish the bounded fibre formula for the Borel tests used in truncation.

\Needspace{15\baselineskip}
\begin{lemma}\label{lem:bounded-product-fibre-fubini}
Let $Q\subset\mathbb C^r$ and $D\subset\mathbb C^m$ be domains, with projection $p_D:Q\times D\to D$. Let $u_1,\ldots,u_r$ be locally bounded psh on $Q\times D$, and put
\[
 M=\ddc u_1\wedge\cdots\wedge\ddc u_r,\qquad
 M_z=\bigwedge_{j=1}^r\ddc_x u_j(\,\cdot\,,z).
\]
For a smooth real $(m,m)$-form $\eta$ on $D$ and a bounded Borel function $h$ of compact support, the fibre integral is Borel measurable and
\begin{equation}\label{eq:bounded-product-fibre-fubini}
 \int_{Q\times D}h\,p_D^*\eta\wedge M
       =\int_D\left(\int_Q h(x,z)\,M_z(x)\right)\eta(z).
\end{equation}
For nonnegative $h$ and $\eta$ the same identity holds with possibly infinite integrals.
\end{lemma}
\begin{proof}
First take $\eta\geq0$ and $h$ continuous of compact support. Smooth decreasing regularizations of the $u_j$ exist on a product neighborhood of that support. In the smooth exterior product, wedging with $p_D^*\eta$ kills every term containing a base differential. Ordinary Fubini therefore gives \eqref{eq:bounded-product-fibre-fubini} for the regularizations. Decreasing Bedford--Taylor continuity gives weak convergence of the products on the total space and on every fibre. The common bounds on the potentials and \cref{thm:cln-bounded} give a uniform fibre mass bound on the compact projection of the support. Dominated convergence proves the continuous-test formula. A finite cover and a partition of unity handle supports requiring more than one such neighborhood.

To justify the Borel tests, fix $U\Subset Q$ and $V\Subset D$. For $a\in C_c(U)$, $z\mapsto\int a\,M_z$ is Borel: it is the pointwise limit of the corresponding continuous smooth-regularization integrals. Every relatively open subset $A\subset U$ admits continuous functions $0\leq a_l\leq1$ of compact support in $U$ increasing to $\mathbf1_A$; construct them from distance to the complement and a compact exhaustion. Monotone convergence proves measurability of $M_z(A)$, including $A=U$. All these measures are finite on $U$, uniformly for $z\in V$. The sets $A$ for which $M_z(A)$ is Borel as a function of $z$ are therefore closed under complements in $U$ and countable disjoint unions. The intersection-closed open sets generate the Borel sets, so the class argument proved in \cref{prop:tonelli-fubini} gives measurability for every Borel $A$.

Apply the same argument to sets $E\subset U\times V$: the class for which $z\mapsto M_z(E_z)$ is Borel contains the Borel rectangles and is closed under complements and countable disjoint unions. It contains all Borel sets. Increasing exhaustions of $Q$ and $D$ give the assertion on the whole product. Hence
\[
 \lambda(E)=\int_D M_z(E_z)\,\eta(z)
\]
is a Borel measure. Countable additivity follows from monotone convergence on each fibre and then on $D$. The uniform local fibre mass bound makes it locally finite, and \cref{prop:borel-regularity} makes it Radon. It agrees with $p_D^*\eta\wedge M$ on continuous compactly supported tests by the first paragraph. Radon uniqueness makes the measures equal. Simple approximation proves the formula for nonnegative Borel functions and then for bounded signed compactly supported functions. On a compact base neighborhood an arbitrary smooth real top form is a difference of two smooth nonnegative top forms, which proves the signed assertion. The cases $r=0$ or $m=0$ reduce to counting measure and the empty product.
\end{proof}
\index{fibre integration!Bedford--Taylor products}

\begin{proposition}\label{prop:nonpluripolar-degree-one-restriction}
For every local psh $u$, its first-degree non-pluripolar product is defined and
\begin{equation}\label{eq:nonpluripolar-degree-one-restriction}
 \langle\ddc u\rangle=\mathbf1_{\{u>-\infty\}}\ddc u,
\end{equation}
where the Hessian on the right is distributional. In particular the indicated restriction is closed.
\end{proposition}
\begin{proof}
On a smaller neighborhood normalize $u\leq0$. Its truncations $u_l=\max\{u,-l\}$ converge in $L^1$ to $u$, so $\ddc u_l\to\ddc u$ weakly, with locally bounded positive trace masses. For fixed $k$ and all $l\geq k$, plurifine locality gives
$\mathbf1_{\{u>-k\}}\ddc u_k\leq\ddc u_l$.
Passing to the weak limit makes each restricted current at most $\ddc u$. Their local masses are thus bounded, and their non-pluripolar limit $S$ exists with $0\leq S\leq\ddc u$. It has no polar mass; hence it is at most the right side of \eqref{eq:nonpluripolar-degree-one-restriction}.

First work in one complex dimension and write $\mu=\ddc u$, a positive measure. Choose a smooth nonnegative cutoff $\eta$ equal to one on a smaller open set $V$. With $\nu=\eta\mu$ and the fundamental kernel $E_2$, \Cref{thm:riesz-decomposition} gives
\[
 u=4\pi E_2*\nu+h\quad\text{on }V,
\]
where $h$ is harmonic. The factor $4\pi$ is fixed by $\ddc u=(\Delta u)/(4\pi)$ as a planar measure. Put $p=E_2*\nu$ and $\nu_f=\mathbf1_{\{p>-\infty\}}\nu$. Since the kernel is bounded above on compact sets of differences, removing positive mass can only increase its potential up to a bounded constant. Thus $E_2*\nu_f$ is finite $\nu_f$-almost everywhere. \Cref{thm:potential-continuity-principle} decomposes $\nu_f$ as a sum of compact restrictions with finite continuous potentials. A finite partial sum $\nu^{(N)}$ has a locally bounded continuous subharmonic potential $v_N=4\pi E_2*\nu^{(N)}$. On $V$, $\mu-\nu^{(N)}\geq0$, so $u-v_N$ is subharmonic: its Laplacian is positive, and its ball averages converge to its displayed pointwise values because $v_N$ is continuous. It is therefore the canonical representative. Multilinearity and the bounded-potential case give
\[
 S=\langle\ddc v_N\rangle+\langle\ddc(u-v_N)\rangle
                                  \geq\nu^{(N)}\quad\text{on }V.
\]
Increasing $N$ gives $S\geq\nu_f=\mathbf1_{\{u>-\infty\}}\mu$ on $V$. Combined with the first inequality, this proves the planar assertion.

For higher dimension work in a product polydisc and choose a coordinate direction $x\in\mathbb C$, with remaining coordinates $z$. Almost every fibre $u_z=u(\,\cdot\,,z)$ is not identically $-\infty$ and is locally integrable by Fubini. Its planar Hessian measure is $\mu_z=\ddc_xu_z$. The kernel measurability argument in \cref{lem:bounded-product-fibre-fubini} also applies to these measures: smooth decreasing regularizations converge weakly on every such fibre, giving Borel integrals on continuous tests and then all Borel tests by the same class argument. Their masses on a fixed smaller disc are bounded by $C\int_{Q'}(-u_z)\,\mathrm dA$, using a planar cutoff. This bound is integrable in $z$. Fubini applied to $u$ times the normal test Laplacian therefore identifies
\[
 \ddc u\wedge\mathrm dV_z=\mu_z\,\mathrm dV_z
\]
as Radon measures, including against bounded Borel tests; discarded base fibres form a null set and carry no mass in this identity. Here $\mathrm dV_z$ denotes the pulled-back Euclidean top form in the remaining coordinates.

Apply the bounded fibre formula to $u_k$ with a compactly supported nonnegative Borel test multiplied by $\mathbf1_{\{u>-k\}}$. Its restricted measures increase. Monotone convergence and the planar assertion identify their fibre limit with $\mathbf1_{\{u_z>-\infty\}}\mu_z$. Thus
$S\wedge\mathrm dV_z=\mathbf1_{\{u>-\infty\}}\ddc u\wedge\mathrm dV_z$.
Repeat in every coordinate direction. The sum is the trace measure, up to the fixed positive normalization. Since $\mathbf1_{\{u>-\infty\}}\ddc u-S$ is positive and has zero trace, its coefficient bounds force it to be zero. This proves \eqref{eq:nonpluripolar-degree-one-restriction}. Closedness follows from \cref{thm:nonpluripolar-product-closed}.
\end{proof}
\index{non-pluripolar product!first-degree polar restriction}

\section{Global finiteness and bounded cohomological mass}
\label{sec:nonpluripolar-global-mass}
Let $(X,\omega)$ now be connected compact K\"ahler. For a closed smooth real $(1,1)$-form $\theta$ and a locally bounded $\theta$-psh potential $\varphi$, the local bounded products of $\theta+\ddc\varphi$ glue by smooth-potential independence. Smooth closed forms can be wedged with these currents.

\begin{proposition}\label{prop:bounded-global-cohomological-mass}
Let $\varphi_j\in\PSH(X,\theta_j)$ be locally bounded, $1\leq j\leq p\leq n$, and let $\eta$ be a smooth closed $(n-p,n-p)$-form. Then
\[
 \int_X\bigwedge_{j=1}^p(\theta_j+\ddc\varphi_j)\wedge\eta
       =\int_X\theta_1\wedge\cdots\wedge\theta_p\wedge\eta.
\]
\end{proposition}
\begin{proof}
For any closed positive remaining bounded product $R$, local recursion gives
$(\theta_j+\ddc\varphi_j)\wedge R=\theta_j\wedge R+\ddc(\varphi_jR)$.
Both sides glue, since $\varphi_j$ is a global bounded function. The derivative term integrates to zero against any smooth closed complementary form: its distributional pairing is with that form's zero Hessian. Replace the first factor, then the second, and so on, absorbing the already replaced smooth closed forms into the test form. All remaining nonsmooth products are positive and closed. The last integral is the displayed smooth one.
\end{proof}

\begin{theorem}\label{thm:compact-nonpluripolar-finiteness}
Every tuple of closed positive $(1,1)$-currents $T_1,\ldots,T_p$, $1\leq p\leq n$, on a compact K\"ahler manifold has a finite non-pluripolar product. Write $T_j=\theta_j+\ddc\varphi_j$ and choose $C\geq0$ so that $\tau_j=\theta_j+C\omega$ are positive definite. Then
\begin{equation}\label{eq:shifted-cohomological-mass-bound}
 \int_X\langle T_1\wedge\cdots\wedge T_p\rangle\wedge\omega^{n-p}
       \leq\int_X\tau_1\wedge\cdots\wedge\tau_p\wedge\omega^{n-p}.
\end{equation}
If the chosen $\theta_j$ are already semipositive, the same bound holds with $C=0$.
\end{theorem}
\begin{proof}
The global potential decomposition is \cref{thm:global-ddc-lemma}. Since $\tau_j$ is positive and $\varphi_j$ is $\tau_j$-psh, the global bounded potentials $\max\{\varphi_j,-k\}$ give positive currents
\[
 R_{j,k}=\tau_j+\ddc\max\{\varphi_j,-k\}.
\]
Their full mixed trace mass is the right side of \eqref{eq:shifted-cohomological-mass-bound}, independent of $k$, by \cref{prop:bounded-global-cohomological-mass}.

To bound the original local truncations, choose on a relatively compact coordinate chart smooth potentials $h_j$ for $\theta_j$ and $h$ for $\omega$. There $u_j=\varphi_j+h_j$ is psh. Choose $B$ with $h_j\leq B$ on a neighborhood of the compact test set. On $O_l(u)$, $\varphi_j>-l-B$. For any integer $k\geq l+B$, plurifine locality identifies $R_{j,k}$ there with the Hessian of $\max\{u_j,-l\}+Ch$: both local bounded psh potentials equal $u_j+Ch$ on that set. Their mixed bounded products agree there. Expanding the latter positive factors shows
\[
 \mathbf1_{O_l(u)}\bigwedge_j\ddc\max\{u_j,-l\}
       \leq\mathbf1_{O_l(u)}\bigwedge_j R_{j,k}.
\]
The global mass bound therefore implies every local mass condition. The products glue and are closed by the preceding results. The same argument for $T_j+C\omega$ identifies their non-pluripolar product with the increasing common-global-level restrictions of $\bigwedge_jR_{j,k}$; these levels exhaust all finite points of the $\varphi_j$. It has mass at most the same smooth bound. Multilinearity and positivity give
$\langle\bigwedge_j T_j\rangle\leq\langle\bigwedge_j(T_j+C\omega)\rangle$,
proving \eqref{eq:shifted-cohomological-mass-bound}. For semipositive $\theta_j$, the constant potentials and their maxima are $\theta_j$-psh, so the argument applies directly with $C=0$.
\end{proof}
\index{non-pluripolar product!global finiteness}

\begin{example}\label{ex:nonpluripolar-divisor-zero-mass}
On $\mathbb P^1$ the current $[0]$ has non-pluripolar product $\langle[0]\rangle=0$. Near zero its potential is $\log|z|^2$. The Hessian of its truncation is supported on the boundary level, excluded by $\{\log|z|^2>-k\}$. Away from zero the current is zero. Its ordinary mass is one, while its non-pluripolar mass is zero. Thus discarding poles can change even a first-degree cohomology class. In dimension zero the empty product is the counting measure and every positive-degree product is zero.
\end{example}

\section{Fibre integration and modifications}
\label{sec:nonpluripolar-fibres}
The bounded fibre formula now passes through non-pluripolar truncation on compact manifolds. The resulting identity keeps the bounded difference weight needed for relative energy.

\Needspace{23\baselineskip}
\begin{corollary}\label{cor:compact-nonpluripolar-fibre-fubini}
Let $X,Z$ be connected compact K\"ahler manifolds of dimensions $r,m$, with projections $p_X,p_Z$. Let $\theta$ be a closed smooth real $(1,1)$-form on $X$, $\varphi\in\PSH(X,\theta)$, and let $\gamma\geq0$ be a closed smooth $(1,1)$-form on $Z$. Set
\[
 \Theta=p_X^*\theta+p_Z^*\gamma,\qquad
 \Phi=p_X^*\varphi,\qquad
 T=\Theta+\ddc W,\quad S=\Theta+\ddc\Phi,
\]
where $W\in\PSH(X\times Z,\Theta)$ and $\Phi-C\leq W\leq\Phi+C$ for a finite $C$. Every $W_z=W(\,\cdot\,,z)$ belongs to $\PSH(X,\theta)$. For $0\leq j\leq r$ and any smooth real $(m,m)$-form $\eta$ on $Z$,
\begin{equation}\label{eq:compact-nonpluripolar-fibre-fubini}
 \begin{aligned}
 &\int_{X\times Z}(W-\Phi)\,p_Z^*\eta\wedge
                              \langle T^j\wedge S^{r-j}\rangle\\
 &\qquad=\int_Z\left(\int_X(W_z-\varphi)
            \langle(\theta+\ddc W_z)^j\wedge
                         (\theta+\ddc\varphi)^{r-j}\rangle\right)\eta(z).
 \end{aligned}
\end{equation}
The same identity holds with both difference weights replaced by $1$. The difference weights may be assigned any bounded value at their common poles.
\end{corollary}
\Needspace{6\baselineskip}
\begin{proof}
In product charts choose smooth local potentials $a(x)$ for $\theta$ and $b(z)$ for $\gamma$. The restrictions of $W+a+b$ to fibres are psh, and $W_z$ is not identically $-\infty$ because $W_z\geq\varphi-C$ and $\varphi$ is finite somewhere. Thus the asserted fibre potentials are defined. All products are finite by \cref{thm:compact-nonpluripolar-finiteness}. That theorem also bounds their fibre masses uniformly in $z$: the shifted smooth background can be chosen from the fixed $\theta$, independently of $W_z$.

Put $w=W+a+b$, $v=\Phi+a+b$ and $O_k=\{w>-k\}\cap\{v>-k\}$. Apply \eqref{eq:bounded-product-fibre-fubini} to the bounded truncations of $w,v$, with multiplicities $j,r-j$, and the Borel weight $\rho G\mathbf1_{O_k}$, where $\rho\geq0$ is a smooth compactly supported chart cutoff and $G\geq0$ is bounded Borel. Their restricted mixed measures increase to the non-pluripolar products on the total space and on every fibre. When one multiplicity is zero, the additional finite-level condition still exhausts that product: its omitted pole set is pluripolar and has zero mass. Monotone convergence gives the same fibre formula with the non-pluripolar limits and weight $\rho G$. Take $G=1$ for the unweighted formula and $G=W-\Phi+C$ on their common finite locus for the weighted formula; assign $G=C$ at their poles. Subtract the constant-weight identity. Sum over a finite nonnegative partition of unity and use linearity for signed $\eta$. The uniform mass bound and $|W-\Phi|\leq C$ justify every finite signed integral. Polar sets have zero product mass, so the assigned values do not affect them.
\end{proof}
\index{fibre integration!non-pluripolar products}

\begin{proposition}\label{prop:nonpluripolar-modification-pushforward}
Let $\pi:\widetilde X\to X$ be a modification between compact K\"ahler manifolds and let $T_1,\ldots,T_p$ be closed positive $(1,1)$-currents on $X$, $1\leq p\leq\dim X$. Then
\[
 \pi_*\langle\pi^*T_1\wedge\cdots\wedge\pi^*T_p\rangle
                  =\langle T_1\wedge\cdots\wedge T_p\rangle.
\]
For closed positive $(1,1)$-currents $R_1,\ldots,R_p$ on $\widetilde X$ one also has
\[
 \pi_*\langle R_1\wedge\cdots\wedge R_p\rangle
                   =\langle\pi_*R_1\wedge\cdots\wedge\pi_*R_p\rangle.
\]
In top degree all corresponding masses agree.
\end{proposition}
\begin{proof}
Pullback is defined by pulling back local psh potentials; their pluriharmonic differences pull back as well. A potential cannot become identically $-\infty$ on a component, because a modification is a biholomorphism over a dense open set. Both products are finite. Choose a proper analytic set $A\subset X$ over whose complement $\pi$ is a biholomorphism, as in the definition of modification. Potential independence and coordinate invariance identify the products over $X\setminus A$. The downstairs product gives $A$ zero mass. The upstairs product gives the proper analytic set $\pi^{-1}(A)$ zero mass, so its pushforward also gives $A$ zero coefficient variation. Equality off $A$ therefore extends to all coefficient measures. Pushforward of a closed positive current is closed and positive, since smooth pullback commutes with differentiation and preserves strong positivity of test forms. The same argument on the biholomorphism locus therefore proves the second identity, with the $R_j$ in place of the pulled-back currents. Testing the constant function in top degree proves the mass assertions.
\end{proof}
\index{non-pluripolar product!modification pushforward}

\section{Capacity convergence and loss of mass}
\label{sec:nonpluripolar-capacity-limits}
The bounded operator is continuous in capacity. An unbounded sequence can lose mass into its poles, so its non-pluripolar products satisfy a lower inequality before any total-mass hypothesis is imposed.

\begin{definition}\label{def:extended-psh-capacity-convergence}
For $s,t\in[-\infty,\infty)$ put $d_*(s,t)=|\arctan s-\arctan t|$, with $\arctan(-\infty)=-\pi/2$. Local psh functions $u_k$ converge in \emph{extended capacity} to $u$ if the outer capacity of
$K\cap\{d_*(u_k,u)>\delta\}$ tends to zero for every compact $K$ and $\delta>0$. On a compact manifold this is required of their local psh potentials in a finite coordinate cover. Changing local potentials by the same smooth function does not affect it.
\end{definition}
\index{convergence!capacity for unbounded potentials}

\begin{lemma}\label{lem:extended-capacity-truncations}
For locally uniformly bounded-above psh functions, extended capacity convergence is equivalent to capacity convergence of $\max\{u_k,-C\}$ to $\max\{u,-C\}$ for every $C>0$. Decreasing psh sequences with psh limit converge in this topology. A sequence of global $\theta$-psh potentials converging in extended capacity has a uniform upper bound.
\end{lemma}
\begin{proof}
On $[-\infty,M]$, the function $\max\{s,-C\}$ is a Lipschitz function of $\arctan s$: it is constant below $\arctan(-C)$, and the derivative of the inverse tangent is bounded on the remaining compact interval. Conversely, $|\arctan s-\arctan\max\{s,-C\}|\leq\pi/2-\arctan C$, uniformly in $s$. These two bounds prove the equivalence. Adding a bounded smooth function $h$ is harmless, since the derivative ratio
$(1+s^2)/(1+(s+h)^2)$ is uniformly bounded when $h$ is bounded. The local capacity comparisons in \cref{prop:bt-capacity-basic} handle changes of charts.

For a decreasing sequence, every fixed truncation decreases to a bounded psh limit. Apply \cref{prop:psh-decreasing-capacity} and the preceding equivalence. For the global assertion, normalize $v_k=u_k-\sup_Xu_k$. \Cref{prop:normalized-global-psh-compactness} gives a common bound for $\int_X|v_k|\,\omega^n$. If a subsequence had $\sup_Xu_k\to+\infty$, Markov's inequality would make $u_k\to+\infty$ in volume measure. This contradicts capacity convergence to a potential finite almost everywhere, since local capacity dominates volume. Therefore the suprema are bounded. Apply the argument componentwise if $X$ is disconnected.
\end{proof}

\begin{theorem}\label{thm:nonpluripolar-capacity-lower-limit}
Let $T_{j,k}=\theta_j+\ddc u_{j,k}$ and $T_j=\theta_j+\ddc u_j$ be closed positive currents on compact K\"ahler $X$, $1\leq j\leq p\leq n$. Suppose $u_{j,k}\to u_j$ in extended capacity. Write $R_k=\langle T_{1,k}\wedge\cdots\wedge T_{p,k}\rangle$ and $R=\langle T_1\wedge\cdots\wedge T_p\rangle$. For every smooth strongly positive test form $\eta$ of complementary bidegree,
\[
 \liminf_k\int_X R_k\wedge\eta\geq\int_X R\wedge\eta.
\]
If also $\limsup_k\int_X R_k\wedge\omega^{n-p}\leq\int_X R\wedge\omega^{n-p}$, then $R_k\to R$ weakly. The same lower inequality is local whenever the local products are finite and the potentials have common local upper bounds.
\end{theorem}
\begin{proof}
Work on a relatively compact coordinate ball, add fixed smooth potentials of the $\theta_j$, and denote the resulting psh functions by $v_{j,k},v_j$. They have common upper bounds by the preceding lemma. For $C,\varepsilon>0$ put
\[
 v_{j,k}^C=\max\{v_{j,k},-C\},\qquad
 a_{j,k}=\frac{v_{j,k}^C+C}{v_{j,k}^C+C+\varepsilon},\qquad
 a_k=\prod_j a_{j,k}.
\]
The analogous limit weights $a_j,a$ are bounded and quasi-continuous. The bounded truncations and their weights converge in capacity. Since $a_k$ vanishes wherever any $v_{j,k}\leq-C$, plurifine locality gives
\[
 a_k R_k=a_k\bigwedge_j\ddc v_{j,k}^C.
\]
For a nonnegative compactly supported test, weighted capacity continuity in \cref{thm:bt-capacity-continuity} identifies the limit of this expression with $a\bigwedge_j\ddc v_j^C=aR$. It is bounded above by $R_k$ in the positive-current order. Let $\varepsilon\downarrow0$ and then $C\to\infty$. The weights increase to one outside the common pole set, which $R$ annihilates. Monotone convergence proves the local lower inequality. A finite nonnegative partition of unity proves the global one.

The shifted bound in \cref{thm:compact-nonpluripolar-finiteness} bounds all trace masses uniformly. Positive coefficient bounds and \cref{thm:local-measure-selection} give weak subsequential limits. Every such limit $R'$ is positive and satisfies $R'\geq R$ by the first assertion. The assumed trace inequality makes the positive difference $R'-R$ have zero total trace, so it vanishes. Every subsequence therefore has the same possible limit, proving weak convergence. The local proof needs no compactness assumption on the manifold.
\end{proof}
\index{non-pluripolar product!lower semicontinuity}

\Needspace{8\baselineskip}
\begin{example}\label{ex:nonpluripolar-mass-loss}
On $\mathbb P^1$ with $\int\omega_{\mathrm{FS}}=1$, put $u=\log|z|^2-\log(1+|z|^2)$ and $u_k=\max\{u,-k\}$. The weight extends smoothly at infinity, and Poincar\'e--Lelong gives $\omega_{\mathrm{FS}}+\ddc u=[0]$. Each $u_k$ is bounded and $\omega_{\mathrm{FS}}$-psh, so its measure has mass one by bounded Stokes. The sequence decreases to $u$ in capacity, and its measures converge weakly to $[0]$ by distributional convergence. But $\langle[0]\rangle=0$ by \cref{ex:nonpluripolar-divisor-zero-mass}. The weak limit retains the mass that the non-pluripolar operator discards at the pole. A mass hypothesis is therefore needed for continuity.
\end{example}

\section{Mass comparison and projective fibres}
\label{sec:nonpluripolar-mass-monotonicity}
A common pole set first permits mass comparison by compactly supported differences. A projective construction then replaces arbitrary singularities by potentials bounded outside such a set. Its fibre mass records the original mass with a weight concentrating at the endpoint.

\begin{lemma}\label{lem:small-locus-mixed-mass-comparison}
Suppose $T_j=\theta_j+\ddc\varphi_j$ and $T'_j=\theta_j+\ddc\psi_j$ are positive currents on compact K\"ahler $X$ and $\varphi_j\geq\psi_j-O(1)$. Assume that all these potentials are locally bounded outside a closed set $A=\{\tau=-\infty\}$, where $\tau$ is a global quasi-psh function locally bounded outside $A$. Then
\[
 \int_X\langle T_1\wedge\cdots\wedge T_p\rangle\wedge\omega^{n-p}
 \geq\int_X\langle T'_1\wedge\cdots\wedge T'_p\rangle\wedge\omega^{n-p}.
\]
Bounded differences in every factor give equality.
\end{lemma}
\begin{proof}
Replace factors successively and absorb their finite non-pluripolar product and $\omega^{n-p}$ into a closed positive current $\Theta$ of bidimension $(1,1)$. It gives $A$ zero mass. Normalize the factor potentials so that $\varphi\geq\psi$ and $\tau\leq0$. For fixed $\varepsilon>0$ put
$f_k=\max\{\psi+\varepsilon\tau,\varphi-k\}$. Their common quasi-psh lower bound is supplied by a sufficiently large multiple $\Omega$ of $\omega$. Since
$\psi+\varepsilon\tau-\varphi\leq\varepsilon\tau\to-\infty$ near $A$, $f_k=\varphi-k$ in a neighborhood of $A$.

The bounded function $f_k-\varphi+k$ has compact support in $X\setminus A$. Local recursion with the base $\Theta$ and distributional Stokes therefore give
\[
 \int_{X\setminus A}\ddc f_k\wedge\Theta
          =\int_{X\setminus A}\ddc\varphi\wedge\Theta.
\]
All signed measures here have finite mass: add their common smooth lower background and apply compact non-pluripolar finiteness, then subtract that background. On $X\setminus A$, the $f_k$ decrease to the locally bounded $\psi+\varepsilon\tau$. Fixed-base decreasing continuity and the lower inequality for positive measures give
\[
 \int_{X\setminus A}(\ddc\psi+\varepsilon\ddc\tau+\Omega)\wedge\Theta
 \leq\int_{X\setminus A}(\ddc\varphi+\Omega)\wedge\Theta.
\]
The $\Omega$ terms cancel. The measure $\ddc\tau\wedge\Theta$ likewise has finite variation by adding a smooth positive background. Let $\varepsilon\downarrow0$, restore the common $\theta$, and repeat for each factor. Reverse the roles when their differences are bounded.
\end{proof}

\Needspace{20\baselineskip}
\begin{lemma}\label{lem:projective-simplex-fibre-formula}
Let $u$ be smooth psh and $v$ locally bounded psh on $U\subset\mathbb C^n$. For $N\geq1$ write
$\Sigma_N=\{x\in\mathbb R^N:x_i\geq0,\ t(x)=\sum_i x_i\leq1\}$ and define on $U\times\mathbb C^N$
\[
 F=\sup_{x\in\Sigma_N}
       \left((1-t(x))u+t(x)v+
                    \sum_i x_i\log|z_i|^2-\sum_i x_i^2\right).
\]
A zero coefficient times $-\infty$ is zero. The canonical extension of this function is locally bounded psh, and with $p:U\times\mathbb C^N\to U$,
\begin{equation}\label{eq:projective-simplex-fibre-formula}
 p_*(\ddc F)^{n+N}
       =\binom{n+N}{n}N\int_0^1
                (\ddc((1-t)u+tv))^n t^{N-1}\,\mathrm dt.
\end{equation}
The pushforward is locally finite; over compact subsets of $U$ the measures have a common bounded fibre support when $v-u$ is bounded below.
\end{lemma}
\begin{proof}
First suppose $v$ smooth and work where every $z_i\ne0$. Put $a=v-u$ and $y_i=\log|z_i|^2$. The maximizing point is
\[
 x=P_{\Sigma_N}\bigl((y+a\mathbf1)/2\bigr),
\]
where $P_{\Sigma_N}$ is the Euclidean closest-point projection. Existence follows from compactness and uniqueness from strict concavity. Its variational inequality
$(b-Pb)\cdot(s-Pb)\leq0$ for $s\in\Sigma_N$ follows by differentiating the squared distance along the segment to $s$. Applying it twice proves $|Pb-Pb'|\leq|b-b'|$. Bounding the difference of two suprema with their respective maximizing points shows that the derivative in $y_i$ is $x_i$; the same argument gives the base differential. Consequently $F$ is $C^1$ with locally Lipschitz gradient.

For $b=(y+a\mathbf1)/2$, the projection has coordinates $x_i=\max\{b_i-\lambda,0\}$. If $\sum_i\max\{b_i,0\}\leq1$, take $\lambda=0$; otherwise the strictly decreasing sum on its positive range determines $\lambda>0$ by $\sum_i\max\{b_i-\lambda,0\}=1$. These coordinates satisfy the variational inequality and hence give the projection. On a fixed active set $I$ in the latter case, $\lambda=(\sum_{i\in I}b_i-1)/|I|$. Thus the projection is affine on finitely many polyhedral regions, with rank $|I|$ when $\lambda=0$ and $|I|-1$ otherwise. On the region where $x_i>0$ and $t(x)<1$ for all $i$, one has $x_i=(y_i+a)/2$ and $F=u+\sum_i x_i^2$. On every other open region the fibre projection has rank less than $N$. Region boundaries have Lebesgue measure zero: each is a nonconstant linear equation in $y+a\mathbf1$, and Fubini applies with the base point fixed. The continuous Lipschitz gradient has bounded distributional derivatives, as is seen by testing its bounded difference quotients and using \cref{lem:dominated-measure-density}. Smooth convolution, Lebesgue differentiation and bounded determinant convergence, together with \cref{prop:bt-uniform-continuity} and \eqref{eq:bt-determinant-normalization}, identify the Bedford--Taylor measure with the ordinary measurable Hessian determinant. Thus these boundaries carry no mass. In the lower-rank regions its positive Hessian has a vertical null vector, and hence zero full determinant.

On the interior region direct differentiation gives
\[
 \ddc F=(1-t(x))\ddc u+t(x)\ddc v
                       +2\sum_i\mathrm dx_i\wedge\mathrm d^{\mathrm c}x_i.
\]
The first summand has only base differentials, while the second has rank at most $N$. Expansion of the $(n+N)$th power therefore retains exactly $n$ base factors and $N$ of these rank-one factors, with coefficient $\binom{n+N}{n}$. On a fibre write $z_i=r_i\mathrm e^{\mathrm i\alpha_i}$. Our normalization gives
$\mathrm d^{\mathrm c}y_i=\mathrm d\alpha_i/(2\pi)$ and
$2\,\mathrm dx_i\wedge\mathrm d^{\mathrm c}x_i=\mathrm dx_i\wedge\mathrm d\alpha_i/(2\pi)$ there. Integrating the angular variables pushes the $N$th power to $N!\,\mathrm dx$ on $\Sigma_N$. No general real Monge--Amp\`ere theorem is needed. Finally
$|\{x\in\Sigma_N:t(x)\leq t\}|=t^N/N!$, by scaling the simplex and elementary iterated integration. Integration first in the level $t$ proves \eqref{eq:projective-simplex-fibre-formula}.

The support lies in the closure of the interior region. If $a\geq-C$, its equations give $y_i=2x_i-a\leq2+C$, hence $|z_i|\leq\mathrm e^{1+C/2}$. The function extends across coordinate hyperplanes as a bounded psh function, since the branch $x=0$ bounds it below and the displayed supremum bounds it locally above. Bounded analytic removal identifies the canonical extension. For general bounded $v$, take smooth decreasing approximations $v_l\downarrow v$ on a smaller base neighborhood. The corresponding $F_l$ decrease pointwise to $F$: at finite $v$ the suprema differ by at most $|v_l-v|$. They have common bounds and the same fibre-support bound. Bounded decreasing continuity passes their products and pushforwards to the limit. The right side passes by the same theorem and the uniform CLN bound in $t\in[0,1]$. Exhausting $U$ proves the lemma.
\end{proof}

\Needspace{24\baselineskip}
\begin{proposition}\label{prop:auxiliary-projective-mass}
Suppose $\{\theta\}$ is big and $\varphi_0\in\PSH(X,\theta)$ is smooth outside its proper analytic pole set, with $\mathrm e^{m\varphi_0}$ smooth for some $m>0$. For $\varphi\in\PSH(X,\theta)$ and homogeneous coordinates $[Z_0:\cdots:Z_N]$, put
\[
 s_i=\log\frac{|Z_i|^2}{\sum_{l=0}^N|Z_l|^2},\qquad
 \widetilde\theta=p_X^*\theta+p_{\mathbb P^N}^*\omega_{\mathrm{FS}},
\]
and let $\Phi$ be the upper regularization of
\[
 \sup_{x\in\Sigma_N}
       \left((1-t(x))(\varphi_0+s_0)+t(x)\varphi
                              +\sum_i x_i s_i-\sum_i x_i^2\right).
\]
Then $\Phi$ is $\widetilde\theta$-psh and locally bounded outside
$A=(\{\varphi_0=-\infty\}\times\mathbb P^N)\cup(X\times\{Z_0=0\})$. Its mass satisfies
\begin{equation}\label{eq:auxiliary-projective-mass}
 \begin{aligned}
 &\int_{X\times\mathbb P^N}\langle(\widetilde\theta+\ddc\Phi)^{n+N}\rangle\\
 &\quad=\binom{n+N}{n}N\int_0^1
       \left(\int_X\langle(\theta+\ddc((1-t)\varphi_0+t\varphi))^n\rangle\right)
                                      t^{N-1}\,\mathrm dt.
 \end{aligned}
\end{equation}
Such a reference $\varphi_0$ exists in every big class.
\end{proposition}
\begin{proof}
Each branch is $\widetilde\theta$-psh: the coefficients of its $X$ currents add to one, as do those of its projective hyperplane currents. The family is locally uniformly bounded above. Its zero branch $\varphi_0+s_0$ is finite off $A$, so its envelope is defined and is locally bounded there. The same branch is a global quasi-psh function with exact pole set $A$ and a smooth lower background. For existence, start with a strict current in the big class and apply \cref{lem:logarithmic-cutoff-sum} with sufficiently large $m$, retaining positivity. Its exponential is smooth and its zeros form the proper analytic multiplier-ideal pole set.

Off that pole set choose $\ddc h=\theta$ and work in $Z_0\ne0$, with $z_i=Z_i/Z_0$. Adding $h+\log(1+\sum_i|z_i|^2)$ to $\Phi$ gives the preceding local supremum with $u=\varphi_0+h$ and $v=\varphi+h$. Replace $v$ by $v_C=\max\{v,u-C\}$, which is locally bounded. On the plurifine open base set $\{v>u-C\}$ the two constructions agree, so locality identifies their products over that set. The fibre formula applies to $v_C$. Let $C\to\infty$ in these restrictions. The base sets exhaust all finite points of $v$; both sides give their omitted pluripolar set zero mass. The upstairs product also gives the inverse image of the analytic pole set and the projective hyperplane zero mass. The measure-valued integrand is Borel in $t$: multilinearity makes it a polynomial with finite mixed-current coefficients. Increasing Borel restrictions, monotone convergence and a finite chart partition therefore prove the pushforward formula and then \eqref{eq:auxiliary-projective-mass}. The local suprema equal their canonical extensions: decreasing bounded truncations have pointwise limits by compact maximization; at $v=-\infty$ the limit is the zero branch $u$. No envelope discrepancy changes the restricted products.
\end{proof}

\begin{theorem}\label{thm:nonpluripolar-self-mass-monotonicity}
If $\varphi,\psi\in\PSH(X,\theta)$ and $\varphi\geq\psi-O(1)$, then
\[
 \int_X\langle(\theta+\ddc\varphi)^n\rangle
       \geq\int_X\langle(\theta+\ddc\psi)^n\rangle.
\]
\end{theorem}
\begin{proof}
First let the class be big, choose the reference in the preceding proposition, and construct $\Phi_N,\Psi_N$. The inequality between the original potentials gives $\Phi_N\geq\Psi_N-O(1)$ with a constant independent of $N$, since $0\leq t(x)\leq1$. Both are bounded outside the same pole set $A$. Apply \cref{lem:small-locus-mixed-mass-comparison} upstairs and cancel the common binomial coefficient in \eqref{eq:auxiliary-projective-mass}. The two remaining integrals have weight $Nt^{N-1}\,\mathrm dt$.

By multilinearity, the mass of $\theta+\ddc((1-t)\varphi_0+t\varphi)$ is a polynomial in $t$: expand its positive current as $(1-t)T_0+tT_\varphi$. It is continuous on $[0,1]$. The probability measures $Nt^{N-1}\,\mathrm dt$ concentrate at $1$, since their mass on $[0,1-\delta]$ is $(1-\delta)^N$. Uniform continuity thus makes the weighted integrals tend to the original masses. This proves the big-class assertion.

For a merely pseudo-effective class add $\varepsilon\omega$ to $\theta$. The class becomes big while the same potentials remain admissible. Apply the proved inequality. Multilinearity and compact finiteness make both resulting masses polynomials in $\varepsilon$ with finite coefficients. Let $\varepsilon\downarrow0$. This reduction requires no assertion that non-big classes have zero mass.
\end{proof}
\index{non-pluripolar product!mass monotonicity}

\begin{corollary}\label{cor:nonpluripolar-bounded-type-mixed-mass}
For tuples $T_j=\theta_j+\ddc u_j$ and $T'_j=\theta_j+\ddc v_j$, $1\leq j\leq p\leq n$, bounded differences $u_j-v_j$ imply equality of their mixed masses against $\omega^{n-p}$. If only $u_j\geq v_j-O(1)$, then
\[
 \int_X\langle T_1\wedge\cdots\wedge T_p\rangle\wedge\omega^{n-p}
 \geq\int_X\langle T'_1\wedge\cdots\wedge T'_p\rangle\wedge\omega^{n-p}.
\]
\end{corollary}
\begin{proof}
For bounded differences, apply self-mass monotonicity in both directions to $\sum_j a_ju_j$ and $\sum_j a_jv_j$, with background $\sum_j a_j\theta_j$, for every tuple $a_j>0$. Add $n-p$ identical smooth $\omega$ factors if $p<n$. The equal degree-$n$ mass polynomials have equal coefficients: fixing all but one variable proves this successively by the elementary uniqueness of a one-variable polynomial. The coefficient of the product of the chosen variables is the mixed mass times its positive multinomial factor, proving equality.

For the ordered case put $v_{j,k}=\max\{u_j-k,v_j\}$. After harmless constant shifts $v_j\leq u_j+C_j$, so $u_j-k\leq v_{j,k}\leq u_j+C_j$. Each tuple therefore has the mixed mass of the $u_j$. It decreases to $v_j$ and converges in extended capacity. The lower-limit theorem bounds the mixed mass of the limit by those equal masses. This proves the inequality, including smooth extra factors.
\end{proof}
\index{non-pluripolar product!bounded-perturbation invariance}

\begin{corollary}\label{cor:nonpluripolar-increasing-continuity}
Suppose $u_{j,k}\in\PSH(X,\theta_j)$ increase almost everywhere to $u_j\in\PSH(X,\theta_j)$, $1\leq j\leq p\leq n$. Then their non-pluripolar products converge weakly to the product of the limits, and their mixed masses against $\omega^{n-p}$ converge.
\end{corollary}
\begin{proof}
Canonical representatives give $u_{j,k}\leq u_j$ pointwise, by applying uniqueness to their maximum. Add smooth local potentials on coordinate charts and truncate below a fixed constant. The resulting locally uniformly bounded psh sequences converge increasingly almost everywhere to the bounded truncations of the limits. \Cref{thm:bt-increasing-continuity,lem:extended-capacity-truncations} therefore give extended-capacity convergence of the original sequences. Mixed mass monotonicity bounds every trace mass by that of the limit product. The lower-limit theorem in \cref{thm:nonpluripolar-capacity-lower-limit} proves both weak convergence and convergence of the masses. This applies in particular to the canonical regularization of an increasing supremum, whose discrepancy set is pluripolar by \cref{cor:psh-pluripolar-discrepancy}.
\end{proof}
\index{non-pluripolar product!increasing continuity}

\section{Maximal mass and comparison}
\label{sec:nonpluripolar-full-mass}
Minimal singularities give the largest mixed mass in big classes. Perturbation by a K\"ahler class extends that number to the pseudo-effective boundary, where a current need not attain it.

\begin{definition}\label{def:positive-mixed-intersection-number}
For big classes $\alpha_j=\{\theta_j\}$, $1\leq j\leq p\leq n$, define the positive mixed intersection number, with the fixed test class $\{\omega\}$, by
\[
 I_\omega(\alpha_1,\ldots,\alpha_p)
 =\int_X\left\langle\bigwedge_{j=1}^p
           (\theta_j+\ddc V_{\theta_j})\right\rangle\wedge\omega^{n-p}.
\]
For pseudo-effective classes define it by the decreasing limit of
$I_\omega(\alpha_1+\varepsilon\{\omega\},\ldots,\alpha_p+\varepsilon\{\omega\})$ as $\varepsilon\downarrow0$. In top degree put $\operatorname{vol}(\alpha)=I_\omega(\alpha,\ldots,\alpha)$; the test factor is then the empty product.
\end{definition}
\index{positive intersection!mixed number}\index{volume!positive current class}

\begin{proposition}\label{prop:positive-mixed-mass-bound}
These definitions are independent of the chosen representatives and minimal potentials. The limit exists and is independent of the K\"ahler class used for the perturbation. For positive $T_j\in\alpha_j$,
\[
 \int_X\langle T_1\wedge\cdots\wedge T_p\rangle\wedge\omega^{n-p}
                       \leq I_\omega(\alpha_1,\ldots,\alpha_p).
\]
The numbers are homogeneous in each class and increase when a pseudo-effective class is added to any big argument. For nef classes they equal the ordinary cohomological intersection numbers. Big classes have positive volume.
\end{proposition}
\begin{proof}
All minimal potentials in a fixed big class differ by a bounded function. Mixed bounded-type invariance proves independence, including a change of smooth representative, which changes each potential by a smooth function. The same theorem and minimality prove the upper bound in big classes.

If $S\geq0$ represents a pseudo-effective class $\gamma$, add it to a minimal current in a big class $\alpha$. The sum belongs to $\alpha+\gamma$ and is at least as singular as its minimal current. Mixed mass monotonicity and positivity of the expanded sum show that replacing $\alpha$ by $\alpha+\gamma$ increases the number. Scaling a minimal potential and its background proves homogeneity. In particular the perturbed numbers decrease as $\varepsilon\downarrow0$; they are nonnegative and bounded above by the shifted smooth bound in \cref{thm:compact-nonpluripolar-finiteness}. Their limit exists. Two K\"ahler forms satisfy $c^{-1}\omega\leq\omega'\leq c\omega$. Class monotonicity sandwiches perturbation by $\varepsilon\omega'$ between perturbations by $c^{-1}\varepsilon\omega$ and $c\varepsilon\omega$, with the same fixed test $\omega^{n-p}$. This proves perturbation independence. The same sandwich, replacing an allowance by its fixed scalar multiple in one argument, extends homogeneity to the boundary. Changing $\omega$ within its fixed class changes its powers by a smooth exact form, whose integral against the closed product is zero by Stokes. For arbitrary $T_j$, apply the big-class bound to $T_j+\varepsilon\omega$ and use multilinearity as $\varepsilon\downarrow0$.

If the $\alpha_j$ are nef, their perturbations are K\"ahler. Minimal potentials there are bounded, so bounded global Stokes identifies the perturbed numbers with the smooth cup-product integrals. Their limits are the ordinary intersections. Finally choose a rational analytic strict current $T_0\in\alpha$ with $T_0\geq\delta\omega$, supplied by \cref{lem:logarithmic-cutoff-sum}. Its mass is at least $\delta^n\int_X\omega^n$ by positive multilinearity. Minimality and mass monotonicity bound the volume below by this positive number.
\end{proof}

\begin{definition}\label{def:full-nonpluripolar-mass}
A positive current $T\in\alpha$ has \emph{full Monge--Amp\`ere mass} if $\int_X\langle T^n\rangle=\operatorname{vol}(\alpha)$. For $\phi\in\PSH(X,\theta)$, the class $\mathcal E(X,\theta,\phi)$ consists of the $\theta$-psh potentials $u$ with $u\leq\phi+O(1)$ and
\[
 \int_X\langle(\theta+\ddc u)^n\rangle
     =\int_X\langle(\theta+\ddc\phi)^n\rangle.
\]
These potentials have full mass relative to $\phi$. The notation $\mathcal E(X,\theta)$ denotes the absolute full-mass class. In a big class it equals $\mathcal E(X,\theta,V_\theta)$.
\end{definition}
\index{full Monge--Amp\`ere mass}\index{full Monge--Amp\`ere mass!relative to a potential}

\begin{corollary}\label{cor:relative-full-mass-capacity-convergence}
Suppose $u_k\leq\phi+C_k$ are $\theta$-psh, $u\in\mathcal E(X,\theta,\phi)$, and $u_k\to u$ in extended capacity. Then
\[
 \langle(\theta+\ddc u_k)^n\rangle
                   \longrightarrow\langle(\theta+\ddc u)^n\rangle
\]
weakly and their masses converge. In particular this holds for decreasing sequences with full-mass limit, using absolute full mass in a big class.
\end{corollary}
\begin{proof}
Monotonicity bounds every mass by that of $\phi$, which equals the limit mass. Apply \cref{thm:nonpluripolar-capacity-lower-limit}. In the absolute case use the positive mixed mass bound; no common constant $C_k$ is required.
\end{proof}

\Needspace{10\baselineskip}
\begin{proposition}\label{prop:global-nonpluripolar-comparison}
Write $\mu_u=\langle(\theta+\ddc u)^n\rangle$ for $u\in\PSH(X,\theta)$. For any $u,v\in\PSH(X,\theta)$ one has
\[
 \int_{\{u<v\}}\mu_v
 \leq\int_{\{u<v\}}\mu_u+\operatorname{vol}(\{\theta\})-\mu_u(X).
\]
The defect term may be omitted if $u$ has full mass, or if $u\geq v-O(1)$.
\end{proposition}
\begin{proof}
For $\varepsilon>0$, $w=\max\{u,v-\varepsilon\}$ is $\theta$-psh. Locality on the disjoint plurifine open sets $\{u<v-\varepsilon\}$ and $\{u>v-\varepsilon\}$ gives
\[
 \mu_w(X)\geq
 \mu_v(\{u<v-\varepsilon\})+\mu_u(\{u>v-\varepsilon\}).
\]
The left side is at most the volume. Subtract $\mu_u(X)$ and use
$\{u\leq v-\varepsilon\}\subset\{u<v\}$. Let $\varepsilon\downarrow0$; the first sets increase to $\{u<v\}$ apart from common poles, which both measures annihilate. If $u$ has full mass the defect is zero. If $u\geq v-O(1)$, then $w-u$ is bounded, so $\mu_w(X)=\mu_u(X)$ and the same computation gives the zero-defect inequality directly.
\end{proof}
\index{comparison principle!non-pluripolar global measures}

\begin{corollary}\label{cor:nonpluripolar-domination-transfer}
Suppose $u,v,\rho\in\PSH(X,\theta)$, $u\geq v-O(1)$ and $u\geq\rho-O(1)$. If $u\geq v$ almost everywhere for $\mu_u$, then it holds almost everywhere for $\mu_\rho$. If the class is big and $u$ has minimal singularities, then $u\geq v$ everywhere.
\end{corollary}
\begin{proof}
Shift $\rho$ so that $\rho\leq u$. For $0<t<1$ put $q_t=(1-t)v+t\rho$. Positivity and comparison give
\[
 t^n\mu_\rho(\{u<q_t\})
   \leq\mu_{q_t}(\{u<q_t\})
   \leq\mu_u(\{u<q_t\})=0,
\]
since $u$ is less singular than $q_t$ and $\{u<q_t\}\subset\{u<v\}$. As $t\downarrow0$, every finite point with $u<v$ eventually belongs to these sets; Fatou proves its zero $\mu_\rho$ mass. Poles have zero mass. For minimal $u$ in a big class choose the strict analytic reference $\rho$, which is more singular than $u$ and has $\mu_\rho\geq\delta^n\omega^n$. The first assertion gives $u\geq v$ in volume measure. Local psh ball averages, with the common smooth background added, then give the pointwise inequality everywhere.
\end{proof}
