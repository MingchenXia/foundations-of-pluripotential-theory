\section{Mixed products for a common measure}
\label{sec:common-measure-mixed-products}
The measure in a density lower bound may be singular with respect to volume. Convolution then preserves neither of the two prescribed lower bounds. Instead we solve equations for cube averages of the pure Monge--Amp\`ere measures. H\"older's inequality compares their densities with the cube averages of the desired mixed lower bound, and the energy estimates justify taking limits.

\begin{theorem}\label{thm:bounded-common-measure-mixed}
Let $\Omega\subset\mathbb C^n$ be open, $n\geq1$, and let $u,v$ be locally bounded plurisubharmonic functions. Let $\mu$ be a positive Radon measure and $f,g\geq0$ belong to $L^1_{\loc}(\mu)$. If
\[
 (\ddc u)^n\geq f\mu,\qquad(\ddc v)^n\geq g\mu,
\]
then, for $1\leq k\leq n-1$,
\begin{equation}\label{eq:bounded-common-measure-mixed}
 (\ddc u)^k\wedge(\ddc v)^{n-k}
                     \geq f^{k/n}g^{(n-k)/n}\mu.
\end{equation}
The endpoint bounds are the given pure-measure inequalities. For $n=1$ there are only these endpoints. Densities are understood up to $\mu$-null sets.
\end{theorem}
\begin{proof}
We may assume $n\geq2$ and fix $1\leq k\leq n-1$. Put $a=k/n$. H\"older's inequality makes $f^ag^{1-a}\mu$ locally finite. We first prove the result for $u,v\in\mathcal E_0(B)$, and then modify arbitrary locally bounded potentials near the boundary of a ball.

\emph{Step 1. Compactly supported pure measures.}
Choose increasing compact concentric balls $K_l\Subset B$ exhausting $B$, and put
\[
 \alpha_l=\mathbf1_{K_l}(\ddc u)^n,\qquad
 \beta_l=\mathbf1_{K_l}(\ddc v)^n.
\]
They have finite masses and bounded subsolutions $u,v$. By \cref{thm:compact-bounded-subsolution-ball-dirichlet}, there are $A_l,B_l\in\mathcal E_0(B)$ with zero boundary values and pure measures $\alpha_l,\beta_l$. The pointwise envelope assertion gives
\[
 u\leq A_{l+1}\leq A_l\leq0,\qquad
 v\leq B_{l+1}\leq B_l\leq0.
\]
Indeed $u$ is a subsolution for every $\alpha_l$, and $A_{l+1}$ is a subsolution for $\alpha_l$; similarly for $v,B_l$.

Let $A_\infty=\lim_lA_l$. It is bounded and psh, and its boundary limits are zero because $u\leq A_\infty\leq0$. Bounded decreasing continuity and $\alpha_l\uparrow(\ddc u)^n$ give
\[
 (\ddc A_\infty)^n=(\ddc u)^n.
\]
Their common total mass is finite. For $\rho=|z-a_0|^2-R^2$ on $B=B(a_0,R)$, compare $u$ with $A_\infty+\varepsilon\rho$. The strict quadratic term gives
\[
 \varepsilon^n\int_{\{u<A_\infty+\varepsilon\rho\}}(\ddc\rho)^n
 \leq0
\]
after cancelling the two finite integrals of $(\ddc u)^n$. Canonical representatives give $u\geq A_\infty+\varepsilon\rho$ everywhere. Let $\varepsilon\downarrow0$ to obtain $A_\infty=u$. This uniqueness argument uses finite mass and comparison, without requiring compact support of the limiting measure. The same argument proves $B_l\downarrow v$.

\emph{Step 2. The inequality on a grid.}
Fix $l$ and use a common sequence of half-open cube partitions with side lengths tending to zero. For all sufficiently fine partitions, every cube meeting $K_l$ has closure in $B$. Let $\alpha_{l,j},\beta_{l,j}$ be the cube averages of $\alpha_l,\beta_l$, and let $A_{l,j},B_{l,j}$ be their zero-boundary solutions. The compact-measure theorem gives
\[
 A_{l,j}\longrightarrow A_l,\qquad
 B_{l,j}\longrightarrow B_l
 \quad\text{locally in capacity.}
\]
The grid energy estimate gives a common bound for their first energies. It can be chosen independently of $l$: the dominating potentials are the fixed $u,v$, and the masses of $\alpha_l,\beta_l$ are bounded by their fixed total masses. Zero measures give zero solutions. \Cref{cor:mixed-products-energy-capacity-continuity} therefore gives
\begin{equation}\label{eq:common-measure-grid-mixed-limit}
 (\ddc A_{l,j})^k\wedge(\ddc B_{l,j})^{n-k}
       \longrightarrow(\ddc A_l)^k\wedge(\ddc B_l)^{n-k}
\end{equation}
weakly at fixed $l$.

Write $\lambda_l=\mathbf1_{K_l}f^ag^{1-a}\mu$, and let $\lambda_{l,j}$ be its cube averages. For each cube $Q$,
\[
 \alpha_l(Q)\geq\int_{Q\cap K_l}f\,\mathrm d\mu,
 \qquad
 \beta_l(Q)\geq\int_{Q\cap K_l}g\,\mathrm d\mu.
\]
H\"older's inequality gives
\begin{equation}\label{eq:common-measure-grid-holder}
 \left(\frac{\alpha_l(Q)}{|Q|}\right)^a
 \left(\frac{\beta_l(Q)}{|Q|}\right)^{1-a}
 \geq\frac1{|Q|}\int_{Q\cap K_l}f^ag^{1-a}\,\mathrm d\mu
       =\frac{\lambda_l(Q)}{|Q|}.
\end{equation}
On the interior of each cube meeting $K_l$, the two pure densities are constant. Apply \cref{cor:bounded-mixed-continuous-density} there. Its reference measure is $(\ddc|z|^2)^n=c_n\,\mathrm dV$, $c_n=n!\pi^{-n}$; dividing both constant densities by $c_n$ and multiplying the resulting mixed bound by $c_n$ gives exactly the left side of \eqref{eq:common-measure-grid-holder} as a density against $\mathrm dV$. The desired right-hand density is zero on all other cubes. Positivity of the mixed measure and the volume-null cube boundaries consequently give
\[
 (\ddc A_{l,j})^k\wedge(\ddc B_{l,j})^{n-k}
                                      \geq\lambda_{l,j}
\]
on $B$. Only finitely many cubes have nonzero right-hand density. For a continuous compact test, its cube average differs from its value by its modulus of continuity at the cube diameter. Thus $\lambda_{l,j}\to\lambda_l$ weakly, without any regularity assumption on $f,g$ or the measure. Passing $j\to\infty$ in \eqref{eq:common-measure-grid-mixed-limit} proves
\[
 (\ddc A_l)^k\wedge(\ddc B_l)^{n-k}\geq\lambda_l.
\]
Now $A_l\downarrow u$, $B_l\downarrow v$ with common bounds, while $\lambda_l\uparrow f^ag^{1-a}\mu$. Bounded decreasing continuity and monotone convergence on nonnegative compact tests prove the theorem for $\mathcal E_0(B)$.

\emph{Step 3. Localization.}
Return to the given $u,v$ on $\Omega$. Choose $B=B(a_0,R)$ with $\overline B\subset\Omega$, put $\rho=|z-a_0|^2-R^2$, and fix $K\Subset B$. Subtract constants so that both potentials lie below $-1$ on $B$. For sufficiently large positive constants $C,D$, the potentials
\[
 \widetilde u=\max\{u,C\rho\},\qquad
 \widetilde v=\max\{v,D\rho\}
\]
agree with $u,v$ on a neighborhood of $K$ and are quadratic on boundary collars. They are in $\mathcal E_0(B)$, since their collar measures are smooth and their interior measures are locally finite. Replace $\mu$ by $\mathbf1_K\mu$. Maximum locality preserves both pure lower bounds for this restricted measure, so the proved case gives the desired mixed lower bound on $K$. The constant shifts do not change any product. Such balls and inner compact neighborhoods cover $\Omega$, proving the measure inequality everywhere.
\end{proof}

\Needspace{10\baselineskip}
\begin{corollary}\label{cor:bounded-common-measure-sum}
Under the hypotheses of the theorem,
\[
 (\ddc(u+v))^n\geq(f^{1/n}+g^{1/n})^n\mu.
\]
\end{corollary}
\begin{proof}
Expand the left side by multilinearity. For each mixed term use the theorem; for the two endpoints use the original pure bounds. The binomial formula gives the asserted density. This also proves the result when $n=1$ by direct additivity. The densities are locally integrable: each interior term is bounded by a constant times $f+g$, by the weighted arithmetic--geometric mean inequality.
\end{proof}
