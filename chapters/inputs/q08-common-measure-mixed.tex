\section{Mixed products for a common measure}
\label{sec:common-measure-mixed-products}
The measure in a density lower bound may be singular with respect to volume. Convolution then preserves neither of the prescribed lower bounds. Instead we solve equations for cube averages of all the pure Monge--Amp\`ere measures. H\"older's inequality compares their densities with the cube averages of the desired mixed lower bound, and the energy estimates justify taking limits.

\begin{theorem}\label{thm:bounded-multifactor-common-measure}
Let $\Omega\subset\mathbb C^n$ be open, $n\geq1$, and let $u_1,\ldots,u_n$ be locally bounded plurisubharmonic functions. Let $\mu$ be a positive Radon measure, and let $f_i\geq0$ belong to $L^1_{\loc}(\mu)$. If
\[
 (\ddc u_i)^n\geq f_i\mu\qquad(1\leq i\leq n),
\]
then
\begin{equation}\label{eq:bounded-multifactor-common-measure}
 \ddc u_1\wedge\cdots\wedge\ddc u_n
                  \geq\left(\prod_{i=1}^nf_i^{1/n}\right)\mu.
\end{equation}
Repeated potentials and zero densities are permitted. For $n=1$ this is the hypothesis. Densities are understood up to $\mu$-null sets.
\end{theorem}
\begin{proof}
Suppose $n\geq2$. Generalized H\"older's inequality makes $\prod_i f_i^{1/n}\mu$ locally finite. We first assume all $u_i\in\mathcal E_0(B)$ and then localize by quadratic collars.

\emph{Step 1. Compactly supported pure measures.}
Choose increasing compact concentric balls $K_l\Subset B$ exhausting $B$, and set
\[
 \alpha_{i,l}=\mathbf1_{K_l}(\ddc u_i)^n.
\]
These measures have finite masses and bounded subsolutions $u_i$. By \cref{thm:compact-bounded-subsolution-ball-dirichlet}, their zero-boundary solutions $A_{i,l}$ lie in $\mathcal E_0(B)$. Its pointwise envelope assertion gives
\[
 u_i\leq A_{i,l+1}\leq A_{i,l}\leq0:
\]
$u_i$ is a subsolution for every $\alpha_{i,l}$, and $A_{i,l+1}$ is a subsolution for $\alpha_{i,l}$.

The limit $A_{i,\infty}$ is bounded and psh. Its boundary limits are zero because $u_i\leq A_{i,\infty}\leq0$. Bounded decreasing continuity gives
\[
 (\ddc A_{i,\infty})^n=(\ddc u_i)^n.
\]
Their common total mass is finite. Write $\rho=|z-a_0|^2-R^2$ for $B=B(a_0,R)$. Compare $u_i$ with $A_{i,\infty}+\varepsilon\rho$. Their equal pure measures and positivity of the mixed terms give
\[
 \varepsilon^n\int_{\{u_i<A_{i,\infty}+\varepsilon\rho\}}
                         (\ddc\rho)^n\leq0
\]
after cancelling two finite integrals of the common measure. The comparison applies because both boundary limits are zero. The strict set is volume-null; canonical psh representatives preserve the almost-everywhere ordering. Hence $u_i\geq A_{i,\infty}+\varepsilon\rho$ everywhere. Letting $\varepsilon\downarrow0$ proves $A_{i,l}\downarrow u_i$ for every $i$.

\emph{Step 2. A common grid.}
Fix $l$. Partition $\mathbb R^{2n}$ into half-open cubes with side length tending to zero, using the same partitions for all $i$. For sufficiently fine grids every cube meeting $K_l$ has closure in $B$. Let $\alpha_{i,l,j}$ be the cube averages of $\alpha_{i,l}$ and $A_{i,l,j}$ their zero-boundary solutions. The compact-measure theorem gives $A_{i,l,j}\to A_{i,l}$ locally in capacity. The canonical-grid energy estimate, \cref{prop:canonical-grid-energy-control}, gives a common finite bound for all their first energies at this fixed $l$. There are only finitely many $i$, and each original measure is dominated by the bounded subsolution $u_i$. Zero measures have zero solutions. Thus \cref{cor:mixed-products-energy-capacity-continuity} gives
\begin{equation}\label{eq:common-measure-grid-mixed-limit}
 \bigwedge_{i=1}^n\ddc A_{i,l,j}
                      \longrightarrow\bigwedge_{i=1}^n\ddc A_{i,l}
\end{equation}
weakly on $B$.

Put $\lambda_l=\mathbf1_{K_l}\prod_i f_i^{1/n}\mu$ and let $\lambda_{l,j}$ be its cube averages. For every cube $Q$, pure domination and generalized H\"older give
\begin{equation}\label{eq:common-measure-grid-holder}
 \prod_{i=1}^n\left(\frac{\alpha_{i,l}(Q)}{|Q|}\right)^{1/n}
 \geq\frac1{|Q|}\int_{Q\cap K_l}\prod_{i=1}^nf_i^{1/n}\,\mathrm d\mu
 =\frac{\lambda_l(Q)}{|Q|}.
\end{equation}
On each cube interior the pure densities of $A_{i,l,j}$ are constant. Apply \cref{cor:bounded-multifactor-continuous-density} there. Its reference measure is $\eta^n=c_n\,\mathrm dV$, $c_n=n!\pi^{-n}$. Dividing the $n$ pure densities by $c_n$ and multiplying the mixed density by $c_n$ cancels this factor, since the $n$ exponents sum to one. The result is the left side of \eqref{eq:common-measure-grid-holder} against $\mathrm dV$. The desired density vanishes on cubes with $\lambda_l(Q)=0$. Positivity and the volume-null cube boundaries therefore give on all of $B$
\[
 \bigwedge_{i=1}^n\ddc A_{i,l,j}\geq\lambda_{l,j}.
\]
Only finitely many cubes have nonzero right-hand density. On a compact continuous test, replacing values by cube averages changes its integral by at most the modulus of continuity at the cube diameter times $\lambda_l(B)$. Hence $\lambda_{l,j}\to\lambda_l$ weakly, even for singular $\mu$ and measurable $f_i$. Pass $j\to\infty$ using \eqref{eq:common-measure-grid-mixed-limit}. This proves $\bigwedge_i\ddc A_{i,l}\geq\lambda_l$. Finally $A_{i,l}\downarrow u_i$ with common bounds for each $i$, and $\lambda_l\uparrow\prod_i f_i^{1/n}\mu$. Bounded mixed decreasing continuity and monotone convergence on nonnegative compact tests prove the claimed inequality for $\mathcal E_0(B)$.

\emph{Step 3. Localization.}
For the given locally bounded potentials choose a ball $B=B(a_0,R)$ with $\overline B\subset\Omega$ and a compact $K\Subset B$. Subtract constants so that every $u_i\leq-1$ on $B$. For sufficiently large $C_i>0$,
\[
 \widetilde u_i=\max\{u_i,C_i\rho\},\qquad
 \rho=|z-a_0|^2-R^2,
\]
agrees with $u_i$ near $K$ and is quadratic on a boundary collar. Thus $\widetilde u_i\in\mathcal E_0(B)$: the collar measure is smooth and its interior measure is locally finite. Replace $\mu$ by $\mathbf1_K\mu$. Maximum locality preserves every pure lower bound for this measure. Apply the proved case and restrict to $K$, where all products agree with the original ones. Constant shifts do not affect any product. Such balls and compact neighborhoods cover $\Omega$, so the inequality holds everywhere.
\end{proof}
\index{Monge--Amp\`ere measure!multifactor common-measure inequality}

\begin{corollary}\label{cor:bounded-common-measure-mixed}
Let $u,v$ be locally bounded plurisubharmonic functions on an open $\Omega\subset\mathbb C^n$, $n\geq1$, and let $\mu$ be a positive Radon measure. If $f,g\geq0$ belong to $L^1_{\loc}(\mu)$ and
\[
 (\ddc u)^n\geq f\mu,\qquad(\ddc v)^n\geq g\mu,
\]
then, for each integer $1\leq k\leq n-1$,
\begin{equation}\label{eq:bounded-common-measure-mixed}
 (\ddc u)^k\wedge(\ddc v)^{n-k}
                       \geq f^{k/n}g^{(n-k)/n}\mu.
\end{equation}
The endpoint bounds are the hypotheses. For $n=1$ only these endpoints occur.
\end{corollary}
\begin{proof}
Apply the theorem to $k$ copies of $u$ and $n-k$ copies of $v$, with the corresponding repeated densities. H\"older's inequality gives local finiteness of the resulting measure.
\end{proof}
\index{Monge--Amp\`ere measure!common-measure mixed inequality}

\Needspace{10\baselineskip}
\begin{corollary}\label{cor:bounded-common-measure-sum}
Under the hypotheses of \cref{cor:bounded-common-measure-mixed},
\[
 (\ddc(u+v))^n\geq(f^{1/n}+g^{1/n})^n\mu.
\]
\end{corollary}
\begin{proof}
Expand the left side by multilinearity. For each interior mixed term use \cref{cor:bounded-common-measure-mixed}; for the two endpoints use the original pure bounds. The binomial formula gives the asserted density. This also proves the result when $n=1$ by direct additivity. The densities are locally integrable: each interior term is bounded by a constant times $f+g$, by the weighted arithmetic--geometric mean inequality.
\end{proof}
