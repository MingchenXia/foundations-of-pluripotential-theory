The contact formula involves the weak complex Hessians of an obstacle only on the contact set. Off this set the second derivatives of an obstacle with bounded metric Laplacian may be unbounded, and to give a meaning to the wedge products of the forms $\theta_j+\ddc f_j$ on all of $X$ we need more integrability. We now prove that all weak second derivatives of such an obstacle lie in $L^p$ for every $p<\infty$. This is the $L^p$ theory of the elliptic operator $L$ on the compact manifold $X$, which we derive from the Euclidean estimate \eqref{eq:euclidean-hessian-lp} by freezing coefficients. We write $C^{1,\alpha}(X)$ for the space of functions on $X$ that belong to $C^{1,\alpha}_{\loc}$ in every chart (\cref{sec:hessian-singular-integrals}). As there, matrices carry the Hilbert--Schmidt norm, and for a matrix $A=(A^{\mu\sigma})$ of functions and a function $w$ we write $A:D^2w=\sum_{\mu,\sigma}A^{\mu\sigma}\partial_\mu\partial_\sigma w$.

\begin{lemma}\label{lem:frozen-coefficient-lp}
Let $d\geq1$, $1<p<\infty$ and $\lambda>0$. There are constants $\delta_0>0$ and $C$, depending only on $d$, $p$ and $\lambda$, with the following property. Let $B\subset\mathbb R^d$ be a ball with center $x_B$, and let $A$ be a measurable function on $B$ with values in the real symmetric $d\times d$ matrices such that
\[
 A(x_B)\geq\lambda I,\qquad |A(x)-A(x_B)|\leq\delta_0\quad\text{for }x\in B.
\]
Then every $w\in W^{2,p}(\mathbb R^d)$ that vanishes almost everywhere outside a compact subset of $B$ satisfies
\[
 \|D^2w\|_p\leq C\|A:D^2w\|_{L^p(B)}.
\]
\end{lemma}
\begin{proof}
First let $A_0$ be a constant real symmetric matrix with $A_0\geq\lambda I$, and put $M=A_0^{-1/2}$, a symmetric matrix of operator norm at most $\lambda^{-1/2}$. For $v\in C_c^\infty(\mathbb R^d)$ let $\tilde v(y)=v(M^{-1}y)$, so that $v(x)=\tilde v(Mx)$. Then $D^2v(x)=M\,D^2\tilde v(Mx)\,M$ and $A_0:D^2v(x)=\operatorname{tr}(MA_0M\,D^2\tilde v(Mx))=\Delta\tilde v(Mx)$. Since $|MAM|\leq\lambda^{-1}|A|$ for every matrix $A$, the change of variables $y=Mx$ and \eqref{eq:euclidean-hessian-lp} give
\[
 \begin{aligned}
 \|D^2v\|_p&\leq\lambda^{-1}|\det M|^{-1/p}\|D^2\tilde v\|_p\\
 &\leq\lambda^{-1}C_{d,p}|\det M|^{-1/p}\|\Delta\tilde v\|_p
 =\lambda^{-1}C_{d,p}\|A_0:D^2v\|_p.
 \end{aligned}
\]
This estimate extends to every $w\in W^{2,p}(\mathbb R^d)$ that vanishes almost everywhere outside a compact set $K$. Indeed, for $0<\varepsilon\leq1$ the mollifications $w*\rho_\varepsilon$ are test functions supported in the compact set $K'=\{x:\operatorname{dist}(x,K)\leq1\}$, their second derivatives are $(\partial_\mu\partial_\sigma w)*\rho_\varepsilon$, by \cref{prop:distribution-smoothing}(1), and these converge to $\partial_\mu\partial_\sigma w$ in $L^p(K')$, by \cref{lem:mollifier-density}(2), hence in $L^p(\mathbb R^d)$, since all these functions vanish almost everywhere outside $K'$. Applying the estimate to $v=w*\rho_\varepsilon$ and letting $\varepsilon\to0$, we obtain it for $w$.

Now let $A$ and $w$ be as in the statement, put $C_1=\lambda^{-1}C_{d,p}$, $\delta_0=1/(2C_1)$, and $A_0=A(x_B)$. Since $|\sum_{\mu,\sigma}m_{\mu\sigma}m'_{\mu\sigma}|\leq|m|\,|m'|$ for matrices $m$ and $m'$, we have $|A_0:D^2w-A:D^2w|\leq\delta_0|D^2w|$ almost everywhere on $B$, and both sides vanish outside $B$. Hence
\[
 \|D^2w\|_p\leq C_1\|A_0:D^2w\|_p\leq C_1\|A:D^2w\|_{L^p(B)}+\tfrac12\|D^2w\|_p,
\]
and since $\|D^2w\|_p<\infty$ we obtain the lemma with $C=2C_1$.
\end{proof}
\index{Hessian!$L^p$ estimate!frozen coefficients}

We apply the lemma to complex Hessian operators. Let $U\subset\mathbb C^n=\mathbb R^{2n}$ be open, with real coordinates $x_{2i-1}=\Real z_i$, $x_{2i}=\Img z_i$, and let $(a^{i\bar j})$ be a function on $U$ with values in $n\times n$ matrices such that $\overline{a^{i\bar j}}=a^{j\bar i}$. Since $\partial_{z_i}=\frac12(\partial_{2i-1}-\mathrm i\partial_{2i})$ and $\partial_{\bar z_j}=\frac12(\partial_{2j-1}+\mathrm i\partial_{2j})$, the operator $\sum_{i,j}a^{i\bar j}\partial_{z_i}\partial_{\bar z_j}$ is $\sum_{\mu,\sigma}c^{\mu\sigma}\partial_\mu\partial_\sigma$ with complex coefficients satisfying $\sum_{\mu,\sigma}c^{\mu\sigma}\xi_\mu\xi_\sigma=\frac14\sum_{i,j}a^{i\bar j}\zeta_i\bar\zeta_j$ for $\xi\in\mathbb R^{2n}$, where $\zeta_i=\xi_{2i-1}-\mathrm i\xi_{2i}$. It maps real functions to real functions, because $\overline{a^{i\bar j}\partial_{z_i}\partial_{\bar z_j}w}=a^{j\bar i}\partial_{\bar z_i}\partial_{z_j}w$ for real $w$. Hence, on real functions,
\begin{equation}\label{eq:complex-hessian-real-form}
 \sum_{i,j}a^{i\bar j}\partial_{z_i}\partial_{\bar z_j}w=A:D^2w,\qquad
 A\xi\cdot\xi=\frac14\sum_{i,j}a^{i\bar j}\zeta_i\bar\zeta_j,
\end{equation}
where $A$ is the symmetric part of the real part of $(c^{\mu\sigma})$. Thus $A$ is real symmetric, $|A(x)-A(y)|\leq|a(x)-a(y)|$, and since $|\zeta|=|\xi|$, a lower bound $\sum a^{i\bar j}\zeta_i\bar\zeta_j\geq\lambda|\zeta|^2$ gives $A\geq(\lambda/4)I$. By \eqref{eq:metric-laplacian-coordinates}, in a holomorphic chart the metric Laplacian acts on real functions as $L=-4\pi\,A:D^2$, where $A$ is obtained in this way from $a^{i\bar j}=h^{i\bar j}$; in particular $L$ has no terms of order less than two.

Later we also need difference quotients. For a coordinate unit vector $e$ of $\mathbb R^d$ and $h\neq0$ we write $\tau_hZ(x)=Z(x+he)$ and $\delta_hZ=(\tau_hZ-Z)/h$.

\begin{lemma}\label{lem:difference-quotients}
Let $D'\Subset D$ be open subsets of $\mathbb R^d$, let $e$ be a coordinate unit vector, and put $h_0=\operatorname{dist}(D',\mathbb R^d\setminus D)$, with $h_0=\infty$ if $D=\mathbb R^d$.
\begin{enumerate}
\item Let $1\leq p<\infty$, and let $Z\in L^1_{\loc}(D)$ be such that the distribution $\partial_eZ$ belongs to $L^p(D)$. Then, for $0<|h|<h_0$, $\|\delta_hZ\|_{L^p(D')}\leq\|\partial_eZ\|_{L^p(D)}$, and $\delta_hZ\to\partial_eZ$ in $L^p(D')$ as $h\to0$.
\item Let $2\leq p<\infty$, let $Z\in L^p(D)$, and suppose that $\|\delta_hZ\|_{L^p(D')}\leq M$ for $0<|h|<h_1$, where $0<h_1\leq h_0$. Then $\partial_eZ\in L^p(D')$ and $\|\partial_eZ\|_{L^p(D')}\leq M$.
\end{enumerate}
\end{lemma}
\begin{proof}
(1) Fix $h$ with $0<|h|<h_0$. We claim that $\delta_hZ(x)=\int_0^1\partial_eZ(x+the)\,\mathrm dt$ for almost every $x\in D'$. For smooth $Z$ this is the fundamental theorem of calculus. In general, let $K\subset D'$ be compact; the set $K'=\{x+the:x\in K,\ 0\leq t\leq1\}$ is a compact subset of $D$. The mollifications $Z_\varepsilon=Z*\rho_\varepsilon$ are smooth near $K'$ for small $\varepsilon$, they converge to $Z$ in $L^1(K')$, and $\partial_eZ_\varepsilon=(\partial_eZ)*\rho_\varepsilon\to\partial_eZ$ in $L^1(K')$, by \cref{prop:distribution-smoothing}(1) and \cref{lem:mollifier-density}(2). Hence the identity for $Z_\varepsilon$ passes to the limit in $L^1(K)$; for the right side we use $\int_K\int_0^1|F(x+the)|\,\mathrm dt\,\mathrm dx\leq\int_{K'}|F|$, by Tonelli's theorem. This proves the claim. By H\"older's inequality on $[0,1]$ and Tonelli's theorem,
\[
 \int_{D'}|\delta_hZ|^p\leq\int_0^1\int_{D'}|\partial_eZ(x+the)|^p\,\mathrm dx\,\mathrm dt\leq\|\partial_eZ\|_{L^p(D)}^p.
\]
Similarly, since $\delta_hZ(x)-\partial_eZ(x)=\int_0^1\bigl(\partial_eZ(x+the)-\partial_eZ(x)\bigr)\,\mathrm dt$ for almost every $x\in D'$, H\"older's inequality and Tonelli's theorem give, with $F$ the extension of $\partial_eZ$ by zero to $\mathbb R^d$,
\[
 \|\delta_hZ-\partial_eZ\|_{L^p(D')}\leq\sup_{|s|\leq|h|}\|F(\cdot+se)-F\|_{L^p(\mathbb R^d)},
\]
which tends to zero as $h\to0$ by the continuity of translation in $L^p(\mathbb R^d)$ (\cref{lem:mollifier-density}(2)).

(2) Let $\chi\in C_c^\infty(D')$ with $0\leq\chi\leq1$, extend $\chi Z$ by zero, and let $0<|h|<\min\{h_1,\operatorname{dist}(\Supp\chi,\mathbb R^d\setminus D')\}$. Then $\delta_h(\chi Z)$ vanishes outside $D'$, and on $D'$ it equals $(\tau_h\chi)\delta_hZ+(\delta_h\chi)Z$. Since $D'$ is bounded and $p\geq2$, H\"older's inequality gives
\[
 \|\delta_h(\chi Z)\|_{L^2(\mathbb R^d)}\leq|D'|^{1/2-1/p}\bigl(M+\|\nabla\chi\|_\infty\|Z\|_{L^p(D)}\bigr)=:M'.
\]
The function $\chi Z$ lies in $L^1\cap L^2$, so its $L^2$ Fourier transform is given by the integral (\cref{thm:plancherel}), and $\widehat{\delta_h(\chi Z)}=m_h\widehat{\chi Z}$ with $m_h(\xi)=(\mathrm e^{2\pi\mathrm ih\xi_e}-1)/h$, where $\xi_e$ is the component of $\xi$ in the direction $e$. By Plancherel's theorem, $\int|m_h|^2|\widehat{\chi Z}|^2\leq M'^2$, and as $m_h(\xi)\to2\pi\mathrm i\xi_e$, Fatou's lemma gives $\int4\pi^2\xi_e^2|\widehat{\chi Z}|^2\leq M'^2$. Let $F\in L^2$ be the inverse $L^2$ transform of $2\pi\mathrm i\xi_e\widehat{\chi Z}$. For a test function $\phi$ we have $\widehat{\partial_e\phi}=2\pi\mathrm i\xi_e\widehat\phi$, by integration by parts, so Parseval's identity, which extends to $L^2$ because the transform is unitary, gives
\[
 \int F\,\overline\phi=\int2\pi\mathrm i\xi_e\widehat{\chi Z}\,\overline{\widehat\phi}=-\int\widehat{\chi Z}\,\overline{\widehat{\partial_e\phi}}=-\int\chi Z\,\overline{\partial_e\phi}.
\]
Thus $\partial_e(\chi Z)=F\in L^2$. Since $\chi$ may equal $1$ near any given compact subset of $D'$, it follows that $\partial_eZ\in L^2_{\loc}(D')$.

Now let $D''\Subset D'''\Subset D'$ be open. By (1), applied with $D'''$ in place of $D$ and $p=2$, we have $\delta_hZ\to\partial_eZ$ in $L^2(D'')$, so along a sequence $h_k\to0$ the functions $\delta_{h_k}Z$ converge to $\partial_eZ$ almost everywhere on $D''$. Fatou's lemma gives $\|\partial_eZ\|_{L^p(D'')}\leq\varliminf_k\|\delta_{h_k}Z\|_{L^p(D'')}\leq M$. Exhausting $D'$ by such sets $D''$ and using monotone convergence, we obtain $\|\partial_eZ\|_{L^p(D')}\leq M$.
\end{proof}

\begin{proposition}\label{prop:compact-metric-laplacian-lp}
Let $1<p<\infty$, and let $f\in L^p(X)$ be such that its distributional metric Laplacian $Lf$ belongs to $L^p(X)$. Then $f\in W^{2,p}(X)$, and
\begin{equation}\label{eq:compact-metric-laplacian-lp}
 \|f\|_{W^{2,p}(X)}\leq C_{X,\omega,p}
       \bigl(\|Lf\|_{L^p(X)}+\|f\|_{L^p(X)}\bigr),
\end{equation}
where the constant $C_{X,\omega,p}$ depends only on $X$, $\omega$ and $p$ (and on the fixed charts defining the norm). In particular, a continuous function on $X$ with bounded metric Laplacian belongs to $W^{2,p}(X)$ for every finite $p\geq1$ and to $C^{1,\alpha}(X)$ for every $0<\alpha<1$.
\end{proposition}
\begin{proof}
\emph{Step 1: the estimate for smooth functions.} Let $d=2n$. Since $L$ is a real operator, it suffices to treat real functions. Fix finitely many holomorphic charts and compact subsets $K_\iota$ of their domains whose interiors cover $X$. By \eqref{eq:complex-hessian-real-form}, in these charts $L=-4\pi\,A:D^2$ on real functions, with smooth real symmetric matrices $A$; by compactness there is $\lambda>0$ with $A\geq\lambda I$ on every $K_\iota$. Let $\delta_0$ and $C$ be the constants of \cref{lem:frozen-coefficient-lp} for $d$, $p$ and $\lambda$. Since $A$ is uniformly continuous on a compact neighborhood of each $K_\iota$, there is $r>0$ such that, for every $x\in K_\iota$, the ball $B(x,r)$ lies in that neighborhood and $|A(y)-A(x)|\leq\delta_0$ for $y\in B(x,r)$. Cover $X$ by finitely many such balls $B_\nu$, and take a smooth partition of unity $(\chi_\nu)$ with $\Supp\chi_\nu$ a compact subset of $B_\nu$ (\cref{prop:manifold-cutoffs-stokes}).

Let $f$ be smooth and real, and let $v_\nu=\chi_\nu f$, written in the chart of $B_\nu$ and extended by zero. By \cref{lem:frozen-coefficient-lp}, $\|D^2v_\nu\|_p\leq C\|A:D^2v_\nu\|_p=(C/4\pi)\|Lv_\nu\|_p$. By \eqref{eq:metric-laplacian-commutator}, $Lv_\nu=\chi_\nu Lf+[L,\chi_\nu]f$, where the commutator involves only $f$ and its first derivatives. Summation therefore gives
\[
 S_2:=\sum_\nu\|D^2v_\nu\|_p
 \leq C\bigl(\|Lf\|_p+\|\nabla f\|_p+\|f\|_p\bigr).
\]
Here and below the $L^p$ norms of $f$ and $\nabla f$ on $X$ may be computed in the charts or with the metric of $\omega$; on the fixed finite cover these choices give equivalent norms. Since $f=\sum_\nu v_\nu$, the norm $\|\nabla f\|_p$ is at most a fixed multiple of $\sum_\nu\|\nabla v_\nu\|_p$. By \cref{lem:gradient-lp-interpolation}, for every $\varepsilon>0$,
\[
 \sum_\nu\|\nabla v_\nu\|_p\leq\varepsilon S_2+C_\varepsilon\|f\|_p.
\]
Combining the last three estimates and choosing $\varepsilon$ small, we can absorb $S_2$, which is finite because $f$ is smooth. This gives $S_2\leq C(\|Lf\|_p+\|f\|_p)$ and then $\|\nabla f\|_p\leq C(\|Lf\|_p+\|f\|_p)$. Since the norm of $W^{2,p}(X)$ may be computed, up to equivalence, with the charts of the balls $B_\nu$ and the partition of unity $(\chi_\nu)$ (\cref{sec:hessian-singular-integrals}), this proves \eqref{eq:compact-metric-laplacian-lp} for smooth $f$.

\emph{Step 2: an $L^p$ bound for the inverse.} Fix $c>0$ and put $P=L+c$. For a smooth real function $\psi$, \cref{lem:compact-elliptic-foundations}(3) gives a unique distributional solution $u=P^{-1}\psi$, which lies in every $H^r$ and is therefore smooth, by \cref{lem:compact-sobolev-spaces}(3); it is real since $P$ is a real operator. For $\eta>0$ put
\[
 \phi_\eta(s)=s(s^2+\eta)^{(p-2)/2},
 \qquad
 \phi_\eta'(s)=
 \bigl((p-1)s^2+\eta\bigr)(s^2+\eta)^{(p-4)/2}\geq0,
 \qquad s\in\mathbb R.
\]
The function $\phi_\eta$ is smooth, so we may multiply the equation $Lu+cu=\psi$ by $\phi_\eta(u)$ and integrate. Since $\mathrm d^*$ is the formal adjoint of $\mathrm d$, we have $\int_X(Lu)\phi_\eta(u)\,\mathrm dV=\int_X\phi_\eta'(u)|\mathrm du|^2\,\mathrm dV\geq0$, and therefore
\[
 c\int_Xu\phi_\eta(u)\,\mathrm dV\leq\int_X\psi\phi_\eta(u)\,\mathrm dV.
\]
As $\eta\downarrow0$, $u\phi_\eta(u)\to|u|^p$ and $\phi_\eta(u)\to u|u|^{p-2}$ pointwise, the latter being $0$ where $u=0$. For $p\geq2$ and $\eta\leq1$ both functions are bounded by $(1+u^2)^{p/2}$, which is bounded since $u$ is continuous on the compact manifold $X$; for $1<p<2$ we have $(u^2+\eta)^{(p-2)/2}\leq|u|^{p-2}$ where $u\neq0$, so they are bounded by $|u|^p$ and $|u|^{p-1}$. Dominated convergence and H\"older's inequality therefore give $c\|u\|_p^p\leq\|\psi\|_p\|u\|_p^{p-1}$, that is (if $u\neq0$; otherwise there is nothing to prove),
\[
 \|P^{-1}\psi\|_p\leq c^{-1}\|\psi\|_p.
\]
Applying Step 1 to $u$, for which $Lu=\psi-cu$, we conclude that $\|P^{-1}\psi\|_{W^{2,p}}\leq C\|\psi\|_p$.

\emph{Step 3: the general case.} Let $f$ be real, put $\psi=Lf+cf\in L^p(X)$, and choose smooth real functions $\psi_j\to\psi$ in $L^p(X)$, using chart cutoffs and mollification (\cref{lem:mollifier-density}). By Step 2 the solutions $u_j=P^{-1}\psi_j$ form a Cauchy sequence in $W^{2,p}(X)$. This space is complete: in each chart the derivatives of order at most two converge in $L^p$, and testing against smooth functions identifies the limits with the weak derivatives of the limit function. Let $u$ be the limit. Then $Pu_j\to Pu$ in the sense of distributions, so $Pu=\psi$. Thus $P(u-f)=0$, and since $P$ is injective on distributions (\cref{lem:compact-elliptic-foundations}(3)), $u=f$. Passing to the limit in the estimate of Step 2 gives $\|f\|_{W^{2,p}}\leq C\|\psi\|_p\leq C(\|Lf\|_p+\|f\|_p)$. For complex $f$ we apply this to the real and imaginary parts, since $L$ is a real operator.

\emph{Step 4: continuous functions with bounded Laplacian.} If $f$ is continuous and $Lf$ is bounded, then $f$ and $Lf$ lie in every $L^p(X)$ with $p<\infty$, which proves the assertion for $1<p<\infty$; the case $p=1$ follows from the case $p=2$, since $X$ has finite volume. Given $0<\alpha<1$, choose $p>d/(1-\alpha)$. In every chart, $f$ belongs to $W^{2,p}_{\loc}$ by \eqref{eq:sobolev-chart-restriction}, so Morrey's inequality (\cref{lem:morrey-sobolev-representative}) gives a representative of $f$ in $C^{1,1-d/p}_{\loc}$, hence in $C^{1,\alpha}_{\loc}$. This representative agrees with the continuous function $f$ almost everywhere, hence everywhere.
\end{proof}
\index{Laplacian!$W^{2,p}$ regularity}

Consequently the Hessian forms of obstacles with bounded metric Laplacian can be multiplied on all of $X$, the products being taken pointwise almost everywhere.

\begin{corollary}\label{cor:metric-laplacian-barrier-determinants}
Let $\theta_1,\ldots,\theta_n$ be closed smooth real $(1,1)$-forms on $X$, and let $f_1,\ldots,f_n$ be continuous real functions on $X$ with bounded metric Laplacian. Then each $\theta_j+\ddc f_j$ is a real $(1,1)$-form whose coefficients, computed from the weak second derivatives of $f_j$, lie in $L^p$ for every $p<\infty$. Their pointwise wedge product is defined almost everywhere and satisfies
\[
 \bigwedge_{j=1}^n(\theta_j+\ddc f_j)
       =F\,\mathrm dV,\qquad F\in L^q(X)\quad\text{for every }1\leq q<\infty;
\]
in particular it is a finite signed Radon measure. On a coordinate ball, with smooth real $\rho_j$ such that $\ddc\rho_j=\theta_j$, it equals $n!\pi^{-n}D(B_1,\ldots,B_n)\,\mathrm dV_{2n}$, where $B_j=((\rho_j+f_j)_{a\bar b})_{a,b}$. If moreover $u_j$ are continuous $\theta_j$-psh functions on $X$ with bounded metric Laplacian, then on $E=\bigcap_j\{u_j=f_j\}$ this signed measure agrees with the positive measure $\mathbf1_E\bigwedge_j(\theta_j+\ddc u_j)$ of \cref{prop:metric-laplacian-regular-contact-density}.
\end{corollary}
\begin{proof}
By \cref{prop:compact-metric-laplacian-lp} and \eqref{eq:sobolev-chart-restriction}, every $f_j$ belongs to $W^{2,p}_{\loc}$ in every chart, for every finite $p$. In a chart, the current $\theta_j+\ddc f_j$ is therefore given by integration against the form $\theta_j+\frac{\mathrm i}{2\pi}\sum_{a,b}(f_j)_{a\bar b}\,\mathrm dz_a\wedge\mathrm d\bar z_b$, where the $(f_j)_{a\bar b}$ are the weak derivatives, as in \eqref{eq:complex-hessian-normalization}; its coefficients lie in $L^p_{\loc}$. On the overlap of two charts, the forms obtained in the two charts represent the same current. For a current with locally integrable coefficients, \eqref{eq:current-coefficient-change}, in which $\delta$ is the Jacobian determinant relating the two Lebesgue measures, says that the coefficients in the two charts are related, as distributions and hence almost everywhere (\cref{prop:distribution-smoothing}(4)), by the pointwise change of variables of forms. Thus the two forms agree almost everywhere as pointwise forms. Hence the pointwise wedge product of the forms $\theta_j+\ddc f_j$ is a well-defined $(n,n)$-form, almost everywhere on $X$, and we write it as $F\,\mathrm dV$. A coefficient of the wedge product in a chart is a sum of products of $n$ coefficients of the factors and of smooth functions; by H\"older's inequality, with each factor in $L^{nq}$ on compact subsets of the chart, these products lie in $L^q$ there, for every $1\leq q<\infty$. Covering $X$ by finitely many compact subsets of charts, we obtain $F\in L^q(X)$. A real function in $L^1(X)$ defines a finite signed Radon measure through its positive and negative parts.

On a coordinate ball, $\theta_j+\ddc f_j=\ddc(\rho_j+f_j)=\frac{\mathrm i}{2\pi}\sum_{a,b}(B_j)_{ab}\,\mathrm dz_a\wedge\mathrm d\bar z_b$ almost everywhere, and the identity $\omega_{A_1}\wedge\cdots\wedge\omega_{A_n}=D(A_1,\ldots,A_n)\,\omega_I^n$ of \cref{lem:mixed-discriminant-basic}(2), applied pointwise with $A_j=B_j/(2\pi)$, together with $\omega_I^n=n!\,2^n\,\mathrm dV_{2n}$, gives $\bigwedge_j(\theta_j+\ddc f_j)=n!\pi^{-n}D(B_1,\ldots,B_n)\,\mathrm dV_{2n}$. Finally, on every coordinate ball $W$ the second identity of \cref{prop:metric-laplacian-regular-contact-density}, applied with $U=X$, shows that $\mathbf1_E\bigwedge_j(\theta_j+\ddc u_j)=\mathbf1_EF\,\mathrm dV$ on $W$. Since coordinate balls cover $X$, the two measures agree on $X$.
\end{proof}
