The contact formula involves the weak complex Hessians of an obstacle only on the contact set. Off this set the coefficients of a general obstacle with bounded metric Laplacian may be unbounded, and to give a meaning to the wedge products of the forms $\theta_j+\ddc f_j$ on all of $X$ we need more integrability. We now prove that all weak second derivatives of such an obstacle lie in $L^p$ for every $p<\infty$. This is the $L^p$ theory of the elliptic operator $L$ on the compact manifold $X$; we derive it from the Euclidean estimate \eqref{eq:euclidean-hessian-lp} by freezing coefficients.

\begin{proposition}\label{prop:compact-metric-laplacian-lp}
Let $1<p<\infty$, and let $f\in L^p(X)$ be such that its distributional metric Laplacian $Lf$ belongs to $L^p(X)$. Then $f\in W^{2,p}(X)$, and
\begin{equation}\label{eq:compact-metric-laplacian-lp}
 \|f\|_{W^{2,p}(X)}\leq C_{X,\omega,p}
       \bigl(\|Lf\|_{L^p(X)}+\|f\|_{L^p(X)}\bigr),
\end{equation}
where the constant $C_{X,\omega,p}$ depends only on $X$, $\omega$ and $p$ (and on the fixed charts defining the norm). In particular, a continuous function on $X$ with bounded metric Laplacian belongs to $W^{2,p}(X)$ for every finite $p\geq1$, and it belongs to $C^{1,\alpha}(X)$ for every $0<\alpha<1$, that is, it is of class $C^{1,\alpha}$ on compact subsets of every chart.
\end{proposition}
\begin{proof}
\emph{Step 1: the estimate for smooth functions.} Let $d=2n$. In real local coordinates write
\[
 L=-\sum_{i,j}a^{ij}(x)\partial_i\partial_j
               +\sum_i b^i(x)\partial_i,
\]
with smooth coefficients and a symmetric positive definite matrix $a=(a^{ij})$. On the compact supports of the cutoff functions of the fixed atlas, the eigenvalues of $a$ lie between two fixed positive constants. Fix a point $x_0$ and consider the operator $L_0=-\sum_{i,j}a^{ij}(x_0)\partial_i\partial_j$ with constant coefficients. A linear change of coordinates transforms $L_0$ into $-\Delta$, and \eqref{eq:euclidean-hessian-lp} then gives
\[
 \|D^2v\|_p\leq C\|L_0v\|_p,\qquad v\in C^\infty_c(\mathbb R^d),
\]
where $D^2v$ is the Hessian matrix; the constant $C$ is uniform as long as the eigenvalues of $a(x_0)$ stay between the fixed bounds. Let $\delta>0$. On a sufficiently small ball around $x_0$ we have $|a(x)-a(x_0)|\leq\delta$, and the radius can be chosen independently of $x_0$ by uniform continuity of $a$. Writing $L_0v=Lv+\sum_{i,j}(a^{ij}(x)-a^{ij}(x_0))\partial_i\partial_jv-\sum_ib^i\partial_iv$, we obtain
\[
 \|D^2v\|_p\leq C\bigl(\|Lv\|_p+
           \delta\|D^2v\|_p+\|\nabla v\|_p\bigr)
\]
for smooth $v$ supported in such a ball, with $C$ independent of $\delta$. Choosing $\delta$ so small that $C\delta\leq1/2$, we can absorb the Hessian term and get $\|D^2v\|_p\leq C(\|Lv\|_p+\|\nabla v\|_p)$.

Now let $f$ be smooth on $X$. Take a finite smooth partition of unity $(\chi_\nu)$ subordinate to a cover of $X$ by such balls, and apply the last estimate to $v_\nu=\chi_\nu f$, extended by zero in its chart. Since $Lv_\nu=\chi_\nu Lf+[L,\chi_\nu]f$, where the commutator $[L,\chi_\nu]$ is a first-order operator, both $Lv_\nu-\chi_\nu Lf$ and $\nabla v_\nu$ involve only $f$ and its first derivatives. Summation therefore gives
\[
 S_2:=\sum_\nu\|D^2v_\nu\|_p
 \leq C\bigl(\|Lf\|_p+\|\nabla f\|_p+\|f\|_p\bigr).
\]
Here and below the $L^p$ norms of $f$ and $\nabla f$ on $X$ may be computed in the charts or with the metric of $\omega$; these choices give equivalent norms on the fixed finite cover. Since $f=\sum_\nu v_\nu$, the norm $\|\nabla f\|_p$ is at most a fixed multiple of $\sum_\nu\|\nabla v_\nu\|_p$. By \cref{lem:gradient-lp-interpolation}, for every $\varepsilon>0$,
\[
 \sum_\nu\|\nabla v_\nu\|_p\leq\varepsilon S_2+C_\varepsilon\|f\|_p.
\]
Combining the last three estimates and choosing $\varepsilon$ small, we can absorb $\|\nabla f\|_p$, which is finite because $f$ is smooth. This gives $\|\nabla f\|_p\leq C(\|Lf\|_p+\|f\|_p)$, then $S_2\leq C(\|Lf\|_p+\|f\|_p)$, and therefore \eqref{eq:compact-metric-laplacian-lp} for smooth $f$.

\emph{Step 2: an $L^p$ bound for the inverse.} Let $P=L+c$, with the fixed $c>0$ used in the proof of \cref{lem:compact-elliptic-foundations}. For a smooth real function $\psi$, that proof gives a smooth solution $u=P^{-1}\psi$, which is real since $P$ is a real operator and the solution is unique. For $\eta>0$ put
\[
 \phi_\eta(s)=s(s^2+\eta)^{(p-2)/2},
 \qquad
 \phi_\eta'(s)=
 \bigl((p-1)s^2+\eta\bigr)(s^2+\eta)^{(p-4)/2}\geq0,
 \qquad s\in\mathbb R.
\]
The function $\phi_\eta$ is smooth, so we may multiply the equation $Lu+cu=\psi$ by $\phi_\eta(u)$ and integrate. Since $L=\mathrm d^*\mathrm d$, Stokes' formula gives $\int_X(Lu)\phi_\eta(u)\,\mathrm dV=\int_X\phi_\eta'(u)|\mathrm du|^2\,\mathrm dV\geq0$, and therefore
\[
 c\int_Xu\phi_\eta(u)\,\mathrm dV\leq\int_X\psi\phi_\eta(u)\,\mathrm dV.
\]
As $\eta\downarrow0$, $u\phi_\eta(u)\to|u|^p$ and $\phi_\eta(u)\to u|u|^{p-2}$ pointwise, the latter being $0$ where $u=0$. For $p\geq2$ and $\eta\leq1$ both functions are bounded by $(1+u^2)^{p/2}$, which is bounded since $u$ is continuous on the compact manifold $X$; for $1<p<2$ we have $(u^2+\eta)^{(p-2)/2}\leq|u|^{p-2}$, so they are bounded by $|u|^p$ and $|u|^{p-1}$. Dominated convergence and H\"older's inequality therefore give $c\|u\|_p^p\leq\|\psi\|_p\|u\|_p^{p-1}$, that is (the case $u=0$ being trivial),
\[
 \|P^{-1}\psi\|_p\leq c^{-1}\|\psi\|_p.
\]
Applying Step 1 to $u$, for which $Lu=\psi-cu$, we conclude that $\|P^{-1}\psi\|_{W^{2,p}}\leq C\|\psi\|_p$.

\emph{Step 3: the general case.} Let $f$ be real, put $\psi=Lf+cf\in L^p(X)$, and choose smooth real functions $\psi_j\to\psi$ in $L^p(X)$, using chart cutoffs and mollification (\cref{lem:mollifier-density}). By Step 2 the solutions $u_j=P^{-1}\psi_j$ form a Cauchy sequence in $W^{2,p}(X)$. This space is complete: in each chart the derivatives of order at most two converge in $L^p$, and testing against smooth functions identifies the limits with the weak derivatives of the limit function. Let $u$ be the limit. Then $Pu_j\to Pu$ in the sense of distributions, so $Pu=\psi$. Thus $P(u-f)=0$, and since $P^{-1}P=1$ on distributions (\cref{lem:compact-elliptic-foundations}), $u=f$. Passing to the limit in the estimate of Step 2 gives $\|f\|_{W^{2,p}}\leq C\|\psi\|_p\leq C(\|Lf\|_p+\|f\|_p)$. For complex $f$ we apply this to the real and imaginary parts, since $L$ is a real operator.

\emph{Step 4: continuous functions with bounded Laplacian.} If $f$ is continuous and $Lf$ is bounded, then $f$ and $Lf$ lie in every $L^p(X)$ with $p<\infty$, and the assertion for $1<p<\infty$ is proved. For $p=1$ it follows from the case $p=2$, since $X$ has finite volume. Given $0<\alpha<1$, choose $p>d/(1-\alpha)$. In each chart, Morrey's inequality (\cref{lem:morrey-sobolev-representative}) gives a representative of $f$ of class $C^{1,1-d/p}$, hence of class $C^{1,\alpha}$ on compact subsets. This representative agrees with the continuous function $f$ almost everywhere, hence everywhere.
\end{proof}
\index{Laplacian!$W^{2,p}$ regularity}

Consequently the Hessian forms of obstacles with bounded metric Laplacian can be multiplied on all of $X$, the products being taken pointwise almost everywhere.

\begin{corollary}\label{cor:metric-laplacian-barrier-determinants}
Let $\theta_1,\ldots,\theta_n$ be closed smooth real $(1,1)$-forms on $X$, and let $f_1,\ldots,f_n$ be continuous real functions on $X$ with bounded metric Laplacian. Then each $\theta_j+\ddc f_j$ is a real $(1,1)$-form whose coefficients, computed from the weak second derivatives of $f_j$, lie in $L^p$ for every $p<\infty$. Their pointwise wedge product is defined almost everywhere and satisfies
\[
 \bigwedge_{j=1}^n(\theta_j+\ddc f_j)
       =F\,\mathrm dV,\qquad F\in L^q(X)\quad\text{for every }1\leq q<\infty;
\]
in particular it is a finite signed Radon measure. On a coordinate ball, with smooth real $\rho_j$ such that $\ddc\rho_j=\theta_j$, it equals $n!\pi^{-n}D(B_1,\ldots,B_n)\,\mathrm dV_{2n}$, where $B_j=(\partial_a\bar\partial_b(\rho_j+f_j))_{a,b}$. If moreover $u_j$ are continuous $\theta_j$-psh functions on $X$ with bounded metric Laplacian, then on $E=\bigcap_j\{u_j=f_j\}$ this signed measure agrees with the positive measure $\mathbf1_E\bigwedge_j(\theta_j+\ddc u_j)$ of \cref{prop:metric-laplacian-regular-contact-density}.
\end{corollary}
\begin{proof}
By \cref{prop:compact-metric-laplacian-lp}, all weak second derivatives of every $f_j$ lie in $L^p$ for every finite $p$. A coefficient of the wedge product is a sum of products of $n$ such coefficients and of smooth functions, and by H\"older's inequality, with each factor in $L^{nq}$, these products lie in $L^q$ for every $1\leq q<\infty$. Weak derivatives obey the chain rule under smooth changes of coordinates, so the weak forms transform like smooth forms and these polynomial expressions agree almost everywhere on overlapping charts; this defines $F$. In a coordinate ball, the identity $\bigwedge_j\ddc(\rho_j+f_j)=n!\pi^{-n}D(B_1,\ldots,B_n)\,\mathrm dV_{2n}$ is the algebraic identity for smooth functions proved at the beginning of the proof of \cref{prop:bounded-complex-hessian-density}, applied pointwise to the weak Hessian matrices. A real function in $L^1(X)$ defines a finite signed Radon measure through its positive and negative parts. Finally, on $E$ the two measures agree by the second identity of \cref{prop:metric-laplacian-regular-contact-density}.
\end{proof}
