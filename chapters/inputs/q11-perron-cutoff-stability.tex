We apply this to the cutoffs of a singular potential $u\leq\phi$. The truncations $u_T=\max\{u,\phi-T\}$ have bounded difference from $\phi$, so for each $T>0$ the Perron segment from $\phi$ to $u_T$ is a weak geodesic, which we parametrize by $[0,T]$. As $T$ grows these segments increase, and at a fixed time $s$ their regularized limit $R^u_s$ is a $\theta$-psh potential between $\phi-s$ and $\phi$. By \cref{prop:perron-geodesic-order-and-limits}, a bounded change of $u$ alters the segment of length $T$ at time $s$ by at most a constant multiple of $s/T$. Consequently the limits $R^u_s$ depend only on the singularity type of $u$.

\Needspace{18\baselineskip}
\begin{proposition}\label{prop:long-perron-cutoff-type-stability}
Let $u\in\PSH(X,\theta)$ with $u\leq\phi$. For $T>0$ and $0\leq s\leq T$ put
\[
 u_T=\max\{u,\phi-T\},\qquad
 L_s^{u,T}=G_{s/T}(\phi,u_T).
\]
For fixed $s\geq0$, the potentials $L_s^{u,T}$ increase with $T\geq s$ and satisfy $\phi-s\leq L_s^{u,T}\leq\phi$. Their regularized limit
\[
 R_s^u=\left(\sup_{T>s}L_s^{u,T}\right)^*
\]
is $\theta$-psh. If $v\in\PSH(X,\theta)$ satisfies $v\leq\phi$, and $u\leq v+C$ for a constant $C\geq0$, then
\[
 L_s^{u,T}\leq L_s^{v,T}+\frac{Cs}{T},\qquad R_s^u\leq R_s^v.
\]
In particular, if $u\sim v$, then $R_s^u=R_s^v$ for every $s\geq0$. Consequently, for an arbitrary $u\in\PSH(X,\theta)$ with $u\preceq\phi$, the potentials $R_s^{u-c}$ do not depend on the constant $c$ with $u-c\leq\phi$. Denoting them by $R_s^u$, we have $R_s^u\leq R_s^v$ whenever $u\preceq v\preceq\phi$.
\end{proposition}
\begin{proof}
Since $\phi-T\leq u_T\leq\phi$, the potential $u_T$ has bounded difference from $\phi$; in particular $u_T\in\mathcal E(X,\theta,\phi)$. By \cref{thm:relative-full-mass-perron-geodesic}, the Perron segment from $\phi$ to $u_T$ is a weak geodesic. We regard it as a curve on $[0,T]$, so that its slice at time $s$ is $L_s^{u,T}$.

\emph{Bounds.} For $0\leq s\leq T$ put $q_s=\max\{u,\phi-s\}$. After the affine change of parameter, the complexification of the curve $(q_s)_{0\leq s\leq T}$ is $(x,\zeta)\mapsto\max\{u(x),\phi(x)-T\Real\zeta\}$, the maximum of two $p^*\theta$-psh functions, because $\Real\zeta$ is pluriharmonic. Moreover $q_0=\phi$, since $u\leq\phi$, and $q_T=u_T$. As $|q_s-q_t|\leq|s-t|$ on the finite locus of $\phi$, the curve tends to $\phi$ and $u_T$ in $L^1$ at the two ends. Thus $(q_s)$ is a subgeodesic with endpoints $\phi$ and $u_T$, and by the definition of the Perron segment, $L_s^{u,T}\geq q_s\geq\phi-s$. On the other hand, \cref{thm:relative-full-mass-perron-geodesic} gives $L_s^{u,T}\leq(1-s/T)\phi+(s/T)u_T\leq\phi$.

\emph{Monotonicity in $T$.} Let $T'>T$. The same subgeodesic, now on $[0,T']$, shows that $L_T^{u,T'}\geq q_T=u_T$. By the last assertion of \cref{thm:relative-full-mass-perron-geodesic}, the restriction of the segment of length $T'$ to $[0,T]$ is the weak geodesic from $\phi$ to $L_T^{u,T'}$. Since weak geodesics increase with their endpoints (\cref{prop:perron-geodesic-order-and-limits}), this restriction dominates the weak geodesic from $\phi$ to $u_T$. Hence $L_s^{u,T'}\geq L_s^{u,T}$ for $0\leq s\leq T$.

\emph{The limit.} Fix $s$. The family $(L_s^{u,T})_{T>s}$ is increasing, bounded above by $\phi$ and bounded below by $\phi-s$. By \cref{thm:psh-upper-envelopes}, applied after adding local potentials of $\theta$, its regularized supremum $R_s^u$ is $\theta$-psh, and $\phi-s\leq R_s^u\leq\phi$. Since the family is increasing, the supremum may be taken over $T\geq A$ for any $A>s$ without changing $R_s^u$. For $s=0$, all the potentials $L_0^{u,T}$ and $R_0^u$ equal $\phi$.

\emph{Comparison.} Suppose that $v\leq\phi$ and $u\leq v+C$ with $C\geq0$. Then
\[
 u_T\leq\max\{v+C,\phi-T\}\leq v_T+C.
\]
By the monotonicity of weak geodesics in their endpoints and their behaviour under affine endpoint shifts (\cref{prop:perron-geodesic-order-and-limits}), $G_t(\phi,u_T)\leq G_t(\phi,v_T+C)=G_t(\phi,v_T)+tC$. Taking $t=s/T$ gives the bound $L_s^{u,T}\leq L_s^{v,T}+Cs/T$. Now fix $s$ and $A>s$. For $T\geq A$ we get $L_s^{u,T}\leq R_s^v+Cs/A$, and taking the regularized supremum over $T\geq A$ gives
\[
 R_s^u\leq R_s^v+\frac{Cs}{A}.
\]
Letting $A\to\infty$ yields $R_s^u\leq R_s^v$.

If $u\sim v$, then $u\leq v+C$ and $v\leq u+C$ for some $C\geq0$, and the comparison applied in both directions gives $R_s^u=R_s^v$. If $u\preceq\phi$ is arbitrary, any two constants $c,c'$ with $u-c\leq\phi$ and $u-c'\leq\phi$ give potentials $u-c\sim u-c'$, so $R_s^{u-c}=R_s^{u-c'}$. Finally, if $u\preceq v\preceq\phi$, choose constants $c,c'$ with $u-c\leq\phi$ and $v-c'\leq\phi$. Then $u-c\leq v-c'+C$ for some $C\geq0$, and the comparison gives $R_s^u=R_s^{u-c}\leq R_s^{v-c'}=R_s^v$.
\end{proof}
