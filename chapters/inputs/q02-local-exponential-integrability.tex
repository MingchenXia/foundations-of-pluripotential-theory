The same Riesz decomposition also bounds an exponential of a negative subharmonic function. For a psh function, restrictions to complex lines through one finite point then give an estimate on a ball. The exponent can be chosen uniformly when the local $L^1$ norm is bounded.

\begin{lemma}\label{lem:negative-subharmonic-disc-exponential}
There are constants $\beta,C>0$ such that every subharmonic $v\leq0$ on the unit disc $D\subset\mathbb C$, with $v(0)\geq-1$, satisfies
\[
 \int_{D(0,1/2)}\mathrm e^{-\beta v}\,\mathrm dV\leq C.
\]
\end{lemma}
\begin{proof}
The cutoff Riesz decomposition in the proof of \cref{lem:negative-subharmonic-disc-lp} gives
\[
 v=P+h,\qquad
 P(z)=\frac1{2\pi}\int\log|z-\zeta|\,\mathrm d\nu(\zeta)
\]
on $D(0,4/5)$, where $\nu$ is a positive measure supported in $D(0,9/10)$ and $h$ is harmonic. That proof bounds both $M=\nu(D)$ and $\sup_{D(0,1/2)}|h|$ by a universal $C_0\geq1$, since $|v(0)|\leq1$. Choose $0<\beta\leq1$ with $\beta C_0/(2\pi)\leq1$.

If $M>0$, put $b=\beta M/(2\pi)\in(0,1]$. Jensen's inequality for the probability measure $\nu/M$ gives
\[
 \mathrm e^{-\beta P(z)}
       \leq\frac1M\int|z-\zeta|^{-b}\,\mathrm d\nu(\zeta).
\]
To include poles, first replace $\log|z-\zeta|$ by its maximum with $-N$, apply Jensen, and let $N\to\infty$ by monotone convergence. For every $\zeta$ in the support of $\nu$, translation and $r^{-b}\leq1+r^{-1}$ give
\[
 \int_{D(0,1/2)}|z-\zeta|^{-b}\,\mathrm dV(z)
   \leq\int_{D(0,2)}(1+|z|^{-1})\,\mathrm dV(z)=8\pi.
\]
Tonelli therefore bounds the integral of $\mathrm e^{-\beta P}$ by $8\pi$. Multiplication by $\mathrm e^{-\beta h}\leq\mathrm e^{\beta C_0}$ proves the assertion. If $M=0$, then $P=0$ and the bound for $h$ gives the assertion directly.
\end{proof}

\begin{proposition}\label{prop:negative-psh-ball-exponential}
For $n\geq1$ there are $\alpha_n,C_n>0$ such that every psh $v\leq0$ on the unit ball $B\subset\mathbb C^n$, with $\|v\|_{L^1(B)}\leq1$, satisfies
\begin{equation}\label{eq:negative-psh-exponential-lone-bound}
 \int_{\frac14B}\mathrm e^{-\alpha_n v}\,\mathrm dV\leq C_n.
\end{equation}
More generally, if $\|v\|_{L^1(B)}\leq M$ with $M\geq1$, the exponent $\alpha_n/M$ gives the same bound.
\end{proposition}
\begin{proof}
Choose a finite point $a\in\frac18B$ with $|v(a)|\leq A_n$, where $A_n=\max\{1,2/|\frac18B|\}$. Integration supplies such a point. For each unit vector $e\in\mathbb C^n$, the function
\[
 \zeta\longmapsto A_n^{-1}v(a+\tfrac34\zeta e)
\]
is negative subharmonic on $D$ and has value at least $-1$ at zero. Its finite center rules out an identically $-\infty$ restriction. Set $\alpha_n=\beta/A_n$. The preceding lemma, with the planar change of variables, gives a uniform bound for
\[
 \int_{|\zeta|<3/8}\mathrm e^{-\alpha_n v(a+\zeta e)}\,\mathrm dV(\zeta).
\]

We pass from these line integrals to volume. Let $\sigma$ be probability surface measure on the unit sphere and $s_{2n-1}$ its area. For a nonnegative Borel function $F$ and $r>0$, polar integration and invariance of $\sigma$ under multiplication by $\mathrm e^{\mathrm i t}$ give
\[
 \begin{split}
 \int_{|z-a|<r}F(z)\,\mathrm dV(z)
 &=s_{2n-1}\int\!\int_0^r F(a+te)t^{2n-1}\,\mathrm dt\,\mathrm d\sigma(e)\\
 &\leq\frac{s_{2n-1}r^{2n-2}}{2\pi}
       \int\!\int_{|\zeta|<r}F(a+\zeta e)\,\mathrm dV(\zeta)\,\mathrm d\sigma(e).
 \end{split}
\]
Indeed $t^{2n-1}\leq r^{2n-2}t$, and averaging the phase of $e$ produces the angular integral in the planar disc. Tonelli applies even when $F$ takes the value $+\infty$. Use $F=\mathrm e^{-\alpha_n v}$ and $r=3/8$. Since $\frac14B\subset B(a,3/8)$, the uniform line bound proves \eqref{eq:negative-psh-exponential-lone-bound}. Applying it to $v/M$ proves the last assertion.
\end{proof}
