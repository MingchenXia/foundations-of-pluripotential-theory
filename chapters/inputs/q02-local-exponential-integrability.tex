The same Riesz decomposition gives exponential integrability: if $v\leq0$ is plurisubharmonic on a ball, then $\mathrm e^{-\alpha v}$ is integrable on a smaller concentric ball for every sufficiently small $\alpha>0$, and the admissible $\alpha$ and the bound for the integral depend only on a bound for the $L^1$ norm of $v$. In one variable this follows from Jensen's inequality applied to the fundamental potential in the Riesz decomposition. In several variables we integrate the one-variable estimates over all complex lines through a suitable point. For $t>0$ we write $tB$ for the ball of radius $t$ centered at $0$ in $\mathbb C^n$; thus $B$ is the unit ball.

\begin{lemma}\label{lem:negative-subharmonic-disc-exponential}
There are constants $\beta,C>0$ such that every subharmonic function $v\leq0$ on $D$ with $v(0)\geq-1$ satisfies
\[
 \int_{D(0,1/2)}\mathrm e^{-\beta v}\,\mathrm dV\leq C.
\]
\end{lemma}
\begin{proof}
Since $v(0)\geq-1>-\infty$, \cref{lem:negative-subharmonic-disc-lp} gives a finite positive measure $\nu$ with support in $D(0,9/10)$ and a harmonic function $h$ on $D(0,4/5)$ such that
\[
 v=P+h,\qquad
 P(z)=\frac1{2\pi}\int\log|z-\zeta|\,\mathrm d\nu(\zeta)
\]
at every point of $D(0,4/5)$. Since $|v(0)|\leq1$, the bounds \eqref{eq:negative-subharmonic-disc-riesz-bounds} give $M=\nu(D)\leq C_0$ and $\sup_{D(0,1/2)}|h|\leq C_0$, where $C_0\geq1$ is the universal constant of that lemma. Fix $\beta\in(0,1]$ with $\beta C_0\leq2\pi$.

We show that $\int_{D(0,1/2)}\mathrm e^{-\beta P}\,\mathrm dV\leq8\pi$. If $M=0$, then $P=0$, and this integral is $\pi/4$. Assume $M>0$ and put $b=\beta M/(2\pi)\in(0,1]$. Fix $z\in D(0,1/2)$. For $N\geq1$ the Borel function $f_N(\zeta)=-b\max\{\log|z-\zeta|,-N\}$ is bounded on $D(0,9/10)$, so Jensen's inequality (\cref{lem:jensen}), applied to the convex function $\exp$ on $\mathbb R$ and the probability measure $\nu/M$, gives
\[
 \exp\Bigl(\int f_N\,\frac{\mathrm d\nu}M\Bigr)\leq\int\mathrm e^{f_N}\,\frac{\mathrm d\nu}M\leq\frac1M\int|z-\zeta|^{-b}\,\mathrm d\nu(\zeta).
\]
As $N\to\infty$, the functions $f_N$ increase to $-b\log|z-\zeta|$, which may take the value $+\infty$, and they are bounded below by the bounded function $f_1$. By monotone convergence, $\int f_N\,\mathrm d\nu/M$ increases to $\int-b\log|z-\zeta|\,\mathrm d\nu(\zeta)/M=-\beta P(z)$, and therefore
\[
 \mathrm e^{-\beta P(z)}\leq\frac1M\int|z-\zeta|^{-b}\,\mathrm d\nu(\zeta)
\]
for every $z\in D(0,1/2)$, with the convention $\mathrm e^{\infty}=\infty$. For every $\zeta\in D(0,9/10)$ the translated disc $D(0,1/2)-\zeta$ lies in $D(0,2)$, and $r^{-b}\leq1+r^{-1}$ for $r>0$ because $0<b\leq1$; hence
\[
 \int_{D(0,1/2)}|z-\zeta|^{-b}\,\mathrm dV(z)
   \leq\int_{D(0,2)}(1+|z|^{-1})\,\mathrm dV(z)=8\pi.
\]
Integrating the bound for $\mathrm e^{-\beta P(z)}$ over $z\in D(0,1/2)$ with Tonelli's theorem, we obtain $\int_{D(0,1/2)}\mathrm e^{-\beta P}\,\mathrm dV\leq8\pi$. Finally, since $\mathrm e^{-\beta v}=\mathrm e^{-\beta P}\mathrm e^{-\beta h}\leq\mathrm e^{\beta C_0}\mathrm e^{-\beta P}$ on $D(0,1/2)$, the lemma follows with $C=8\pi\mathrm e^{C_0}$.
\end{proof}

\begin{proposition}\label{prop:negative-psh-ball-exponential}
Let $n\geq1$, and let $B$ be the unit ball of $\mathbb C^n$. There are constants $\alpha_n,C_n>0$ such that every plurisubharmonic function $v\leq0$ on $B$ with $\|v\|_{L^1(B)}\leq1$ satisfies
\begin{equation}\label{eq:negative-psh-exponential-lone-bound}
 \int_{\frac14B}\mathrm e^{-\alpha_n v}\,\mathrm dV\leq C_n.
\end{equation}
More generally, if $\|v\|_{L^1(B)}\leq M$ for some $M>0$, the same bound holds with the exponent $\alpha_n/M$ in place of $\alpha_n$.
\end{proposition}
\begin{proof}
Put $A_n=\max\{1,2/|\frac18B|\}$. The mean of $|v|$ over $\frac18B$ is at most $1/|\frac18B|<A_n$. Hence the set of points of $\frac18B$ with $|v|\leq A_n$ has positive measure, since otherwise $|v|>A_n$ almost everywhere on $\frac18B$ and the mean would be at least $A_n$; so there is a point $a\in\frac18B$ with $|v(a)|\leq A_n$. For each unit vector $e\in\mathbb C^n$, the disc $\{a+\tfrac34\zeta e:|\zeta|<1\}$ lies in $B$, so the function
\[
 \zeta\longmapsto A_n^{-1}v(a+\tfrac34\zeta e)
\]
is subharmonic and nonpositive on $D$, with value at least $-1$ at $0$. Set $\alpha_n=\beta/A_n$, with $\beta$ as in \cref{lem:negative-subharmonic-disc-exponential}. That lemma and the change of variables $\zeta\mapsto\frac34\zeta$ give a bound, independent of $e$ and $v$, for
\[
 \int_{|\zeta|<3/8}\mathrm e^{-\alpha_n v(a+\zeta e)}\,\mathrm dV(\zeta).
\]

It remains to pass from these integrals over discs in complex lines to an integral over a ball. Let $\tau$ and $s_{2n-1}$ be as in \eqref{eq:complex-line-polar-coordinates}. By that identity, applied with center $a$, and since $|\zeta|^{2n-2}\leq r^{2n-2}$ for $|\zeta|<r$, every Borel function $F\geq0$, possibly taking the value $+\infty$, satisfies
\[
 \int_{|z-a|<r}F(z)\,\mathrm dV(z)
 \leq\frac{s_{2n-1}r^{2n-2}}{2\pi}
       \int_{S^{2n-1}}\int_{|\zeta|<r}F(a+\zeta e)\,\mathrm dV(\zeta)\,\mathrm d\tau(e)
\]
for every $r>0$. We apply this with $F=\mathrm e^{-\alpha_n v}$ and $r=3/8$. Since $\frac14B\subset B(a,3/8)$, the uniform bound for the line integrals proves \eqref{eq:negative-psh-exponential-lone-bound}. Finally, if $\|v\|_{L^1(B)}\leq M$ with $M>0$, then $v/M$ is a nonpositive plurisubharmonic function with $\|v/M\|_{L^1(B)}\leq1$, and \eqref{eq:negative-psh-exponential-lone-bound} for $v/M$ is the last assertion.
\end{proof}

Consequently, every plurisubharmonic function $u$ on a complex manifold $X$ is locally exponentially integrable: every point $x\in X$ has a neighborhood $W$ for which there is $\alpha_0>0$ such that $\mathrm e^{-\alpha u}$ is integrable on $W$ for $0<\alpha\leq\alpha_0$, the integrals being computed in local coordinates. In dimension zero we may take $W=\{x\}$. In dimension $n\geq1$, choose a chart in which $x=0$ and the closed ball $\overline{B(0,R)}$ lies in the chart domain, for some $R>0$, and let $c$ be an upper bound for $u$ on this ball. The function $v(\zeta)=u(R\zeta)-c$ is plurisubharmonic and nonpositive on the unit ball $B$, and since $u$ is locally integrable there is $A>0$ with $\|v\|_{L^1(B)}\leq A$. Put $\alpha_0=\alpha_n/A$ and $W=B(0,R/4)$. For $0<\alpha\leq\alpha_0$ we have $\mathrm e^{-\alpha v}\leq\mathrm e^{-\alpha_0v}$, because $v\leq0$, so the change of variables $z=R\zeta$ and the last assertion of \cref{prop:negative-psh-ball-exponential} give
\[
 \int_W\mathrm e^{-\alpha u}\,\mathrm dV=R^{2n}\mathrm e^{-\alpha c}\int_{\frac14B}\mathrm e^{-\alpha v}\,\mathrm dV\leq R^{2n}\mathrm e^{-\alpha c}C_n<\infty.
\]
