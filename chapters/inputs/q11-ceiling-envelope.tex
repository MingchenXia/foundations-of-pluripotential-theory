\section{Singularity envelopes at arbitrary mass}
\label{sec:arbitrary-mass-singularity-envelopes}
Throughout this section $\theta$ is a smooth closed real $(1,1)$-form on $X$ with big class, and we put $V=V_\theta$ and $\Omega=\int_X\omega^n$; volumes of Borel sets are taken with respect to $\omega^n$. The notation $\theta_q$, $\mu_q$ and $M(q)$ for $q\in\PSH(X,\theta)$ is that of \cref{sec:singular-bounded-difference-integration}. No model potential is fixed in this section.

For a potential $u\leq0$ of positive mass, \cref{thm:positive-mass-model-characterization} describes $P_\theta[u]$ as the largest potential $v\leq0$ with $u\preceq v$ and $M(v)=M(u)$, and it shows that $P_\theta[P_\theta[u]]=P_\theta[u]$. Its proof relies on the domination principle, which requires positive mass. For arbitrary $u$ we replace the mass by the collection of all mixed masses of $u$ against $V$; this leads to the singularity ceiling of \cref{def:singularity-ceiling}. We prove that the ceiling equals $P_\theta[u]$ and that $P_\theta[\cdot]$ is idempotent, without any assumption on $M(u)$.

The main step, \cref{prop:positive-mass-singularity-envelope-limit}, perturbs $u$ to $w_\varepsilon=(1-\varepsilon)u+\varepsilon V$, which has positive mass $m_\varepsilon$, and shows that $P_\theta[w_\varepsilon]$ decreases to $P_\theta[u]$. Since $m_\varepsilon$ tends to zero when $M(u)=0$, the estimates must not deteriorate as the mass degenerates. We therefore use a tail bound normalized by $m_\varepsilon$, which is uniform in $\varepsilon$, and convert it into a volume bound by means of constant-density solutions of Monge--Amp\`ere equations. Historical remarks on these results are in \cref{sec:history-energy}.

\begin{proposition}\label{prop:positive-mass-singularity-envelope-limit}
Let $u\in\PSH(X,\theta)$ with $u\leq0$. For $0<\varepsilon<1$ put
\[
 w_\varepsilon=(1-\varepsilon)u+\varepsilon V,
 \qquad \phi_\varepsilon=P_\theta[w_\varepsilon].
\]
Then $\phi_\varepsilon$ decreases to $P_\theta[u]$, pointwise and in $L^1(X)$, as $\varepsilon\downarrow0$.
\end{proposition}
\begin{proof}
\emph{Step 1: the limit exists.} Put $\phi=P_\theta[u]$. Since $u\leq0$, we have $u\leq V$, so $w_\varepsilon=u+\varepsilon(V-u)$ decreases to $u$ as $\varepsilon\downarrow0$ (at the poles of $u$ all these functions equal $-\infty$). In particular $u\leq w_\varepsilon\leq0$. By the monotonicity of $P_\theta[\cdot]$ in \cref{prop:p-envelope-calculus} and the bound $P_\theta[\cdot]\leq V$ of \cref{def:singularity-p-envelope}, we get $\phi\leq\phi_\varepsilon\leq V$, and $\phi_\varepsilon$ decreases as $\varepsilon$ decreases. Hence $q=\lim_{\varepsilon\downarrow0}\phi_\varepsilon$ exists pointwise; as a decreasing limit of $\theta$-psh potentials bounded below by $\phi$, it is $\theta$-psh, and $\phi\leq q\leq V$. Since $\phi$ and $V$ are integrable, dominated convergence gives $\phi_\varepsilon\to q$ in $L^1(X)$. It remains to prove that $q\leq\phi$.

\emph{Step 2: masses and a normalized tail bound.} For $0\leq k\leq n$ consider the finite non-pluripolar measures
\[
 \nu_k=\langle\theta_u^k\wedge\theta_V^{n-k}\rangle,
 \qquad a_k=\nu_k(X).
\]
Since $\theta_{w_\varepsilon}=(1-\varepsilon)\theta_u+\varepsilon\theta_V$, multilinearity (\cref{prop:nonpluripolar-local-calculus}) gives
\[
 \begin{aligned}
 \mu_{w_\varepsilon}
   &=\sum_{k=0}^n\binom nk(1-\varepsilon)^k\varepsilon^{n-k}\nu_k,
 \\
 m_\varepsilon:=M(w_\varepsilon)
   &=\sum_{k=0}^n\binom nk(1-\varepsilon)^k\varepsilon^{n-k}a_k.
 \end{aligned}
\]
Here $a_0=M(V)>0$ by \cref{prop:positive-mixed-mass-bound}, so $m_\varepsilon\geq\varepsilon^na_0>0$. Since $w_\varepsilon\leq0$, the mass preservation in \cref{prop:p-envelope-calculus} and \cref{thm:positive-mass-model-characterization} show that $\phi_\varepsilon$ is a model potential of mass $m_\varepsilon$.

In the rest of the proof, sets defined by inequalities between potentials, such as $\{u<V-A\}$, are understood pointwise, with values in $[-\infty,+\infty)$. The pole sets of all potentials involved are pluripolar, hence null for all the non-pluripolar measures below and for $\omega^n$. For $A>0$ put
\[
 r(A)=\max_{k:\,a_k>0}\frac{\nu_k(\{u<V-A\})}{a_k}.
\]
The maximum is over a finite set which contains $k=0$. As $A\to\infty$, the sets $\{u<V-A\}$ decrease to a subset of the pole set of $u$, which is $\nu_k$-null; since the measures $\nu_k$ are finite, continuity from above gives $r(A)\downarrow0$. Moreover $w_\varepsilon-V=(1-\varepsilon)(u-V)$, so $\{w_\varepsilon<V-A\}\subset\{u<V-A\}$. Using the expansion of $\mu_{w_\varepsilon}$ and the bound $\nu_k(\{u<V-A\})\leq r(A)a_k$, valid for every $k$ (when $a_k=0$ both sides vanish), we obtain the estimate
\begin{equation}\label{eq:normalized-singularity-tail}
 \frac{\mu_{w_\varepsilon}(\{w_\varepsilon<V-A\})}{m_\varepsilon}
                  \leq r(A),\qquad 0<\varepsilon<1,
\end{equation}
which is uniform in $\varepsilon$.

\emph{Step 3: rooftops.} For fixed $A>0$ put
\[
 p_{\varepsilon,A}=P_\theta(w_\varepsilon+A,V),
 \qquad p_A=P_\theta(u+A,V).
\]
The potential $w_\varepsilon$ is a common minorant of $w_\varepsilon+A$ and $V$, so $w_\varepsilon\leq p_{\varepsilon,A}$. On the other hand $p_{\varepsilon,A}\leq V\leq0$ and $p_{\varepsilon,A}\leq w_\varepsilon+A$, so $p_{\varepsilon,A}$ is a competitor in the definition of $P_\theta[w_\varepsilon]$, and $p_{\varepsilon,A}\leq\phi_\varepsilon$. As $M(w_\varepsilon)=M(\phi_\varepsilon)=m_\varepsilon$, the monotonicity of mass, \cref{thm:nonpluripolar-self-mass-monotonicity}, gives $M(p_{\varepsilon,A})=m_\varepsilon$; thus $p_{\varepsilon,A}\in\mathcal E(X,\theta,\phi_\varepsilon)$. Since $w_\varepsilon+A$ decreases to $u+A$, \cref{prop:rooftop-algebra} gives $p_{\varepsilon,A}\downarrow p_A$ as $\varepsilon\downarrow0$. In the same way as above, $p_A$ is a competitor in the definition of $P_\theta[u]$, so $p_A\leq\phi$.

\emph{Step 4: auxiliary solutions.} Apply \cref{thm:bounded-density-relative-prescribed-measure} with the model potential $\phi_\varepsilon$ of mass $m_\varepsilon$ in place of $\phi$, the volume form $\omega^n$ and the constant density $m_\varepsilon/\Omega$. It gives $\chi_\varepsilon\in\mathcal E^1(X,\theta,\phi_\varepsilon)$ with
\[
 \mu_{\chi_\varepsilon}=\frac{m_\varepsilon}{\Omega}\,\omega^n,
 \qquad \sup_X\chi_\varepsilon=0.
\]
By \cref{lem:model-normalized-supremum}, $\chi_\varepsilon\leq\phi_\varepsilon$. By \cref{prop:normalized-global-psh-compactness}, normalized $\theta$-psh potentials are uniformly bounded in $L^1$, so there is a constant $C_1$, independent of $\varepsilon$, with $\int_X(-\chi_\varepsilon)\,\omega^n\leq C_1$.

\emph{Step 5: a volume bound uniform in $\varepsilon$.} Fix $\delta>0$ and $B>1$, put $\tau=\min\{1/2,\delta/(4B)\}$, and define
\[
 \begin{aligned}
  G_{\varepsilon,B}&=\{\chi_\varepsilon>\phi_\varepsilon-B\},\\
  \psi_\varepsilon&=(1-\tau)\phi_\varepsilon
                       +\tau\chi_\varepsilon-\delta/2,\\
  E_{\varepsilon,A,\delta}&=\{\phi_\varepsilon>p_{\varepsilon,A}+\delta\}.
 \end{aligned}
\]
Since $\phi_\varepsilon\leq0$, the complement of $G_{\varepsilon,B}$ lies in $\{\chi_\varepsilon\leq-B\}$, whose volume is at most $C_1/B$ by Markov's inequality. Moreover,
\[
 \chi_\varepsilon-\delta/2\leq\psi_\varepsilon
                      \leq\phi_\varepsilon-\delta/2,
\]
and since $\chi_\varepsilon$ and $\phi_\varepsilon$ have the same mass, the monotonicity of mass places $\psi_\varepsilon$ in $\mathcal E(X,\theta,\phi_\varepsilon)$. On $G_{\varepsilon,B}$ we have $\psi_\varepsilon-\phi_\varepsilon=\tau(\chi_\varepsilon-\phi_\varepsilon)-\delta/2>-\tau B-\delta/2\geq-3\delta/4$, whereas $p_{\varepsilon,A}<\phi_\varepsilon-\delta$ on $E_{\varepsilon,A,\delta}$. Hence
\[
 E_{\varepsilon,A,\delta}\cap G_{\varepsilon,B}\subset\{p_{\varepsilon,A}<\psi_\varepsilon\}.
\]
By multilinearity, $\mu_{\psi_\varepsilon}\geq\tau^n\mu_{\chi_\varepsilon}=\tau^n(m_\varepsilon/\Omega)\omega^n$. As $p_{\varepsilon,A}$ and $\psi_\varepsilon$ both belong to $\mathcal E(X,\theta,\phi_\varepsilon)$, the comparison principle of \cref{prop:relative-mixed-comparison} applies to them, and we obtain
\[
 \tau^n\frac{m_\varepsilon}{\Omega}
       \int_{E_{\varepsilon,A,\delta}\cap G_{\varepsilon,B}}\omega^n
 \leq\int_{\{p_{\varepsilon,A}<\psi_\varepsilon\}}\mu_{\psi_\varepsilon}
 \leq\int_{\{p_{\varepsilon,A}<\psi_\varepsilon\}}\mu_{p_{\varepsilon,A}}.
\]
To estimate the right side, we use the contact inequality for rooftops, \cref{cor:rooftop-contact-inequality}:
\[
 \mu_{p_{\varepsilon,A}}
 \leq\mathbf1_{\{p_{\varepsilon,A}=w_\varepsilon+A\}}\mu_{w_\varepsilon}
       +\mathbf1_{\{p_{\varepsilon,A}=V\}}\mu_V.
\]
The second contact set does not meet $\{p_{\varepsilon,A}<\psi_\varepsilon\}$: at a common point we would have $V<\psi_\varepsilon\leq\phi_\varepsilon-\delta/2\leq V-\delta/2$. At a point of $\{p_{\varepsilon,A}<\psi_\varepsilon\}$ in the first contact set, $w_\varepsilon+A<\psi_\varepsilon\leq V-\delta/2$, hence $w_\varepsilon<V-A$. Therefore, by \eqref{eq:normalized-singularity-tail},
\[
 \int_{\{p_{\varepsilon,A}<\psi_\varepsilon\}}\mu_{p_{\varepsilon,A}}
 \leq\mu_{w_\varepsilon}(\{w_\varepsilon<V-A\})\leq m_\varepsilon r(A).
\]
Dividing by the positive number $\tau^nm_\varepsilon/\Omega$ and adding the volume of the complement of $G_{\varepsilon,B}$, we arrive at
\begin{equation}\label{eq:ceiling-volume-recovery}
 \int_{E_{\varepsilon,A,\delta}}\omega^n
       \leq\frac{C_1}{B}+\Omega\tau^{-n}r(A).
\end{equation}
This bound is uniform in $\varepsilon$, even if $m_\varepsilon\to0$.

\emph{Step 6: passage to the limit.} Take a sequence $\varepsilon_j\downarrow0$. If $q(x)>p_A(x)+\delta$, then $\phi_{\varepsilon_j}(x)\geq q(x)$ and $p_{\varepsilon_j,A}(x)\downarrow p_A(x)$, so $x\in E_{\varepsilon_j,A,\delta}$ for all large $j$. Thus
\[
 \{q>p_A+\delta\}\subset
                 \liminf_j E_{\varepsilon_j,A,\delta},
\]
and Fatou's lemma together with \eqref{eq:ceiling-volume-recovery} shows that the volume of $\{q>p_A+\delta\}$ is at most $C_1/B+\Omega\tau^{-n}r(A)$. Since $p_A\leq\phi$, the same bound holds for the set $\{q>\phi+\delta\}$, which depends neither on $A$ nor on $B$. Here $\tau$ depends on $B$ and $\delta$ but not on $A$. Letting first $A\to\infty$, so that $r(A)\to0$, and then $B\to\infty$, we see that $\{q>\phi+\delta\}$ has zero volume. As $\delta>0$ was arbitrary, $q\leq\phi$ almost everywhere, and hence everywhere, because both are $\theta$-psh and have unique canonical representatives. Together with $\phi\leq q$, this gives $q=\phi$.
\end{proof}

\begin{definition}\label{def:singularity-ceiling}
For $u\in\PSH(X,\theta)$ let $\mathcal F_u$ be the set of potentials $v\in\PSH(X,\theta)$ such that $v\leq0$, $u\preceq v$ and
\[
 \int_X\langle\theta_v^k\wedge\theta_V^{n-k}\rangle
       =\int_X\langle\theta_u^k\wedge\theta_V^{n-k}\rangle
                         \qquad\text{for }0\leq k\leq n.
\]
The \emph{singularity ceiling} of $u$ is $C_\theta(u)=(\sup_{v\in\mathcal F_u}v)^*$.
\end{definition}
The family $\mathcal F_u$ contains $u-\sup_Xu$ and is bounded above by zero, so its regularized supremum $C_\theta(u)$ is a $\theta$-psh potential. Both $C_\theta(u)$ and $P_\theta[u]$ are unchanged when a constant is added to $u$.
\index{singularity ceiling}

\begin{theorem}\label{thm:arbitrary-mass-ceiling-envelope}
Let $\theta$ be a smooth closed real $(1,1)$-form on $X$ with big class. Then for every $u\in\PSH(X,\theta)$,
\[
 C_\theta(u)=P_\theta[u],
 \qquad P_\theta[P_\theta[u]]=P_\theta[u].
\]
This includes potentials $u$ with $M(u)=0$.
\end{theorem}
\begin{proof}
By the remark following \cref{def:singularity-ceiling}, both identities are unchanged when a constant is added to $u$, so we may assume $u\leq0$. Put $\phi=P_\theta[u]$.

We first show $\phi\leq C_\theta(u)$. By \cref{def:singularity-p-envelope}, $\phi\leq0$, and $u\leq\phi$ because $u$ is itself a competitor. Applying the mixed mass preservation of \cref{prop:p-envelope-calculus} to the tuple consisting of $k$ copies of $u$ and $n-k$ copies of $V$, and using $P_\theta[V]=V$, we see that $\phi$ has the same mixed masses against $V$ as $u$. Hence $\phi\in\mathcal F_u$, and $\phi\leq C_\theta(u)$.

Conversely, let $v\in\mathcal F_u$, and for $0<\varepsilon<1$ put
\[
 w_\varepsilon=(1-\varepsilon)u+\varepsilon V,
 \qquad z_\varepsilon=(1-\varepsilon)v+\varepsilon V.
\]
Expanding by multilinearity and using the mixed mass identities that define $\mathcal F_u$, we get $M(w_\varepsilon)=M(z_\varepsilon)\geq\varepsilon^nM(V)>0$. Moreover $w_\varepsilon\preceq z_\varepsilon$, since $u\preceq v$. Let $\zeta_\varepsilon=P_\theta[z_\varepsilon]$. By \cref{thm:positive-mass-model-characterization}, $\zeta_\varepsilon$ is a model potential with $M(\zeta_\varepsilon)=M(z_\varepsilon)=M(w_\varepsilon)$. As $w_\varepsilon\preceq z_\varepsilon\leq\zeta_\varepsilon$, the potential $w_\varepsilon$ belongs to $\mathcal E(X,\theta,\zeta_\varepsilon)$, and \cref{prop:relative-envelope-full-mass-characterization} gives
\[
 P_\theta[w_\varepsilon]=P_\theta[\zeta_\varepsilon]=\zeta_\varepsilon
                                      =P_\theta[z_\varepsilon].
\]
Now let $\varepsilon\downarrow0$. \Cref{prop:positive-mass-singularity-envelope-limit}, applied to $u$ and to $v$, gives $P_\theta[u]=P_\theta[v]$. Since $v\leq0$, the potential $v$ is itself a competitor in the definition of $P_\theta[v]$, so $v\leq P_\theta[v]=\phi$. Taking the supremum over $v\in\mathcal F_u$ and its upper semicontinuous regularization, we obtain $C_\theta(u)\leq\phi$.

For idempotence, let $\phi_\varepsilon=P_\theta[w_\varepsilon]$ be as in \cref{prop:positive-mass-singularity-envelope-limit}. Since $\phi\leq\phi_\varepsilon$, the monotonicity of $P_\theta[\cdot]$ and \cref{thm:positive-mass-model-characterization}, which applies to $w_\varepsilon$ because $w_\varepsilon\leq0$ and $M(w_\varepsilon)>0$, give
\[
 P_\theta[\phi]\leq P_\theta[\phi_\varepsilon]
                                    =\phi_\varepsilon.
\]
Letting $\varepsilon\downarrow0$ and using the proposition, we obtain $P_\theta[\phi]\leq\phi$. Conversely, $\phi\leq0$ is a competitor in the definition of $P_\theta[\phi]$, so $\phi\leq P_\theta[\phi]$. This proves idempotence.
\end{proof}
