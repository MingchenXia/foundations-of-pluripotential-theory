\subsection{Mixed energy and truncation}
\label{sec:mixed-local-energy}
In the next section we approximate several potentials at once by canonical approximants and pass to the limit in their mixed Monge--Amp\`ere products. The approximants converge in capacity and have bounded energies, but we have no uniform bound for them, so \cref{thm:bt-capacity-continuity} does not apply to them directly. We therefore compare the mixed products with those of the truncated potentials, and for this we need to know that the sets where one of the potentials is very negative carry little mass of a mixed product. This follows from the mixed energy inequality of \cref{thm:bounded-mixed-local-energy}, which bounds a mixed energy by a constant multiple of the geometric mean of the energies $e_1$ of \cref{def:bounded-zero-boundary-energy}.

\begin{theorem}\label{thm:bounded-mixed-local-energy}
Let $u_0,\ldots,u_n\in\mathcal E_0(B)$. Then
\begin{equation}\label{eq:bounded-mixed-local-energy}
 \int_B(-u_0)\,\ddc u_1\wedge\cdots\wedge\ddc u_n
 \leq C_n\prod_{l=0}^n e_1(u_l)^{1/(n+1)},
 \qquad C_n=2^{n+2}.
\end{equation}
In particular the integral is finite, and it vanishes if one of the energies $e_1(u_l)$ vanishes.
\end{theorem}
\begin{proof}
Write $B=B(a,R)$ and $\rho(z)=|z-a|^2-R^2$. We say that $v\in\mathcal E_0(B)$ has a \emph{quadratic collar} if $v=c\rho$ on a neighborhood of $\partial B$ in $B$ for some constant $c>0$. We first prove the inequality with constant one for potentials with quadratic collars (Steps 1 and 2), and then approximate arbitrary elements of $\mathcal E_0(B)$ by such potentials while keeping their energies under control (Steps 3 and 4). For functions $b_0,\ldots,b_n$ which either all have quadratic collars, or are smooth real functions on a neighborhood of $\overline B$ with $b_2,\ldots,b_n$ plurisubharmonic, put
\[
 I(b_0,\ldots,b_n)
       =\int_B(-b_0)\,\ddc b_1\wedge\cdots\wedge\ddc b_n.
\]
The integral is finite: in the first case every mixed product of the $b_l$ is smooth near $\partial B$ and locally finite on the rest of $B$, and in the second the integrand is a smooth form on a neighborhood of $\overline B$.

\emph{Step 1. Smooth functions.}
Let $b_0,\ldots,b_n$ be smooth real functions on a neighborhood of $\overline B$, with $b_2,\ldots,b_n$ plurisubharmonic and $b_0=0$ on $\partial B$, and put $S=\ddc b_2\wedge\cdots\wedge\ddc b_n$, with $S=1$ if $n=1$. With $\mathrm d^{\mathrm c}$ as in \eqref{eq:dc-operator}, we claim that
\begin{equation}\label{eq:mixed-local-energy-gradient-form}
 I(b_0,b_1,b_2,\ldots,b_n)
          =\int_B\mathrm db_0\wedge\mathrm d^{\mathrm c}b_1\wedge S.
\end{equation}
Fix a smooth function $\theta\colon\mathbb R\to[0,1]$ with $\theta=1$ on $(-\infty,-2]$ and $\theta=0$ on $[-1,\infty)$, and put $\chi_k=\theta(k\rho)$ for $k\geq1$; then $\chi_k\in C^\infty_c(B)$. The $(2n-1)$-form $\chi_kb_0\,\mathrm d^{\mathrm c}b_1\wedge S$ is smooth with compact support in $B$, and $S$ is closed, so \cref{prop:manifold-cutoffs-stokes} and the Leibniz rule give
\[
 0=\int_B\chi_k\,\mathrm db_0\wedge\mathrm d^{\mathrm c}b_1\wedge S
   +\int_Bb_0\,\mathrm d\chi_k\wedge\mathrm d^{\mathrm c}b_1\wedge S
   +\int_B\chi_kb_0\,\ddc b_1\wedge S.
\]
As $k\to\infty$, $\chi_k\to1$ pointwise on $B$, and the forms $\mathrm db_0\wedge\mathrm d^{\mathrm c}b_1\wedge S$ and $b_0\,\ddc b_1\wedge S$ have bounded coefficients on $B$; by dominated convergence, the first and the last integral tend to $\int_B\mathrm db_0\wedge\mathrm d^{\mathrm c}b_1\wedge S$ and $-I(b_0,\ldots,b_n)$. In the middle integral, $\mathrm d\chi_k=k\theta'(k\rho)\,\mathrm d\rho$ vanishes outside the set where $-2\leq k\rho\leq-1$. Let $\Lambda$ be the maximum of $|\nabla b_0|$ on $\overline B$. If $z\in B$ and $\zeta\in\partial B$ is a nearest boundary point, then $b_0(\zeta)=0$ and $|z-\zeta|=R-|z-a|$, so the mean value theorem on the segment from $z$ to $\zeta$ gives
\[
 |b_0(z)|\leq\Lambda(R-|z-a|)\leq\frac{\Lambda}{R}(R^2-|z-a|^2)=\frac{\Lambda}{R}|\rho(z)|.
\]
Hence $|b_0\,k\theta'(k\rho)|\leq(\Lambda/R)|k\rho|\,|\theta'(k\rho)|\leq2\Lambda\sup|\theta'|/R$ on $B$, uniformly in $k$, and $b_0\,k\theta'(k\rho)\to0$ pointwise on $B$. Since $\mathrm d\rho\wedge\mathrm d^{\mathrm c}b_1\wedge S$ has bounded coefficients on $B$, dominated convergence shows that the middle integral tends to zero. This proves \eqref{eq:mixed-local-energy-gradient-form}.

As observed at the beginning of \cref{sec:bt-monotone}, $\mathrm db\wedge\mathrm d^{\mathrm c}c\wedge S=\mathrm dc\wedge\mathrm d^{\mathrm c}b\wedge S$ for smooth real $b,c$. Moreover, $\mathrm db\wedge\mathrm d^{\mathrm c}b=(\mathrm i/2\pi)\partial b\wedge\bar\partial b$ is a positive $(1,1)$-form, and so are the forms $\ddc b_l$, $l\geq2$, since the complex Hessian of a smooth plurisubharmonic function is positive semidefinite (\cref{prop:smooth-psh-criterion}). Positive $(1,1)$-forms are strongly positive (\cref{sec:positive-currents}), so by \cref{prop:positivity-cones} the $(n,n)$-form $\mathrm db\wedge\mathrm d^{\mathrm c}b\wedge S$ is a nonnegative multiple of the volume form. Consequently the real bilinear form
\[
 Q(b,c)=\int_B\mathrm db\wedge\mathrm d^{\mathrm c}c\wedge S,
\]
defined for smooth real functions $b,c$ on a neighborhood of $\overline B$ vanishing on $\partial B$, is symmetric and positive semidefinite, and by \eqref{eq:mixed-local-energy-gradient-form} we have $Q(b,c)=I(b,c,b_2,\ldots,b_n)$. The Cauchy--Schwarz inequality for $Q$, which follows from the nonnegativity of the quadratic polynomial $t\mapsto Q(b+tc,b+tc)$, therefore gives
\begin{equation}\label{eq:mixed-local-energy-two-slot}
 I(b,c,b_2,\ldots,b_n)^2
 \leq I(b,b,b_2,\ldots,b_n)I(c,c,b_2,\ldots,b_n).
\end{equation}
The symmetry of $Q$ allows us to interchange the first two arguments of $I$, and the symmetry of the products allows us to permute the last $n$ arguments. Hence $I$ is symmetric in all $n+1$ arguments when they are smooth plurisubharmonic functions on a neighborhood of $\overline B$ vanishing on $\partial B$.

\emph{Step 2. Quadratic collars.}
Let $u_0,\ldots,u_n$ have quadratic collars, say $u_l=c_l\rho$ near $\partial B$. We first show that the assertions of Step 1 persist for them. Extend each $u_l$ by $c_l\rho$ to a plurisubharmonic function on a neighborhood of $\overline B$. Let $\vartheta_\varepsilon$ be the radial mollifiers of \cref{sec:convolution}, denoted $\rho_\varepsilon$ there; we change the letter because $\rho$ is now the quadratic function above. Because $\rho$ is a quadratic polynomial and $\vartheta_\varepsilon$ is radial, $\rho*\vartheta_\varepsilon=\rho+\gamma\varepsilon^2$ with the constant $\gamma=\int|y|^2\vartheta_1(y)\,\mathrm dV(y)$. Hence
\[
 u_{l,\varepsilon}=u_l*\vartheta_\varepsilon-c_l\gamma\varepsilon^2
\]
equals $c_l\rho$ on a fixed neighborhood of $\partial B$ for all small $\varepsilon$. By \cref{prop:smooth-psh-criterion}, the convolutions $u_l*\vartheta_\varepsilon$ are smooth and plurisubharmonic on a neighborhood of $\overline B$ for small $\varepsilon$, and they decrease to $u_l$ as $\varepsilon\downarrow0$. Hence each $u_{l,\varepsilon}$, which differs from $u_l*\vartheta_\varepsilon$ by a constant, is smooth and plurisubharmonic on a neighborhood of $\overline B$ and vanishes on $\partial B$, although the $u_{l,\varepsilon}$ need not decrease. By \cref{thm:bt-decreasing-continuity},
\[
 (-u_0*\vartheta_\varepsilon)\,\ddc u_{1,\varepsilon}\wedge\cdots\wedge\ddc u_{n,\varepsilon}
 \longrightarrow(-u_0)\,\ddc u_1\wedge\cdots\wedge\ddc u_n
\]
weakly, and the constant $c_0\gamma\varepsilon^2$ changes the integral against a fixed test function only by an amount tending to zero, by the local mass bounds of \cref{thm:cln-bounded}. Near $\partial B$, all the $u_{l,\varepsilon}$ and $u_l$ coincide with $c_l\rho$. Splitting the integrals over $B$ by a cutoff in $C_c(B)$ which equals one outside this neighborhood of $\partial B$, we conclude that $I(u_{0,\varepsilon},\ldots,u_{n,\varepsilon})\to I(u_0,\ldots,u_n)$, and likewise for every rearrangement of the potentials. Passing to the limit in the smooth case proves that $I$ is symmetric and satisfies \eqref{eq:mixed-local-energy-two-slot} for potentials with quadratic collars.

We now prove the inequality with constant one for these potentials. Since $u_l=c_l\rho$ near $\partial B$ with $c_l>0$, the energy $e_1(u_l)$ is positive: already its integral over a neighborhood of $\partial B$ is positive. Since $I$ is positively homogeneous of degree one in each argument and $e_1(tv)=t^{n+1}e_1(v)$ for $t>0$, both sides of the inequality are multiplied by the same factor when each $u_l$ is replaced by $e_1(u_l)^{-1/(n+1)}u_l$. We may therefore assume that $e_1(u_l)=1$ for every $l$. Consider the finitely many ordered $(n+1)$-tuples whose entries are taken from the functions $u_0,\ldots,u_n$, and let $F$ be the largest value of $I$ on these tuples. Since $I(u_l,\ldots,u_l)=e_1(u_l)=1$, we have $F\geq1$. Among the tuples on which $I$ equals $F$, choose one for which the sum of the squares of the multiplicities of its distinct entries is largest.

Suppose that this tuple contains two distinct functions $b$ and $c$, with multiplicities $r$ and $s$. By symmetry we may place them in the first two slots. In \eqref{eq:mixed-local-energy-two-slot}, the left side is $F^2$, while the two factors on the right are nonnegative values of $I$ on tuples of our family, hence at most $F$. Therefore both factors equal $F$. Replacing one entry $c$ by $b$ changes the sum of the squares of the multiplicities by $(r+1)^2+(s-1)^2-r^2-s^2=2(r-s)+2$, and replacing one entry $b$ by $c$ changes it by $2(s-r)+2$. One of these numbers is positive, which contradicts the choice of the tuple. Hence the chosen tuple consists of a single function repeated $n+1$ times, and $F=1$. In particular $I(u_0,\ldots,u_n)\leq1$, which is the inequality with constant one.

\emph{Step 3. Approximation with controlled energy.}
Let $u\in\mathcal E_0(B)$, put $m_\rho=\int_B(\ddc\rho)^n$, and for $j\geq1$ let
\[
 \delta_j=j^{-n-1},\qquad v_j=\max\{u-\delta_j,j\rho\}.
\]
Since $u-\delta_j\leq-\delta_j$ while $j\rho>-\delta_j$ near $\partial B$, the function $v_j$ equals $j\rho$ on a neighborhood of $\partial B$ in $B$. Since $v_j$ and $j\rho$ coincide outside a compact subset of $B$, \cref{lem:compact-perturbation-mass} gives $\int_B(\ddc v_j)^n=j^nm_\rho$, and hence $v_j\in\mathcal E_0(B)$. Moreover, $v_j\to u$ uniformly on $B$. Indeed, let $s>0$, and choose a neighborhood $V$ of $\partial B$ with $u>-s$ on $V\cap B$. Since $u-\delta_j\leq v_j\leq0$, we have $|v_j-u|\leq s+\delta_j$ on $V\cap B$. On the set $B\setminus V$, whose closure is a compact subset of $B$, the function $\rho$ is bounded away from zero and $u$ is bounded, so $j\rho<u-\delta_j$, and hence $v_j=u-\delta_j$, there for all large $j$.

Next we estimate $e_1(v_j)$. Let $t>4\delta_j$. Since $u\leq v_j+\delta_j$ and $v_j\leq0$,
\[
 \{v_j<-t\}\subset\{u<v_j/2-t/4\}
                       \subset\{u<-t/4\};
\]
for the first inclusion note that on $\{v_j<-t\}$ we have $v_j/2<-t/2<-t/4-\delta_j$, hence $u\leq v_j+\delta_j<v_j/2-t/4$. As $u$ tends to $0$ and $v_j/2-t/4$ tends to $-t/4$ at $\partial B$, the comparison theorem (\cref{thm:bt-comparison}) applies to $u$ and $v_j/2-t/4$ and gives
\[
 \int_{\{v_j<-t\}}(\ddc v_j)^n
       \leq2^n\int_{\{u<v_j/2-t/4\}}\bigl(\ddc(v_j/2)\bigr)^n
       \leq2^n\int_{\{u<-t/4\}}(\ddc u)^n.
\]
For $0<t\leq4\delta_j$ we use the total mass $j^nm_\rho$ instead. Integrating over $t$ by the layer-cake formula $e_1(v)=\int_0^\infty(\ddc v)^n(\{v<-t\})\,\mathrm dt$, we obtain
\begin{equation}\label{eq:quadratic-collar-energy-approximation}
 e_1(v_j)\leq4\delta_j j^n m_\rho+2^{n+2}e_1(u)
                 =\frac{4m_\rho}j+C_ne_1(u).
\end{equation}

\emph{Step 4. The general case.}
Apply the construction of Step 3 to each $u_l$, obtaining functions $v_{l,j}\in\mathcal E_0(B)$ which equal $j\rho$ near $\partial B$. They are uniformly bounded, since $u_l-1\leq v_{l,j}\leq0$, and they converge uniformly, hence locally in capacity, to $u_l$. Let $\chi\in C_c(B)$ with $0\leq\chi\leq1$. The weight $-u_0$ is bounded and quasi-continuous (\cref{thm:psh-capacity-quasicontinuity}), so the weighted form of \cref{thm:bt-capacity-continuity} and Step 2 give
\[
 \begin{aligned}
 \int_B\chi(-u_0)\,\ddc u_1\wedge\cdots\wedge\ddc u_n
 &=\lim_{j\to\infty}\int_B\chi(-v_{0,j})\,\ddc v_{1,j}\wedge\cdots\wedge\ddc v_{n,j}\\
 &\leq\varlimsup_{j\to\infty}\prod_{l=0}^ne_1(v_{l,j})^{1/(n+1)}.
 \end{aligned}
\]
By \eqref{eq:quadratic-collar-energy-approximation}, the right side is at most
\[
 \lim_{j\to\infty}\prod_{l=0}^n\left(\frac{4m_\rho}j+C_ne_1(u_l)\right)^{1/(n+1)}
 =C_n\prod_{l=0}^ne_1(u_l)^{1/(n+1)}.
\]
Letting $\chi$ increase to one proves \eqref{eq:bounded-mixed-local-energy}. Vanishing energies need no separate treatment, since the energies of the $v_{l,j}$ are positive.
\end{proof}

\begin{corollary}\label{cor:mixed-products-energy-capacity-continuity}
Let $E\geq0$, and for $1\leq l\leq n$ and $j\geq1$ let $u_{l,j}\in\mathcal E_0(B)$ with $e_1(u_{l,j})\leq E$. Suppose that, for each $l$, $u_{l,j}\to u_l$ locally in capacity as $j\to\infty$, where $u_l$ is a locally bounded plurisubharmonic function on $B$. Then
\[
 \ddc u_{1,j}\wedge\cdots\wedge\ddc u_{n,j}
          \longrightarrow\ddc u_1\wedge\cdots\wedge\ddc u_n
\]
weakly on $B$.
\end{corollary}
\begin{proof}
We compare the products with those of the truncations at a level $-T$: the latter converge by \cref{thm:bt-capacity-continuity}, and by \cref{thm:bounded-mixed-local-energy} the two differ by a term of order $1/T$. For $T>0$ put $a_{l,j}=\max\{u_{l,j},-T\}$, and write
\[
 R_j=\ddc u_{1,j}\wedge\cdots\wedge\ddc u_{n,j},\qquad
 R_{j,T}=\ddc a_{1,j}\wedge\cdots\wedge\ddc a_{n,j}.
\]
By \cref{lem:zero-boundary-truncation-mass}, $a_{l,j}\in\mathcal E_0(B)$ and $e_1(a_{l,j})\leq E$. For each $l$, \cref{thm:bounded-mixed-local-energy} with $u_0=u_{l,j}$, respectively $u_0=a_{l,j}$, gives
\[
 \int_B(-u_{l,j})\,R_j\leq C_nE,\qquad\int_B(-a_{l,j})\,R_{j,T}\leq C_nE.
\]
Let $D_{j,T}=\bigcup_{l=1}^n\{u_{l,j}\leq-T\}$. On $\{u_{l,j}\leq-T\}$ both $-u_{l,j}$ and $-a_{l,j}$ are at least $T$, so Chebyshev's inequality gives
\[
 R_j(D_{j,T})\leq nC_nE/T,\qquad R_{j,T}(D_{j,T})\leq nC_nE/T.
\]
On the complement of $D_{j,T}$, every $u_{l,j}$ is greater than $-T$. For $0\leq k\leq n$ put
\[
 P_k=\ddc a_{1,j}\wedge\cdots\wedge\ddc a_{k,j}\wedge\ddc u_{k+1,j}\wedge\cdots\wedge\ddc u_{n,j},
\]
so that $P_0=R_j$ and $P_n=R_{j,T}$. By the symmetry of the products, $P_k=\ddc a_{k,j}\wedge\Theta_k$ and $P_{k-1}=\ddc u_{k,j}\wedge\Theta_k$ with the same closed positive current $\Theta_k$ formed by the other $n-1$ factors, and maximum locality (\cref{prop:bt-maximum-locality} with $p=1$, $u=u_{k,j}$, $v=-T$ and base current $\Theta_k$) gives $P_k=P_{k-1}$ on $\{u_{k,j}>-T\}$. Hence $R_j=R_{j,T}$ on the complement of $D_{j,T}$, and for every $\chi\in C_c(B)$,
\[
 \left|\int_B\chi\,(R_j-R_{j,T})\right|
                    \leq2nC_nE\|\chi\|_\infty/T.
\]
Now fix $\chi\in C_c(B)$, choose an open set $B'\Subset B$ containing $\Supp\chi$, and let $T>\max_l\sup_{B'}|u_l|$. By \eqref{eq:truncation-level-inclusions}, the uniformly bounded functions $a_{l,j}$ converge to the locally bounded plurisubharmonic function $\max\{u_l,-T\}$ locally in capacity, and \cref{thm:bt-capacity-continuity} shows that $R_{j,T}\to\bigwedge_l\ddc\max\{u_l,-T\}$ weakly on $B$. On $B'$ we have $\max\{u_l,-T\}=u_l$, so by locality (\cref{prop:bt-positive-recursion}) this limit equals $\ddc u_1\wedge\cdots\wedge\ddc u_n$ there. Consequently
\[
 \varlimsup_{j\to\infty}\left|\int_B\chi\,R_j-\int_B\chi\,\ddc u_1\wedge\cdots\wedge\ddc u_n\right|\leq2nC_nE\|\chi\|_\infty/T
\]
for all large $T$, and letting $T\to\infty$ completes the proof.
\end{proof}
