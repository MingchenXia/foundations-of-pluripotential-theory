\section{Gradients against positive currents}
\label{sec:bt-current-gradients}
Several later arguments differentiate the product $v\Theta$ of a bounded plurisubharmonic function $v$ and a closed positive current $\Theta$; examples are the proof that non-pluripolar products are closed (\cref{thm:nonpluripolar-product-closed}) and cutoff arguments on compact manifolds. For smooth $v$ the derivative is $\mathrm dv\wedge\Theta$, and its size is controlled by the gradient measure $\mathrm dv\wedge\mathrm d^{\mathrm c}v\wedge\Theta$, where $\mathrm d^{\mathrm c}$ is the real operator of \eqref{eq:dc-operator}, so that $\mathrm d\mathrm d^{\mathrm c}=\ddc$. In this section we extend both objects, together with the chain rule, to locally bounded plurisubharmonic functions by decreasing approximation.

Throughout, $U\subset\mathbb C^n$ is open and $\Theta$ is a closed positive current of bidimension $(1,1)$ on $U$, with trace measure $\mu=\Theta\wedge\beta$. We first describe $\Theta$ by a matrix of densities. By \cref{thm:positive-current-order-zero}, the coefficient measures of $\Theta$ are complex Radon measures whose total variations are bounded by a constant multiple of $\mu$. Applying \cref{lem:dominated-measure-density} to the positive and negative parts of their real and imaginary parts, we obtain locally bounded Borel functions $h_{jk}$ with
\[
 \frac{\mathrm i}{2\pi}\,\mathrm dz_k\wedge\mathrm d\bar z_j\wedge\Theta=h_{jk}\,\mu
 \qquad(1\leq j,k\leq n).
\]
Since $\Theta$ is real, the matrix $H=(h_{jk})$ is Hermitian $\mu$-almost everywhere. We identify a $(1,0)$-form $\xi=\sum_k\xi_k\,\mathrm dz_k$ with its coefficient vector $(\xi_1,\ldots,\xi_n)\in\mathbb C^n$. For a constant $(1,0)$-form $\xi$ we have $(\mathrm i/2\pi)\,\xi\wedge\bar\xi\wedge\Theta=\langle H\xi,\xi\rangle\mu$, where $\langle a,b\rangle=\sum_ja_j\bar b_j$ on $\mathbb C^n$. These measures are positive because $\Theta$ is positive, and testing a countable dense set of vectors $\xi$ shows that $H$ is positive semidefinite $\mu$-almost everywhere. Changing $H$ on a $\mu$-null set, we may assume that it is Hermitian and positive semidefinite everywhere. For a real $C^1$ function $w$, the form $\partial w$ corresponds to the vector $(\partial w/\partial z_1,\ldots,\partial w/\partial z_n)$, which we also denote by $\partial w$. Since $\mathrm dw\wedge\mathrm d^{\mathrm c}w=(\mathrm i/2\pi)\,\partial w\wedge\bar\partial w$, it follows that
\begin{equation*}
 \mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta=\bigl|H^{1/2}\partial w\bigr|^2\mu,
\end{equation*}
where $H^{1/2}$ is the positive semidefinite square root, a locally bounded Borel function of $H$. Thus, for a compact set $K\subset U$, the square of the \emph{energy seminorm} $\|H^{1/2}\partial w\|_{L^2(K,\mu)}$ is the mass that $\mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta$ gives to $K$. Similarly, let $\eta$ be a real one-form with $(1,0)$-part $\sum_ke_k\,\mathrm dz_k$, and put $e=(e_1,\ldots,e_n)$ and $w_k=\partial w/\partial z_k$. Only the $(1,1)$-part $\sum_{j,k}(e_k\bar w_j-\bar e_jw_k)\,\mathrm dz_k\wedge\mathrm d\bar z_j$ of $\eta\wedge\mathrm dw$ contributes to $\eta\wedge\mathrm dw\wedge\Theta$, so that $\eta\wedge\mathrm dw\wedge\Theta=-2\pi\mathrm i\bigl(\langle He,\partial w\rangle-\langle H\partial w,e\rangle\bigr)\mu$. Since $H$ is Hermitian, this gives
\begin{equation*}
 \eta\wedge\mathrm dw\wedge\Theta=4\pi\Img\bigl\langle H^{1/2}e,H^{1/2}\partial w\bigr\rangle\,\mu.
\end{equation*}

\begin{lemma}\label{lem:bounded-psh-current-chain-rule}
Let $\Theta$ be a closed positive current of bidimension $(1,1)$ on an open set $U\subset\mathbb C^n$, and let $v$ be a locally bounded plurisubharmonic function on $U$. Then the current $\mathrm d(v\Theta)$ has measure coefficients; we denote it by $\mathrm dv\wedge\Theta$. The current
\begin{equation}\label{eq:bounded-current-gradient-measure}
 E_\Theta(v)=\mathrm dv\wedge\mathrm d^{\mathrm c}v\wedge\Theta
       =\tfrac12\ddc(v^2\Theta)-v\,\ddc v\wedge\Theta
\end{equation}
is a positive measure, the \emph{gradient measure} of $v$ with respect to $\Theta$. If smooth plurisubharmonic functions $v_l$ decrease to $v$ on a neighborhood of a compact set $K\subset U$, then their gradients $\partial v_l$ converge in the energy seminorm on $K$, that is, $H^{1/2}\partial v_l$ converges in $L^2(K,\mu)$. For every smooth $f\colon\mathbb R\to\mathbb R$,
\begin{equation}\label{eq:bounded-current-chain-rule}
 \mathrm d(f(v)\Theta)=f'(v)\,\mathrm dv\wedge\Theta.
\end{equation}
Finally, let $\eta$ be a compactly supported smooth real one-form on $U$, and let $\tau\in C^\infty_c(U)$, $\tau\geq0$, be equal to one near $\Supp\eta$. There is a constant $C_\eta$, depending only on $\eta$ and $\Theta$, such that for every bounded Borel function $a$,
\begin{equation}\label{eq:current-gradient-cauchy-schwarz}
 \left|\int a\eta\wedge\mathrm dv\wedge\Theta\right|^2
 \leq C_\eta\left(\int\tau E_\Theta(v)\right)
                 \left(\int_{\Supp\eta}|a|^2\,
                                         \Theta\wedge\beta\right).
\end{equation}
\end{lemma}
\begin{proof}
All assertions are local, so we may work near a fixed compact set $K\subset U$, on a neighborhood of which smooth plurisubharmonic functions $v_l$ decrease to $v$; for instance, radial mollifications of $v$. Then the $v_l$ are uniformly bounded near $K$. Let $\sigma\in C^\infty_c(U)$ be nonnegative and equal to one near $K$, with support in the region where the $v_l$ are defined.

\emph{Step 1. Convergence of the gradients.}
For a smooth real function $w$, we have $\ddc(w^2)=2\,\mathrm dw\wedge\mathrm d^{\mathrm c}w+2w\,\ddc w$, and $\ddc(w^2\Theta)=\ddc(w^2)\wedge\Theta$ because $\Theta$ is closed. Hence
\[
 \mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta
       =\tfrac12\ddc(w^2\Theta)-w\,\ddc w\wedge\Theta.
\]
Apply this identity to $w=v_l-v_m$ and integrate against $\sigma$:
\[
 \int\sigma\,\mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta
 =\tfrac12\int(v_l-v_m)^2\ddc\sigma\wedge\Theta
  -\int\sigma(v_l-v_m)\,\ddc(v_l-v_m)\wedge\Theta.
\]
Both terms on the right tend to zero as $l,m\to\infty$. For the first, $|v_l-v_m|\leq v_N-v$ for $l,m\geq N$, and $v_N-v\to0$ pointwise and boundedly; since $\ddc\sigma\wedge\Theta$ is a signed measure with total variation bounded by a multiple of $\mu$ on $\Supp\sigma$, dominated convergence applies. For the second, expand it into the four integrals $\int\sigma v_p\,\ddc v_q\wedge\Theta$ with $p,q\in\{l,m\}$, taken with the signs $+,-,-,+$. Given indices $l_k,m_k\to\infty$, pass to a subsequence along which both are strictly increasing; then $v_{l_k}$ and $v_{m_k}$ decrease to $v$, and the weighted form of \cref{thm:bt-decreasing-continuity}, with the base current $\Theta$, shows that each of the four integrals tends to $\int\sigma v\,\ddc v\wedge\Theta$. Their signed sum therefore tends to zero, and since this holds for a subsequence of every such pair of index sequences, the second term tends to zero as $l,m\to\infty$. Consequently $H^{1/2}\partial v_l$ is a Cauchy sequence in $L^2(K,\mu)$; let $\Gamma$ be its limit.

\emph{Step 2. The first derivative.}
Let $\eta$ be a smooth real one-form with support in $K$, with $(1,0)$-part $\sum_ke_k\,\mathrm dz_k$. By the formula preceding the lemma,
\[
 \int\eta\wedge\mathrm dv_l\wedge\Theta
 =4\pi\int\Img\bigl\langle H^{1/2}e,H^{1/2}\partial v_l\bigr\rangle\,\mathrm d\mu
 \longrightarrow4\pi\int\Img\bigl\langle H^{1/2}e,\Gamma\bigr\rangle\,\mathrm d\mu.
\]
On the other hand, $\mathrm dv_l\wedge\Theta=\mathrm d(v_l\Theta)$ since $\Theta$ is closed, and $v_l\Theta\to v\Theta$ as currents by dominated convergence on the coefficient measures; hence $\mathrm d(v_l\Theta)\to\mathrm d(v\Theta)$. Therefore $\int\eta\wedge\mathrm d(v\Theta)$ equals the last integral. Since $H^{1/2}e$ is bounded and $\Gamma\in L^2(K,\mu)$, this shows that $\mathrm d(v\Theta)$ has measure coefficients, absolutely continuous with respect to $\mu$, and that for a bounded Borel function $a$,
\[
 \int a\eta\wedge\mathrm dv\wedge\Theta=4\pi\int a\Img\bigl\langle H^{1/2}e,\Gamma\bigr\rangle\,\mathrm d\mu.
\]

\emph{Step 3. The gradient measure.}
Since $H^{1/2}\partial v_l\to\Gamma$ in $L^2(K,\mu)$, the densities $|H^{1/2}\partial v_l|^2$ converge to $|\Gamma|^2$ in $L^1(K,\mu)$, so the measures $\mathrm dv_l\wedge\mathrm d^{\mathrm c}v_l\wedge\Theta$ converge weakly near $K$ to the positive measure $|\Gamma|^2\mu$. On the other hand, by the identity of Step 1,
\[
 \mathrm dv_l\wedge\mathrm d^{\mathrm c}v_l\wedge\Theta
 =\tfrac12\ddc(v_l^2\Theta)-v_l\,\ddc v_l\wedge\Theta
 \longrightarrow\tfrac12\ddc(v^2\Theta)-v\,\ddc v\wedge\Theta:
\]
the first term converges by dominated convergence on the coefficient measures of $v_l^2\Theta$, and the second by \cref{thm:bt-decreasing-continuity}. Hence the right side of \eqref{eq:bounded-current-gradient-measure} is the positive measure $|\Gamma|^2\mu$ near $K$. Since this right side does not involve the approximations, neither does the limit.

\emph{Step 4. The Cauchy--Schwarz inequality.}
Let $\eta$, $\tau$ and $a$ be as in the statement, and apply Steps 1--3 with a compact set $K$ containing $\Supp\tau$. For smooth $w$, the formula for $\eta\wedge\mathrm dw\wedge\Theta$ and the Cauchy--Schwarz inequality give
\[
 \left|\int a\eta\wedge\mathrm dw\wedge\Theta\right|^2
 \leq16\pi^2\int_{\Supp\eta}|a|^2\bigl|H^{1/2}e\bigr|^2\,\mathrm d\mu
      \int_{\Supp\eta}\bigl|H^{1/2}\partial w\bigr|^2\,\mathrm d\mu.
\]
Since $H$ and $e$ are bounded on $\Supp\eta$, we have $16\pi^2|H^{1/2}e|^2\leq C_\eta$ there for a suitable constant $C_\eta$, and the second integral is at most $\int\tau\,\mathrm dw\wedge\mathrm d^{\mathrm c}w\wedge\Theta$, because $\tau\geq0$ and $\tau=1$ on $\Supp\eta$. Applying this to $w=v_l$ and letting $l\to\infty$, using Steps 2 and 3, proves \eqref{eq:current-gradient-cauchy-schwarz}.

\emph{Step 5. The chain rule.}
The functions $v_l$ and $v$ take values in a compact interval $J$ near $K$. Since $f(v_l)\to f(v)$ boundedly, $f(v_l)\Theta\to f(v)\Theta$ by dominated convergence, and hence $\mathrm d(f(v_l)\Theta)\to\mathrm d(f(v)\Theta)$. By the classical chain rule, $\mathrm d(f(v_l)\Theta)=f'(v_l)\,\mathrm dv_l\wedge\Theta$. Moreover $f'(v_l)H^{1/2}\partial v_l\to f'(v)\Gamma$ in $L^2(K,\mu)$: the difference is
\[
 f'(v_l)\bigl(H^{1/2}\partial v_l-\Gamma\bigr)+\bigl(f'(v_l)-f'(v)\bigr)\Gamma,
\]
where the first term tends to zero because $f'$ is bounded on $J$, and the second by dominated convergence. Now let $\eta$ be as in Step 2. The formula preceding the lemma, applied to the smooth form $f'(v_l)\eta$, and Step 2 with $a=f'(v)$ give
\[
 \int\eta\wedge f'(v_l)\,\mathrm dv_l\wedge\Theta
 =4\pi\int\Img\bigl\langle H^{1/2}e,f'(v_l)H^{1/2}\partial v_l\bigr\rangle\,\mathrm d\mu
 \longrightarrow\int\eta\wedge f'(v)\,\mathrm dv\wedge\Theta.
\]
Since the left side also tends to $\int\eta\wedge\mathrm d(f(v)\Theta)$, the currents $\mathrm d(f(v)\Theta)$ and $f'(v)\,\mathrm dv\wedge\Theta$ agree on forms supported in $K$. As $K$ was arbitrary, this proves \eqref{eq:bounded-current-chain-rule}.
\end{proof}
