\subsection{Extending strictly positive potentials}
\label{sec:strict-positive-potential-extension}
We keep the compact K\"ahler manifold $(X,\omega)$ and the submanifold $Y\subset X$, but now we extend potentials instead of sections. Let $\varphi\in\PSH(Y,\omega|_Y)$ be such that $\omega|_Y+\ddc\varphi$ is a K\"ahler current on $Y$, that is, dominates a positive multiple of $\omega|_Y$. We look for $\widetilde\varphi\in\PSH(X,\omega)$ that restricts exactly to $\varphi$ and for which $\omega+\ddc\widetilde\varphi$ is again a K\"ahler current (\cref{thm:holder-positive-potential-extension}). Near $Y$, local extensions are obtained by pulling $\varphi$ back by a holomorphic projection onto $Y$ and adding a large multiple of a function vanishing to second order along $Y$. The difficulty is to glue them, since two projections give different pullbacks. We glue by a maximum after lowering each local extension by a cutoff penalty near the edge of its chart. The maximum is then $\omega$-plurisubharmonic near $Y$ and keeps the prescribed values on $Y$, provided that near every point of $Y$ on the edge of a chart the penalized extension from that chart is dominated by the extension from another chart. This comparison is obtained by adding to $\mathrm e^\varphi$ the exponential of a small logarithmic pole along $Y$: when $\mathrm e^\varphi$ is H\"older continuous, the difference between two pullbacks of $\mathrm e^\varphi$ is small compared with this exponential near $Y$. By the following elementary lemma, the logarithm of such a sum is again quasi-plurisubharmonic.

\begin{lemma}\label{lem:background-logarithmic-sum}
Let $U$ be an open subset of a complex manifold, let $\theta$ be a smooth closed real $(1,1)$-form on $U$, and let $a,b\in\PSH(U,\theta)$. Then
\[
 \log(\mathrm e^a+\mathrm e^b)\in\PSH(U,\theta),
\]
where $\mathrm e^{-\infty}=0$ and $\log0=-\infty$, so that this function equals $-\infty$ precisely at the common poles of $a$ and $b$. Moreover
\[
 \max\{a,b\}\leq\log(\mathrm e^a+\mathrm e^b)\leq\max\{a,b\}+\log2.
\]
\end{lemma}
\begin{proof}
Both assertions are local. Let $B\Subset U$ be a coordinate ball, let $s$ be a smooth real function on $B$ with $\ddc s=\theta$ (\cref{lem:smooth-potential-on-ball}), and let $B'\Subset B$ be a smaller concentric ball. Then $a+s$ and $b+s$ are plurisubharmonic on $B$, and
\[
 \log(\mathrm e^a+\mathrm e^b)+s=H(a+s,b+s),\qquad H(x,y)=\log(\mathrm e^x+\mathrm e^y).
\]
The function $H$ is smooth on $\mathbb R^2$ and increasing in each variable: its partial derivatives are $p$ and $1-p$, where $p=\mathrm e^x/(\mathrm e^x+\mathrm e^y)$, and its Hessian matrix is $p(1-p)\left(\begin{smallmatrix}1&-1\\-1&1\end{smallmatrix}\right)$, so $H$ is also convex. If $f,g$ are smooth plurisubharmonic functions, the chain rule gives
\[
 \ddc H(f,g)=p\,\ddc f+(1-p)\,\ddc g
      +p(1-p)\,\frac{\mathrm i}{2\pi}\,\partial(f-g)\wedge\bar\partial(f-g)\geq0,
\]
with $p$ evaluated at $(f,g)$; hence $H(f,g)$ is plurisubharmonic. In general, let $f_k\downarrow a+s$ and $g_k\downarrow b+s$ be the smooth plurisubharmonic functions on $B'$ obtained by convolution with radial mollifiers (\cref{thm:psh-distribution-representative}). Since $H$ extends continuously to $[-\infty,\infty)^2$ and remains increasing in each variable, the functions $H(f_k,g_k)$ decrease to $H(a+s,b+s)$. This limit is at least $a+s$, so it is not identically $-\infty$, and it is plurisubharmonic by \cref{thm:psh-decreasing-nets}. Consequently $\log(\mathrm e^a+\mathrm e^b)=H(a+s,b+s)-s$ belongs to $\PSH(B',\theta)$; since every point of $U$ is the center of such a ball $B'$, the first assertion follows.

For the inequalities fix a point and suppose that $a\geq b$ there; the other case is symmetric. If $a=-\infty$, then $b=-\infty$ as well, and all three expressions equal $-\infty$. Otherwise $\log(\mathrm e^a+\mathrm e^b)=a+\log(1+\mathrm e^{b-a})$, and $0\leq\log(1+\mathrm e^{b-a})\leq\log2$.
\end{proof}
\index{quasi-plurisubharmonic function!logarithm of a sum of exponentials}

\begin{theorem}\label{thm:holder-positive-potential-extension}
Let $Y\subset X$ be a nonempty connected closed complex submanifold, and let $\varphi\in\PSH(Y,\omega|_Y)$ satisfy
\[
 \omega|_Y+\ddc\varphi\geq\delta\,\omega|_Y
\]
for some $\delta>0$. Suppose that $\mathrm e^\varphi$, defined to be zero at the poles of $\varphi$, is H\"older continuous on $Y$. Then there are $\widetilde\varphi\in\PSH(X,\omega)$ and $\varepsilon>0$ such that
\[
 \widetilde\varphi|_Y=\varphi,\qquad
 \omega+\ddc\widetilde\varphi\geq\varepsilon\omega.
\]
Moreover $\widetilde\varphi$ is continuous on $X\setminus Y$, and $\mathrm e^{\widetilde\varphi}$ is H\"older continuous on $X$. If $\varphi$ has rational analytic singularities, then so does $\widetilde\varphi$.
\end{theorem}
\begin{proof}
If $Y=X$, take $\widetilde\varphi=\varphi$ and $\varepsilon=\delta$. If $Y$ is a point $y$, then $\varphi(y)$ is a real number, since quasi-plurisubharmonic functions on a point are real-valued, and the constant function $\widetilde\varphi=\varphi(y)$ has all the stated properties, with $\varepsilon=1$. We may therefore assume that $0<\dim Y<n$. Fix $\varepsilon$ with $0<\varepsilon<\min\{\delta/4,1/4\}$, and let $\alpha\in(0,1]$ be a H\"older exponent of $\mathrm e^\varphi$. Distances are taken with respect to the Riemannian metric of $\omega$, and $r(x)$ denotes the distance from $x\in X$ to $Y$.

\emph{Step 1. A small normal pole.} Let $R=R_Y$ be the function of \cref{lem:global-normal-pole}, and let $\kappa\geq0$ be the constant called $b$ there, so that $\ddc\log R\geq-\kappa\omega$. Choose a rational number $c$ with $0<c<\alpha$ and $c\kappa/2\leq1-4\varepsilon$; this is possible because $1-4\varepsilon>0$. Put $F=(c/2)\log R$. Then
\[
 \omega+\ddc F\geq\Bigl(1-\frac{c\kappa}2\Bigr)\omega\geq4\varepsilon\omega.
\]
The function $F$ is smooth on $X\setminus Y$, equals $-\infty$ on $Y$, and has rational analytic singularities, since $\log R$ has analytic singularities of type $(1,\mathcal I_Y)$ by \cref{lem:global-normal-pole} and $c/2$ is rational. Since $R$ vanishes exactly on $Y$ and its Hessian is positive definite in the normal directions, Taylor's formula in local coordinates $(z',w)$ with $Y=\{w=0\}$ shows that $R$ and $r^2$ are both comparable to $|w|^2$ near $Y$. As $Y$ is compact, there is therefore a constant $C\geq1$ such that, near $Y$,
\begin{equation}\label{eq:normal-exponential-lower-bound}
 C^{-1}r^2\leq R\leq Cr^2,\qquad \mathrm e^F=R^{c/2}\geq C^{-1}r^c.
\end{equation}
Moreover $\mathrm e^F=R^{c/2}$ is H\"older continuous on $X$, because $R$ is smooth and $t\mapsto t^{c/2}$ is H\"older continuous on $[0,\max_XR]$.

\emph{Step 2. Local extensions.} Cover $Y$ by finitely many coordinate charts $W_j$ of the following form: in coordinates $(z',w)\in\mathbb C^m\times\mathbb C^{n-m}$, where $m=\dim Y$, the chart is a product $W_j=B'_j\times B''_j$ of balls and $Y\cap W_j=B'_j\times\{0\}$. Let $\pi_j\colon W_j\to Y$, $\pi_j(z',w)=(z',0)$, be the holomorphic projection. We may assume that each $W_j$ is relatively compact in a larger product chart of the same form, to which $\pi_j$ extends. Below we shall shrink the normal factors $B''_j$ several times; this does not change $Y\cap W_j$, so the charts keep covering $Y$. We claim that there is $A>0$ such that, after such a shrinking,
\[
 a_j=\varphi\circ\pi_j+AR,\qquad b_j=F+AR
\]
belong to $\PSH(W_j,(1-2\varepsilon)\omega)$ for every $j$.

Since $4\varepsilon\leq\delta$, the hypothesis gives $\ddc\varphi\geq-(1-\delta)\omega|_Y\geq-(1-4\varepsilon)\omega|_Y$. Holomorphic pullback preserves plurisubharmonicity (\cref{prop:psh-holomorphic-pullback}); applying it to the sum of $\varphi$ and a smooth potential of $(1-4\varepsilon)\omega|_Y$ on $Y\cap W_j$ (\cref{lem:smooth-potential-on-ball}), we obtain $\ddc(\varphi\circ\pi_j)\geq-(1-4\varepsilon)\pi_j^*(\omega|_Y)$. Here $\varphi\circ\pi_j$ is not identically $-\infty$, because $\pi_j$ maps $W_j$ onto the open set $Y\cap W_j$, on which $\varphi$ is locally integrable. For $a_j$ it therefore suffices that the smooth form
\[
 Q_j=(1-2\varepsilon)\omega-(1-4\varepsilon)\pi_j^*(\omega|_Y)+A\ddc R
\]
be positive on $W_j$. At a point of $Y$, split the tangent space into the tangential directions $\partial/\partial z'$ and the normal directions $\partial/\partial w$, and write the coefficient matrices of real $(1,1)$-forms in the corresponding blocks. Since $\pi_j$ is the identity on $Y$ and its differential vanishes on the normal directions, $\pi_j^*(\omega|_Y)$ has zero mixed and normal blocks, and the tangential block of $Q_j$ is $2\varepsilon\,\omega|_Y$, which is positive definite. The function $R\geq0$ vanishes on $Y$, so its first derivatives vanish there as well; differentiating the identities $R(z',0)=0$ and $\partial R/\partial\bar w(z',0)=0$ with respect to $z'$ shows that $\ddc R$ has zero tangential and mixed blocks along $Y$, while its normal block $N_R$ is positive definite because the Hessian of $R$ is positive definite in the normal directions. Let $\Omega_M$ and $\Omega_N$ denote the mixed and normal blocks of $(1-2\varepsilon)\omega$. By the Schur complement criterion, $Q_j$ is positive definite at a point of $Y$ if and only if
\[
 \Omega_N+AN_R-\Omega_M^*\bigl(2\varepsilon\,\omega|_Y\bigr)^{-1}\Omega_M
\]
is positive definite there. All blocks depend continuously on the point, $N_R$ is positive definite, and the sets $Y\cap\overline{W_j}$ are compact and finite in number; hence one $A$ achieves this at every point of every $Y\cap\overline{W_j}$. Fix such an $A$. Since the coefficients of $Q_j$ are continuous and positive definiteness is an open condition, $Q_j$ remains positive definite on a neighborhood of $Y\cap\overline{W_j}$, and we shrink the normal factor $B''_j$ so that $W_j$ lies in this neighborhood. This proves the claim for $a_j$. Along $Y$ the form $\ddc R$ is positive semidefinite, so after shrinking the normal factors once more we also have $A\ddc R\geq-\varepsilon\omega$ on every $W_j$. Together with $\ddc F\geq-(1-4\varepsilon)\omega$ this gives
\[
 (1-2\varepsilon)\omega+\ddc b_j\geq(1-2\varepsilon)\omega-(1-4\varepsilon)\omega-\varepsilon\omega=\varepsilon\omega\geq0,
\]
which proves the claim for $b_j$.

By \cref{lem:background-logarithmic-sum}, applied with $\theta=(1-2\varepsilon)\omega$, the functions
\[
 \varphi_j=\log(\mathrm e^{a_j}+\mathrm e^{b_j})
          =\log(\mathrm e^{\varphi\circ\pi_j}+\mathrm e^F)+AR
\]
belong to $\PSH(W_j,(1-2\varepsilon)\omega)$. On $Y$ we have $F=-\infty$, $R=0$ and $\varphi\circ\pi_j=\varphi$, so $\varphi_j=\varphi$ on $Y\cap W_j$, also at the poles of $\varphi$. Off $Y$ the function $\mathrm e^F$ is positive and continuous and $\mathrm e^{\varphi\circ\pi_j}$ is continuous, so $\varphi_j$ is continuous on $W_j\setminus Y$. Finally $\mathrm e^{\varphi_j}=(\mathrm e^\varphi\circ\pi_j+\mathrm e^F)\,\mathrm e^{AR}$ is locally H\"older continuous on $W_j$.

\emph{Step 3. Comparison on overlaps.} We claim that for every $p\in Y\cap W_i\cap W_j$ and every $\eta>0$ there is a neighborhood of $p$ on which
\begin{equation}\label{eq:holder-extension-overlap}
 \varphi_i\leq\varphi_j+\eta.
\end{equation}
At points of $Y$ it holds because $\varphi_i=\varphi=\varphi_j$ there, including at the poles of $\varphi$; so we only have to consider points off $Y$. Work in a coordinate chart of $X$ around $p$, in which the Riemannian distance is comparable to the Euclidean distance $|\cdot|$. Both projections are smooth near $p$ and restrict to the identity on $Y$. If $x$ is close to $p$ and $y\in Y$ is a point with $\operatorname{dist}(x,y)=r(x)$, then $y$ is also close to $p$ and $\pi_i(y)=y=\pi_j(y)$, so the Lipschitz bounds for $\pi_i$ and $\pi_j$ near $p$ give $|\pi_i(x)-\pi_j(x)|\leq C_1r(x)$. H\"older continuity of $\mathrm e^\varphi$ near $p$ therefore yields
\[
 \bigl|\mathrm e^{\varphi\circ\pi_i}-\mathrm e^{\varphi\circ\pi_j}\bigr|\leq C_2r^\alpha
\]
near $p$. Since $c<\alpha$, the lower bound in \eqref{eq:normal-exponential-lower-bound} gives $C_2r^\alpha\leq(\mathrm e^\eta-1)C^{-1}r^c\leq(\mathrm e^\eta-1)\mathrm e^F$ at all points sufficiently close to $p$. There
\[
 \mathrm e^{\varphi\circ\pi_i}+\mathrm e^F
 \leq\mathrm e^{\varphi\circ\pi_j}+\mathrm e^\eta\mathrm e^F
 \leq\mathrm e^\eta\bigl(\mathrm e^{\varphi\circ\pi_j}+\mathrm e^F\bigr).
\]
Taking logarithms and adding $AR$ to both sides gives \eqref{eq:holder-extension-overlap}.

\emph{Step 4. Gluing near $Y$.} Choose open sets $W'_j\Subset U_j\Subset T_j\Subset W_j$ such that the $W'_j$ still cover $Y$, and smooth functions $f_j\geq0$ with compact support in $W_j$ such that $f_j=0$ on a neighborhood of $\overline{W'_j}$ and $f_j\geq1$ on a neighborhood of $\overline{T_j}\setminus U_j$. Fix $\eta>0$ so small that $\eta\ddc f_j\leq\varepsilon\omega$ for every $j$. Since $(1-\varepsilon)\omega+\ddc\varphi_j\geq\varepsilon\omega$, the functions
\[
 \widehat\varphi_j=\varphi_j-\eta f_j
\]
belong to $\PSH(W_j,(1-\varepsilon)\omega)$. On $W=\bigcup_jT_j$ define
\[
 \psi(x)=\max\{\widehat\varphi_j(x):x\in T_j\}.
\]
The set of indices in this maximum changes with $x$. We show that near $Y$ the maximum is nevertheless taken over a fixed set of indices. Let $p\in Y$, and choose $j$ with $p\in W'_j$; then $f_j=0$ near $p$, and $j$ occurs in the maximum near $p$. Let $l$ be an index with $p\in\partial T_l$. Then $p\in\overline{T_l}\setminus U_l$, so $f_l\geq1$ near $p$. Since $p\in Y\cap W_l\cap W_j$, the inequality \eqref{eq:holder-extension-overlap} with $\eta/2$ in place of $\eta$ gives, near $p$,
\[
 \widehat\varphi_l\leq\varphi_l-\eta\leq\varphi_j-\eta/2\leq\widehat\varphi_j.
\]
These are inequalities between functions with values in $[-\infty,\infty)$, valid also at common poles. Hence there is a neighborhood $O_p$ of $p$ on which the indices $l$ with $p\in\partial T_l$ never affect the maximum, the indices with $p\notin\overline{T_l}$ do not occur, and the indices with $p\in T_l$ occur at every point. On $O_p$ the function $\psi$ is therefore the maximum of the $\widehat\varphi_l$ over the fixed finite set of indices $l$ with $p\in T_l$.

Let $W_0=\bigcup_{p\in Y}O_p$. A finite maximum of functions in $\PSH(O_p,(1-\varepsilon)\omega)$ belongs to this class, as one sees by adding a local potential of $(1-\varepsilon)\omega$ and applying \cref{prop:psh-max-gluing}. Hence $\psi\in\PSH(W_0,(1-\varepsilon)\omega)$; moreover $\psi$ is continuous on $W_0\setminus Y$, and $\mathrm e^\psi$ is locally H\"older continuous. On $Y$ every $\widehat\varphi_l$ restricts to $\varphi-\eta f_l\leq\varphi$, while $\widehat\varphi_j=\varphi$ near $p$ for the index $j$ chosen above. Consequently $\psi|_Y=\varphi$.

\emph{Step 5. The global extension.} Choose an open neighborhood $G$ of $Y$ with $\overline G\subset W_0$. Its boundary $\partial G$ is compact and disjoint from $Y$, and both $\psi$ and $F$ are finite and continuous near $\partial G$. Choose a constant $M$ such that $F+M>\psi$ on a neighborhood of $\partial G$, and set
\[
 \widetilde\varphi=
 \begin{cases}
  \max\{\psi,F+M\}&\text{on }G,\\
  F+M&\text{on }X\setminus G.
 \end{cases}
\]
The two expressions agree near $\partial G$. The function $F+M$ belongs to $\PSH(X,(1-4\varepsilon)\omega)\subset\PSH(X,(1-\varepsilon)\omega)$, and $\psi\in\PSH(W_0,(1-\varepsilon)\omega)$. The gluing statement of \cref{prop:psh-max-gluing}, applied after adding local potentials of $(1-\varepsilon)\omega$, therefore gives $\widetilde\varphi\in\PSH(X,(1-\varepsilon)\omega)$, that is, $\omega+\ddc\widetilde\varphi\geq\varepsilon\omega$. Since $F=-\infty$ on $Y$, we have $\widetilde\varphi|_Y=\psi|_Y=\varphi$. The function $\widetilde\varphi$ is continuous on $X\setminus Y$ because $\psi$ and $F$ are. Its exponential, which equals $\max\{\mathrm e^\psi,\mathrm e^M\mathrm e^F\}$ on $G$ and $\mathrm e^M\mathrm e^F$ elsewhere, is locally H\"older continuous. On the compact manifold $X$ this implies H\"older continuity, with the smallest of finitely many local exponents: two points closer than a Lebesgue number of a finite cover lie in one set of the cover, and for points farther apart it suffices that $\mathrm e^{\widetilde\varphi}$ is bounded.

\emph{Step 6. Analytic singularities.} Finally suppose that $\varphi$ has rational analytic singularities (\cref{def:rational-analytic-singularities}). By \cref{prop:rational-analytic-operations}, so does the holomorphic pullback $\varphi\circ\pi_j$, and the function $F$ has rational analytic singularities by Step 1. By the same proposition, $\max\{\varphi\circ\pi_j,F\}+AR$ has rational analytic singularities, and by \cref{lem:background-logarithmic-sum},
\[
 \max\{\varphi\circ\pi_j,F\}+AR\leq\varphi_j\leq\max\{\varphi\circ\pi_j,F\}+AR+\log2.
\]
Hence a local presentation \eqref{eq:rational-analytic-singularities} of the maximum yields one of $\varphi_j$, with the same holomorphic functions and a bounded remainder changed by at most $\log2$; so $\varphi_j$ has rational analytic singularities. Subtracting the smooth function $\eta f_j$ and taking the locally fixed finite maxima of Step 4 preserve this property, so $\psi$ has rational analytic singularities on $W_0$. On $G$, the function $\widetilde\varphi$ is the maximum of $\psi$ and $F+M$, and near $X\setminus G$ it equals $F+M$, which is smooth there. Therefore $\widetilde\varphi$ has rational analytic singularities on $X$.
\end{proof}
