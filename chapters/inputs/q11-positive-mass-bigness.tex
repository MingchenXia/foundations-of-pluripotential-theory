\subsection{Positive mass and bigness}
\label{sec:positive-mass-bigness}

So far the background class has been big, and big classes have positive volume by \cref{prop:positive-mixed-mass-bound}. The equations solved above give the converse for pseudo-effective classes. In this subsection $\theta$ denotes an arbitrary smooth closed real $(1,1)$-form on $X$; its class need not be big, and no model potential is fixed. Recall from \cref{sec:positive-currents} that weak and strong positivity coincide for real $(n-1,n-1)$-forms, so that we speak simply of positive forms, and that such a form is strictly positive if at every point it lies in the interior of the cone of positive forms. For instance, $\omega^{n-1}$ is strictly positive: at a point $x$, choose linear coordinates on the tangent space in which $\omega(x)$ is the standard form $\beta$; then $\omega(x)^{n-1}=\beta^{n-1}$ is an interior point of the cone by \cref{prop:positivity-cones}, and complex-linear isomorphisms preserve positivity. For real $(n-1,n-1)$-forms we write $\eta\leq\tau$ if $\tau-\eta$ is positive.

Let $\alpha=\{\theta\}$ be a pseudo-effective class whose volume, in the sense of \cref{def:positive-mixed-intersection-number}, is positive. The classes $\alpha+\varepsilon\{\omega\}$, $\varepsilon>0$, are big, and in each of them we solve the Monge--Amp\`ere equation whose right-hand side is a fixed smooth measure rescaled to the total mass $\operatorname{vol}(\alpha+\varepsilon\{\omega\})$, which is at least $\operatorname{vol}(\alpha)$. A pointwise inequality between traces, integrated against these solutions, yields lower bounds for the pairings $\int_X\theta\wedge\tau$ with strictly positive $\ddc$-closed $(n-1,n-1)$-forms $\tau$, uniformly in $\varepsilon$. Lamari's duality criterion, which we prove first, turns such bounds into a K\"ahler current in $\alpha$.

\begin{lemma}[Lamari's criterion]\label{lem:kahler-lamari-criterion}
Let $\theta$ be a smooth closed real $(1,1)$-form on $X$. Then $\PSH(X,\theta)\neq\varnothing$ if and only if
\begin{equation}\label{eq:lamari-positive-tests}
 \int_X\theta\wedge\tau\geq0
\end{equation}
for every smooth strictly positive real $(n-1,n-1)$-form $\tau$ on $X$ with $\ddc\tau=0$. Consequently, the class $\{\theta\}$ is big if and only if there is $\delta>0$ such that every such form $\tau$ satisfies
\begin{equation}\label{eq:lamari-strict-tests}
 \int_X\theta\wedge\tau\geq\delta\int_X\omega\wedge\tau.
\end{equation}
\end{lemma}
\begin{proof}
\emph{Necessity.} If $u\in\PSH(X,\theta)$, then $T=\theta+\ddc u\geq0$. For $\tau$ as in the statement, the definition of $\ddc u$ as a current gives $\int_X\ddc u\wedge\tau=\int_Xu\,\ddc\tau=0$, hence $\int_X\theta\wedge\tau=\int_XT\wedge\tau$, which is nonnegative by the definition of a positive current, since $\tau$ is positive, hence strongly positive.

\emph{Sufficiency: a positive extension.} Conversely, assume \eqref{eq:lamari-positive-tests} for all forms $\tau$ as in the statement. Let $F$ be the real vector space of smooth real $(n-1,n-1)$-forms on $X$, let $S=\{\tau\in F:\ddc\tau=0\}$, and put $e=\omega^{n-1}$, with $e=1$ if $n=1$. Since $\omega$ is closed, $e\in S$, and $e$ is strictly positive. Define $\ell_0(\tau)=\int_X\theta\wedge\tau$ for $\tau\in S$. By hypothesis $\ell_0(\tau)\geq0$ when $\tau\in S$ is strictly positive. If $\tau\in S$ is merely positive, then $\tau+te\in S$ is strictly positive for $t>0$: at each point $x$, if the cone of positive forms contains a ball, for some norm, about $e(x)$, then it contains the ball of $t$ times the radius about $\tau(x)+te(x)$, since it is stable under sums and positive multiples. Hence $\ell_0(\tau)+t\ell_0(e)\geq0$; letting $t\downarrow0$ gives $\ell_0(\tau)\geq0$.

We next compare the forms in $F$ with $e$. Fix finitely many holomorphic charts $U_1,\ldots,U_N$ and compact sets $K_a\subset U_a$ with $\bigcup_aK_a=X$. In $U_a$ write $\eta\in F$ as $\eta=\sum_{I,J}\eta^a_{IJ}\,\mathrm dz_I\wedge\mathrm d\bar z_J$, the sum running over increasing multi-indices of length $n-1$, and put
\[
 \|\eta\|=\max_{1\leq a\leq N}\ \max_{I,J}\ \sup_{K_a}|\eta^a_{IJ}|.
\]
We claim that there is a constant $K>0$ such that
\begin{equation}\label{eq:lamari-uniform-domination}
 -K\|\eta\|e\leq\eta\leq K\|\eta\|e\qquad\text{for every }\eta\in F.
\end{equation}
Let $Z$ be the finite-dimensional real vector space of real $(n-1,n-1)$-forms with constant coefficients on $\mathbb C^n$, normed by the maximum modulus of the coefficients, and let $P\subset Z$ be the closed convex cone of positive forms (\cref{prop:positivity-cones}). For $x\in U_a$, the differential of the chart is a complex-linear isomorphism from the tangent space at $x$ onto $\mathbb C^n$ that maps $\mathrm dz_j$ to the $j$th coordinate form. Since complex-linear isomorphisms preserve positivity, a real $(n-1,n-1)$-form $\eta$ is positive at $x$ if and only if its coefficient form $\eta^a(x)=\sum_{I,J}\eta^a_{IJ}(x)\,\mathrm dz_I\wedge\mathrm d\bar z_J\in Z$ lies in $P$. Because $e$ is strictly positive, the continuous map $x\mapsto e^a(x)$ sends the compact set $K_a$ into the interior of $P$; the complement of this interior is closed and disjoint from the compact image, so it has positive distance $r_a$ from it. Put $r=\frac12\min_ar_a$. If $\eta\in F$ satisfies $\|\eta\|\leq r$ and $x\in K_a$, then $e^a(x)\pm\eta^a(x)$ lies within distance $r<r_a$ of $e^a(x)$, hence in $P$; thus $e\pm\eta$ is positive at $x$. Since the sets $K_a$ cover $X$, the forms $e\pm\eta$ are positive. Applying this to $r\eta/\|\eta\|$ for $\eta\neq0$ proves \eqref{eq:lamari-uniform-domination} with $K=1/r$.

For $\eta\in F$, a form $\tau\in S$ with $\tau\geq\eta$ is called an \emph{$S$-majorant} of $\eta$, and we put
\[
 p(\eta)=\inf\{\ell_0(\tau):\tau\text{ is an $S$-majorant of }\eta\}.
\]
Let $\eta\in F$ and $C=K\|\eta\|$, so that $-Ce\leq\eta\leq Ce$ by \eqref{eq:lamari-uniform-domination}. The form $Ce$ is an $S$-majorant of $\eta$, so $p(\eta)\leq C\ell_0(e)$. If $\tau$ is an $S$-majorant of $\eta$, then $\tau+Ce\geq\eta+Ce\geq0$, so $\tau+Ce\in S$ is positive and $\ell_0(\tau)\geq-C\ell_0(e)$. Thus $p$ is finite. It is sublinear: the sum of $S$-majorants of $\eta_1$ and $\eta_2$ is an $S$-majorant of $\eta_1+\eta_2$, and for $\lambda>0$ the $S$-majorants of $\lambda\eta$ are $\lambda$ times those of $\eta$. If $\eta\in S$, then $\eta$ is its own $S$-majorant, and every $S$-majorant $\tau$ of $\eta$ satisfies $\ell_0(\tau)-\ell_0(\eta)=\ell_0(\tau-\eta)\geq0$ because $\tau-\eta\in S$ is positive. Hence $p=\ell_0$ on $S$.

Next we extend $\ell_0$ to a linear functional on $F$ dominated by $p$; this is the Hahn--Banach theorem\index{Hahn--Banach theorem}, whose proof we recall. Let $\ell$ be a linear functional on a subspace $W\supset S$ which extends $\ell_0$ and satisfies $\ell\leq p$ on $W$, and let $x\in F\setminus W$. For $y,z\in W$,
\[
 \ell(y)+\ell(z)=\ell(y+z)\leq p(y+z)\leq p(y-x)+p(z+x),
\]
so
\[
 \sup_{y\in W}\bigl(\ell(y)-p(y-x)\bigr)
 \leq\inf_{z\in W}\bigl(p(z+x)-\ell(z)\bigr).
\]
Taking $y=z=0$ shows that the supremum is greater than $-\infty$ and the infimum is less than $+\infty$; hence both are finite, and we may choose a real number $a$ between them. Define $\ell(y+tx)=\ell(y)+ta$ for $y\in W$ and $t\in\mathbb R$. For $t>0$, the upper bound for $a$ with $z=y/t$, multiplied by $t$, gives $\ell(y)+ta\leq p(y+tx)$; for $t<0$, the lower bound for $a$ with $y$ replaced by $y/|t|$ gives the same inequality. Thus the extension to $W+\mathbb Rx$ is still dominated by $p$. The union of a chain of such extensions is again such an extension, so Zorn's lemma gives a maximal one, and by the step just proved its domain is $F$. We denote it again by $\ell$.

The functional $\ell$ is positive: if $\eta\in F$ is positive, then $0$ is an $S$-majorant of $-\eta$, so $\ell(-\eta)\leq p(-\eta)\leq0$. It is also bounded: by \eqref{eq:lamari-uniform-domination}, the form $K\|\eta\|e$ is an $S$-majorant of both $\eta$ and $-\eta$, so
\begin{equation}\label{eq:lamari-functional-bound}
 |\ell(\eta)|\leq K\ell_0(e)\|\eta\|\qquad(\eta\in F).
\end{equation}
We now turn $\ell$ into a current. Every smooth form on the compact manifold $X$ is a test form. For a complex form $\varphi$ of degree $2n-2$, let $\varphi'$ be its component of bidegree $(n-1,n-1)$, write $\varphi'=\varphi_1+\mathrm i\varphi_2$ with the real forms $\varphi_1=(\varphi'+\overline{\varphi'})/2$ and $\varphi_2=(\varphi'-\overline{\varphi'})/(2\mathrm i)$ in $F$, and put $T(\varphi)=\ell(\varphi_1)+\mathrm i\ell(\varphi_2)$. This is a complex-linear functional: replacing $\varphi$ by $\mathrm i\varphi$ replaces $(\varphi_1,\varphi_2)$ by $(-\varphi_2,\varphi_1)$. It has the continuity property of a current of degree $2$. Indeed, let $L$ be a compact subset of a chart $U$ with real coordinates $t$, and let $\varphi=\sum_J\varphi_J\,\mathrm dt_J$ be supported in $L$. On the compact set $K_a\cap L$, the coefficients $(\varphi_1)^a_{IJ}$ and $(\varphi_2)^a_{IJ}$ are linear combinations of the functions $\varphi_J$ and $\overline{\varphi_J}$ with coefficients that are smooth on $U\cap U_a$, and they vanish on $K_a\setminus L$. Hence $\|\varphi_1\|+\|\varphi_2\|\leq C_L\sum_Jp_{L,0}(\varphi_J)$ with a constant $C_L$ independent of $\varphi$, and \eqref{eq:lamari-functional-bound} gives $|T(\varphi)|\leq KC_L\ell_0(e)\sum_Jp_{L,0}(\varphi_J)$. By construction $T$ vanishes on forms of bidegree other than $(n-1,n-1)$, so it is a current of bidegree $(1,1)$. It is real, because $\overline\varphi$ has the components $\varphi_1$ and $-\varphi_2$, so that $T(\overline\varphi)=\overline{T(\varphi)}$. It is positive, because a strongly positive test form $\varphi$ of bidegree $(n-1,n-1)$ is real and positive, so that $T(\varphi)=\ell(\varphi)\geq0$. Since $\ell$ extends $\ell_0$, the real $(1,1)$-current $R=T-\theta$ annihilates $S$.

\emph{The current $R$ is $\ddc$-exact.} If $\xi$ is a smooth form of bidegree $(n-2,n-1)$, then $\ddc\partial\xi=\frac{\mathrm i}{2\pi}\partial\bar\partial\partial\xi=-\frac{\mathrm i}{2\pi}\partial\partial\bar\partial\xi=0$, and $\ddc\bar\partial\xi'=\frac{\mathrm i}{2\pi}\partial\bar\partial\bar\partial\xi'=0$ for $\xi'$ of bidegree $(n-1,n-2)$. Since $\ddc$ is a real operator, the conjugate of a $\ddc$-closed complex $(n-1,n-1)$-form $\sigma$ is $\ddc$-closed, so the real forms $(\sigma+\overline\sigma)/2$ and $(\sigma-\overline\sigma)/(2\mathrm i)$ lie in $S$; as $R$ is complex-linear and annihilates $S$, it annihilates $\sigma$. In particular $R(\partial\xi)=R(\bar\partial\xi')=0$. For a smooth $(2n-3)$-form $\gamma$, the $(n-1,n-1)$-component of $\mathrm d\gamma$ is $\partial\gamma^{n-2,n-1}+\bar\partial\gamma^{n-1,n-2}$; since $R$ has bidegree $(1,1)$, it follows that $R(\mathrm d\gamma)=0$. Therefore $\mathrm dR=0$. Moreover $\int_XR\wedge\omega^{n-1}=0$ because $e\in S$. Let $\Lambda$ denote contraction with $\omega$, as in \cref{lem:kahler-trace-star}; the identity $R\wedge\omega^{n-1}=(\Lambda R)\omega^n/n$ shows that the trace $\Lambda R$ has mean zero. By \cref{lem:compact-elliptic-foundations}(2), applied to the Laplacian $L=\mathrm d^*\mathrm d$ of \cref{sec:global-trace-equation}, there is a real distribution $f$ with $Lf=-4\pi\Lambda R$. By \cref{lem:kahler-trace-star}, $\Lambda\ddc f=\Lambda R$, so $R'=R-\ddc f$ is a closed real $(1,1)$-current with zero trace. By the last assertion of \cref{lem:kahler-trace-star}, $R'$ is harmonic, and by \cref{lem:compact-elliptic-foundations}(1) it is smooth. If $n=1$, a $(1,1)$-form with zero trace vanishes, so $R'=0$. If $n\geq2$, \eqref{eq:kahler-trace-star} gives $*R'=-R'\wedge\omega^{n-2}/(n-2)!$, a smooth closed $(n-1,n-1)$-form; hence $*R'\in S$. Since $R$ annihilates $S$ and $\ddc{*R'}=0$,
\[
 \|R'\|_{L^2}^2
   =\int_XR'\wedge *R'
   =R(*R')-\int_Xf\,\ddc{*R'}=0.
\]
Thus $R'=0$ and $R=\ddc f$ in every dimension.

\emph{Conclusion.} The current $T=\theta+\ddc f$ is closed and positive, and it lies in the class $\{\theta\}$, since it differs from $\theta$ by $\ddc$ of a distribution. By \cref{cor:current-theta-psh-potential}, there is $u\in\PSH(X,\theta)$ with $\theta+\ddc u=T$. This proves sufficiency.

For the last assertion, $\{\theta\}$ is big exactly when $\PSH(X,\theta-\delta\omega)\neq\varnothing$ for some $\delta>0$. Indeed, if $T\geq\delta\omega$ is a K\"ahler current in $\{\theta\}$ (\cref{def:cohomology-positivity-cones}), then $T-\delta\omega$ is a closed positive current in the class $\{\theta-\delta\omega\}$, so $T-\delta\omega=\theta-\delta\omega+\ddc u$ with $u\in\PSH(X,\theta-\delta\omega)$, by \cref{cor:current-theta-psh-potential}. Conversely, if $u\in\PSH(X,\theta-\delta\omega)$, then $\theta+\ddc u\geq\delta\omega$ is a K\"ahler current in $\{\theta\}$. Applying the first part to $\theta-\delta\omega$ shows that this holds if and only if \eqref{eq:lamari-strict-tests} holds for that $\delta$.
\end{proof}

The proof of the bigness criterion rests on the following lower bound for the product of two trace masses of a current with prescribed Monge--Amp\`ere measure.

\begin{lemma}\label{lem:prescribed-density-mixed-trace}
Let $A$ be a closed subset of $X$ of Lebesgue measure zero in every chart, and let $S$ be a closed positive $(1,1)$-current on $X$ whose local potentials are locally bounded on $X\setminus A$. Let $\tau$ be a smooth strictly positive real $(n-1,n-1)$-form, and put $\nu=\omega\wedge\tau$. If
\[
 \langle S^n\rangle=m\nu\qquad\text{for some }m\geq0,
\]
then
\begin{equation}\label{eq:prescribed-density-mixed-trace}
 \left(\int_X S\wedge\tau\right)
 \left(\int_X\langle S^{n-1}\rangle\wedge\omega\right)
               \geq\frac mn\left(\int_X\nu\right)^2.
\end{equation}
For $n=1$ the second factor is $\int_X\omega$.
\end{lemma}
\begin{proof}
\emph{Step 1: a pointwise inequality.} Let $\sigma$ be a smooth positive definite real $(1,1)$-form. At a given point, choose linear coordinates on the tangent space that diagonalize the Hermitian matrices of $\omega$ and $\sigma$ simultaneously, so that $\omega=\sum_j\gamma_j$ and $\sigma=\sum_j\lambda_j\gamma_j$ with $\lambda_j>0$, where $\gamma_j=\mathrm i\,\mathrm dz_j\wedge\mathrm d\bar z_j$, and put $\mathrm dV=\gamma_1\wedge\cdots\wedge\gamma_n$. Since $\tau$ is strictly positive (\cref{sec:positive-currents}), the form $\tau-c\,\omega^{n-1}$ is positive at this point for some $c>0$. The product of the strongly positive form $\gamma_j$ with this positive form is a nonnegative multiple of $\mathrm dV$ (\cref{prop:positivity-cones}), and $\gamma_j\wedge\omega^{n-1}=(n-1)!\,\mathrm dV$, so $\gamma_j\wedge\tau\geq c(n-1)!\,\mathrm dV$; thus $\gamma_j\wedge\tau=t_j\,\mathrm dV$ with $t_j>0$. Expanding the products, we obtain
\[
 \frac{\sigma\wedge\tau}{\omega\wedge\tau}
       =\frac{\sum_j\lambda_jt_j}{\sum_jt_j},\qquad
 \frac{\sigma^{n-1}\wedge\omega}{\sigma^n}
       =\frac1n\sum_j\lambda_j^{-1}.
\]
Since $\lambda_j\sum_k\lambda_k^{-1}\geq1$ for every $j$, the product of these two ratios is at least $1/n$. If $\sigma$ is only semipositive, we apply this to $\sigma+t\omega$ and let $t\downarrow0$. Thus, for every smooth semipositive real $(1,1)$-form $\sigma$, as densities with respect to any fixed smooth volume form,
\begin{equation}\label{eq:pointwise-mixed-trace}
 (\sigma\wedge\tau)(\sigma^{n-1}\wedge\omega)
                     \geq\frac1n\sigma^n\nu.
\end{equation}

\emph{Step 2: regularization on a ball.} Since the local potentials of $S$ are locally bounded on the open set $X\setminus A$, every point of $X\setminus A$ is the center of a coordinate ball $B\Subset X\setminus A$ on which $S=\ddc v$ with $v$ plurisubharmonic and bounded. Fix such a ball, put $\eta=\ddc|z|^2$ as in \cref{sec:bt-density-lower-bounds}, and write $\nu=g\eta^n$ with $g$ smooth and positive on $B$. Since $v$ is bounded on $B$, the non-pluripolar powers of $S$ on $B$ are the Bedford--Taylor powers of $v$ (see the remark following \cref{def:nonpluripolar-product}); in particular $(\ddc v)^n=mg\eta^n$. Let $v_\varepsilon=v*\rho_\varepsilon$ be the regularizations of $v$ by the radial mollifiers of \cref{sec:convolution}. On each ball $B'$ concentric with $B$ and of smaller radius, the functions $v_\varepsilon$ are defined for $\varepsilon$ smaller than the difference of the radii, and by \cref{prop:smooth-psh-criterion}(2) they are smooth and plurisubharmonic on $B'$ and decrease to $v$ as $\varepsilon\downarrow0$. As recalled at the beginning of \cref{sec:bt-density-lower-bounds}, $(\ddc v_\varepsilon)^n=\det\mathcal H(v_\varepsilon)\,\eta^n$. The last assertion of \cref{prop:bounded-monge-ampere-linear-density}, applied on $B$ to $v$ and the continuous density $mg$, therefore gives
\[
 (\ddc v_\varepsilon)^n=F_\varepsilon\eta^n,
 \qquad
 F_\varepsilon=\det\mathcal H(v_\varepsilon)\geq
       \bigl((mg)^{1/n}*\rho_\varepsilon\bigr)^n.
\]
The right side converges to $mg$ uniformly on compact subsets of $B$, by \cref{lem:mollifier-density}(1) applied to the continuous function $(mg)^{1/n}$. Hence $\sqrt{F_\varepsilon g}$ is bounded below by functions converging uniformly to $\sqrt mg$ on compact sets, and for every nonnegative $\chi\in C^\infty_c(B)$,
\begin{equation}\label{eq:mixed-trace-local-root-limit}
 \varliminf_{\varepsilon\downarrow0}
       \int\chi\sqrt{F_\varepsilon g}\,\eta^n
                  \geq\sqrt m\int\chi\nu.
\end{equation}
Moreover, for every $\chi\in C^\infty_c(B)$, as $\varepsilon\downarrow0$ along any sequence,
\[
 \int\chi\,\ddc v_\varepsilon\wedge\tau\longrightarrow\int\chi\,S\wedge\tau,
 \qquad
 \int\chi\,(\ddc v_\varepsilon)^{n-1}\wedge\omega\longrightarrow\int\chi\,\langle S^{n-1}\rangle\wedge\omega.
\]
For the first convergence, note that $\int\chi\,\ddc v_\varepsilon\wedge\tau=\int v_\varepsilon\,\ddc(\chi\tau)$, by the definition of $\ddc$ on currents (\cref{sec:positive-currents}); since $v_\varepsilon\to v$ in $L^1$ on $\Supp\chi$ (\cref{lem:mollifier-density}(2)), this tends to $\int v\,\ddc(\chi\tau)=\int\chi\,S\wedge\tau$. For the second, choose a ball $B'$ as above containing $\Supp\chi$, and a sequence $\varepsilon_j\downarrow0$. By \cref{thm:bt-decreasing-continuity}, applied on $B'$ to the closed positive current $\omega$ of bidimension $(n-1,n-1)$ and to $n-1$ copies of the decreasing sequence $(v_{\varepsilon_j})$, the currents $(\ddc v_{\varepsilon_j})^{n-1}\wedge\omega$ converge weakly to the Bedford--Taylor product $(\ddc v)^{n-1}\wedge\omega$. By \cref{prop:bt-positive-recursion}(1) and (4), these products are the exterior products with $\omega$ of the smooth forms $(\ddc v_{\varepsilon_j})^{n-1}$ and of the current $(\ddc v)^{n-1}$, which equals $\langle S^{n-1}\rangle$ on $B$. (For $n=1$ both sides of the second convergence equal $\int\chi\,\omega$.)

\emph{Step 3: globalization.} Let $\chi$ be a smooth function with compact support in $X\setminus A$ and $0\leq\chi\leq1$. Cover $\Supp\chi$ by finitely many balls $B_l$ as in Step~2, and write $\chi=\sum_l\chi_l$ with nonnegative $\chi_l\in C^\infty_c(B_l)$; for instance, multiply $\chi$ by a partition of unity with compact supports subordinate to the cover of $X$ by the balls $B_l$ and $X\setminus\Supp\chi$ (\cref{prop:manifold-cutoffs-stokes}). In each $B_l$ we use the notation of Step~2 with an index $l$ and put
\[
 a_{l,\varepsilon}=\int\chi_l\,\ddc v_{l,\varepsilon}\wedge\tau,\qquad
 b_{l,\varepsilon}=\int\chi_l\,(\ddc v_{l,\varepsilon})^{n-1}\wedge\omega.
\]
Let $\alpha_{l,\varepsilon}$ and $\beta_{l,\varepsilon}$ be the densities of $\ddc v_{l,\varepsilon}\wedge\tau$ and $(\ddc v_{l,\varepsilon})^{n-1}\wedge\omega$ with respect to $\eta_l^n$. By \eqref{eq:pointwise-mixed-trace} with $\sigma=\ddc v_{l,\varepsilon}$, $\alpha_{l,\varepsilon}\beta_{l,\varepsilon}\geq F_{l,\varepsilon}g_l/n$, so the Cauchy--Schwarz inequality gives
\[
 \int\chi_l\sqrt{F_{l,\varepsilon}g_l}\,\eta_l^n
   \leq\sqrt n\int\chi_l\sqrt{\alpha_{l,\varepsilon}\beta_{l,\varepsilon}}\,\eta_l^n
   \leq\sqrt{n\,a_{l,\varepsilon}b_{l,\varepsilon}}.
\]
Summing over $l$ and applying the Cauchy--Schwarz inequality to the finite sum $\sum_l\sqrt{a_{l,\varepsilon}b_{l,\varepsilon}}$, we obtain
\[
 \left(\sum_l a_{l,\varepsilon}\right)
 \left(\sum_l b_{l,\varepsilon}\right)
 \geq\frac1n\left(\sum_l
       \int\chi_l\sqrt{F_{l,\varepsilon}g_l}\,\eta_l^n\right)^2.
\]
Now let $\varepsilon\downarrow0$ along a sequence. By Step~2, the two sums on the left converge to $\int_X\chi\,S\wedge\tau$ and $\int_X\chi\langle S^{n-1}\rangle\wedge\omega$. These are at most the corresponding integrals over $X$ without $\chi$, because $S\wedge\tau$ and $\langle S^{n-1}\rangle\wedge\omega$ are positive measures and $0\leq\chi\leq1$. By \eqref{eq:mixed-trace-local-root-limit} and $g_l\eta_l^n=\nu$, the lower limit of the right side is at least $(m/n)(\int_X\chi\nu)^2$. Therefore
\[
 \left(\int_X S\wedge\tau\right)
 \left(\int_X\langle S^{n-1}\rangle\wedge\omega\right)
 \geq\frac mn\left(\int_X\chi\nu\right)^2.
\]
Finally, fix a metric $d$ inducing the topology of $X$ (\cref{lem:compact-manifold-metric}), and for $k\geq1$ let $K_k=\{x\in X:d(x,A)\geq1/k\}$, with $K_k=X$ if $A=\varnothing$. These compact subsets of $X\setminus A$ increase to $X\setminus A$, because $A$ is closed. By \cref{prop:manifold-cutoffs-stokes} there are smooth functions $\chi_k\colon X\to[0,1]$ with compact support in $X\setminus A$ and $\chi_k=1$ on $K_k$. Since $\nu$ has a smooth density and $A$ has Lebesgue measure zero, $\nu(K_k)\to\nu(X\setminus A)=\int_X\nu$, so $\int_X\chi_k\nu\to\int_X\nu$. Applying the last inequality to $\chi=\chi_k$ and letting $k\to\infty$ proves \eqref{eq:prescribed-density-mixed-trace}. Both factors on its left are finite: the first is the pairing of a current with a smooth form on the compact manifold $X$, and the second is finite by \cref{thm:compact-nonpluripolar-finiteness}. For $n=1$ the factor $(\ddc v_{l,\varepsilon})^{n-1}$ is the constant $1$, and the argument applies without change.
\end{proof}
\index{Monge--Amp\`ere equation!mixed trace estimate}

\begin{theorem}\label{thm:positive-nonpluripolar-mass-bigness}
A pseudo-effective class $\alpha\in H^{1,1}(X,\mathbb R)$ is big if and only if $\operatorname{vol}(\alpha)>0$. In particular, if a closed positive $(1,1)$-current $T$ satisfies $\int_X\langle T^n\rangle>0$, then its class is big.
\end{theorem}
\begin{proof}
By \cref{prop:positive-mixed-mass-bound}, big classes have positive volume, and every closed positive $(1,1)$-current $T$ satisfies $\int_X\langle T^n\rangle\leq\operatorname{vol}(\{T\})$. It therefore suffices to show that a pseudo-effective class $\alpha$ with $m=\operatorname{vol}(\alpha)>0$ is big. Write $\alpha=\{\theta\}$ with $\theta$ smooth and closed, choose $C>0$ such that $\theta+C\omega$ is positive definite, and put
\[
 C_0=\int_X(\theta+(C+1)\omega)^{n-1}\wedge\omega>0;
\]
the integrand is a positive multiple of the volume form at every point, by the positivity of mixed discriminants of positive definite matrices (\cref{lem:mixed-discriminant-basic}(2),(3)).

For $0<\varepsilon\leq1$ let $\theta_\varepsilon=\theta+\varepsilon\omega$. Since $\alpha$ is pseudo-effective, there is $q\in\PSH(X,\theta)$ (\cref{cor:current-theta-psh-potential}), and $\theta_\varepsilon+\ddc q\geq\varepsilon\omega$; thus $\{\theta_\varepsilon\}=\alpha+\varepsilon\{\omega\}$ is big. By \cref{prop:p-envelope-calculus}, $P_{\theta_\varepsilon}[V_{\theta_\varepsilon}]=V_{\theta_\varepsilon}$, and the mass of $V_{\theta_\varepsilon}$ is
\[
 m_\varepsilon=\operatorname{vol}(\alpha+\varepsilon\{\omega\})
                                                        \geq m>0;
\]
the inequality holds because $\operatorname{vol}(\alpha+\varepsilon\{\omega\})$ decreases to $\operatorname{vol}(\alpha)$ as $\varepsilon\downarrow0$, by \cref{def:positive-mixed-intersection-number,prop:positive-mixed-mass-bound}. Hence $V_{\theta_\varepsilon}$ is a model potential of mass $m_\varepsilon$.

Fix a smooth strictly positive real $(n-1,n-1)$-form $\tau$ with $\ddc\tau=0$, normalized by $\int_X\omega\wedge\tau=1$. The measure $m_\varepsilon\omega\wedge\tau$ has a smooth density with respect to $\omega^n$ and total mass $m_\varepsilon$. By \cref{cor:relative-lp-capacity-boundedness}, applied in the big class $\{\theta_\varepsilon\}$ with the reference $V_{\theta_\varepsilon}$, there are $u_\varepsilon\in\PSH(X,\theta_\varepsilon)$ and finite constants $B_\varepsilon$ such that
\[
 \begin{gathered}
 S_\varepsilon=\theta_\varepsilon+\ddc u_\varepsilon\geq0,
 \qquad \langle S_\varepsilon^n\rangle
                =m_\varepsilon\omega\wedge\tau,\\
 V_{\theta_\varepsilon}-B_\varepsilon
                   \leq u_\varepsilon\leq V_{\theta_\varepsilon}.
 \end{gathered}
\]
The constants $B_\varepsilon$ need not be bounded uniformly in $\varepsilon$; we only use that $u_\varepsilon$ is locally bounded outside a closed set of measure zero. By \cref{lem:big-class-bounded-locus}, applied to the big class $\{\theta_\varepsilon\}$, there is a closed pluripolar set $A_\varepsilon\subset X$ such that $V_{\theta_\varepsilon}$ is locally bounded on $X\setminus A_\varepsilon$; being pluripolar, $A_\varepsilon$ has Lebesgue measure zero in every chart (see the remarks after \cref{def:pluripolar-set}). Hence $u_\varepsilon$, which lies between $V_{\theta_\varepsilon}-B_\varepsilon$ and $V_{\theta_\varepsilon}$, is locally bounded on $X\setminus A_\varepsilon$, and so are the local potentials of $S_\varepsilon$, which differ from $u_\varepsilon$ by smooth local potentials of $\theta_\varepsilon$ (\cref{lem:smooth-dbar-poincare}).

We may thus apply \cref{lem:prescribed-density-mixed-trace} to $S_\varepsilon$, with $A_\varepsilon$, $\nu=\omega\wedge\tau$ and $m_\varepsilon$ in place of $A$, $\nu$ and $m$:
\[
 \left(\int_X S_\varepsilon\wedge\tau\right)
        \left(\int_X\langle S_\varepsilon^{n-1}\rangle\wedge\omega\right)
 \geq\frac{m_\varepsilon}{n}\geq\frac mn.
\]
We bound the two factors on the left. Since $\ddc\tau=0$, the first factor equals the cohomological pairing $\int_X\theta_\varepsilon\wedge\tau=\int_X\theta\wedge\tau+\varepsilon$. If $n\geq2$, the form $\theta_\varepsilon+C\omega$ is positive definite, so \cref{thm:compact-nonpluripolar-finiteness} gives
\[
 \int_X\langle S_\varepsilon^{n-1}\rangle\wedge\omega
 \leq\int_X(\theta+(C+\varepsilon)\omega)^{n-1}\wedge\omega
 \leq C_0;
\]
the last inequality holds because $\theta+(C+1)\omega=(\theta+(C+\varepsilon)\omega)+(1-\varepsilon)\omega$ and products of positive $(1,1)$-forms are positive. If $n=1$, the second factor equals $\int_X\omega=C_0$. As the first factor is nonnegative, in both cases
\[
 \left(\int_X\theta\wedge\tau+\varepsilon\right)C_0\geq\frac mn.
\]
Letting $\varepsilon\downarrow0$ gives $\int_X\theta\wedge\tau\geq m/(nC_0)$. Removing the normalization of $\tau$, we conclude that every smooth strictly positive $\ddc$-closed $(n-1,n-1)$-form $\tau$ satisfies \eqref{eq:lamari-strict-tests} with $\delta=m/(nC_0)>0$. By Lamari's criterion, \cref{lem:kahler-lamari-criterion}, the class $\alpha$ is big.
\end{proof}
\index{big class!positive non-pluripolar mass}\index{volume!bigness criterion}

If $\{\theta\}$ is pseudo-effective but not big, the theorem shows that every $\theta$-psh potential has zero non-pluripolar mass. This allows us to extend the contact theorems of \cref{chap:q10} from big to pseudo-effective classes.

\begin{corollary}\label{cor:pseudoeffective-obstacle-contact}
Let $\theta$ be a smooth closed real $(1,1)$-form, and let $f\colon X\to[-\infty,+\infty]$ be a quasi-continuous Borel function. Suppose that the family $\mathcal A_\theta(f)$ of \cref{def:global-obstacle-envelope} is nonempty and uniformly bounded above. Then $u=P_\theta(f)$ satisfies
\[
 \mu_u\bigl(X\setminus\{u=f\}\bigr)=0.
\]
If $\{\theta\}$ is not big, then $\mu_u=0$. Moreover, if $r\geq1$ and $u_1,\ldots,u_r\in\PSH(X,\theta)$ have a rooftop $w=P_\theta(\min_{1\leq j\leq r}u_j)$ that is not identically $-\infty$, then
\[
 \mu_w\leq\sum_{j=1}^r\mathbf1_{\{w=u_j\}}\mu_{u_j}.
\]
\end{corollary}
\begin{proof}
By \cref{prop:obstacle-envelope-maximal-member}, $u\in\mathcal A_\theta(f)$; that is, $u\in\PSH(X,\theta)$ and $u\leq f$ quasi-everywhere. If $\{\theta\}$ is not big, then $\mu_u(X)=0$ by \cref{thm:positive-nonpluripolar-mass-bigness}, since otherwise $\theta+\ddc u$ would be a closed positive current of positive non-pluripolar mass in $\{\theta\}$. Hence $\mu_u=0$, and in particular the first assertion holds.

If $\{\theta\}$ is big, the first assertion is \cref{thm:quasicontinuous-obstacle-contact}.

For the rooftop, the function $\min_ju_j$ is quasi-continuous, as a finite minimum of potentials (see the beginning of \cref{sec:global-obstacle-envelopes}), and it is bounded above; its envelope is $w$. Since $w$ is not identically $-\infty$, the family $\mathcal A_\theta(\min_ju_j)$ is nonempty, and it is uniformly bounded above because the obstacle is. By the first assertion, $\mu_w$ is carried by $\{w=\min_ju_j\}\subset\bigcup_j\{w=u_j\}$. Since $w\leq u_j$ everywhere, \cref{lem:ordered-potential-contact-measure} gives $\mathbf1_{\{w=u_j\}}\mu_w\leq\mathbf1_{\{w=u_j\}}\mu_{u_j}$ for every $j$. Summing over $j$ proves the rooftop inequality.
\end{proof}
\index{envelope!pseudo-effective contact concentration}

\begin{corollary}\label{cor:pseudoeffective-regular-obstacle-density}
Let $\theta$ be a smooth closed real $(1,1)$-form with $\PSH(X,\theta)\neq\varnothing$, and let $f$ be a continuous real function on $X$ with bounded metric Laplacian. Then
\[
 \langle(\theta+\ddc P_\theta(f))^n\rangle
   =\mathbf1_{\{P_\theta(f)=f\}}(\theta+\ddc f)^n,
\]
where the right side is understood as in \cref{thm:irregular-mixed-contact-density}. In particular $\langle(\theta+\ddc V_\theta)^n\rangle=\mathbf1_{\{V_\theta=0\}}\theta^n$. If $\{\theta\}$ is not big, both sides of these two identities vanish.
\end{corollary}
\begin{proof}
The potential $V_\theta+\min_Xf$ lies below $f$, so $\mathcal A_\theta(f)$ is nonempty, and it is bounded above by $\max_Xf$. Put $u=P_\theta(f)$. Since $f$ is continuous, $u\leq f$ everywhere by \cref{prop:obstacle-envelope-maximal-member}. The scalar case of \cref{thm:irregular-mixed-contact-density}, with all $\theta_j=\theta$, $u_j=u$ and $f_j=f$, gives
\[
 \mathbf1_{\{u=f\}}\langle\theta_u^n\rangle=\mathbf1_{\{u=f\}}(\theta+\ddc f)^n,
\]
and \cref{cor:pseudoeffective-obstacle-contact} shows that $\langle\theta_u^n\rangle$ is carried by $\{u=f\}$. This proves the first identity. For $f=0$ the envelope $P_\theta(0)$ is $V_\theta$, by \cref{def:global-minimal-potential}, which gives the second. If $\{\theta\}$ is not big, the left sides vanish by \cref{cor:pseudoeffective-obstacle-contact}, and hence so do the right sides.
\end{proof}
