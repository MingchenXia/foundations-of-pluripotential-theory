\section{Density lower bounds and mixed products}
\label{sec:bt-density-lower-bounds}

We now turn to the equation $(\ddc u)^n=\mu$ with a prescribed measure $\mu$ on a ball, and to the lower bounds that mixed products inherit from lower bounds of the pure Monge--Amp\`ere measures. From here to the end of \cref{sec:common-measure-mixed-products}, $\Omega\subset\mathbb C^n$ is open, $n\ge1$, and
\[
 \eta=\ddc|z|^2=\beta/\pi,\qquad\text{so that}\qquad\eta^n=\frac{n!}{\pi^n}\,\mathrm dV_{2n}
\]
by \eqref{eq:bt-determinant-normalization}. Densities of Monge--Amp\`ere measures are taken with respect to $\eta^n$; thus, by \eqref{eq:bt-determinant-normalization} and the remark following \cref{prop:bt-uniform-continuity}, a plurisubharmonic function $u\in C^2(\Omega)$ satisfies $(\ddc u)^n=\det(u_{j\bar k})\,\eta^n$.

A lower bound $(\ddc u)^n\ge f\eta^n$ is a nonlinear condition on $u$, and it is not evident that it is preserved by convolution. We show that, for a continuous density $f$, it can be tested pointwise with $C^2$ functions touching $u$ from above, and that it is equivalent to a family of linear elliptic inequalities, one for each positive definite Hermitian matrix of determinant one (\cref{prop:bounded-monge-ampere-linear-density}). Linear inequalities are preserved by convolution, so the determinant inequalities for mixed products of smooth functions pass to locally bounded potentials. The same characterization will make subsolutions of the Dirichlet problem stable under the operations used in \cref{sec:bt-continuous-ball-dirichlet}.

Let $u$ be a real function on an open subset of $\mathbb R^d$. A real $C^2$ function $q$ defined near a point $x$ \emph{touches $u$ from above at $x$} if $q(x)=u(x)$ and $q\ge u$ on a neighborhood of $x$; we then call $q$ an \emph{upper test function} for $u$ at $x$.
\index{upper test function}

\begin{lemma}\label{lem:determinant-linearization-mixed}
Let $A$ be a positive semidefinite Hermitian $n\times n$ matrix. Then
\[
 (\det A)^{1/n}=\inf\left\{\frac{\operatorname{tr}(BA)}n:\ B\text{ positive definite Hermitian},\ \det B=1\right\}.
\]
If $C$ is another positive semidefinite Hermitian $n\times n$ matrix, define the numbers $M_k(A,C)$, $0\le k\le n$, by
\[
 \det(tA+sC)=\sum_{k=0}^n\binom nk M_k(A,C)\,t^ks^{n-k}\qquad(t,s\in\mathbb R).
\]
Then $M_n(A,C)=\det A$, $M_0(A,C)=\det C$, and for each integer $1\le k\le n-1$,
\[
 M_k(A,C)\ge(\det A)^{k/n}(\det C)^{(n-k)/n}.
\]
\end{lemma}
\begin{proof}
Let $B$ be positive definite with $\det B=1$. The positive semidefinite matrix $B^{1/2}AB^{1/2}$ has trace $\operatorname{tr}(BA)$ and determinant $\det A$, so the arithmetic--geometric mean inequality for its eigenvalues gives $\operatorname{tr}(BA)/n\ge(\det A)^{1/n}$. If $A$ is positive definite, equality holds for $B=(\det A)^{1/n}A^{-1}$. If $A$ is singular, the matrices $B_\varepsilon=\det(A+\varepsilon I)^{1/n}(A+\varepsilon I)^{-1}$, $\varepsilon>0$, are admissible and
\[
 0\le\operatorname{tr}(B_\varepsilon A)\le\operatorname{tr}\bigl(B_\varepsilon(A+\varepsilon I)\bigr)=n\det(A+\varepsilon I)^{1/n}\longrightarrow0
\]
as $\varepsilon\downarrow0$. This proves the infimum formula in both cases.

The formulas for $M_n$ and $M_0$ follow from $\det(tA)=t^n\det A$ and $\det(sC)=s^n\det C$. Suppose first that $A$ and $C$ are positive definite, and let $\lambda_1,\ldots,\lambda_n>0$ be the eigenvalues of $A^{-1/2}CA^{-1/2}$. Since $\det(tA+sC)=\det A\prod_{i=1}^n(t+s\lambda_i)$, comparison of coefficients gives
\[
 M_k(A,C)=\frac{\det A}{\binom nk}\sum_{|J|=n-k}\prod_{j\in J}\lambda_j,
\]
the sum being over the subsets $J\subset\{1,\ldots,n\}$ with $n-k$ elements. Each eigenvalue occurs in exactly $\binom{n-1}{n-k-1}$ of these $\binom n{n-k}$ products, so the geometric mean of the products is $(\lambda_1\cdots\lambda_n)^{(n-k)/n}$. Since $\lambda_1\cdots\lambda_n=\det C/\det A$, the arithmetic--geometric mean inequality gives
\[
 M_k(A,C)\ge\det A\left(\frac{\det C}{\det A}\right)^{(n-k)/n}=(\det A)^{k/n}(\det C)^{(n-k)/n}.
\]
For semidefinite $A$ and $C$, apply this to $A+\varepsilon I$ and $C+\varepsilon I$ and let $\varepsilon\downarrow0$; both sides depend continuously on the matrices.
\end{proof}
\index{determinant!trace linearization}

\begin{lemma}\label{lem:constant-elliptic-upper-tests}
Let $d\ge1$, let $(c_{kl})$ be a real symmetric positive definite $d\times d$ matrix, and put $L=\sum_{k,l=1}^dc_{kl}\partial_k\partial_l$. Let $U\subset\mathbb R^d$ be open, let $w\colon U\to\mathbb R$ be upper semicontinuous, and let $F\colon U\to\mathbb R$ be continuous. Suppose that, for every $x\in U$, every upper test function $q$ for $w$ at $x$ satisfies $Lq(x)\ge F(x)$. Then $w$ is locally integrable, and $Lw\ge F$ in the sense of distributions on $U$.
\end{lemma}
\begin{proof}
We first reduce to the case $L=\Delta$. Let $C=(c_{kl})$, let $P$ be the positive definite square root of $C^{-1}$, so that $PCP=I$, and put $\tilde U=P(U)$, $\tilde w=w\circ P^{-1}$ and $\tilde F=F\circ P^{-1}$. If $q$ is a $C^2$ function near $x\in U$ and $\tilde q=q\circ P^{-1}$, then $\partial_k\partial_lq(x)=\sum_{a,b}P_{ak}P_{bl}(\partial_a\partial_b\tilde q)(Px)$ by the chain rule, and hence
\[
 (Lq)(x)=\sum_{a,b}(PCP)_{ab}(\partial_a\partial_b\tilde q)(Px)=(\Delta\tilde q)(Px).
\]
The map $q\mapsto\tilde q$ is a bijection from the upper test functions for $w$ at $x$ onto the upper test functions for $\tilde w$ at $Px$, so $\tilde w$, $\Delta$ and $\tilde F$ satisfy the hypothesis of the lemma on $\tilde U$. Suppose that the lemma holds for them. Then $w=\tilde w\circ P$ is locally integrable, and for every nonnegative test function $\chi$ on $U$ the change of variables $y=Px$, with $\tilde\chi=\chi\circ P^{-1}$ and $L\chi=(\Delta\tilde\chi)\circ P$, gives
\[
 \langle Lw,\chi\rangle=\int_Uw\,L\chi\,\mathrm dx
 =\frac1{|\det P|}\int_{\tilde U}\tilde w\,\Delta\tilde\chi\,\mathrm dy
 \ge\frac1{|\det P|}\int_{\tilde U}\tilde F\tilde\chi\,\mathrm dy=\int_UF\chi\,\mathrm dx;
\]
the first equality is the definition of distributional derivatives. We may therefore assume that $L=\Delta$.

Suppose first that $F=0$. We show that $w$ satisfies assertion (1) of \cref{thm:subharmonic-comparison} for every ball $D=B(y,r)$ with $\overline{B(y,r)}\subset U$. Let $h$ be a real function which is harmonic on $B(y,r)$ and continuous on $\overline{B(y,r)}$, with $\varlimsup_{B(y,r)\ni x\to\zeta}w(x)\le h(\zeta)$ for every $\zeta\in\partial B(y,r)$, and suppose that $w(x_1)>h(x_1)$ for some $x_1\in B(y,r)$. Choose $\varepsilon>0$ so small that the upper semicontinuous function
\[
 g(x)=w(x)-h(x)-\varepsilon(r^2-|x-y|^2),\qquad x\in B(y,r),
\]
satisfies $g(x_1)>0$. Since $h$ is continuous on the closed ball, $\varlimsup_{B(y,r)\ni x\to\zeta}g(x)\le0$ for every $\zeta\in\partial B(y,r)$. Hence every $\zeta\in\partial B(y,r)$ has an open neighborhood $N_\zeta$ with $g<g(x_1)/2$ on $N_\zeta\cap B(y,r)$; finitely many of these neighborhoods cover the sphere, and the complement $K$ of their union in $\overline{B(y,r)}$ is a compact subset of $B(y,r)$ containing $x_1$. The upper semicontinuous function $g$ attains its maximum over $K$ at a point $x_0$, and $c=g(x_0)\ge g(x_1)>0$; since $g<g(x_1)/2$ outside $K$, we have $g\le c$ on $B(y,r)$. The function $q=h+\varepsilon(r^2-|x-y|^2)+c$ is smooth on $B(y,r)$, satisfies $q-w=c-g\ge0$ there and $q(x_0)=w(x_0)$, so it is an upper test function for $w$ at $x_0$. But $\Delta q=-2d\varepsilon<0$, which contradicts the hypothesis. Hence $w\le h$ on $B(y,r)$. By the converse part of \cref{thm:subharmonic-comparison}, $w$ is subharmonic on $U$. Being real-valued, it is not identically $-\infty$ on any connected component of $U$, so \cref{thm:subharmonic-distribution}(1) shows that $w$ is locally integrable and $\Delta w\ge0$.

In the general case, let $V\Subset U$ be open. Every compact subset of $U$ lies in such a set $V$, so it suffices to show that $w$ is locally integrable on $V$ and that $\Delta w\ge F$ on $V$. Choose an open set $V'$ with $\overline V\subset V'\Subset U$. By \cref{lem:mollifier-density}(1), the mollifications of $F$ converge to $F$ uniformly on $\overline{V'}$; subtracting from each of them its maximal deviation from $F$ on $\overline{V'}$, we obtain smooth functions $F_j$ on a neighborhood of $\overline{V'}$ such that $F_j\le F$ on $\overline{V'}$ and $F_j\to F$ uniformly there. Let $\chi\in C^\infty_c(V')$ be equal to one on an open set $W\supset\overline V$ (\cref{lem:mollifier-density}(4)). By \cref{prop:laplacian-fundamental-solution}, the convolution $h_j=E_d*(\chi F_j)$, where $\chi F_j$ is extended by zero, is a smooth function on $\mathbb R^d$ with $\Delta h_j=\chi F_j$; in particular $\Delta h_j=F_j$ on $W$. If $q$ is an upper test function for $w-h_j$ at $x\in V$, then $q+h_j$ is an upper test function for $w$ at $x$, so that $\Delta q(x)+F_j(x)\ge F(x)$, that is, $\Delta q(x)\ge F(x)-F_j(x)\ge0$. The case already treated, applied to the real upper semicontinuous function $w-h_j$ on $V$, shows that $w-h_j$ is locally integrable on $V$ and $\Delta(w-h_j)\ge0$ there. Hence $w$ is locally integrable on $V$ and $\Delta w\ge F_j$ on $V$. Testing against a nonnegative test function on $V$ and letting $j\to\infty$, we obtain $\Delta w\ge F$ on $V$.
\end{proof}
\index{upper test function!distributional inequality}

For a real $C^2$ function $q$ on an open subset of $\mathbb C^n$, we write $\mathcal H(q)=(q_{j\bar k})$ for its complex Hessian matrix, normalized as in \eqref{eq:complex-hessian-normalization}. For a positive definite Hermitian matrix $B$ with $\det B=1$, put
\[
 L_Bq=\operatorname{tr}\bigl(B\mathcal H(q)\bigr).
\]
This is a real operator with constant coefficients, since $B$ and $\mathcal H(q)$ are Hermitian, and it is elliptic. Indeed, write $B=U\operatorname{diag}(b_1,\ldots,b_n)U^*$ with $U$ unitary and $b_1,\ldots,b_n>0$, and introduce coordinates $w$ by $z=\bar Uw$. By the chain rule, the complex Hessian of $\tilde q(w)=q(\bar Uw)$ is $U^*\mathcal H(q)U$, so that
\[
 L_Bq=\operatorname{tr}\bigl(\operatorname{diag}(b_1,\ldots,b_n)\,U^*\mathcal H(q)U\bigr)=\sum_{j=1}^nb_j\tilde q_{j\bar j}=\frac14\sum_{j=1}^nb_j\Delta_j\tilde q
\]
by \eqref{eq:complex-hessian-normalization}, where $\Delta_j$ is the Laplacian in the variable $w_j$. The real-linear map $z\mapsto w$ is orthogonal on $\mathbb C^n=\mathbb R^{2n}$, with matrix $O$ say. Hence, in the real coordinates of $\mathbb C^n$, the chain rule writes $L_B$ in the form $\sum_{k,l}c_{kl}\partial_k\partial_l$ with $(c_{kl})=O^t\Lambda O$, where $O^t$ is the transpose of $O$ and $\Lambda$ is the diagonal matrix with entries $b_1/4,b_1/4,\ldots,b_n/4,b_n/4$. This matrix is symmetric and positive definite, so \cref{lem:constant-elliptic-upper-tests} applies to $L_B$.

\begin{proposition}\label{prop:bounded-monge-ampere-linear-density}
Let $u$ be a locally bounded plurisubharmonic function on $\Omega$, and let $f\ge0$ be a continuous function on $\Omega$. The following conditions are equivalent:
\begin{enumerate}
\item $(\ddc u)^n\ge f\eta^n$ as measures on $\Omega$;
\item $\det\mathcal H(q)(x)\ge f(x)$ for every $x\in\Omega$ and every upper test function $q$ for $u$ at $x$;
\item $L_Bu\ge nf^{1/n}$ in the sense of distributions on $\Omega$, for every positive definite Hermitian matrix $B$ with $\det B=1$.
\end{enumerate}
If these conditions hold, then for every $\varepsilon>0$ the mollification $u_\varepsilon=u*\rho_\varepsilon$ satisfies $\det\mathcal H(u_\varepsilon)\ge(f^{1/n}*\rho_\varepsilon)^n$ on $\Omega_\varepsilon$.
\end{proposition}
\begin{proof}
We first observe that the complex Hessian $\mathcal H(q)(x)$ of an upper test function $q$ for $u$ at $x$ is positive semidefinite. Indeed, let $\xi\in\mathbb C^n\setminus\{0\}$, and let $r>0$ be so small that the closed disc $\{x+\zeta\xi:|\zeta|\le r\}$ lies in the neighborhood of $x$ on which $q\ge u$. The function $\zeta\mapsto u(x+\zeta\xi)$ is subharmonic near this disc and finite at $\zeta=0$, so \cref{thm:subharmonic-comparison}(2) gives
\[
 q(x)=u(x)\le\frac1{2\pi}\int_0^{2\pi}u(x+r\mathrm e^{\mathrm i\theta}\xi)\,\mathrm d\theta\le\frac1{2\pi}\int_0^{2\pi}q(x+r\mathrm e^{\mathrm i\theta}\xi)\,\mathrm d\theta.
\]
Write $q_j=\partial_{z_j}q$ and $q_{jk}=\partial_{z_j}\partial_{z_k}q$. Since $q$ is real and of class $C^2$, Taylor's formula gives
\[
 q(x+h)=q(x)+2\Real\sum_jq_j(x)h_j+\Real\sum_{j,k}q_{jk}(x)h_jh_k+\sum_{j,k}q_{j\bar k}(x)h_j\bar h_k+o(|h|^2).
\]
For $h=r\mathrm e^{\mathrm i\theta}\xi$ the terms containing $\mathrm e^{\mathrm i\theta}$ and $\mathrm e^{2\mathrm i\theta}$ have mean zero over $\theta\in[0,2\pi]$, so the right-hand side of the preceding display equals $q(x)+r^2\sum_{j,k}q_{j\bar k}(x)\xi_j\bar\xi_k+o(r^2)$. Dividing by $r^2$ and letting $r\downarrow0$, we obtain $\sum_{j,k}q_{j\bar k}(x)\xi_j\bar\xi_k\ge0$.

\emph{(1) implies (2).} Assume (1), let $q$ be an upper test function for $u$ at $x$, and suppose that $\det\mathcal H(q)(x)<f(x)$. Since $\mathcal H(q)(x)$ is positive semidefinite, we can choose $\delta>0$ so small that $q_\delta=q+\delta|z-x|^2$ satisfies $\mathcal H(q_\delta)(x)=\mathcal H(q)(x)+\delta I>0$ and still $\det\mathcal H(q_\delta)(x)<f(x)$. Since $q$ is of class $C^2$ and $f$ is continuous, there is $r>0$ such that $\overline{B(x,r)}\subset\Omega$, $q\ge u$ on $\overline{B(x,r)}$, the complex Hessian $\mathcal H(q_\delta)$ is positive definite on $\overline{B(x,r)}$, and $\det\mathcal H(q_\delta)<f$ on $\overline{B(x,r)}$; in particular, $q_\delta$ is plurisubharmonic on $B(x,r)$ by \cref{prop:smooth-psh-criterion}. Put $b=q_\delta-\delta r^2/2$. Then $b=q+\delta r^2/2>u$ on the sphere $\partial B(x,r)$, while $b(x)=u(x)-\delta r^2/2<u(x)$.

The Borel set $S=\{z\in B(x,r):b(z)<u(z)\}$ has positive Lebesgue measure. Otherwise $\max\{u,b\}=b$ almost everywhere on $B(x,r)$; since both functions are plurisubharmonic, the uniqueness assertion of \cref{thm:psh-distribution-representative} gives $\max\{u,b\}=b$ everywhere, contradicting $b(x)<u(x)$. The function $b-u$ is lower semicontinuous and positive on $\partial B(x,r)$, so $\varliminf_{z\to\zeta}(b(z)-u(z))>0$ for every $\zeta\in\partial B(x,r)$. The comparison principle \cref{thm:bt-comparison}, applied on $B(x,r)$ with the bounded plurisubharmonic functions $b$ and $u$ in place of $u$ and $v$, therefore gives
\[
 \int_S f\eta^n\le\int_S(\ddc u)^n\le\int_S(\ddc b)^n=\int_S\det\mathcal H(q_\delta)\,\eta^n.
\]
Here $(\ddc b)^n=\det\mathcal H(q_\delta)\,\eta^n$ on $B(x,r)$ because $b$ is a plurisubharmonic $C^2$ function there (see the beginning of this section), and the integrals are finite because $(\ddc u)^n$ is a Radon measure on $\Omega$ and $\overline{B(x,r)}$ is a compact subset of $\Omega$. Since $\det\mathcal H(q_\delta)<f$ on $S$ and $S$ has positive measure, the right side is strictly smaller than the left side, a contradiction. Thus (2) holds.

\emph{(2) implies (3).} Let $q$ be an upper test function for $u$ at $x$, and let $B$ be positive definite with $\det B=1$. Since $\mathcal H(q)(x)$ is positive semidefinite, the infimum formula of \cref{lem:determinant-linearization-mixed} and (2) give
\[
 L_Bq(x)=\operatorname{tr}\bigl(B\mathcal H(q)(x)\bigr)\ge n\bigl(\det\mathcal H(q)(x)\bigr)^{1/n}\ge nf(x)^{1/n}.
\]
Since $u$ is real-valued and upper semicontinuous, \cref{lem:constant-elliptic-upper-tests}, applied to $L_B$ and the continuous function $nf^{1/n}$, gives (3).

\emph{(3) implies (1).} The mollifications $u_\varepsilon=u*\rho_\varepsilon$ of \cref{sec:convolution} are smooth plurisubharmonic functions on $\Omega_\varepsilon$ which decrease to $u$ as $\varepsilon\downarrow0$ (\cref{prop:smooth-psh-criterion}). Fix $B$ as in (3). The distribution $L_Bu-nf^{1/n}$ is positive, so its mollification is a nonnegative function on $\Omega_\varepsilon$ (\cref{prop:distribution-smoothing}(3)). Since $L_B$ has constant coefficients, \cref{prop:distribution-smoothing}(1) identifies the mollification of $L_Bu$ with $L_Bu_\varepsilon$, and we obtain the pointwise inequality
\[
 \operatorname{tr}\bigl(B\mathcal H(u_\varepsilon)\bigr)\ge n\,(f^{1/n}*\rho_\varepsilon)\quad\text{on }\Omega_\varepsilon\qquad(B>0,\ \det B=1).
\]
Since $\mathcal H(u_\varepsilon)$ is positive semidefinite (\cref{prop:smooth-psh-criterion}), taking the infimum over $B$ and using \cref{lem:determinant-linearization-mixed}, we obtain $\det\mathcal H(u_\varepsilon)\ge(f^{1/n}*\rho_\varepsilon)^n$, which is the last assertion of the proposition; that is,
\[
 (\ddc u_\varepsilon)^n\ge(f^{1/n}*\rho_\varepsilon)^n\eta^n\quad\text{on }\Omega_\varepsilon.
\]
As $\varepsilon\downarrow0$, the density on the right converges locally uniformly to $f$ by \cref{lem:mollifier-density}, and $(\ddc u_\varepsilon)^n\to(\ddc u)^n$ weakly by \cref{thm:bt-decreasing-continuity}, applied along any sequence $\varepsilon_j\downarrow0$. Testing against nonnegative functions in $C_c(\Omega)$ gives (1).
\end{proof}

\begin{corollary}\label{cor:bounded-mixed-continuous-density}
Let $u,v$ be locally bounded plurisubharmonic functions on $\Omega$, and let $f,g\ge0$ be continuous functions on $\Omega$ such that $(\ddc u)^n\ge f\eta^n$ and $(\ddc v)^n\ge g\eta^n$. Then, for each integer $1\le k\le n-1$,
\[
 (\ddc u)^k\wedge(\ddc v)^{n-k}\ge f^{k/n}g^{(n-k)/n}\eta^n.
\]
For $k=0$ and $k=n$ the same inequality is the hypothesis.
\end{corollary}
\begin{proof}
Let $u_\varepsilon=u*\rho_\varepsilon$ and $v_\varepsilon=v*\rho_\varepsilon$, with the same mollifiers for both functions. By the last assertion of \cref{prop:bounded-monge-ampere-linear-density}, applied to $u$ with the density $f$ and to $v$ with the density $g$,
\[
 \det\mathcal H(u_\varepsilon)\ge(f^{1/n}*\rho_\varepsilon)^n,
 \qquad
 \det\mathcal H(v_\varepsilon)\ge(g^{1/n}*\rho_\varepsilon)^n.
\]
For real $t$ and $s$, \eqref{eq:bt-determinant-normalization}, applied to the smooth function $tu_\varepsilon+sv_\varepsilon$, gives $(t\ddc u_\varepsilon+s\ddc v_\varepsilon)^n=\det\bigl(t\mathcal H(u_\varepsilon)+s\mathcal H(v_\varepsilon)\bigr)\eta^n$ for the exterior powers of smooth forms. Expanding the left side by the binomial formula, which applies because $2$-forms commute, and comparing the coefficients of $\binom nkt^ks^{n-k}$, we find, since Bedford--Taylor products of smooth functions are exterior products of smooth forms (\cref{prop:bt-positive-recursion}),
\[
 (\ddc u_\varepsilon)^k\wedge(\ddc v_\varepsilon)^{n-k}=M_k\bigl(\mathcal H(u_\varepsilon),\mathcal H(v_\varepsilon)\bigr)\eta^n\ge(f^{1/n}*\rho_\varepsilon)^k(g^{1/n}*\rho_\varepsilon)^{n-k}\eta^n,
\]
where the inequality is the second assertion of \cref{lem:determinant-linearization-mixed}, applied to the matrices $\mathcal H(u_\varepsilon)$ and $\mathcal H(v_\varepsilon)$, which are positive semidefinite by \cref{prop:smooth-psh-criterion}. As $\varepsilon\downarrow0$, the density on the right converges locally uniformly to $f^{k/n}g^{(n-k)/n}$, while the left side converges weakly to $(\ddc u)^k\wedge(\ddc v)^{n-k}$ by \cref{thm:bt-decreasing-continuity}, since $u_\varepsilon$ and $v_\varepsilon$ are smooth plurisubharmonic functions decreasing to $u$ and $v$ (\cref{prop:smooth-psh-criterion}).
\end{proof}
