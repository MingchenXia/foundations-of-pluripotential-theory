We next show that plurisubharmonic functions belong to $L^p_{\loc}$ for every finite $p$, and that $L^1_{\loc}$ convergence of plurisubharmonic functions implies $L^p_{\loc}$ convergence. We first treat a subharmonic function $v\leq0$ on the unit disc. By the Riesz decomposition (\cref{thm:riesz-decomposition}), $v$ is locally the sum of a harmonic function and the fundamental potential $E_2*\mu$ of a measure, and the kernel $E_2(z)=(2\pi)^{-1}\log|z|$ belongs locally to every $L^p$ with $p<\infty$; this gives an $L^p$ bound in terms of the value of $v$ at the center. Applying this bound in one complex variable at a time then gives the estimate for plurisubharmonic functions on polydiscs.

We write $D(a,r)=\{z\in\mathbb C:|z-a|<r\}$ and $D=D(0,1)$, and for $t>0$ we write $tD=D(0,t)$; thus $D^n$ is the unit polydisc in $\mathbb C^n$ and $(tD)^n$ the polydisc of radius $t$ centered at $0$. Integrals over discs and polydiscs are taken with respect to Lebesgue measure $\mathrm dV$.

\begin{lemma}\label{lem:negative-subharmonic-disc-lp}
There is a constant $C_0\geq1$ with the following property. Let $v\leq0$ be a subharmonic function on $D$ with $v(0)>-\infty$. Then there are a finite positive Borel measure $\nu$ with support in $D(0,9/10)$ and a harmonic function $h$ on $D(0,4/5)$ such that
\begin{equation}\label{eq:negative-subharmonic-disc-riesz-bounds}
 \begin{gathered}
 v(z)=P(z)+h(z)\quad(z\in D(0,4/5)),\qquad P(z)=\frac1{2\pi}\int\log|z-\zeta|\,\mathrm d\nu(\zeta),\\
 \nu(D)\leq C_0|v(0)|,\qquad\sup_{\overline{D(0,1/2)}}|h|\leq C_0|v(0)|.
 \end{gathered}
\end{equation}
Moreover, for every $1\leq p<\infty$ there is a constant $C_p>0$ such that every such $v$ satisfies
\[
 \|v\|_{L^p(D(0,1/2))}\leq C_p|v(0)|.
\]
\end{lemma}
\begin{proof}
In this proof $C$ denotes constants that depend only on $p$ and on two cutoff functions $\chi$ and $\eta$, fixed once and for all below; we say so when a constant does not depend on $p$. Since $v(0)>-\infty$, the function $v$ is not identically $-\infty$, and hence it is locally integrable by \cref{prop:subharmonic-local-integrability}. For $0<r<1$, the ball submean inequality (\cref{thm:subharmonic-comparison}) and $v\leq0$ give
\begin{equation}\label{eq:negative-subharmonic-disc-lone}
 \|v\|_{L^1(D(0,r))}=-\int_{D(0,r)}v\,\mathrm dV\leq\pi r^2|v(0)|.
\end{equation}
Let $\mu=\Delta v$ be the Riesz measure of $v$, and choose a smooth cutoff $\chi\geq0$ supported in $D(0,19/20)$ and equal to one on $D(0,9/10)$. By the definition of the distributional Laplacian and \eqref{eq:negative-subharmonic-disc-lone},
\[
 \mu(D(0,9/10))\leq\int\chi\,\mathrm d\mu
   =\int v\Delta\chi\,\mathrm dV\leq\sup|\Delta\chi|\,\|v\|_{L^1(D(0,19/20))}\leq C|v(0)|.
\]
Next choose a smooth cutoff $0\leq\eta\leq1$ supported in $D(0,9/10)$ and equal to one on a neighborhood of $\overline{D(0,4/5)}$. The fundamental solution in the plane is $(2\pi)^{-1}\log|z|$, so \cref{thm:riesz-decomposition}, applied with the measure $\nu=\eta\mu$, which is finite and supported in $\Supp\eta\subset D(0,9/10)$, gives
\[
 v(z)=P(z)+h(z),\qquad
 P(z)=\frac1{2\pi}\int\log|z-\zeta|\,\mathrm d\nu(\zeta),
\]
for every $z\in D(0,4/5)$, with $h$ harmonic on $D(0,4/5)$.

We estimate $P$. For $|\zeta|\leq9/10$, the translated disc $D(0,4/5)-\zeta$ lies in $D(0,2)$, so
\[
 \int_{D(0,4/5)}|\log|z-\zeta||^p\,\mathrm dV(z)
     \leq L_p=\int_{D(0,2)}|\log|z||^p\,\mathrm dV(z)<\infty;
\]
the integral $L_p$ is finite because its part over $D$ equals $2\pi\int_0^1r|\log r|^p\,\mathrm dr$, whose integrand is bounded on $(0,1]$, while $|\log|z||\leq\log2$ on the rest of $D(0,2)$. Let $m=\nu(D)=\int\eta\,\mathrm d\mu$, so that $m\leq\mu(D(0,9/10))\leq C|v(0)|$, with a constant that does not depend on $p$. If $m=0$, then $P=0$. Otherwise, $|P(z)|\leq\frac m{2\pi}\int|\log|z-\zeta||\,\mathrm d\nu(\zeta)/m$, and H\"older's inequality for the probability measure $\nu/m$ gives
\[
 |P(z)|^p\leq\Bigl(\frac m{2\pi}\Bigr)^p\int|\log|z-\zeta||^p\,\frac{\mathrm d\nu(\zeta)}m,
\]
and integrating over $z\in D(0,4/5)$ with Tonelli's theorem yields $\|P\|_{L^p(D(0,4/5))}\leq L_p^{1/p}m/(2\pi)\leq C|v(0)|$.

We now estimate $h$. The case $p=1$ of the bound for $P$ and \eqref{eq:negative-subharmonic-disc-lone} give $\|h\|_{L^1(D(0,4/5))}\leq C|v(0)|$, again with a constant that does not depend on $p$. For $|z|\leq1/2$, the closed disc $\overline{D(z,1/4)}$ lies in $D(0,4/5)$, and the ball mean-value property of the harmonic function $h$ (\cref{prop:spherical-mean-derivative}) gives $|h(z)|\leq(16/\pi)\|h\|_{L^1(D(0,4/5))}$. We have thus found a constant $C_0\geq1$, depending only on the cutoff functions $\chi$ and $\eta$, such that $\nu(D)=m\leq C_0|v(0)|$ and $\sup_{\overline{D(0,1/2)}}|h|\leq C_0|v(0)|$; this proves \eqref{eq:negative-subharmonic-disc-riesz-bounds}. Adding the estimates for $P$ and $h$ on $D(0,1/2)$, we obtain
\[
 \|v\|_{L^p(D(0,1/2))}\leq\|P\|_{L^p(D(0,4/5))}+(\pi/4)^{1/p}C_0|v(0)|\leq C|v(0)|,
\]
which proves the last assertion.
\end{proof}

\begin{proposition}\label{prop:negative-psh-polydisc-lp}
Let $n\geq1$ and $1\leq p<\infty$. There are constants $C_{n,p},A_{n,p}>0$ such that every plurisubharmonic function $v\leq0$ on the unit polydisc $D^n$ satisfies
\[
 \|v\|_{L^p((\tfrac12D)^n)}\leq C_{n,p}|v(0)|
                         \quad\text{if }v(0)>-\infty,
\]
and
\begin{equation}\label{eq:negative-psh-lp-lone-bound}
 \|v\|_{L^p((\tfrac14D)^n)}\leq A_{n,p}\|v\|_{L^1(D^n)}.
\end{equation}
Here $\|v\|_{L^1(D^n)}$ may be infinite, in which case there is nothing to prove.
\end{proposition}
\begin{proof}
We prove the first estimate by induction on $n$. For $n=1$ it is \cref{lem:negative-subharmonic-disc-lp}. Let $n\geq2$, write $z=(z_1,z')$ with $z'\in\mathbb C^{n-1}$, and let $v$ be as in the statement with $v(0)>-\infty$. The function $w(z')=v(0,z')$ is the composition of $v$ with the holomorphic map $z'\mapsto(0,z')$, so by \cref{prop:psh-holomorphic-pullback} it is plurisubharmonic or identically $-\infty$ on $D^{n-1}$. Since $w(0)=v(0)>-\infty$, it is plurisubharmonic, hence locally integrable and finite outside a Lebesgue null set $E\subset D^{n-1}$. For $z'\in D^{n-1}\setminus E$, the function $z_1\mapsto v(z_1,z')$ is subharmonic and nonpositive on $D$, with the finite value $w(z')$ at the center, so \cref{lem:negative-subharmonic-disc-lp} gives
\[
 \int_{\frac12D}|v(z_1,z')|^p\,\mathrm dV(z_1)
                              \leq C_p^p|w(z')|^p.
\]
We integrate over $z'\in(\tfrac12D)^{n-1}$ using Tonelli's theorem; the null set $E$, on which the inner integral may be infinite, does not contribute. The induction hypothesis applied to $w$ then gives
\[
 \|v\|_{L^p((\frac12D)^n)}^p\leq C_p^p\|w\|_{L^p((\frac12D)^{n-1})}^p\leq C_p^pC_{n-1,p}^p|v(0)|^p,
\]
which is the first estimate with $C_{n,p}=C_pC_{n-1,p}$.

For the second estimate, let $Q=(\tfrac1{10}D)^n$ and $|Q|$ its volume. If $\|v\|_{L^1(D^n)}=0$, then $v=0$ almost everywhere, hence $v\equiv0$ by \cref{thm:psh-distribution-representative}, and there is nothing to prove; we may also assume that $\|v\|_{L^1(D^n)}<\infty$. Put $\lambda=2|Q|^{-1}\|v\|_{L^1(D^n)}>0$. By Chebyshev's inequality, the set of points of $Q$ with $|v|>\lambda$ has measure at most $\lambda^{-1}\int_Q|v|\,\mathrm dV\leq|Q|/2$, so its complement in $Q$ has positive measure. Since $v$ is finite almost everywhere, there is a point $a\in Q$ with $v(a)>-\infty$ and
\[
 |v(a)|\leq2|Q|^{-1}\|v\|_{L^1(D^n)}.
\]
The polydisc of radius $4/5$ centered at $a$ lies in $D^n$, and the polydisc of radius $2/5$ centered at $a$ contains $(\tfrac14D)^n$, since $\tfrac14+\tfrac1{10}<\tfrac25$. We apply the first estimate to the plurisubharmonic function $\zeta\mapsto v(a+\tfrac45\zeta)\leq0$ on $D^n$ and change variables:
\[
 \|v\|_{L^p((\frac14D)^n)}\leq\bigl(\tfrac45\bigr)^{2n/p}C_{n,p}|v(a)|\leq2\bigl(\tfrac45\bigr)^{2n/p}C_{n,p}|Q|^{-1}\|v\|_{L^1(D^n)}.
\]
This is \eqref{eq:negative-psh-lp-lone-bound}.
\end{proof}

\begin{theorem}\label{thm:psh-local-lp-integrability}
Let $X$ be a complex manifold and $1\leq p<\infty$. Every plurisubharmonic function on $X$ belongs to $L^p_{\loc}(X)$. If a sequence of plurisubharmonic functions on $X$ converges in $L^1_{\loc}(X)$, then it converges in $L^p_{\loc}(X)$ to the same limit.
\end{theorem}
\begin{proof}
If $X$ has dimension zero, every compact subset of $X$ is finite, plurisubharmonic functions are real-valued, and convergence in $L^p_{\loc}$ is pointwise convergence; so both assertions hold. Let $X$ have dimension $n\geq1$. Every compact subset of $X$ is covered by finitely many coordinate polydiscs, so it suffices to work near a point $x\in X$ in a chart in which $x=0$ and the closed polydisc $\overline{(RD)^n}$ lies in the chart, for some $R>0$.

Let $u$ be plurisubharmonic. Being upper semicontinuous, $u$ is bounded above on $\overline{(RD)^n}$ by some $M$. The function $\zeta\mapsto u(R\zeta)-M$ is plurisubharmonic and nonpositive on $D^n$, and its $L^1$ norm on $D^n$ is finite because $u$ is locally integrable. By \eqref{eq:negative-psh-lp-lone-bound}, $u$ belongs to $L^p$ on the polydisc $(\tfrac R4D)^n$. This proves the first assertion.

Now let $(u_j)$ be a sequence of plurisubharmonic functions converging in $L^1_{\loc}(X)$. For $p=1$ there is nothing to prove, so let $p>1$. By \cref{thm:psh-compactness-hartogs} applied in charts, together with the uniqueness in \cref{thm:psh-distribution-representative}, the limit is represented by a plurisubharmonic function $u$. The norms $\|u_j\|_{L^1((RD)^n)}$ are bounded because the sequence converges. If $y\in(\tfrac R2D)^n$, the ball $B(y,R/2)$ lies in $(RD)^n$, so the ball submean inequality gives $u_j(y)\leq|B(y,R/2)|^{-1}\|u_j\|_{L^1((RD)^n)}$. Hence there is a constant $M$, independent of $j$, with $u_j\leq M$ and $u\leq M$ on $(\tfrac R2D)^n$. Applying \eqref{eq:negative-psh-lp-lone-bound} to $\zeta\mapsto u_j(\tfrac R2\zeta)-M$ and to $\zeta\mapsto u(\tfrac R2\zeta)-M$, whose $L^1$ norms on $D^n$ are bounded independently of $j$, we obtain a bound for the $L^{2p}$ norms of $u_j$ and $u$ on $(\tfrac R8D)^n$ that is uniform in $j$. By a finite covering, for every compact set $K\subset X$ the norms $\|u_j\|_{L^{2p}(K)}$ and $\|u\|_{L^{2p}(K)}$ are bounded independently of $j$.

We interpolate between $L^1$ and $L^{2p}$. For a function $f$ on $K$, write $|f|^p=|f|^{p/(2p-1)}\cdot|f|^{p(2p-2)/(2p-1)}$ and apply H\"older's inequality with the conjugate exponents $(2p-1)/p$ and $(2p-1)/(p-1)$; since $\frac{p(2p-2)}{2p-1}\cdot\frac{2p-1}{p-1}=2p$, this gives
\[
 \int_K|f|^p\,\mathrm dV\leq\Bigl(\int_K|f|\,\mathrm dV\Bigr)^{p/(2p-1)}\Bigl(\int_K|f|^{2p}\,\mathrm dV\Bigr)^{(p-1)/(2p-1)},
\]
that is, $\|f\|_{L^p(K)}\leq\|f\|_{L^1(K)}^{1/(2p-1)}\|f\|_{L^{2p}(K)}^{(2p-2)/(2p-1)}$. With $f=u_j-u$, whose $L^{2p}(K)$ norms are bounded independently of $j$, we obtain
\[
 \|u_j-u\|_{L^p(K)}
 \leq\|u_j-u\|_{L^1(K)}^{1/(2p-1)}
       \|u_j-u\|_{L^{2p}(K)}^{(2p-2)/(2p-1)}\longrightarrow0.
\]
\end{proof}
