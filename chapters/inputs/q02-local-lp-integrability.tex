Write $D(a,r)=\{z\in\mathbb C:|z-a|<r\}$ and $D=D(0,1)$. The measures in the disc and polydisc estimates below are Euclidean volume.

The logarithmic kernel on a complex line belongs locally to every finite $L^p$ space. Riesz decomposition transfers this integrability to negative subharmonic functions, with a bound at a finite center. Applying that bound one variable at a time gives the stronger integrability of psh functions.

\Needspace{12\baselineskip}
\begin{lemma}\label{lem:negative-subharmonic-disc-lp}
For every $1\leq p<\infty$, there is $C_p>0$ such that a subharmonic function $v\leq0$ on the unit disc $D\subset\mathbb C$, with $v(0)>-\infty$, satisfies
\[
 \|v\|_{L^p(D(0,1/2))}\leq C_p|v(0)|.
\]
\end{lemma}
\begin{proof}
The ball submean inequality gives $\|v\|_{L^1(D(0,r))}\leq\pi r^2|v(0)|$ for $0<r<1$. Write $\mu=\Delta v$ and choose a nonnegative smooth cutoff $\chi$ supported in $D(0,19/20)$, equal to one on $D(0,9/10)$. Distributional integration gives
\[
 \mu(D(0,9/10))\leq\int\chi\,\mathrm d\mu
   =\int v\Delta\chi\,\mathrm dV\leq C|v(0)|.
\]
Choose a smooth cutoff $0\leq\eta\leq1$ supported in $D(0,9/10)$ and equal to one near $\overline{D(0,4/5)}$. By \cref{thm:riesz-decomposition},
\[
 v(z)=P(z)+h(z),\qquad
 P(z)=\frac1{2\pi}\int\log|z-\zeta|\,\eta(\zeta)\,\mathrm d\mu(\zeta),
\]
on $D(0,4/5)$, with $h$ harmonic. For $|\zeta|\leq9/10$, translation and polar integration bound
\[
 \int_{D(0,4/5)}|\log|z-\zeta||^p\,\mathrm dV(z)
     \leq\int_{D(0,2)}|\log|z||^p\,\mathrm dV(z)<\infty.
\]
Indeed the integral near zero is $2\pi\int_0^1r|\log r|^p\,\mathrm dr$. Jensen's inequality for the finite measure $\eta\mu$, followed by Tonelli, therefore gives $\|P\|_{L^p(D(0,4/5))}\leq C_p|v(0)|$. If this measure is zero, $P=0$ and the same bound holds. The bounds for $v$ and $P$ in $L^1$ give $\|h\|_{L^1(D(0,4/5))}\leq C|v(0)|$. For $|z|\leq1/2$, the harmonic mean formula on a fixed-radius ball about $z$ bounds $|h(z)|$ by this last norm times a fixed constant. Adding the two estimates proves the assertion.
\end{proof}

\begin{proposition}\label{prop:negative-psh-polydisc-lp}
Let $D^n\subset\mathbb C^n$ be the unit polydisc, $n\geq1$. For every $1\leq p<\infty$ there are constants $C_{n,p},A_{n,p}>0$ such that every psh $v\leq0$ on $D^n$ satisfies
\[
 \|v\|_{L^p((\tfrac12D)^n)}\leq C_{n,p}|v(0)|
                         \quad\text{if }v(0)>-\infty,
\]
and
\begin{equation}\label{eq:negative-psh-lp-lone-bound}
 \|v\|_{L^p((\tfrac14D)^n)}\leq A_{n,p}\|v\|_{L^1(D^n)}.
\end{equation}
The second estimate applies whenever its right side is finite.
\end{proposition}
\begin{proof}
For the first assertion apply \cref{lem:negative-subharmonic-disc-lp} in the first variable with the other variables fixed. The center function $z'\mapsto v(0,z')$ is psh by \cref{prop:psh-holomorphic-pullback} and is finite at the origin. It is therefore locally integrable and finite almost everywhere. Off its null pole set the line restriction has a finite center, so
\[
 \int_{\frac12D}|v(z_1,z')|^p\,\mathrm dV(z_1)
                              \leq C_p^p|v(0,z')|^p.
\]
Tonelli and induction on $n$ prove the first estimate, including the possibility of infinite slice integrals over that null exceptional set.

For the second, choose $a\in(\tfrac1{10}D)^n$ with $v(a)>-\infty$ and
\[
 |v(a)|\leq2|(\tfrac1{10}D)^n|^{-1}\|v\|_{L^1(D^n)}.
\]
Such a point exists by integration; when the norm is zero, the canonical representative is zero and the estimate follows. The polydisc of radius $4/5$ about $a$ lies in $D^n$. Its half-sized polydisc contains $(\tfrac14D)^n$. Apply the first estimate after translation and rescaling. The Euclidean volume factor is fixed, giving \eqref{eq:negative-psh-lp-lone-bound}.
\end{proof}

\begin{theorem}\label{thm:psh-local-lp-integrability}
On a complex manifold, every psh function belongs to $L^p_{\loc}$ for $1\leq p<\infty$. A sequence of psh functions converging in $L^1_{\loc}$ converges in $L^p_{\loc}$ for every such $p$.
\end{theorem}
\begin{proof}
In a relatively compact coordinate polydisc, subtract a constant upper bound to make the function negative. Local $L^1$ integrability and the rescaled estimate \eqref{eq:negative-psh-lp-lone-bound} give $L^p$ integrability on a smaller polydisc. These smaller polydiscs cover each compact subset, proving the first assertion.

Suppose $u_j\to u$ in $L^1_{\loc}$. Their norms on a fixed larger coordinate neighborhood are bounded. The ball submean inequality bounds $u_j$ uniformly above on each relatively compact smaller neighborhood; the same holds for the canonical limit $u$. Subtract a common upper bound and apply \eqref{eq:negative-psh-lp-lone-bound}. A finite covering gives uniform $L^{2p}$ bounds for $u_j,u$ on a given compact set $K$. For $p>1$, H\"older's inequality gives
\[
 \|u_j-u\|_{L^p(K)}
 \leq\|u_j-u\|_{L^1(K)}^{1/(2p-1)}
       \|u_j-u\|_{L^{2p}(K)}^{(2p-2)/(2p-1)}\longrightarrow0.
\]
For $p=1$ this is the given convergence. In dimension zero the assertion is the convergence of finite values on the finitely many points of each compact set.
\end{proof}
