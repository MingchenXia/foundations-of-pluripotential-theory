\section{Bounded measurable densities}
\label{sec:bt-measurable-ball-dirichlet}

We now allow an arbitrary bounded nonnegative measurable density $f$, approximate it by continuous densities $f_j$ obtained by convolution, and pass to the limit in the solutions given by \cref{thm:continuous-density-ball-dirichlet}. Since $f_j$ need not converge to $f$ uniformly, the stability estimate \eqref{eq:ball-dirichlet-root-stability} is of no use here. However, the $f_j$ have a common bound and converge in $L^1$, and we show that this suffices for convergence of the solutions in capacity. A comparison argument (\cref{lem:local-comparison-capacity-sublevels}) bounds the capacity of the set where an approximate solution lies far below the limit in terms of the common bound and the $L^1$ convergence, while Hartogs' lemma and quasi-continuity control the set where it lies far above. The convergence theorem (\cref{thm:bt-capacity-continuity}) then identifies the Monge--Amp\`ere measure of the limit.

\begin{lemma}\label{lem:local-comparison-capacity-sublevels}
Let $D\subset\mathbb C^n$ be a bounded open set, let $u,v$ be locally bounded plurisubharmonic functions on $D$, and let $s\in\mathbb R$ and $t>0$. If $\{u<v-s\}\Subset D$, then
\[
 t^n\operatorname{Cap}_D(\{u<v-s-t\})
 \le\int_{\{u<v-s\}}(\ddc u)^n.
\]
\end{lemma}
\begin{proof}
Let $w\in\PSH(D)$ with $-1\le w\le0$. Then
\[
 \{u<v-s-t\}\subset\{u<v-s+tw\}\subset\{u<v-s\},
\]
so the middle set is relatively compact in $D$. Hence every $x\in\partial D$ has a neighborhood on whose intersection with $D$ we have $u\ge v-s+tw$, and in particular $\varliminf_{D\ni z\to x}\bigl(u(z)-(v(z)-s+tw(z))\bigr)\ge0$. Expanding $(\ddc v+t\ddc w)^n$ by additivity and symmetry of the products (\cref{prop:bt-positive-recursion,thm:bt-decreasing-continuity}), and using that all the mixed products of the locally bounded plurisubharmonic functions $v$ and $w$ are positive, we get $\bigl(\ddc(v-s+tw)\bigr)^n\ge t^n(\ddc w)^n$. The comparison principle \cref{thm:bt-comparison}, applied to $u$ and $v-s+tw$, therefore gives
\[
 \begin{aligned}
 t^n\int_{\{u<v-s-t\}}(\ddc w)^n
 &\le\int_{\{u<v-s+tw\}}\bigl(\ddc(v-s+tw)\bigr)^n\\
 &\le\int_{\{u<v-s+tw\}}(\ddc u)^n
 \le\int_{\{u<v-s\}}(\ddc u)^n.
 \end{aligned}
\]
As $w$ ranges over these functions, $w+1$ ranges over the competitors for $\operatorname{Cap}_D$ (\cref{def:monge-ampere-capacity}). Taking the supremum over $w$ proves the lemma.
\end{proof}

\begin{theorem}\label{thm:bounded-density-ball-dirichlet}
Let $B=B(a,R)\subset\mathbb C^n$ with $n\ge1$ and $R>0$, let $f\in L^\infty(B)$ be nonnegative, and let $\varphi\in C(\partial B)$ be real. There is a unique bounded plurisubharmonic function $U_{f,\varphi}$ on $B$ such that
\[
 (\ddc U_{f,\varphi})^n=f\eta^n,
 \qquad\lim_{B\ni z\to x}U_{f,\varphi}(z)=\varphi(x)
       \quad(x\in\partial B).
\]
It is the pointwise supremum of the bounded plurisubharmonic functions $v$ on $B$ such that $(\ddc v)^n\ge f\eta^n$ and $\varlimsup_{B\ni z\to x}v(z)\le\varphi(x)$ for every $x\in\partial B$. The stability estimate \eqref{eq:ball-dirichlet-root-stability} also holds with $U_{f,\varphi}$ and $U_{g,\chi}$ in place of $H_{f,\varphi}$ and $H_{g,\chi}$, for every nonnegative $g\in L^\infty(B)$ and real $\chi\in C(\partial B)$, the norm of $f^{1/n}-g^{1/n}$ being the essential supremum.

Moreover, let $f_j\in L^\infty(B)$ and $\varphi_j\in C(\partial B)$ be such that $0\le f_j\le M$ almost everywhere for a constant $M$, $f_j\to f$ in $L^1(B)$ and $\varphi_j\to\varphi$ uniformly on $\partial B$. Then
\[
 U_{f_j,\varphi_j}\longrightarrow U_{f,\varphi}
 \quad\text{in }L^1(B)\text{ and locally in capacity}.
\]
\end{theorem}
\begin{proof}
Throughout the proof, $H_{g,\chi}$ denotes the solution of \cref{thm:continuous-density-ball-dirichlet} for continuous data $(g,\chi)$.

\emph{Step 1. Approximation and barriers.}
Extend $f$ by zero outside $B$, let $M=\|f\|_{L^\infty(B)}$, and convolve $f$ with mollifiers. The resulting smooth functions $f_j$ on $\mathbb C^n$ satisfy $0\le f_j\le M$, and $f_j\to f$ in $L^1(B)$ by \cref{lem:mollifier-density}(2), applied on $\mathbb C^n$ with the compact set $\overline B$; thus they satisfy the hypotheses of the second part of the theorem. Let $u_j=H_{f_j,\varphi}$. The constant density $M$ and the zero density give two barriers. Indeed, since $M\eta^n\ge f_j\eta^n\ge0$, the function $H_{M,\varphi}$ belongs to the family whose supremum is $u_j$ in \cref{thm:continuous-density-ball-dirichlet}, and $u_j$ belongs to the family whose supremum is $H_{0,\varphi}$. Hence
\[
 H_{M,\varphi}\le u_j\le H_{0,\varphi}.
\]
Both barriers are continuous on $\overline B$ and equal $\varphi$ on $\partial B$.

By \cref{thm:psh-compactness-hartogs}, after passing to a subsequence (not relabeled), $u_j$ converges in $L^1_{\loc}(B)$ to a plurisubharmonic function $U$; convergence to $-\infty$ is excluded by the lower barrier. For every compact set $K\subset B$ we have $(H_{M,\varphi}-U)_+\le|u_j-U|$ and $(U-H_{0,\varphi})_+\le|u_j-U|$, so the integrals of these two functions over $K$ are bounded by $\|u_j-U\|_{L^1(K)}\to0$ and hence vanish. Thus $H_{M,\varphi}\le U\le H_{0,\varphi}$ almost everywhere, and hence everywhere by the uniqueness assertion of \cref{thm:psh-distribution-representative}, applied to $\max\{U,H_{M,\varphi}\}$ and $U$, and to $\max\{U,H_{0,\varphi}\}$ and $H_{0,\varphi}$. Consequently $U$ is bounded and $\lim_{B\ni z\to x}U(z)=\varphi(x)$ for every $x\in\partial B$. Since $u_j$ and $U$ lie between the two bounded barriers, $|u_j-U|$ is bounded independently of $j$, so the convergence also holds in $L^1(B)$: given $\varepsilon>0$, choose a compact set $K\subset B$ with $|B\setminus K|<\varepsilon$ and use the convergence in $L^1(K)$.

\emph{Step 2. Convergence in capacity.}
Fix a compact set $K\subset B$ and $\delta>0$. We first show that the outer capacity of $K\cap\{u_j>U+\delta\}$ tends to zero. Let $\varepsilon>0$. By \cref{thm:psh-capacity-quasicontinuity}, there is an open set $G\subset B$ with $\operatorname{Cap}_B(G)<\varepsilon$ such that the restriction of $U$ to $B\setminus G$ is continuous. We claim that $u_j<U+\delta$ on the compact set $K\setminus G$ for all sufficiently large $j$. Let $x\in K\setminus G$. By upper semicontinuity, $U<U(x)+\delta/4$ on a neighborhood of a closed ball $\overline{B'}$ centered at $x$, so Hartogs' lemma (\cref{thm:psh-compactness-hartogs}), applied with the constant majorant $U(x)+\delta/4$, gives $u_j<U(x)+\delta/4$ on $\overline{B'}$ for all large $j$. Shrinking $B'$, we may also assume that $U>U(x)-\delta/4$ on $(K\setminus G)\cap B'$, since the restriction of $U$ to $B\setminus G$ is continuous. Hence $u_j-U<\delta/2$ on $(K\setminus G)\cap B'$ for all large $j$, and a finite cover of $K\setminus G$ by such balls proves the claim. Consequently $K\cap\{u_j>U+\delta\}\subset G$ for all large $j$, so that $\varlimsup_j\operatorname{Cap}_B^*(K\cap\{u_j>U+\delta\})\le\operatorname{Cap}_B(G)<\varepsilon$. Since $\varepsilon>0$ was arbitrary, the limit is zero.

Next we consider the set where $u_j$ lies below $U$. Since $H_{M,\varphi}\le u_j$ and $U\le H_{0,\varphi}$, the set $\{u_j<U-\delta\}$ is contained in $\{H_{M,\varphi}-H_{0,\varphi}\le-\delta\}$, which is a compact subset of $B$ because the continuous function $H_{M,\varphi}-H_{0,\varphi}$ on $\overline B$ vanishes on $\partial B$. \Cref{lem:local-comparison-capacity-sublevels}, applied with $s=t=\delta$ to $u_j$ and $U$, gives
\[
 \begin{split}
 \delta^n\operatorname{Cap}_B(\{u_j<U-2\delta\})
 &\le\int_{\{u_j<U-\delta\}}f_j\eta^n\\
 &\le \frac M\delta\int_B(U-u_j)_+\eta^n\longrightarrow0,
 \end{split}
\]
because $(\ddc u_j)^n=f_j\eta^n\le M\eta^n$, $U-u_j>\delta$ on the domain of integration, $\eta^n=n!\pi^{-n}\,\mathrm dV_{2n}$, and $u_j\to U$ in $L^1(B)$. To pass from capacity to the outer capacity of \cref{def:local-capacity-convergence}, we use quasi-continuity. The function $(U-u_j)_+$ is bounded and quasi-continuous: $U$ and $u_j$ are real and quasi-continuous by \cref{thm:psh-capacity-quasicontinuity}, so their difference, and its composition with the continuous map $t\mapsto t_+$, are quasi-continuous by the remark following \cref{def:quasi-continuity}. Let $L\subset B$ be a compact neighborhood of $K$. By \cref{lem:quasi-continuous-level-capacity},
\[
 \begin{aligned}
 \operatorname{Cap}_B^*(K\cap\{u_j<U-4\delta\})
 &\le\operatorname{Cap}_B(L\cap\{(U-u_j)_+>2\delta\})\\
 &\le\operatorname{Cap}_B(\{u_j<U-2\delta\})\longrightarrow0.
 \end{aligned}
\]
As $K$ and $\delta$ are arbitrary, $u_j\to U$ locally in capacity in the sense of \cref{def:local-capacity-convergence}.

\emph{Step 3. The equation, the envelope and uniqueness.}
The functions $u_j$ are uniformly bounded and converge to $U$ locally in capacity, so \cref{thm:bt-capacity-continuity} gives $(\ddc u_j)^n\to(\ddc U)^n$ weakly. Since $f_j\to f$ in $L^1(B)$, also $f_j\eta^n\to f\eta^n$ weakly, and therefore $(\ddc U)^n=f\eta^n$.

Let $v$ be a bounded plurisubharmonic function on $B$ with $(\ddc v)^n\ge f\eta^n$ and boundary upper limits at most $\varphi$, and put $v_\varepsilon=v+\varepsilon(|z-a|^2-R^2)$ for $\varepsilon>0$. Expanding $(\ddc v+\varepsilon\eta)^n$ and using positivity of the mixed products, we get $(\ddc v_\varepsilon)^n\ge(f+\varepsilon^n)\eta^n$. Since $\varliminf_{z\to x}(U(z)-v_\varepsilon(z))\ge0$ for every $x\in\partial B$, the comparison principle \cref{thm:bt-comparison} gives
\[
 \int_{\{U<v_\varepsilon\}}(f+\varepsilon^n)\eta^n\le\int_{\{U<v_\varepsilon\}}f\eta^n.
\]
These integrals are finite, so $\{U<v_\varepsilon\}$ is Lebesgue null. By the uniqueness assertion of \cref{thm:psh-distribution-representative}, applied to $\max\{U,v_\varepsilon\}$ and $U$, we obtain $U\ge v_\varepsilon$ everywhere, and letting $\varepsilon\downarrow0$ gives $U\ge v$. Since $U$ itself is such a function $v$, it is the pointwise supremum in the statement; in particular, it does not depend on the subsequence chosen in Step 1. If $U'$ is another bounded solution with boundary limits $\varphi$, then $U'\le U$, and the same argument with $U'$ in place of $U$ gives $U\le U'$. This proves existence and uniqueness, and we write $U_{f,\varphi}=U$.

\emph{Step 4. Stability.}
Let $f_j$, $\varphi_j$ and $M$ be as in the second part of the theorem, and suppose first that $\varphi_j=\varphi$ for all $j$. By the envelope characterization just proved, $H_{M,\varphi}\le U_{f_j,\varphi}\le H_{0,\varphi}$. Steps~1--3 apply without change to the sequence $U_{f_j,\varphi}$, since they used only these barrier inequalities, the equations $(\ddc U_{f_j,\varphi})^n=f_j\eta^n$ with $0\le f_j\le M$, and the convergence $f_j\to f$ in $L^1(B)$. Hence every subsequence of $U_{f_j,\varphi}$ has a further subsequence converging in $L^1(B)$ and locally in capacity to a bounded solution with data $(f,\varphi)$, that is, to $U_{f,\varphi}$. Therefore the whole sequence converges to $U_{f,\varphi}$ in $L^1(B)$ and locally in capacity.

For varying boundary data we use that, for a fixed density $f$, the function $U_{f,\chi}-\|\varphi-\chi\|_{L^\infty(\partial B)}$ belongs to the family whose supremum is $U_{f,\varphi}$; by symmetry, $\|U_{f,\varphi}-U_{f,\chi}\|_{L^\infty(B)}\le\|\varphi-\chi\|_{L^\infty(\partial B)}$. Applied with $f_j$ in place of $f$, this gives $\|U_{f_j,\varphi_j}-U_{f_j,\varphi}\|_{L^\infty(B)}\le\|\varphi_j-\varphi\|_{L^\infty(\partial B)}\to0$. Since $B$ is bounded, it follows that $U_{f_j,\varphi_j}-U_{f_j,\varphi}\to0$ in $L^1(B)$. Moreover, for every $\delta>0$ and all large $j$, the set $\{U_{f_j,\varphi_j}>U_{f,\varphi}+\delta\}$ is contained in $\{U_{f_j,\varphi}>U_{f,\varphi}+\delta/2\}$, and similarly for the sets where the functions lie below $U_{f,\varphi}-\delta$. The general case therefore follows from the case of fixed boundary data.

Finally, we prove \eqref{eq:ball-dirichlet-root-stability} for bounded measurable densities. Let $f,g\in L^\infty(B)$ be nonnegative and $\varphi,\chi\in C(\partial B)$. Extend $f^{1/n}$ and $g^{1/n}$ by zero outside $B$, and convolve them with the same mollifiers. The resulting smooth functions $F_j,G_j\ge0$ satisfy $\|F_j-G_j\|_{L^\infty(B)}\le\|f^{1/n}-g^{1/n}\|_{L^\infty(B)}$. Moreover $F_j\to f^{1/n}$ and $G_j\to g^{1/n}$ in $L^1(B)$ with a common bound, so $F_j^n\to f$ and $G_j^n\to g$ in $L^1(B)$ with a common bound. By \cref{thm:continuous-density-ball-dirichlet},
\[
 \|H_{F_j^n,\varphi}-H_{G_j^n,\chi}\|_{L^\infty(B)}\le c,
 \qquad c=\|\varphi-\chi\|_{L^\infty(\partial B)}+R^2\|f^{1/n}-g^{1/n}\|_{L^\infty(B)}.
\]
By uniqueness, $H_{F_j^n,\varphi}=U_{F_j^n,\varphi}$ and $H_{G_j^n,\chi}=U_{G_j^n,\chi}$, and by the stability just proved these functions converge in $L^1(B)$ to $U_{f,\varphi}$ and $U_{g,\chi}$. Since $|H_{F_j^n,\varphi}-H_{G_j^n,\chi}|\le c$ on $B$, the triangle inequality gives
\[
 \bigl(|U_{f,\varphi}-U_{g,\chi}|-c\bigr)_+\le|U_{f,\varphi}-H_{F_j^n,\varphi}|+|H_{G_j^n,\chi}-U_{g,\chi}|,
\]
and integrating over $B$ and letting $j\to\infty$, we obtain $|U_{f,\varphi}-U_{g,\chi}|\le c$ almost everywhere. The uniqueness assertion of \cref{thm:psh-distribution-representative}, applied to $\max\{U_{f,\varphi},U_{g,\chi}+c\}$ and $U_{g,\chi}+c$, and to $\max\{U_{g,\chi},U_{f,\varphi}+c\}$ and $U_{f,\varphi}+c$, shows that this inequality holds everywhere.
\end{proof}
