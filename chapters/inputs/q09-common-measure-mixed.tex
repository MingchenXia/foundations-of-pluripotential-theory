\subsection{Mixed lower bounds}
\label{sec:nonpluripolar-common-measure-mixed}
The bounded common-measure inequality passes through the simultaneous finite-level restrictions defining a non-pluripolar product. If the reference measure has a polar part, domination by each pure non-pluripolar measure makes its prescribed density vanish on that part.

\begin{proposition}\label{prop:nonpluripolar-multifactor-common-measure}
Let $T_1,\ldots,T_n$ be closed positive $(1,1)$-currents on a compact K\"ahler manifold $X$ of dimension $n\geq1$. Let $\nu$ be a finite positive Radon measure and let $f_i\geq0$ belong to $L^1(\nu)$ with
\[
 \langle T_i^n\rangle\geq f_i\nu\qquad(1\leq i\leq n).
\]
Then
\[
 \langle T_1\wedge\cdots\wedge T_n\rangle
                    \geq\left(\prod_{i=1}^n f_i^{1/n}\right)\nu.
\]
Repeated currents are permitted. All products are finite by \cref{thm:compact-nonpluripolar-finiteness}.
\end{proposition}
\begin{proof}
For $n=1$ this is the hypothesis. Otherwise work on a relatively compact coordinate domain where $T_i=\ddc a_i$ with psh potentials bounded above. Put
\[
 a_{i,j}=\max\{a_i,-j\},\qquad
 G_j=\bigcap_{i=1}^n\{a_i>-j\}.
\]
Truncation locality on $G_j$ identifies each pure bounded product with its pure non-pluripolar measure. Positivity gives on the whole coordinate domain
\[
 (\ddc a_{i,j})^n\geq\mathbf1_{G_j}f_i\nu
                         \qquad(1\leq i\leq n).
\]
Apply \cref{thm:bounded-multifactor-common-measure} and restrict to $G_j$. The defining simultaneous truncation identity yields
\[
 \langle T_1\wedge\cdots\wedge T_n\rangle
               \geq\mathbf1_{G_j}\prod_i f_i^{1/n}\nu.
\]
The complement of $\bigcup_jG_j$ is the union of the local pole sets. Every pure non-pluripolar measure annihilates this entire pluripolar union, not merely its own pole set. Its domination of $f_i\nu$ therefore gives $f_i=0$ $\nu$-almost everywhere on the union, for every $i$. Generalized H\"older makes the product density integrable. Monotone convergence on nonnegative compact tests removes the indicator. Coordinate domains cover $X$, proving the asserted measure inequality.
\end{proof}
\index{non-pluripolar product!multifactor common-measure inequality}

\begin{corollary}\label{cor:nonpluripolar-common-measure-mixed}
Let $T,S$ be closed positive $(1,1)$-currents on compact K\"ahler $X$ of dimension $n\geq1$. If $\nu$ is a finite positive Radon measure, $f,g\geq0$ belong to $L^1(\nu)$ and
\[
 \langle T^n\rangle\geq f\nu,\qquad
 \langle S^n\rangle\geq g\nu,
\]
then, for every integer $1\leq k\leq n-1$,
\[
 \langle T^k\wedge S^{n-k}\rangle
                        \geq f^{k/n}g^{(n-k)/n}\nu.
\]
The endpoints are the hypotheses; for $n=1$ there are only endpoints.
\end{corollary}
\begin{proof}
Use the proposition with $k$ copies of $T$ and $n-k$ copies of $S$, and the corresponding repeated densities.
\end{proof}
\index{non-pluripolar product!common-measure mixed inequality}

\begin{corollary}\label{cor:nonpluripolar-common-measure-convex}
Under these hypotheses, for $0\leq t\leq1$,
\[
 \left\langle\big((1-t)T+tS\big)^n\right\rangle
       \geq\big((1-t)f^{1/n}+tg^{1/n}\big)^n\nu.
\]
\end{corollary}
\begin{proof}
Positive multilinearity expands the left side. Apply \cref{cor:nonpluripolar-common-measure-mixed} to the interior mixed coefficients and use the two pure bounds for the endpoints. The binomial formula gives the right side. Endpoint values of $t$ use the pure hypotheses directly; for $n=1$ use additivity. Weighted arithmetic--geometric mean bounds each interior density by a multiple of $f+g$, so the right-hand measure is finite.
\end{proof}
