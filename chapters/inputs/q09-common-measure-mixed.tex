\subsection{Mixed lower bounds}
\label{sec:nonpluripolar-common-measure-mixed}
Lower bounds $\langle T_i^n\rangle\geq f_i\nu$ by densities with respect to a common measure $\nu$ pass to mixed products, as they do for bounded potentials (\cref{thm:bounded-multifactor-common-measure}): we apply the bounded inequality to the truncated potentials and restrict to the set where no truncation is active. A pluripolar set charged by $\nu$ causes no difficulty, because there the densities $f_i$ vanish $\nu$-almost everywhere.

\begin{proposition}\label{prop:nonpluripolar-multifactor-common-measure}
Let $T_1,\ldots,T_n$ be closed positive $(1,1)$-currents on a compact K\"ahler manifold $X$ of dimension $n\geq1$, not necessarily distinct. Let $\nu$ be a finite positive Radon measure on $X$, and let $f_1,\ldots,f_n\geq0$ belong to $L^1(\nu)$ and satisfy
\[
 \langle T_i^n\rangle\geq f_i\nu\qquad(1\leq i\leq n).
\]
Then
\[
 \langle T_1\wedge\cdots\wedge T_n\rangle
                    \geq\left(\prod_{i=1}^n f_i^{1/n}\right)\nu.
\]
All the products are finite by \cref{thm:compact-nonpluripolar-finiteness}.
\end{proposition}
\begin{proof}
For $n=1$ the conclusion is the hypothesis, so let $n\geq2$. Work on a relatively compact coordinate domain on which $T_i=\ddc a_i$ with plurisubharmonic potentials $a_i$ bounded above, and put
\[
 a_{i,j}=\max\{a_i,-j\},\qquad
 G_j=\bigcap_{i=1}^n\{a_i>-j\}.
\]
Since $G_j\subset\{a_i>-j\}$, plurifine locality and \cref{lem:nonpluripolar-truncation-limit} show that the bounded measure $(\ddc a_{i,j})^n$ agrees with $\langle T_i^n\rangle$ on $G_j$. By positivity, therefore, on the whole coordinate domain
\[
 (\ddc a_{i,j})^n\geq\mathbf1_{G_j}(\ddc a_{i,j})^n\geq\mathbf1_{G_j}f_i\nu
                         \qquad(1\leq i\leq n).
\]
Now \cref{thm:bounded-multifactor-common-measure}, applied to the densities $\mathbf1_{G_j}f_i$, gives $\ddc a_{1,j}\wedge\cdots\wedge\ddc a_{n,j}\geq\mathbf1_{G_j}\prod_if_i^{1/n}\nu$. Restricting to $G_j$, where the left side agrees with the non-pluripolar product of the $T_i$ by definition, we obtain
\[
 \langle T_1\wedge\cdots\wedge T_n\rangle
               \geq\mathbf1_{G_j}\prod_i f_i^{1/n}\nu .
\]
The complement of $\bigcup_jG_j$ is the union $P$ of the pole sets of the $a_i$. Every $\langle T_i^n\rangle$ gives zero mass to the whole pluripolar set $P$, not only to the pole set of $a_i$, so the inequality $\langle T_i^n\rangle\geq f_i\nu$ forces $f_i=0$ $\nu$-almost everywhere on $P$, for every $i$. Hence $\mathbf1_{G_j}\prod_if_i^{1/n}$ increases to $\prod_if_i^{1/n}$ $\nu$-almost everywhere as $j\to\infty$. The limit is $\nu$-integrable by the generalized H\"older inequality, and monotone convergence, tested against nonnegative continuous functions with compact support, removes the indicator. Since such coordinate domains cover $X$, the asserted inequality of measures follows.
\end{proof}
\index{non-pluripolar product!multifactor common-measure inequality}

\begin{corollary}\label{cor:nonpluripolar-common-measure-mixed}
Let $T,S$ be closed positive $(1,1)$-currents on a compact K\"ahler manifold $X$ of dimension $n\geq1$. Let $\nu$ be a finite positive Radon measure on $X$, and let $f,g\geq0$ belong to $L^1(\nu)$ and satisfy
\[
 \langle T^n\rangle\geq f\nu,\qquad
 \langle S^n\rangle\geq g\nu.
\]
Then for every integer $1\leq k\leq n-1$,
\[
 \langle T^k\wedge S^{n-k}\rangle
                        \geq f^{k/n}g^{(n-k)/n}\nu.
\]
The endpoint cases $k=0$ and $k=n$ are the hypotheses; for $n=1$ there are no other cases.
\end{corollary}
\begin{proof}
Apply \cref{prop:nonpluripolar-multifactor-common-measure} with $k$ factors equal to $T$, with density $f$, and $n-k$ factors equal to $S$, with density $g$.
\end{proof}
\index{non-pluripolar product!common-measure mixed inequality}

\begin{corollary}\label{cor:nonpluripolar-common-measure-convex}
Under the hypotheses of \cref{cor:nonpluripolar-common-measure-mixed}, for every $0\leq t\leq1$,
\[
 \left\langle\big((1-t)T+tS\big)^n\right\rangle
       \geq\big((1-t)f^{1/n}+tg^{1/n}\big)^n\nu.
\]
\end{corollary}
\begin{proof}
For $t=0$ and $t=1$ this is the hypothesis. Otherwise, multilinearity expands the left side as $\sum_k\binom nk(1-t)^kt^{n-k}\langle T^k\wedge S^{n-k}\rangle$, and the argument of \cref{cor:bounded-common-measure-sum} applies: we bound the terms with $1\leq k\leq n-1$ by \cref{cor:nonpluripolar-common-measure-mixed} and the terms with $k=0,n$ by the hypotheses, and the binomial formula gives the right side. The measure on the right is finite, since each density $f^{k/n}g^{(n-k)/n}$ is at most $f+g$.
\end{proof}
