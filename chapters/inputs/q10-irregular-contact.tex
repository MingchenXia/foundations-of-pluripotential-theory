\subsection{Contact with an arbitrary singular potential}
\label{sec:irregular-potential-contact}
The metric-Laplacian regularity in \cref{thm:kahler-metric-laplacian-rooftop} applies to an envelope even when the potential below the obstacle is singular. We insert such envelopes between the two functions, compare their regular contact densities, and pass to the singular limit with a bounded level weight. This avoids any assertion of measure convergence for an untruncated decreasing sequence.

\begin{lemma}\label{lem:smooth-decreasing-usc-approximation}
Every upper semicontinuous $u:X\to[-\infty,\infty)$ that is not identically $-\infty$ is the pointwise decreasing limit of smooth real functions $\phi_j$. The approximants need not be quasi-psh.
\end{lemma}
\begin{proof}
Choose a distance $d$ for the compact manifold topology and put $M=\max_Xu<\infty$. For each positive integer $j$ define
\[
 v_j(x)=\max_{y\in X}\{u(y)-j d(x,y)\}.
\]
These are finite $j$-Lipschitz functions: a point where $u$ is finite bounds the maximum below, and the triangle inequality bounds their differences. They decrease and dominate $u$. If $u(x)$ is finite, a maximizing $y_j$ satisfies $j d(x,y_j)\leq M-u(x)$, so $y_j\to x$; upper semicontinuity proves $v_j(x)\downarrow u(x)$. If $u(x)=-\infty$, for every $N$ a small ball about $x$ has $u<-N$, while outside it $M-jd(x,y)<-N$ for large $j$. Thus the same limit holds there.

Set $\phi_0=M+1$ and $\eta_j=4^{-j}$. Inductively suppose $\phi_{j-1}\geq u+\eta_{j-1}$. The continuous function
\[
 q_j=\min\{\phi_{j-1}-2\eta_j,\ v_j+2\eta_j\}
\]
is at least $u+2\eta_j$. Chart convolution and a finite smooth partition approximate a continuous function uniformly: on each smaller chart convolution converges uniformly, and the partition sums the errors with total weight one. Choose a smooth real $\phi_j$ with $\|\phi_j-q_j\|_\infty<\eta_j$. Then
\[
 u+\eta_j\leq\phi_j\leq\phi_{j-1}-\eta_j,
 \qquad \phi_j\leq v_j+3\eta_j.
\]
These bounds prove the induction, monotonicity and the required limit, including the $-\infty$ locus.
\end{proof}

\begin{theorem}\label{thm:irregular-mixed-contact-density}
Let $\theta_1,\ldots,\theta_n$ be closed smooth real $(1,1)$-forms. For each $j$ let $u_j\in\PSH(X,\theta_j)$ and let $f_j$ be continuous and real with bounded distributional Laplacian for the fixed metric $\omega$. Suppose $u_j\leq f_j$ everywhere, and put $E=\bigcap_j\{u_j=f_j\}$. Then
\begin{equation}\label{eq:irregular-mixed-contact-density}
 \mathbf1_E\left\langle\bigwedge_{j=1}^n
                    (\theta_j+\ddc u_j)\right\rangle
   =\mathbf1_E\bigwedge_{j=1}^n(\theta_j+\ddc f_j).
\end{equation}
The right side is the restriction of the signed weak-form measure in \cref{cor:metric-laplacian-barrier-determinants}; on $E$ its density is nonnegative and bounded. No regularity of the $u_j$ is assumed.
\end{theorem}
\begin{proof}
First let $\kappa$ be K\"ahler and let $u\leq f$, where $u$ is $\kappa$-psh and $f$ is a continuous $\kappa$-psh function with bounded $L f$. Write $F=\{u=f\}$ and $\mu_a=\langle(\kappa+\ddc a)^n\rangle$. Take the preceding smooth $\phi_j\downarrow u$ and set
\[
 \psi_j=P_\kappa(\min\{\phi_j,f\}).
\]
Each envelope has bounded metric Laplacian by \cref{thm:kahler-metric-laplacian-rooftop}. Its defining bounds give $u\leq\psi_j\leq f$ and $\psi_j\leq\phi_j$, so $\psi_j\downarrow u$. On $F$ it equals $f$. Regular contact locality, \cref{prop:metric-laplacian-regular-contact-density}, therefore gives
\[
 \mu_{\psi_j}\geq\lambda,
 \qquad \lambda=\mathbf1_F\mu_f.
\]
Here $\mu_f$ is a positive measure, so $\lambda$ is positive too.

Choose $C>0$ with $\min_Xf>-C$, fix $\varepsilon>0$, and put
\[
 \begin{aligned}
 \psi_j^C&=\max\{\psi_j,-C\},&
 u^C&=\max\{u,-C\},\\
 b_j&=\frac{\psi_j^C+C}{\psi_j^C+C+\varepsilon},&
 b&=\frac{u^C+C}{u^C+C+\varepsilon}.
 \end{aligned}
\]
The bounded $\kappa$-psh functions $\psi_j^C$ decrease to $u^C$ and converge locally in capacity by \cref{prop:psh-decreasing-capacity}. The weights are quasi-continuous, bounded by one and converge in capacity, since the scalar map has Lipschitz constant at most $1/\varepsilon$. Weighted bounded continuity in \cref{thm:bt-capacity-continuity}, with smooth local potentials for $\kappa$ and a finite partition, gives
\[
 b_j\mu_{\psi_j^C}\longrightarrow b\mu_{u^C}
 \quad\text{weakly on }X.
\]
Plurifine locality and the vanishing of $b_j$ on $\{\psi_j\leq-C\}$ identify the left measure with $b_j\mu_{\psi_j}$. The limit is similarly $b\mu_u$. On $F$, all weights equal the same strictly positive continuous value
$b_f=(f+C)/(f+C+\varepsilon)$. Thus $b_j\mu_{\psi_j}\geq b_f\lambda$ for every $j$, and the limit gives $b\mu_u\geq b_f\lambda$. Restrict to $F$ and divide by $b_f$, whose reciprocal is bounded on $X$, to obtain $\mathbf1_F\mu_u\geq\lambda$. The ordered-contact inequality in \cref{lem:ordered-potential-contact-measure} gives the reverse inequality. This proves the scalar identity in the K\"ahler case with a psh barrier. A single fixed level weight suffices; no limit in $C$ or $\varepsilon$ is needed.

Next let $\theta$ be any closed smooth real background, let $u\in\PSH(X,\theta)$, and suppose $u\leq f$ and $f$ is $(\theta+A\omega)$-psh for some $A>0$. Increase $A$ so that $\theta+A\omega$ is K\"ahler. For every $t>A$ the first case applies to the K\"ahler background $\theta+t\omega$, with the same fixed measuring metric $\omega$. For a continuous real test $g$ the difference
\[
 Q_g(t)=\int_{\{u=f\}}g\,
        \langle(\theta+t\omega+\ddc u)^n\rangle
        -\int_{\{u=f\}}g\,(\theta+t\omega+\ddc f)^n
\]
is a polynomial in $t$. Indeed positive multilinearity and compatibility with smooth bounded factors expand the first integrand; expansion of the weak forms gives the second. Every coefficient is a finite measure by the earlier global product bounds and \cref{cor:metric-laplacian-barrier-determinants}. The first case gives $Q_g(t)=0$ for all $t>A$, so every coefficient vanishes, in particular $Q_g(0)=0$. Continuous tests determine finite signed measures, proving the scalar identity for this shifted-psh barrier.

For a general $f$ with bounded $Lf$, choose $A>0$ with $\kappa=\theta+A\omega$ K\"ahler and insert
\[
 u\leq v=P_\kappa(f)\leq f.
\]
The new envelope $v$ has bounded metric Laplacian by \cref{thm:kahler-metric-laplacian-rooftop}, and is $\kappa$-psh. The preceding case with barrier $v$ gives
\[
 \mathbf1_{\{u=v\}}\langle(\theta+\ddc u)^n\rangle
   =\mathbf1_{\{u=v\}}(\theta+\ddc v)^n.
\]
Both $v$ and $f$ are $W^{2,p}$ for every finite $p$. Weak Hessian locality in \cref{prop:bounded-complex-hessian-contact} makes their weak forms agree almost everywhere on $\{v=f\}$. Since $\{u=f\}=\{u=v\}\cap\{v=f\}$, restriction of the last identity proves the scalar assertion for $f$. This step does not assume that $f$ becomes psh after a smooth background shift.

Finally, for $\lambda_j>0$ put
$\theta_\lambda=\sum_j\lambda_j\theta_j$,
$u_\lambda=\sum_j\lambda_ju_j$ and
$f_\lambda=\sum_j\lambda_jf_j$.
Then $u_\lambda$ is $\theta_\lambda$-psh, $u_\lambda\leq f_\lambda$, and $E\subset\{u_\lambda=f_\lambda\}$. Restrict the scalar identity to $E$ and expand both sides as measure-valued polynomials in $\lambda$. Testing against a continuous function and comparing the coefficient of $\lambda_1\cdots\lambda_n$ gives $n!$ times each side of \eqref{eq:irregular-mixed-contact-density}. Cancel this factor. The finite barriers exclude all pole points from $E$.

To check boundedness of the contact density, insert for each $j$ its regular envelope $v_j=P_{\theta_j+A_j\omega}(f_j)$. On $\{u_j=f_j\}$ it equals $f_j$, and their weak Hessians agree almost everywhere. The positive form $\theta_j+A_j\omega+\ddc v_j$ has bounded coefficients, so $\theta_j+\ddc f_j$ has bounded coefficients on this contact set. Their mixed determinant is bounded on $E$; its nonnegativity follows from equality with the positive non-pluripolar product.
\end{proof}
\index{contact set!arbitrary singular potentials}

\begin{corollary}\label{cor:big-regular-obstacle-density}
If $\{\theta\}$ is big and $f$ is continuous and real with bounded metric Laplacian, then
\[
 \langle(\theta+\ddc P_\theta(f))^n\rangle
   =\mathbf1_{\{P_\theta(f)=f\}}(\theta+\ddc f)^n.
\]
In particular $\langle(\theta+\ddc V_\theta)^n\rangle
=\mathbf1_{\{V_\theta=0\}}\theta^n$. The same identities hold in every pseudo-effective class after the zero-mass argument in \cref{cor:pseudoeffective-regular-obstacle-density}.
\end{corollary}
\begin{proof}
The family is nonempty because a constant shift of $V_\theta$ lies below $f$. Its envelope is pointwise at most the continuous obstacle. Apply the scalar contact theorem and \cref{thm:quasicontinuous-obstacle-contact}. The case $f=0$ is the definition of $V_\theta$.
\end{proof}
